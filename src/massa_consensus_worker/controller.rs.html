27/// The retrieval of data is made using a shared state and modifications are asked by sending message to a channel.
28/// This is done mostly to be able to:
29///
30/// - send commands through the channel without waiting for them to be processed from the point of view of the sending thread, and channels are very much optimal for that (much faster than locks)
31/// - still be able to read the current state of the graph as processed so far (for this we need a shared state)
32///
33/// Note that sending commands and reading the state is done from different, mutually-asynchronous tasks and they can have data that are not sync yet.
34///
35/// The command channel is bounded and the sends below use `try_send`: when it is full, the command
36/// is dropped rather than waited on. This is deliberate, and waiting instead would be worse:
37///
38/// - the consensus worker blocks on the protocol block-retrieval channel (`send_wishlist_delta`)
39///   while processing these very commands, and the block-retrieval thread is what both drains that
40///   channel and calls `register_block_header`, so waiting here without a deadline deadlocks the
41///   two threads against each other;
42/// - `register_block` is called by the block factory, which does not propagate the block we just
43///   produced until the command has gone through, so waiting would delay the announcement of our
44///   own block precisely when the node is already struggling.
45///
46/// Dropping is acceptable because consensus is already its own retry mechanism: a block whose
47/// dependencies are missing is held in `WaitingForDependencies` and re-requested from peers through
48/// the wishlist. A header that was merely announced to us is therefore fetched again if it turns
49/// out to matter, so layering another retry queue on top of the graph would duplicate that
50/// mechanism rather than add robustness. Channel saturation is visible through the
51/// `consensus_command_channel_actual_size` metric.
52#[derive(Clone)]
53pub struct ConsensusControllerImpl {
54    command_sender: MassaSender<ConsensusCommand>,
55    broadcasts: ConsensusBroadcasts,
56    shared_state: Arc<RwLock<ConsensusState>>,
57    bootstrap_part_size: u64,
58    broadcast_enabled: bool,
59}

120    /// Get a part of the graph to send to a node so that he can setup his graph.
121    /// Used for bootstrap.
122    ///
123    /// # Arguments:
124    /// * `cursor`: streaming cursor containing the current state of bootstrap and what blocks have been to the client already
125    /// * `execution_cursor`: streaming cursor of the final state to ensure that last slot of the bootstrap info corresponds
126    ///
127    /// # Returns:
128    /// * A portion of the graph
129    /// * The list of outdated block ids
130    /// * The streaming step value after the current iteration
131    fn get_bootstrap_part(
132        &self,
133        mut cursor: StreamingStep<PreHashSet<BlockId>>,
134        execution_cursor: StreamingStep<Slot>,
135    ) -> Result<
136        (
137            BootstrapableGraph,
138            PreHashSet<BlockId>,
139            StreamingStep<PreHashSet<BlockId>>,
140        ),
141        ConsensusError,
142    > {
143        let mut final_blocks: Vec<ExportActiveBlock> = Vec::new();
144        let mut retrieved_ids: PreHashSet<BlockId> = PreHashSet::default();
145        let read_shared_state = self.shared_state.read();
146        let required_blocks: PreHashSet<BlockId> = match execution_cursor {
147            StreamingStep::Ongoing(slot) | StreamingStep::Finished(Some(slot)) => {
148                // end_slot must be at least the last genesis slot so that every thread
149                // has a final block with block.slot <= end_slot (see list_required_active_blocks).
150                let last_genesis_slot = Slot::new(
151                    read_shared_state.config.last_start_period,
152                    read_shared_state.config.thread_count.saturating_sub(1),
153                );
154                if slot < last_genesis_slot {
155                    return Err(ConsensusError::ContainerInconsistency(format!(
156                        "bootstrap end_slot {} is before the last genesis slot {}; cannot list a latest final block in every thread",
157                        slot, last_genesis_slot
158                    )));
159                }
160                read_shared_state.list_required_active_blocks(Some(slot))?
161            }
162            _ => PreHashSet::default(),
163        };
164
165        let (current_ids, previous_ids, outdated_ids) = match cursor {
166            StreamingStep::Started => (
167                required_blocks,
168                PreHashSet::default(),
169                PreHashSet::default(),
170            ),
171            StreamingStep::Ongoing(ref cursor_ids) => (
172                // ids that are contained in required_blocks but not in the download cursor => current_ids
173                required_blocks.difference(cursor_ids).cloned().collect(),
174                // ids previously downloaded => previous_ids
175                cursor_ids.clone(),
176                // ids previously downloaded but not contained in required_blocks anymore => outdated_ids
177                cursor_ids.difference(&required_blocks).cloned().collect(),
178            ),
179            StreamingStep::Finished(_) => {
180                return Ok((
181                    BootstrapableGraph { final_blocks },
182                    PreHashSet::default(),
183                    cursor,
184                ))
185            }
186        };
187
188        for b_id in &current_ids {
189            if let Some(BlockStatus::Active {
190                a_block,
191                storage_or_block,
192            }) = read_shared_state.blocks_state.get(b_id)
193            {
194                if final_blocks.len() as u64 >= self.bootstrap_part_size {
195                    break;
196                }
197                match execution_cursor {
198                    StreamingStep::Ongoing(slot) | StreamingStep::Finished(Some(slot)) => {
199                        if a_block.slot > slot {
200                            continue;
201                        }
202                    }
203                    _ => (),
204                }
205                if a_block.is_final {
206                    let export = ExportActiveBlock::from_active_block(a_block, storage_or_block);
207                    final_blocks.push(export);
208                    retrieved_ids.insert(*b_id);
209                }
210            }
211        }
212
213        if final_blocks.is_empty() {
214            cursor = StreamingStep::Finished(None);
215        } else {
216            let pruned_previous_ids = previous_ids.difference(&outdated_ids);
217            retrieved_ids.extend(pruned_previous_ids);
218            cursor = StreamingStep::Ongoing(retrieved_ids);
219        }
220
221        Ok((BootstrapableGraph { final_blocks }, outdated_ids, cursor))
222    }

263    fn register_block(&self, block_id: BlockId, slot: Slot, block_storage: Storage, created: bool) {
264        if self.broadcast_enabled {
265            if let Some(verifiable_block) = block_storage.read_blocks().get(&block_id) {
266                let operations: Vec<(OperationId, Option<SecureShare<Operation, OperationId>>)> =
267                    verifiable_block
268                        .content
269                        .operations
270                        .iter()
271                        .map(|operation_id| {
272                            match block_storage.read_operations().get(operation_id).cloned() {
273                                Some(verifiable_operation) => {
274                                    (*operation_id, Some(verifiable_operation))
275                                }
276                                None => (*operation_id, None),
277                            }
278                        })
279                        .collect();
280
281                if let Err(err) = self.broadcasts.block_sender.send(verifiable_block.clone()) {
282                    trace!(
283                        "error, failed to broadcast block with id {} due to: {}",
284                        block_id,
285                        err
286                    );
287                }
288
289                if let Err(err) = self.broadcasts.filled_block_sender.send(FilledBlock {
290                    header: verifiable_block.content.header.clone(),
291                    operations,
292                }) {
293                    trace!(
294                        "error, failed to broadcast filled block with id {} due to: {}",
295                        block_id,
296                        err
297                    );
298                }
299            } else {
300                debug!(
301                    "error, no broadcast event sent, block with id {} not found",
302                    block_id
303                );
304            };
305        }
306
307        if let Err(err) = self
308            .command_sender
309            .try_send(ConsensusCommand::RegisterBlock(
310                block_id,
311                slot,
312                block_storage,
313                created,
314            ))
315        {
316            warn!("error trying to register a block: {}", err);
317            // dropping on a full channel is intended, see the note on `ConsensusControllerImpl`
318        }
319    }
