15enum WaitingStatus {
16    Ended,
17    Interrupted,
18    Disconnected,
19}
20
21impl ConsensusWorker {
22    /// Execute a command received from the controller also run an update of the graph after processing the command.
23    ///
24    /// # Arguments:
25    /// * `command`: the command to execute
26    ///
27    /// # Returns:
28    /// An error if the command failed
29    fn manage_command(&mut self, command: ConsensusCommand) -> Result<(), ConsensusError> {
30        let mut write_shared_state = self.shared_state.write();
31        match command {
32            ConsensusCommand::RegisterBlockHeader(block_id, header) => {
33                write_shared_state.register_block_header(block_id, header, self.previous_slot)?;
34                write_shared_state.block_db_changed()
35            }
36            ConsensusCommand::RegisterBlock(block_id, slot, block_storage, created) => {
37                write_shared_state.register_block(
38                    block_id,
39                    slot,
40                    self.previous_slot,
41                    block_storage,
42                    created,
43                )?;
44                write_shared_state.block_db_changed()
45            }
46            ConsensusCommand::MarkInvalidBlock(block_id, header) => {
47                write_shared_state.mark_invalid_block(&block_id, header);
48                Ok(())
49            }
50        }
51    }
52
53    /// Wait and interrupt if we receive a command, a stop signal or we reach the `instant`
54    ///
55    /// # Return:
56    /// WaitingStatus::Interrupted => if a command has been executed
57    /// WaitingStatus::Ended => if we reached the `instant`
58    /// WaitingStatus::Disconnected => if we received a stop signal
59    fn wait_slot_or_command(&mut self, deadline: Instant) -> WaitingStatus {
60        match self.command_receiver.recv_deadline(deadline) {
61            // message received => manage it
62            Ok(command) => {
63                if let Err(err) = self.manage_command(command) {
64                    warn!("Error in consensus: {}", err);
65                }
66                WaitingStatus::Interrupted
67            }
68            // timeout => continue main loop
69            Err(crossbeam::channel::RecvTimeoutError::Timeout) => WaitingStatus::Ended,
70            // channel disconnected (sender dropped) => quit main loop
71            Err(crossbeam::channel::RecvTimeoutError::Disconnected) => WaitingStatus::Disconnected,
72        }
73    }
74
75    /// Gets the next slot and the instant when it will happen.
76    /// Slots can be skipped if we waited too much in-between.
77    /// Extra safety against double-production caused by clock adjustments (this is the role of the `previous_slot` parameter).
78    fn get_next_slot(&self, previous_slot: Option<Slot>) -> (Slot, Instant) {
79        // get current absolute time
80        let now = MassaTime::now();
81
82        // get closest slot according to the current absolute time
83        let mut next_slot = get_closest_slot_to_timestamp(
84            self.config.thread_count,
85            self.config.t0,
86            self.config.genesis_timestamp,
87            now,
88        );
89
90        // protection against double-production on unexpected system clock adjustment
91        if let Some(prev_slot) = previous_slot {
92            if next_slot <= prev_slot {
93                next_slot = prev_slot
94                    .get_next_slot(self.config.thread_count)
95                    .expect("could not compute next slot");
96            }
97        }
98
99        // get the timestamp of the target slot
100        let next_instant = get_block_slot_timestamp(
101            self.config.thread_count,
102            self.config.t0,
103            self.config.genesis_timestamp,
104            next_slot,
105        )
106        .expect("could not get block slot timestamp")
107        .estimate_instant()
108        .expect("could not estimate block slot instant");
109
110        (next_slot, next_instant)
111    }
112
113    /// Performs all per-slot maintenance: cycle logging, `slot_tick`, pruning,
114    /// and advancing to the next slot.
115    ///
116    /// # Arguments
117    /// * `last_prune`: instant when the block DB was last pruned
118    fn process_slot_tick(&mut self, last_prune: &mut Instant) {
119        let previous_cycle = self
120            .previous_slot
121            .map(|s| s.get_cycle(self.config.periods_per_cycle));
122        let observed_cycle = self.next_slot.get_cycle(self.config.periods_per_cycle);
123        if previous_cycle.is_none() {
124            // first cycle observed
125            info!("Massa network has started ! 🎉")
126        }
127        if previous_cycle < Some(observed_cycle) {
128            info!("Started cycle {}", observed_cycle);
129        }
130        // Execute all operations and checks that should be performed at each slot
131        {
132            let mut write_shared_state = self.shared_state.write();
133            if let Err(err) = write_shared_state.slot_tick(self.next_slot) {
134                warn!("Error while processing block tick: {}", err);
135                // Side effects are preserved in `to_propagate`, `attack_attempts`,
136                // `new_final_blocks` and `new_stale_blocks`, so they will be
137                // retried on the next tick. Keep advancing slots to avoid
138                // halting consensus while downstream is under transient pressure.
139            }
140        };
141        if last_prune.elapsed().as_millis()
142            > self.config.block_db_prune_interval.as_millis() as u128
143        {
144            self.shared_state
145                .write()
146                .prune()
147                .expect("Error while pruning");
148            *last_prune = Instant::now();
149        }
150        self.previous_slot = Some(self.next_slot);
151        (self.next_slot, self.next_instant) = self.get_next_slot(Some(self.next_slot));
152    }
153
154    /// Runs in loop forever. This loop must stop every slot to perform operations on stats and graph
155    /// but can be stopped anytime by a command received.
156    pub fn run(&mut self) {
157        let mut last_prune = Instant::now();
158        loop {
159            match self.wait_slot_or_command(self.next_instant) {
160                // When we reached the instant of the next slot
161                WaitingStatus::Ended => {
162                    self.process_slot_tick(&mut last_prune);
163                }
164                WaitingStatus::Disconnected => {
165                    break;
166                }
167                WaitingStatus::Interrupted => {
168                    // A command was processed. If the slot deadline has already passed,
169                    // run the overdue slot maintenance before waiting again so that a
170                    // sustained flood of commands cannot starve time-driven work.
171                    if Instant::now() >= self.next_instant {
172                        self.process_slot_tick(&mut last_prune);
173                    }
174                }
175            };
176        }
177    }
178}
