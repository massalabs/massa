34/// Creates genesis block in given thread.
35///
36/// # Arguments
37/// * `cfg`: consensus configuration
38/// * `thread_number`: thread in which we want a genesis block
39///
40/// # Returns
41/// A genesis block
42pub fn create_genesis_block(
43    cfg: &ConsensusConfig,
44    thread_number: u8,
45) -> Result<SecureShareBlock, ConsensusError> {
46    let keypair = &cfg.genesis_key;
47    let header = BlockHeader::new_verifiable(
48        BlockHeader {
49            current_version: 0,
50            announced_version: None,
51            slot: Slot::new(cfg.last_start_period, thread_number),
52            parents: Vec::new(),
53            operation_merkle_root: Hash::compute_from(&Vec::new()),
54            endorsements: Vec::new(),
55            denunciations: Vec::new(),
56        },
57        BlockHeaderSerializer::new(),
58        keypair,
59        cfg.chain_id,
60    )?;
61
62    Ok(Block::new_verifiable(
63        Block {
64            header,
65            operations: Default::default(),
66        },
67        BlockSerializer::new(),
68        keypair,
69        cfg.chain_id,
70    )?)
71}

73impl ConsensusWorker {
74    /// Creates a new consensus worker.
75    ///
76    /// # Arguments
77    /// * `config`: consensus configuration
78    /// * `command_receiver`: channel to receive commands from controller
79    /// * `channels`: channels to communicate with other workers
80    /// * `shared_state`: shared state with the controller
81    /// * `init_graph`: Optional graph of blocks to initiate the worker
82    /// * `storage`: shared storage
83    ///
84    /// # Returns:
85    /// A `ConsensusWorker`, to interact with it use the `ConsensusController`
86    pub fn new(
87        config: ConsensusConfig,
88        command_receiver: MassaReceiver<ConsensusCommand>,
89        shared_state: Arc<RwLock<ConsensusState>>,
90        init_graph: Option<BootstrapableGraph>,
91        storage: Storage,
92    ) -> Result<Self, ConsensusError> {
93        let now = MassaTime::now();
94        let previous_slot = get_latest_block_slot_at_timestamp(
95            config.thread_count,
96            config.t0,
97            config.genesis_timestamp,
98            now,
99        )
100        .expect("Couldn't get the init slot consensus.");
101
102        // load genesis blocks
103        let mut block_statuses = PreHashMap::default();
104        let mut genesis_block_ids = Vec::with_capacity(config.thread_count as usize);
105        for thread in 0u8..config.thread_count {
106            let block = create_genesis_block(&config, thread).map_err(|err| {
107                ConsensusError::GenesisCreationError(format!("genesis error {}", err))
108            })?;
109            let mut storage = storage.clone_without_refs();
110            storage.store_block(block.clone());
111            genesis_block_ids.push(block.id);
112            block_statuses.insert(
113                block.id,
114                BlockStatus::Active {
115                    a_block: Box::new(ActiveBlock {
116                        creator_address: block.content_creator_address,
117                        parents: Vec::new(),
118                        children: vec![PreHashMap::default(); config.thread_count as usize],
119                        descendants: Default::default(),
120                        is_final: true,
121                        block_id: block.id,
122                        slot: block.content.header.content.slot,
123                        fitness: block.get_fitness(),
124                        same_thread_parent_creator: None,
125                    }),
126                    storage_or_block: StorageOrBlock::Storage(storage),
127                },
128            );
129        }
130
131        let next_slot = previous_slot.map_or(Ok(Slot::new(0u64, 0u8)), |s| {
132            s.get_next_slot(config.thread_count)
133        })?;
134        let next_instant = get_block_slot_timestamp(
135            config.thread_count,
136            config.t0,
137            config.genesis_timestamp,
138            next_slot,
139        )?
140        .estimate_instant()?;
141
142        info!(
143            "Started node at time {}, cycle {}, period {}, thread {}",
144            now.format_instant(),
145            next_slot.get_cycle(config.periods_per_cycle),
146            next_slot.period,
147            next_slot.thread,
148        );
149
150        if config.genesis_timestamp > now {
151            let (days, hours, mins, secs) = config
152                .genesis_timestamp
153                .saturating_sub(now)
154                .days_hours_mins_secs()?;
155            info!(
156                "{} days, {} hours, {} minutes, {} seconds remaining to genesis",
157                days, hours, mins, secs,
158            )
159        }
160
161        if config.last_start_period > 0
162            && config
163                .genesis_timestamp
164                .checked_add(config.t0.checked_mul(config.last_start_period)?)?
165                > now
166        {
167            let (days, hours, mins, secs) = config
168                .genesis_timestamp
169                .checked_add(config.t0.checked_mul(config.last_start_period)?)?
170                .saturating_sub(now)
171                .days_hours_mins_secs()?;
172            info!(
173                "{} days, {} hours, {} minutes, {} seconds remaining to network restart",
174                days, hours, mins, secs,
175            )
176        }
177
178        // add genesis blocks to stats
179        let genesis_addr = Address::from_public_key(&config.genesis_key.get_public_key());
180        let mut final_block_stats = VecDeque::new();
181        for thread in 0..config.thread_count {
182            final_block_stats.push_back((
183                get_block_slot_timestamp(
184                    config.thread_count,
185                    config.t0,
186                    config.genesis_timestamp,
187                    Slot::new(config.last_start_period, thread),
188                )?,
189                genesis_addr,
190                false,
191            ))
192        }
193
194        let mut res_consensus = ConsensusWorker {
195            config: config.clone(),
196            command_receiver,
197            shared_state,
198            previous_slot,
199            next_slot,
200            next_instant,
201        };
202
203        // If the node starts after the genesis timestamp then it has to initialize its graph
204        // with already produced blocks received from the bootstrap.
205        if let Some(BootstrapableGraph { final_blocks }) = init_graph {
206            // load final blocks
207            let final_blocks: Vec<(ActiveBlock, StorageOrBlock)> = final_blocks
208                .into_iter()
209                .map(|export_b| export_b.to_active_block(config.thread_count))
210                .collect::<Result<_, ConsensusError>>()?;
211
212            // Exporters only send blocks with is_final (see get_bootstrap_part). Enforce that
213            // locally so a compromised/misconfigured bootstrap source cannot advance the
214            // finality frontier with non-final blocks.
215            if let Some((b, _)) = final_blocks.iter().find(|(b, _)| !b.is_final) {
216                return Err(ConsensusError::ContainerInconsistency(format!(
217                    "bootstrap final_blocks contains non-final block {} at slot {}",
218                    b.block_id, b.slot
219                )));
220            }
221
222            // compute latest_final_blocks_periods from verified final blocks only
223            let mut latest_final_blocks_periods: Vec<(BlockId, u64)> =
224                genesis_block_ids.iter().map(|id| (*id, 0u64)).collect();
225            for (b, _) in &final_blocks {
226                if let Some(v) = latest_final_blocks_periods.get_mut(b.slot.thread as usize) {
227                    if b.slot.period > v.1 {
228                        *v = (b.block_id, b.slot.period);
229                    }
230                }
231            }
232            // Initialize the shared state between the worker and the interface used by the other modules.
233            {
234                let mut write_shared_state = res_consensus.shared_state.write();
235                write_shared_state.genesis_hashes = genesis_block_ids;
236                write_shared_state.best_parents = latest_final_blocks_periods.clone();
237                write_shared_state.latest_final_blocks_periods = latest_final_blocks_periods;
238                for (b, storage_or_block) in final_blocks {
239                    write_shared_state.blocks_state.transition_map(
240                        &(b.block_id.clone()),
241                        |_, _| {
242                            Some(BlockStatus::Active {
243                                a_block: Box::new(b),
244                                storage_or_block,
245                            })
246                        },
247                    );
248                }
249                write_shared_state.final_block_stats = final_block_stats;
250            }
251
252            res_consensus.claim_parent_refs()?;
253        } else {
254            // Initialize the shared state between the worker and the interface used by the other modules.
255            {
256                let mut write_shared_state = res_consensus.shared_state.write();
257                write_shared_state.latest_final_blocks_periods =
258                    genesis_block_ids.iter().map(|h| (*h, 0)).collect();
259                write_shared_state.best_parents =
260                    genesis_block_ids.iter().map(|v| (*v, 0)).collect();
261                write_shared_state.genesis_hashes = genesis_block_ids;
262                for (b, s) in block_statuses {
263                    write_shared_state
264                        .blocks_state
265                        .transition_map(&b, |_, _| Some(s));
266                }
267                write_shared_state.final_block_stats = final_block_stats;
268            }
269        }
270
271        // Notify execution module of current blockclique and all final blocks.
272        // we need to do this because the bootstrap snapshots of the executor vs the consensus may not have been taken in sync
273        // because the two modules run concurrently and out of sync.
274        {
275            let mut write_shared_state = res_consensus.shared_state.write();
276            let mut block_metadata: PreHashMap<BlockId, ExecutionBlockMetadata> =
277                Default::default();
278            let notify_finals: HashMap<Slot, BlockId> = write_shared_state
279                .get_all_final_blocks()
280                .into_iter()
281                .map(|(b_id, (b_slot, b_metadata))| {
282                    block_metadata.insert(b_id, b_metadata);
283                    (b_slot, b_id)
284                })
285                .collect();
286            let notify_blockclique: HashMap<Slot, BlockId> = write_shared_state
287                .get_blockclique()
288                .iter()
289                .map(|b_id| {
290                    let (a_block, storage_or_block) = write_shared_state
291                        .get_full_active_block(b_id)
292                        .expect("active block missing from block_db");
293                    let storage = match storage_or_block {
294                        StorageOrBlock::Storage(storage) => Some(storage.clone()),
295                        _ => None,
296                    };
297                    block_metadata.insert(
298                        *b_id,
299                        ExecutionBlockMetadata {
300                            same_thread_parent_creator: a_block.same_thread_parent_creator,
301                            storage,
302                        },
303                    );
304                    (a_block.slot, *b_id)
305                })
306                .collect();
307            write_shared_state.prev_blockclique =
308                notify_blockclique.iter().map(|(k, v)| (*v, *k)).collect();
309            write_shared_state
310                .channels
311                .execution_controller
312                .update_blockclique_status(notify_finals, Some(notify_blockclique), block_metadata);
313        }
314
315        Ok(res_consensus)
316    }

318    /// Internal function used at initialization of the `ConsensusWorker` to link blocks with their parents
319    fn claim_parent_refs(&mut self) -> Result<(), ConsensusError> {
320        let mut write_shared_state = self.shared_state.write();
321
322        // list active block parents
323        let active_blocks_map: PreHashMap<BlockId, (Slot, Vec<BlockId>)> = write_shared_state
324            .blocks_state
325            .iter()
326            .filter_map(|(h, s)| {
327                if let BlockStatus::Active { a_block: a, .. } = s {
328                    return Some((*h, (a.slot, a.parents.iter().map(|(ph, _)| *ph).collect())));
329                }
330                None
331            })
332            .collect();
333
334        for (b_id, (b_slot, b_parents)) in active_blocks_map.into_iter() {
335            // update aprenthood relationships
336            write_shared_state.insert_parents_descendants(b_id, b_slot, b_parents);
337        }
338        Ok(())
339    }
