131    /// get the blockclique (or final) block ID at a given slot, if any
132    pub fn get_blockclique_block_at_slot(&self, slot: &Slot) -> Option<BlockId> {
133        // List all blocks at this slot.
134        // The list should be small: make a copy of it to avoid holding the storage lock.
135        let blocks_at_slot = {
136            let storage_read = self.storage.read_blocks();
137            let returned = match storage_read.get_blocks_by_slot(slot) {
138                Some(v) => v.clone(),
139                None => return None,
140            };
141            returned
142        };
143
144        // search for the block in the blockclique
145        let search_in_blockclique = blocks_at_slot
146            .intersection(
147                &self
148                    .max_cliques
149                    .iter()
150                    .find(|c| c.is_blockclique)
151                    .expect("expected one clique to be the blockclique")
152                    .block_ids,
153            )
154            .next();
155        if let Some(found_id) = search_in_blockclique {
156            return Some(*found_id);
157        }
158
159        // block not found in the blockclique: search in the final blocks
160        blocks_at_slot
161            .into_iter()
162            .find(|b_id| match self.blocks_state.get(b_id) {
163                Some(BlockStatus::Active { a_block, .. }) => a_block.is_final,
164                _ => false,
165            })
166    }

168    /// get the latest blockclique (or final) block ID at a given slot, if any
169    pub fn get_latest_blockclique_block_at_slot(&self, slot: &Slot) -> BlockId {
170        let (mut best_block_id, mut best_block_period) = self
171            .latest_final_blocks_periods
172            .get(slot.thread as usize)
173            .unwrap_or_else(|| panic!("unexpected not found latest final block period"));
174
175        self.get_blockclique()
176            .iter()
177            .for_each(|id| match self.blocks_state.get(id) {
178                Some(BlockStatus::Active { a_block, .. }) => {
179                    if a_block.is_final {
180                        panic!(
181                            "unexpected final block on getting latest blockclique block at slot"
182                        );
183                    }
184                    if a_block.slot.thread == slot.thread
185                        && a_block.slot.period < slot.period
186                        && a_block.slot.period > best_block_period
187                    {
188                        best_block_period = a_block.slot.period;
189                        best_block_id = *id;
190                    }
191                }
192                _ => {
193                    panic!("expected to find only active block but found another status")
194                }
195            });
196        best_block_id
197    }

199    pub fn get_block_status(&self, block_id: &BlockId) -> BlockGraphStatus {
200        match self.blocks_state.get(block_id) {
201            None => BlockGraphStatus::NotFound,
202            Some(BlockStatus::Active { a_block, .. }) => {
203                if a_block.is_final {
204                    BlockGraphStatus::Final
205                } else if self
206                    .max_cliques
207                    .iter()
208                    .find(|clique| clique.is_blockclique)
209                    .expect("blockclique absent")
210                    .block_ids
211                    .contains(block_id)
212                {
213                    BlockGraphStatus::ActiveInBlockclique
214                } else {
215                    BlockGraphStatus::ActiveInAlternativeCliques
216                }
217            }
218            Some(BlockStatus::Discarded { .. }) => BlockGraphStatus::Discarded,
219            Some(BlockStatus::Incoming(_)) => BlockGraphStatus::Incoming,
220            Some(BlockStatus::WaitingForDependencies { .. }) => {
221                BlockGraphStatus::WaitingForDependencies
222            }
223            Some(BlockStatus::WaitingForSlot(_)) => BlockGraphStatus::WaitingForSlot,
224        }
225    }

227    /// list the latest final blocks at the given slot
228    ///
229    /// exclusively used by `list_required_active_blocks`
230    fn list_latest_final_blocks_at(
231        &self,
232        slot: Slot,
233    ) -> Result<Vec<(BlockId, u64)>, ConsensusError> {
234        let mut latest: Vec<Option<(BlockId, u64)>> = vec![None; self.config.thread_count as usize];
235        for id in self.blocks_state.active_blocks().iter() {
236            let (block, _storage) = self.try_get_full_active_block(id)?;
237            if let Some((_, p)) = latest[block.slot.thread as usize] {
238                if block.slot.period < p {
239                    continue;
240                }
241            }
242            if block.is_final && block.slot <= slot {
243                latest[block.slot.thread as usize] = Some((*id, block.slot.period));
244            }
245        }
246        latest
247            .into_iter()
248            .enumerate()
249            .map(|(thread, opt)| {
250                opt.ok_or_else(|| {
251                    ConsensusError::ContainerInconsistency(format!(
252                        "could not find latest block for thread {}",
253                        thread
254                    ))
255                })
256            })
257            .collect()
258    }

260    /// list the earliest blocks of the given block id list
261    ///
262    /// exclusively used by `list_required_active_blocks`
263    fn list_earliest_blocks_of(
264        &self,
265        block_ids: &PreHashSet<BlockId>,
266        end_slot: Option<Slot>,
267    ) -> Result<Vec<(BlockId, u64)>, ConsensusError> {
268        let mut earliest: Vec<Option<(BlockId, u64)>> =
269            vec![None; self.config.thread_count as usize];
270        for id in block_ids {
271            let (block, _storage) = self.try_get_full_active_block(id)?;
272            if let Some(slot) = end_slot {
273                if block.slot > slot {
274                    continue;
275                }
276            }
277            if let Some((_, p)) = earliest[block.slot.thread as usize] {
278                if block.slot.period > p {
279                    continue;
280                }
281            }
282            earliest[block.slot.thread as usize] = Some((*id, block.slot.period));
283        }
284        earliest
285            .into_iter()
286            .enumerate()
287            .map(|(thread, opt)| {
288                opt.ok_or_else(|| {
289                    ConsensusError::ContainerInconsistency(format!(
290                        "could not find earliest block for thread {}",
291                        thread
292                    ))
293                })
294            })
295            .collect()
296    }

298    /// adds to the given container every active block coming after the lower bound
299    ///
300    /// exclusively used by `list_required_active_blocks`
301    fn add_active_blocks_after(
302        &self,
303        kept_blocks: &mut PreHashSet<BlockId>,
304        lower_bound: &[(BlockId, u64)],
305        end_slot: Option<Slot>,
306    ) {
307        for id in self.blocks_state.active_blocks().iter() {
308            if let Some((block, _storage)) = self.get_full_active_block(id) {
309                if let Some(slot) = end_slot {
310                    if block.slot > slot {
311                        continue;
312                    }
313                }
314                if block.slot.period >= lower_bound[block.slot.thread as usize].1 {
315                    kept_blocks.insert(*id);
316                }
317            }
318        }
319    }

321    /// Lists the active block IDs that must be kept: the latest finals of each thread, all active
322    /// blocks after them, and exactly 2 rounds of "add parents, then fill holes down to the
323    /// earliest kept block of each thread".
324    ///
325    /// This is deliberately NOT a transitive closure over parents and must not become one:
326    /// - a fixpoint would walk back one ancestor per round until it hits a block already pruned
327    ///   (`Discarded { reason: Final }`), at which point `try_get_full_active_block` errors,
328    ///   `prune()` panics and `get_bootstrap_part` fails. Tests do not catch this because test
329    ///   graphs are never pruned.
330    /// - no consumer needs a closure:
331    ///   - `prune_active` uses this as a floor and additionally keeps
332    ///     `force_keep_final_periods_without_ops` periods of finals per thread. The set is
333    ///     self-sustaining (what it kept last time is still active next time), and a block whose
334    ///     parent was pruned can never become Active (the parent comes back Stale and the child
335    ///     is discarded), so no dangling parent can appear.
336    ///   - `get_bootstrap_part` exports only the final blocks of this set. A valid block compatible
337    ///     with final F_t can only reference F_t's own parent in thread t (grandpa rule), or one
338    ///     level deeper in thread-gap corner cases. Round 1 adds the parents of the finals, round 2
339    ///     covers the deeper case. Blocks validated after bootstrap are all after `end_slot` and
340    ///     are not in this set anyway, so a closure over blocks up to `end_slot` would not help.
341    pub fn list_required_active_blocks(
342        &self,
343        end_slot: Option<Slot>,
344    ) -> Result<PreHashSet<BlockId>, ConsensusError> {
345        // if an end_slot is provided compute the latest final block for that given slot
346        // if not use the latest_final_blocks_periods
347        let effective_latest_finals: Vec<(BlockId, u64)> = if let Some(slot) = end_slot {
348            self.list_latest_final_blocks_at(slot)?
349        } else {
350            self.latest_final_blocks_periods.clone()
351        };
352
353        // init kept_blocks using effective_latest_finals
354        let mut kept_blocks: PreHashSet<BlockId> = effective_latest_finals
355            .iter()
356            .map(|(id, _period)| *id)
357            .collect();
358
359        // add all the active blocks that are after the effective_latest_finals of their thread
360        self.add_active_blocks_after(&mut kept_blocks, &effective_latest_finals, end_slot);
361
362        // do the following 2 times
363        for _ in 0..2 {
364            // extend kept_blocks with the parents of the current kept_blocks
365            let mut cumulated_parents: PreHashSet<BlockId> = PreHashSet::default();
366            for id in kept_blocks.iter() {
367                let parents = self
368                    .try_get_full_active_block(id)?
369                    .0
370                    .parents
371                    .iter()
372                    .map(|(id, _period)| *id);
373                cumulated_parents.extend(parents);
374            }
375            kept_blocks.extend(cumulated_parents);
376            // add all the active blocks whose slot is after the earliest kept_blocks of their thread
377            let earliest_blocks = self.list_earliest_blocks_of(&kept_blocks, end_slot)?;
378            self.add_active_blocks_after(&mut kept_blocks, &earliest_blocks, end_slot);
379        }
380
381        // check that we have the full blocks for every id we are about to return
382        for id in kept_blocks.iter() {
383            self.try_get_full_active_block(id)?;
384        }
385
386        // debug log for an easier diagnostic if needed
387        debug!("list_required_active_blocks return: {:?}", kept_blocks);
388
389        // return kept_blocks
390        Ok(kept_blocks)
391    }

393    pub fn extract_block_graph_part(
394        &self,
395        slot_start: Option<Slot>,
396        slot_end: Option<Slot>,
397    ) -> Result<BlockGraphExport, ConsensusError> {
398        let mut export = BlockGraphExport {
399            genesis_blocks: self.genesis_hashes.clone(),
400            active_blocks: PreHashMap::with_capacity(self.blocks_state.len()),
401            discarded_blocks: PreHashMap::with_capacity(self.blocks_state.len()),
402            best_parents: self.best_parents.clone(),
403            latest_final_blocks_periods: self.latest_final_blocks_periods.clone(),
404            gi_head: self.gi_head.clone(),
405            max_cliques: self.max_cliques.clone(),
406        };
407
408        let filter = |&s| {
409            if let Some(s_start) = slot_start {
410                if s < s_start {
411                    return false;
412                }
413            }
414            if let Some(s_end) = slot_end {
415                if s >= s_end {
416                    return false;
417                }
418            }
419            true
420        };
421
422        for (block_id, block) in self.blocks_state.iter() {
423            match block {
424                BlockStatus::Discarded {
425                    slot,
426                    creator,
427                    parents,
428                    reason,
429                    ..
430                } => {
431                    if filter(slot) {
432                        export.discarded_blocks.insert(
433                            *block_id,
434                            (reason.clone(), (*slot, *creator, parents.clone())),
435                        );
436                    }
437                }
438                BlockStatus::Active {
439                    a_block,
440                    storage_or_block,
441                } => {
442                    if filter(&a_block.slot) {
443                        export.active_blocks.insert(
444                            *block_id,
445                            ExportCompiledBlock {
446                                header: storage_or_block.clone_block(block_id).content.header,
447                                children: a_block
448                                    .children
449                                    .iter()
450                                    .map(|thread| {
451                                        thread.keys().copied().collect::<PreHashSet<BlockId>>()
452                                    })
453                                    .collect(),
454                                is_final: a_block.is_final,
455                            },
456                        );
457                    }
458                }
459                _ => continue,
460            }
461        }
462
463        Ok(export)
464    }

466    /// Gets all stored final blocks, not only the still-useful ones
467    /// This is used when initializing Execution from Consensus.
468    /// Since the Execution bootstrap snapshot is older than the Consensus snapshot,
469    /// we might need to signal older final blocks for Execution to catch up.
470    pub fn get_all_final_blocks(&self) -> HashMap<BlockId, (Slot, ExecutionBlockMetadata)> {
471        self.blocks_state
472            .active_blocks()
473            .iter()
474            .filter_map(|b_id| {
475                if let Some(BlockStatus::Active {
476                    a_block,
477                    storage_or_block,
478                }) = self.blocks_state.get(b_id)
479                {
480                    if !a_block.is_final {
481                        return None;
482                    }
483                    let storage = match storage_or_block {
484                        StorageOrBlock::Storage(storage) => Some(storage.clone()),
485                        _ => None,
486                    };
487                    return Some((
488                        *b_id,
489                        (
490                            a_block.slot,
491                            ExecutionBlockMetadata {
492                                same_thread_parent_creator: a_block.same_thread_parent_creator,
493                                storage,
494                            },
495                        ),
496                    ));
497                }
498                None
499            })
500            .collect()
501    }

503    /// get the current block wish list, including the operations hash.
504    pub fn get_block_wishlist(
505        &self,
506    ) -> Result<PreHashMap<BlockId, Option<SecuredHeader>>, ConsensusError> {
507        let mut wishlist = PreHashMap::<BlockId, Option<SecuredHeader>>::default();
508        for block_id in self.blocks_state.waiting_for_dependencies_blocks().iter() {
509            if let Some(BlockStatus::WaitingForDependencies {
510                unsatisfied_dependencies,
511                ..
512            }) = self.blocks_state.get(block_id)
513            {
514                for unsatisfied_h in unsatisfied_dependencies.iter() {
515                    match self.blocks_state.get(unsatisfied_h) {
516                        Some(BlockStatus::WaitingForDependencies {
517                            header_or_block: HeaderOrBlock::Header(header),
518                            ..
519                        }) => {
520                            wishlist.insert(header.id, Some(header.clone()));
521                        }
522                        None => {
523                            wishlist.insert(*unsatisfied_h, None);
524                        }
525                        _ => {}
526                    }
527                }
528            }
529        }
530
531        Ok(wishlist)
532    }

534    /// Recompute the block wishlist from the currently retained waiting blocks
535    /// and publish the delta (new and removed entries) to protocol.
536    ///
537    /// Invariant: the published wishlist is always derived exclusively from
538    /// currently retained, reachable waiting blocks. This must be called after
539    /// any operation that removes or invalidates waiting-for-dependency blocks
540    /// (e.g. `block_db_changed`, `prune_waiting_for_dependencies`) so that
541    /// protocol does not keep requesting blocks that consensus has already
542    /// discarded as unreachable.
543    ///
544    /// # Returns
545    /// Error if the wishlist could not be computed or sent.
546    pub fn sync_wishlist(&mut self) -> Result<(), ConsensusError> {
547        // compute the block wishlist from currently retained waiting blocks
548        let new_wishlist = self.get_block_wishlist()?;
549        let new_blocks: PreHashMap<BlockId, Option<SecuredHeader>> = new_wishlist
550            .iter()
551            .filter_map(|(id, header)| {
552                if !self.wishlist.contains_key(id) {
553                    Some((*id, header.clone()))
554                } else {
555                    None
556                }
557            })
558            .collect();
559        let remove_blocks: PreHashSet<BlockId> = self
560            .wishlist
561            .iter()
562            .filter_map(|(id, _)| {
563                if !new_wishlist.contains_key(id) {
564                    Some(*id)
565                } else {
566                    None
567                }
568            })
569            .collect();
570        if !new_blocks.is_empty() || !remove_blocks.is_empty() {
571            massa_trace!("consensus.consensus_worker.sync_wishlist.send_wishlist_delta", { "new": new_wishlist, "remove": remove_blocks });
572            self.channels
573                .protocol_controller
574                .send_wishlist_delta(new_blocks, remove_blocks)?;
575            self.wishlist = new_wishlist;
576        }
577        Ok(())
578    }

580    /// Gets a block and all its descendants
581    ///
582    /// # Argument
583    /// * hash : hash of the given block
584    pub fn get_active_block_and_descendants(&self, block_id: &BlockId) -> PreHashSet<BlockId> {
585        let mut to_visit = vec![*block_id];
586        let mut result = PreHashSet::<BlockId>::default();
587        while let Some(visit_h) = to_visit.pop() {
588            if !result.insert(visit_h) {
589                continue; // already visited
590            }
591            match self.blocks_state.get(&visit_h) {
592                Some(BlockStatus::Active { a_block, .. }) => {
593                    a_block.as_ref()
594                    .children.iter()
595                    .for_each(|thread_children| to_visit.extend(thread_children.keys()))
596                },
597                _ => panic!("inconsistency inside block statuses iterating through descendants of {} - missing {}", block_id, visit_h),
598            }
599        }
600        result
601    }
