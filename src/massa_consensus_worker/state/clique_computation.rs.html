10/// Computes max cliques of compatible blocks
11pub fn compute_max_cliques(
12    gi_head: &PreHashMap<BlockId, PreHashSet<BlockId>>,
13) -> Vec<PreHashSet<BlockId>> {
14    let mut max_cliques: Vec<PreHashSet<BlockId>> = Vec::new();
15
16    // algorithm adapted from IK_GPX as summarized in:
17    //   Cazals et al., "A note on the problem of reporting maximal cliques"
18    //   Theoretical Computer Science, 2008
19    //   https://doi.org/10.1016/j.tcs.2008.05.010
20
21    // stack: r, p, x
22    let mut stack: Vec<(
23        PreHashSet<BlockId>,
24        PreHashSet<BlockId>,
25        PreHashSet<BlockId>,
26    )> = vec![(
27        PreHashSet::<BlockId>::default(),
28        gi_head.keys().cloned().collect(),
29        PreHashSet::<BlockId>::default(),
30    )];
31    while let Some((r, mut p, mut x)) = stack.pop() {
32        if p.is_empty() && x.is_empty() {
33            max_cliques.push(r);
34            continue;
35        }
36        // choose the pivot vertex following the GPX scheme:
37        // u_p = node from (p \/ x) that maximizes the cardinality of (P \ Neighbors(u_p, GI))
38        let &u_p = p
39            .union(&x)
40            .max_by_key(|&u| {
41                p.difference(&(&gi_head[u] | &vec![*u].into_iter().collect()))
42                    .count()
43            })
44            .unwrap(); // p was checked to be non-empty before
45
46        // iterate over u_set = (p /\ Neighbors(u_p, GI))
47        let u_set: PreHashSet<BlockId> = &p & &(&gi_head[&u_p] | &vec![u_p].into_iter().collect());
48        for u_i in u_set.into_iter() {
49            p.remove(&u_i);
50            let u_i_set: PreHashSet<BlockId> = vec![u_i].into_iter().collect();
51            let comp_n_u_i: PreHashSet<BlockId> = &gi_head[&u_i] | &u_i_set;
52            stack.push((&r | &u_i_set, &p - &comp_n_u_i, &x - &comp_n_u_i));
53            x.insert(u_i);
54        }
55    }
56    max_cliques
57}

59/// Tests
60#[cfg(test)]
61mod tests {
62    use crate::state::clique_computation::compute_max_cliques;
63    use itertools::Itertools;
64    use massa_models::{
65        block_id::BlockId,
66        prehash::{PreHashMap, PreHashSet},
67    };
68    use rand::Rng;
69
70    #[test]
71    fn test_compute_max_cliques() {
72        // Define the maximum size of the graph and the number of iterations
73        const MAX_SIZE: usize = 10;
74        const ITERATIONS: usize = 1000;
75
76        // Generate random test cases and run the algorithm
77        let mut rng = rand::thread_rng();
78
79        for _ in 0..ITERATIONS {
80            // Generate a random graph size
81            let size = rng.gen_range(0..=MAX_SIZE);
82
83            // Generate random incompatibility relationships
84            let mut gi_head = PreHashMap::default();
85            for i in 0..size {
86                gi_head.insert(
87                    BlockId::generate_from_hash(massa_hash::Hash::compute_from(&i.to_be_bytes())),
88                    PreHashSet::default(),
89                );
90            }
91            for i in 0..size.saturating_sub(1) {
92                for j in (i + 1)..size {
93                    // Generate a random compatibility relationship
94                    let is_compatible = rng.gen_bool(0.5);
95
96                    if !is_compatible {
97                        let i_id = BlockId::generate_from_hash(massa_hash::Hash::compute_from(
98                            &i.to_be_bytes(),
99                        ));
100                        let j_id = BlockId::generate_from_hash(massa_hash::Hash::compute_from(
101                            &j.to_be_bytes(),
102                        ));
103                        // Add the incompatibility relationship to gi_head
104                        gi_head.entry(i_id).or_default().insert(j_id);
105                        gi_head.entry(j_id).or_default().insert(i_id);
106                    }
107                }
108            }
109
110            // Check cliques
111            assert_cliques_valid(&gi_head, &compute_max_cliques(&gi_head));
112        }
113    }
114
115    /// Assert that a set of cliques is valid
116    fn assert_cliques_valid(
117        gi_head: &PreHashMap<BlockId, PreHashSet<BlockId>>,
118        max_cliques: &Vec<PreHashSet<BlockId>>,
119    ) {
120        // Check that there is at least one clique
121        if max_cliques.is_empty() {
122            panic!("max_cliques is empty");
123        }
124
125        // Check that all cliques are unique
126        for (i, clique1) in max_cliques.iter().enumerate() {
127            for (j, clique2) in max_cliques.iter().enumerate() {
128                if i != j && clique1 == clique2 {
129                    panic!("two of the cliques are identical");
130                }
131            }
132        }
133
134        for clique in max_cliques {
135            // Check that all pairs of vertices in the clique are compatible
136            for (v1, v2) in clique.iter().tuple_combinations() {
137                if gi_head[v1].contains(v2) || gi_head[v2].contains(v1) {
138                    panic!("incompatible vertices found within the same clique");
139                }
140            }
141
142            // Check that the clique is maximal
143            for v in gi_head.keys() {
144                if !clique.contains(v) && clique.iter().all(|c| !gi_head[v].contains(c)) {
145                    panic!("a clique is non-maximal");
146                }
147            }
148        }
149
150        // All cliques are valid, unique and maximal
151    }
