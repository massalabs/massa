19    pub fn insert_parents_descendants(
20        &mut self,
21        add_block_id: BlockId,
22        add_block_slot: Slot,
23        parents_hash: Vec<BlockId>,
24    ) {
25        // add as child to parents
26        for parent_h in parents_hash.iter() {
27            if let Some(BlockStatus::Active {
28                a_block: a_parent, ..
29            }) = self.blocks_state.get_mut(parent_h)
30            {
31                a_parent.children[add_block_slot.thread as usize]
32                    .insert(add_block_id, add_block_slot.period);
33            }
34        }
35
36        // add same-thread creator parent
37        let same_thread_parent_creator =
38            parents_hash
39                .get(add_block_slot.thread as usize)
40                .and_then(|parent_id| {
41                    if let Some(BlockStatus::Active { a_block, .. }) =
42                        self.blocks_state.get(parent_id)
43                    {
44                        Some(a_block.creator_address)
45                    } else {
46                        None
47                    }
48                });
49        if let Some(BlockStatus::Active { a_block, .. }) = self.blocks_state.get_mut(&add_block_id)
50        {
51            a_block.same_thread_parent_creator = same_thread_parent_creator;
52        } else {
53            panic!("block status should be active");
54        };
55
56        // add as descendant to ancestors if not final
57        let mut ancestors: VecDeque<BlockId> = parents_hash.iter().copied().collect();
58        let mut visited = PreHashSet::<BlockId>::default();
59        while let Some(ancestor_h) = ancestors.pop_back() {
60            if !visited.insert(ancestor_h) {
61                continue;
62            }
63            if let Some(BlockStatus::Active { a_block: ab, .. }) =
64                self.blocks_state.get_mut(&ancestor_h)
65            {
66                // No need to add descendants if ancestor is final,
67                // because only non-final active blocks are scanned for descendents for finality detection.
68                if ab.is_final {
69                    continue;
70                }
71                ab.descendants.insert(add_block_id);
72                for (ancestor_parent_h, _) in ab.parents.iter() {
73                    ancestors.push_front(*ancestor_parent_h);
74                }
75            }
76        }
77    }

79    pub fn compute_fitness_find_blockclique(
80        &mut self,
81        add_block_id: &BlockId,
82    ) -> Result<usize, ConsensusError> {
83        let block_id_serializer = BlockIdSerializer::new();
84        let mut blockclique_i = 0usize;
85        let mut max_clique_fitness = (0u64, num::BigInt::default());
86        for (clique_i, clique) in self.max_cliques.iter_mut().enumerate() {
87            clique.fitness = 0;
88            clique.is_blockclique = false;
89            let mut sum_hash = num::BigInt::default();
90            for block_h in clique.block_ids.iter() {
91                let fitness = match self.blocks_state.get(block_h) {
92                    Some(BlockStatus::Active { a_block, .. }) => a_block.fitness,
93                    _ => return Err(ConsensusError::ContainerInconsistency(format!("inconsistency inside block statuses computing fitness while adding {} - missing {}", add_block_id, block_h))),
94                };
95                clique.fitness = clique
96                    .fitness
97                    .checked_add(fitness)
98                    .ok_or(ConsensusError::FitnessOverflow)?;
99                let mut bytes = Vec::new();
100                block_id_serializer
101                    .serialize(block_h, &mut bytes)
102                    .map_err(|err| ConsensusError::SerializationError(err.to_string()))?;
103                sum_hash -= num::BigInt::from_bytes_be(num::bigint::Sign::Plus, &bytes);
104            }
105            let cur_fit = (clique.fitness, sum_hash);
106            if cur_fit > max_clique_fitness {
107                blockclique_i = clique_i;
108                max_clique_fitness = cur_fit;
109            }
110        }
111        self.max_cliques[blockclique_i].is_blockclique = true;
112        Ok(blockclique_i)
113    }

115    pub fn list_stale_blocks(&self, fitness_threshold: u64) -> PreHashSet<BlockId> {
116        // iterate from largest to smallest to minimize reallocations
117        let mut indices: Vec<usize> = (0..self.max_cliques.len()).collect();
118        indices.sort_unstable_by_key(|&i| std::cmp::Reverse(self.max_cliques[i].block_ids.len()));
119        let mut high_set = PreHashSet::<BlockId>::default();
120        let mut low_set = PreHashSet::<BlockId>::default();
121        for clique_i in indices.into_iter() {
122            if self.max_cliques[clique_i].fitness >= fitness_threshold {
123                high_set.extend(&self.max_cliques[clique_i].block_ids);
124            } else {
125                low_set.extend(&self.max_cliques[clique_i].block_ids);
126            }
127        }
128        &low_set - &high_set
129    }

131    pub fn remove_block(&mut self, add_block_id: &BlockId, block_id: &BlockId) {
132        let sequence_number = self.blocks_state.sequence_counter();
133        self.blocks_state.transition_map(block_id, |block_status, block_statuses| {
134        if let Some(BlockStatus::Active {
135            a_block: active_block,
136            ..
137        }) = block_status
138        {
139            if active_block.is_final {
140               panic!("inconsistency inside block statuses removing stale blocks adding {} - block {} was already final", add_block_id, block_id);
141            }
142
143            // remove from gi_head
144            if let Some(other_incomps) = self.gi_head.remove(block_id) {
145                for other_incomp in other_incomps.into_iter() {
146                    if let Some(other_incomp_lst) = self.gi_head.get_mut(&other_incomp) {
147                        other_incomp_lst.remove(block_id);
148                    }
149                }
150            }
151
152            // remove from cliques
153            let stale_block_fitness = active_block.fitness;
154            self.max_cliques.iter_mut().for_each(|c| {
155                if c.block_ids.remove(block_id) {
156                    c.fitness -= stale_block_fitness;
157                }
158            });
159            self.max_cliques.retain(|c| !c.block_ids.is_empty()); // remove empty cliques
160            if self.max_cliques.is_empty() {
161                // make sure at least one clique remains
162                self.max_cliques = vec![Clique {
163                    block_ids: PreHashSet::<BlockId>::default(),
164                    fitness: 0,
165                    is_blockclique: true,
166                }];
167            }
168
169            // remove from parent's children
170            for (parent_h, _parent_period) in active_block.parents.iter() {
171                if let Some(BlockStatus::Active {
172                    a_block: parent_active_block,
173                    ..
174                }) = block_statuses.get_mut(parent_h)
175                {
176                    parent_active_block.children[active_block.slot.thread as usize]
177                        .remove(block_id);
178                }
179            }
180
181            massa_trace!("consensus.block_graph.add_block_to_graph.stale", {
182                "hash": block_id
183            });
184
185            // mark as stale
186            self.new_stale_blocks
187                .insert(*block_id, (active_block.creator_address, active_block.slot));
188            Some(
189                BlockStatus::Discarded {
190                    slot: active_block.slot,
191                    creator: active_block.creator_address,
192                    parents: active_block.parents.iter().map(|(h, _)| *h).collect(),
193                    reason: DiscardReason::Stale,
194                    sequence_number,
195                }
196            )
197        } else {
198            panic!("inconsistency inside block statuses removing stale blocks adding {} - block {} is missing", add_block_id, block_id);
199        }
200    });
201    }

203    pub fn list_final_blocks(&self) -> Result<PreHashSet<BlockId>, ConsensusError> {
204        // short-circuiting intersection of cliques from smallest to largest
205        let mut indices: Vec<usize> = (0..self.max_cliques.len()).collect();
206        indices.sort_unstable_by_key(|&i| self.max_cliques[i].block_ids.len());
207        let mut indices_iter = indices.iter();
208        let mut final_candidates = self.max_cliques
209            [*indices_iter.next().expect("expected at least one clique")]
210        .block_ids
211        .clone();
212        for i in indices_iter {
213            final_candidates.retain(|v| self.max_cliques[*i].block_ids.contains(v));
214            if final_candidates.is_empty() {
215                break;
216            }
217        }
218
219        // restrict search to cliques with high enough fitness, sort cliques by fitness (highest to lowest)
220        massa_trace!(
221            "consensus.block_graph.add_block_to_graph.list_final_blocks.restrict",
222            {}
223        );
224        indices.retain(|&i| self.max_cliques[i].fitness > self.config.delta_f0);
225        indices.sort_unstable_by_key(|&i| std::cmp::Reverse(self.max_cliques[i].fitness));
226
227        let mut final_blocks = PreHashSet::<BlockId>::default();
228        for clique_i in indices.into_iter() {
229            massa_trace!(
230                "consensus.block_graph.add_block_to_graph.list_final_blocks.loop",
231                { "clique_i": clique_i }
232            );
233            // check in cliques from highest to lowest fitness
234            if final_candidates.is_empty() {
235                // no more final candidates
236                break;
237            }
238            let clique = &self.max_cliques[clique_i];
239
240            // compute the total fitness of all the descendants of the candidate within the clique
241            let loc_candidates = final_candidates.clone();
242            for candidate_h in loc_candidates.into_iter() {
243                let descendants = match self.blocks_state.get(&candidate_h) {
244                    Some(BlockStatus::Active { a_block, .. }) => &a_block.descendants,
245                    _ => {
246                        return Err(ConsensusError::MissingBlock(format!(
247                            "missing block when computing total fitness of descendants: {}",
248                            candidate_h
249                        )))
250                    }
251                };
252                let desc_fit: u64 = descendants
253                    .intersection(&clique.block_ids)
254                    .map(|h| {
255                        if let Some(BlockStatus::Active { a_block: ab, .. }) =
256                            self.blocks_state.get(h)
257                        {
258                            return ab.fitness;
259                        }
260                        0
261                    })
262                    .sum();
263                if desc_fit > self.config.delta_f0 {
264                    // candidate is final
265                    final_candidates.remove(&candidate_h);
266                    final_blocks.insert(candidate_h);
267                }
268            }
269        }
270        Ok(final_blocks)
271    }

283    pub fn mark_final_blocks(
284        &mut self,
285        add_block_id: &BlockId,
286        final_blocks: PreHashSet<BlockId>,
287    ) -> Result<(), ConsensusError> {
288        for block_id in final_blocks.into_iter() {
289            // remove from gi_head
290            if let Some(other_incomps) = self.gi_head.remove(&block_id) {
291                for other_incomp in other_incomps.into_iter() {
292                    if let Some(other_incomp_lst) = self.gi_head.get_mut(&other_incomp) {
293                        other_incomp_lst.remove(&block_id);
294                    }
295                }
296            }
297
298            // mark as final and update latest_final_blocks_periods
299            if let Some(BlockStatus::Active {
300                a_block: final_block,
301                ..
302            }) = self.blocks_state.get_mut(&block_id)
303            {
304                massa_trace!("consensus.block_graph.add_block_to_graph.final", {
305                    "hash": block_id
306                });
307                final_block.is_final = true;
308                // remove from cliques
309                let final_block_fitness = final_block.fitness;
310                self.max_cliques.iter_mut().for_each(|c| {
311                    if c.block_ids.remove(&block_id) {
312                        c.fitness -= final_block_fitness;
313                    }
314                });
315                self.max_cliques.retain(|c| !c.block_ids.is_empty()); // remove empty cliques
316                if self.max_cliques.is_empty() {
317                    // make sure at least one clique remains
318                    self.max_cliques = vec![Clique {
319                        block_ids: PreHashSet::<BlockId>::default(),
320                        fitness: 0,
321                        is_blockclique: true,
322                    }];
323                }
324                // update latest final blocks
325                if final_block.slot.period
326                    > self.latest_final_blocks_periods[final_block.slot.thread as usize].1
327                {
328                    self.latest_final_blocks_periods[final_block.slot.thread as usize] =
329                        (block_id, final_block.slot.period);
330                }
331                // update new final blocks list
332                self.new_final_blocks.insert(block_id);
333            } else {
334                return Err(ConsensusError::ContainerInconsistency(format!("inconsistency inside block statuses updating final blocks adding {} - block {} is missing", add_block_id, block_id)));
335            }
336        }
337        Ok(())
338    }
