14/// Interface that communicates with the graph worker thread
15#[cfg_attr(feature = "test-exports", mockall_wrap::wrap, mockall::automock)]
16pub trait ConsensusController: Send + Sync {
17    /// Get an export of a part of the graph
18    ///
19    /// # Arguments
20    /// * `start_slot`: the slot to start the export from, if None, the export starts from the genesis
21    /// * `end_slot`: the slot to end the export at, if None, the export ends at the current slot
22    ///
23    /// # Returns
24    /// The export of the graph
25    fn get_block_graph_status(
26        &self,
27        start_slot: Option<Slot>,
28        end_slot: Option<Slot>,
29    ) -> Result<BlockGraphExport, ConsensusError>;
30
31    /// Get statuses of a list of blocks
32    ///
33    /// # Arguments
34    /// * `ids`: the list of block ids to get the status of
35    ///
36    /// # Returns
37    /// The statuses of the blocks sorted by the order of the input list
38    fn get_block_statuses(&self, ids: &[BlockId]) -> Vec<BlockGraphStatus>;
39
40    /// Get all the cliques of the graph
41    ///
42    /// # Returns
43    /// The list of cliques
44    fn get_cliques(&self) -> Vec<Clique>;
45
46    /// Get a part of the graph to send to a node for it to setup its graph.
47    /// Used for bootstrap.
48    ///
49    /// # Arguments:
50    /// * `cursor`: streaming cursor containing the current state of bootstrap and what blocks have previously been sent to the client
51    /// * `execution_cursor`: streaming cursor of the final state to ensure that last slot of the bootstrap info match the slot of the execution
52    ///
53    /// # Returns:
54    /// * A portion of the graph
55    /// * The list of outdated block ids
56    /// * The streaming step value after the current iteration to be saved to be able to use it as parameters and resume the bootstrap
57    #[allow(clippy::type_complexity)]
58    fn get_bootstrap_part(
59        &self,
60        cursor: StreamingStep<PreHashSet<BlockId>>,
61        execution_cursor: StreamingStep<Slot>,
62    ) -> Result<
63        (
64            BootstrapableGraph,
65            PreHashSet<BlockId>,
66            StreamingStep<PreHashSet<BlockId>>,
67        ),
68        ConsensusError,
69    >;
70
71    /// Get the stats of the consensus
72    ///
73    /// # Returns
74    /// The stats of the consensus
75    fn get_stats(&self) -> Result<ConsensusStats, ConsensusError>;
76
77    /// Get the best parents for the next block to be produced
78    ///
79    /// # Returns
80    /// The id of best parents for the next block to be produced along with their period
81    fn get_best_parents(&self) -> Vec<(BlockId, u64)>;
82
83    /// Get the block id of the block at a specific slot in the blockclique
84    ///
85    /// # Arguments
86    /// * `slot`: the slot to get the block id of
87    ///
88    /// # Returns
89    /// The block id of the block at the specified slot if exists
90    fn get_blockclique_block_at_slot(&self, slot: Slot) -> Option<BlockId>;
91
92    /// Get the latest block, that is in the blockclique, in the thread of the given slot and before this `slot`.
93    ///
94    /// # Arguments:
95    /// * `slot`: the slot that will give us the thread and the upper bound
96    ///
97    /// # Returns:
98    /// The block id of the latest block in the thread of the given slot and before this slot
99    fn get_latest_blockclique_block_at_slot(&self, slot: Slot) -> BlockId;
100
101    /// Register a block in the graph
102    ///
103    /// # Arguments
104    /// * `block_id`: the id of the block to register
105    /// * `slot`: the slot of the block
106    /// * `block_storage`: the storage that contains all the objects of the block
107    /// * `created`: is the block created by our node ?
108    fn register_block(&self, block_id: BlockId, slot: Slot, block_storage: Storage, created: bool);
109
110    /// Register a block header in the graph
111    ///
112    /// # Arguments
113    /// * `block_id`: the id of the block to register
114    /// * `header`: the header of the block to register
115    fn register_block_header(&self, block_id: BlockId, header: SecureShare<BlockHeader, BlockId>);
116
117    /// Mark a block as invalid in the graph
118    ///
119    /// # Arguments
120    /// * `block_id`: the id of the block to mark as invalid
121    /// * `header`: the header of the block to mark as invalid
122    fn mark_invalid_block(&self, block_id: BlockId, header: SecureShare<BlockHeader, BlockId>);
123
124    /// Returns a boxed clone of self.
125    /// Useful to allow cloning `Box<dyn ConsensusController>`.
126    fn clone_box(&self) -> Box<dyn ConsensusController>;
127}
128
129/// Allow cloning `Box<dyn ConsensusController>`
130/// Uses `ConsensusController::clone_box` internally
131impl Clone for Box<dyn ConsensusController> {
132    fn clone(&self) -> Box<dyn ConsensusController> {
133        self.clone_box()
134    }
135}

137/// Consensus manager used to stop the consensus thread
138pub trait ConsensusManager {
139    /// Stop the consensus thread
140    /// Note that we do not take self by value to consume it
141    /// because it is not allowed to move out of `Box<dyn ConsensusManager>`
142    /// This will improve if the `unsized_fn_params` feature stabilizes enough to be safely usable.
143    fn stop(&mut self);
144}
