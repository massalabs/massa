15/// Structure gathering all elements needed by the event cache thread
16pub(crate) struct EventCacheWriterThread {
17    // A copy of the input data allowing access to incoming requests
18    input_data: Arc<(Condvar, Mutex<EventCacheWriterInputData>)>,
19    /// Event cache
20    cache: Arc<RwLock<EventCache>>,
21}
22
23impl EventCacheWriterThread {
24    fn new(
25        input_data: Arc<(Condvar, Mutex<EventCacheWriterInputData>)>,
26        event_cache: Arc<RwLock<EventCache>>,
27    ) -> Self {
28        Self {
29            input_data,
30            cache: event_cache,
31        }
32    }
33
34    /// Waits for an event to trigger a new iteration in the event cache main loop.
35    ///
36    /// # Returns
37    /// `ExecutionInputData` representing the input requests,
38    /// and a boolean saying whether we should stop the loop.
39    fn wait_loop_event(&mut self) -> (EventCacheWriterInputData, bool) {
40        loop {
41            // lock input data
42            let mut input_data_lock = self.input_data.1.lock();
43
44            // Only consume the shared input once there is something to do
45            // (events to flush and/or a stop request). Taking the whole input
46            // unconditionally used to consume the `stop` flag together with a
47            // pending batch of events and then drop it when returning early,
48            // which could make the writer wait forever after `stop()`.
49            if input_data_lock.events.is_empty() && !input_data_lock.stop {
50                self.input_data.0.wait(&mut input_data_lock);
51                continue;
52            }
53
54            // take current input data, resetting it
55            let input_data: EventCacheWriterInputData = std::mem::take(&mut *input_data_lock);
56            // Propagate the (durable) stop flag alongside the taken events so a
57            // final queued batch is still flushed before the loop terminates.
58            let stop = input_data.stop;
59            return (input_data, stop);
60        }
61    }
62
63    /// Main loop of the worker
64    pub fn main_loop(&mut self) {
65        loop {
66            let (input_data, stop) = self.wait_loop_event();
67            debug!(
68                "Event cache writer loop triggered, input_data = {:?}",
69                input_data
70            );
71
72            // Always flush any queued events, even if this iteration also
73            // observed a stop request, so no final batch is silently lost.
74            if !input_data.events.is_empty() {
75                let mut lock = self.cache.write();
76                lock.insert_multi_it(input_data.events.into_iter());
77                // drop the lock as early as possible
78                drop(lock);
79            }
80
81            if stop {
82                // we need to stop
83                break;
84            }
85        }
86    }
87}

89/// Event cache manager trait used to stop the event cache thread
90pub trait EventCacheManager {
91    /// Stop the event cache thread
92    /// Note that we do not take self by value to consume it
93    /// because it is not allowed to move out of `Box<dyn ExecutionManager>`
94    /// This will improve if the `unsized_fn_params` feature stabilizes enough to be safely usable.
95    fn stop(&mut self);
96}
97
98/// ... manager
99/// Allows stopping the ... worker
100pub struct EventCacheWriterManagerImpl {
101    /// input data to process in the VM loop
102    /// with a wake-up condition variable that needs to be triggered when the data changes
103    pub(crate) input_data: Arc<(Condvar, Mutex<EventCacheWriterInputData>)>,
104    /// handle used to join the worker thread
105    pub(crate) thread_handle: Option<std::thread::JoinHandle<()>>,
106}
107
108impl EventCacheManager for EventCacheWriterManagerImpl {
109    /// stops the worker
110    fn stop(&mut self) {
111        info!("Stopping Execution controller...");
112        // notify the worker thread to stop
113        {
114            let mut input_wlock = self.input_data.1.lock();
115            input_wlock.stop = true;
116            self.input_data.0.notify_one();
117        }
118        // join the thread
119        if let Some(join_handle) = self.thread_handle.take() {
120            join_handle.join().expect("VM controller thread panicked");
121        }
122        info!("Execution controller stopped");
123    }
124}

126pub fn start_event_cache_writer_worker(
127    cfg: EventCacheConfig,
128) -> (Box<dyn EventCacheManager>, Box<dyn EventCacheController>) {
129    let event_cache = Arc::new(RwLock::new(EventCache::new(
130        cfg.event_cache_path.as_path(),
131        cfg.max_event_cache_length,
132        cfg.snip_amount,
133        cfg.thread_count,
134        cfg.max_call_stack_length,
135        cfg.max_event_data_length,
136        cfg.max_events_per_operation,
137        cfg.max_operations_per_block,
138        cfg.max_events_per_query,
139    )));
140
141    // define the input data interface
142    let input_data = Arc::new((Condvar::new(), Mutex::new(EventCacheWriterInputData::new())));
143    let input_data_clone = input_data.clone();
144
145    // create a controller
146    let controller = EventCacheControllerImpl {
147        input_data: input_data.clone(),
148        cache: event_cache.clone(),
149    };
150
151    let thread_builder = thread::Builder::new().name("event_cache".into());
152    let thread_handle = thread_builder
153        .spawn(move || {
154            EventCacheWriterThread::new(input_data_clone, event_cache).main_loop();
155        })
156        .expect("failed to spawn thread : event_cache");
157
158    // create a manager
159    let manager = EventCacheWriterManagerImpl {
160        input_data,
161        thread_handle: Some(thread_handle),
162    };
163
164    // return the manager and controller pair
165    (Box::new(manager), Box::new(controller))
166}

178    use massa_models::slot::Slot;

182    fn sample_event() -> SCOutputEvent {
183        SCOutputEvent {
184            context: EventExecutionContext {
185                slot: Slot::new(1, 0),
186                block: None,
187                read_only: false,
188                index_in_slot: 0,
189                call_stack: Default::default(),
190                origin_operation_id: None,
191                is_final: true,
192                is_error: false,
193                deferred_call_id: None,
194                async_msg_id: None,
195            },
196            data: "shutdown-race event".to_string(),
197        }
198    }
199
200    /// Reproduces the lost-stop-signal race: a final batch of events is queued
201    /// *together with* a stop request. The writer must flush the batch and then
202    /// terminate, instead of consuming the stop flag with the batch and waiting
203    /// on the condvar forever.
204    #[test]
205    fn stop_is_not_lost_when_a_final_batch_is_queued() {
206        let tmp = TempDir::new().unwrap();
207        let cache = Arc::new(RwLock::new(EventCache::new(
208            tmp.path(),
209            1000,
210            300,
211            THREAD_COUNT,
212            MAX_RECURSIVE_CALLS_DEPTH,
213            MAX_EVENT_DATA_SIZE as u64,
214            MAX_EVENT_PER_OPERATION as u64,
215            MAX_OPERATIONS_PER_BLOCK as u64,
216            5000,
217        )));
218
219        let mut input = EventCacheWriterInputData::new();
220        input.events.push_back(sample_event());
221        input.stop = true;
222        let input_data = Arc::new((Condvar::new(), Mutex::new(input)));
223
224        let mut worker = EventCacheWriterThread::new(input_data, cache);
225        let handle = thread::spawn(move || worker.main_loop());
226
227        let deadline = Instant::now() + Duration::from_secs(10);
228        while !handle.is_finished() {
229            assert!(
230                Instant::now() < deadline,
231                "event cache writer thread did not stop after a stop request"
232            );
233            thread::sleep(Duration::from_millis(10));
234        }
235        handle.join().expect("event cache writer thread panicked");
236    }
