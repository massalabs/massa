229/// Protocol Configuration, read from toml user configuration file
230#[allow(dead_code)]
231#[derive(Debug, Deserialize, Clone)]
232pub struct ProtocolSettings {
233    /// after `ask_block_timeout` milliseconds we try to ask a block to another node
234    pub ask_block_timeout: MassaTime,
235    /// Max known blocks we keep during their propagation
236    pub max_blocks_kept_for_propagation: usize,
237    /// Time during which a block is expected to propagate
238    pub max_block_propagation_time: MassaTime,
239    /// Block propagation tick interval, useful for propagating blocks quickly to newly connected peers.
240    pub block_propagation_tick: MassaTime,
241    /// max known blocks our node keeps in its knowledge cache
242    pub max_known_blocks_size: usize,
243    /// max cache size for which blocks a foreign node knows about
244    pub max_node_known_blocks_size: usize,
245    /// max wanted blocks per node kept in memory
246    pub max_node_wanted_blocks_size: usize,
247    /// max known operations current node kept in memory
248    pub max_known_ops_size: usize,
249    /// size of the buffer of asked operations
250    pub asked_operations_buffer_capacity: usize,
251    /// max known operations of foreign nodes we keep in memory (by node)
252    pub max_node_known_ops_size: usize,
253    /// max known endorsements by our node that we kept in memory
254    pub max_known_endorsements_size: usize,
255    /// max known endorsements of foreign nodes we keep in memory (by node)
256    pub max_node_known_endorsements_size: usize,
257    /// we ask for the same block `max_simultaneous_ask_blocks_per_node` times at the same time
258    pub max_simultaneous_ask_blocks_per_node: usize,
259    /// Max wait time for sending a Network or Node event.
260    pub max_send_wait: MassaTime,
261    /// Maximum number of batches in the memory buffer.
262    /// Dismiss the new batches if overflow
263    pub operation_batch_buffer_capacity: usize,
264    /// Maximum number of operations in the announcement buffer.
265    /// Immediately announce if overflow.
266    pub operation_announcement_buffer_capacity: usize,
267    /// Start processing batches in the buffer each `operation_batch_proc_period` in millisecond
268    pub operation_batch_proc_period: MassaTime,
269    /// Interval at which operations are announced in batches.
270    pub operation_announcement_interval: MassaTime,
271    /// Maximum of operations sent in one message.
272    pub max_operations_per_message: u64,
273    /// MAx number of operations kept for propagation
274    pub max_ops_kept_for_propagation: usize,
275    /// Max number of endorsements merged into a single propagation round
276    pub max_endorsements_per_propagation_round: usize,
277    /// Time threshold after which operation are not propagated
278    pub max_operations_propagation_time: MassaTime,
279    /// Time threshold after which operation are not propagated
280    pub max_endorsements_propagation_time: MassaTime,
281    /// Path for initial peers
282    pub initial_peers_file: PathBuf,
283    /// Keypair
284    pub keypair_file: PathBuf,
285    /// Ip we are bind to listen to
286    pub bind: SocketAddr,
287    /// Ip seen by others. If none the bind ip is used
288    pub routable_ip: Option<IpAddr>,
289    /// Time threshold to have a connection to a node
290    pub connect_timeout: MassaTime,
291    /// Number of tester threads
292    pub thread_tester_count: u8,
293    /// Number of bytes we can read/write by seconds in a connection (must be a 10 multiple)
294    pub read_write_limit_bytes_per_second: u64,
295    /// try connection timer
296    pub try_connection_timer: MassaTime,
297    /// try connection timer for the same peer
298    pub try_connection_timer_same_peer: MassaTime,
299    /// periodically unban every peer
300    pub unban_everyone_timer: MassaTime,
301    /// Timeout connection
302    pub timeout_connection: MassaTime,
303    /// Message timeout
304    pub message_timeout: MassaTime,
305    /// Timeout for the tester operations
306    pub tester_timeout: MassaTime,
307    /// Nb in connections
308    pub max_in_connections: usize,
309    /// Peers limits per category
310    pub peers_categories: HashMap<String, PeerCategoryInfo>,
311    /// Limits for default category
312    pub default_category_info: PeerCategoryInfo,
313    /// Cooldown before testing again an old peer
314    pub test_oldest_peer_cooldown: MassaTime,
315    /// Rate limitation to apply to the data stream (per second)
316    pub rate_limit: u64,
317}

319/// gRPC settings
320/// the gRPC settings
321#[derive(Debug, Deserialize, Clone)]
322pub struct GrpcSettings {
323    /// whether to enable gRPC
324    pub enabled: bool,
325    /// whether to accept HTTP/1.1 requests
326    pub accept_http1: bool,
327    /// whether to enable CORS. Works only if `accept_http1` is true
328    pub enable_cors: bool,
329    /// whether to enable gRPC health service
330    pub enable_health: bool,
331    /// whether to enable gRPC reflection
332    pub enable_reflection: bool,
333    /// whether to enable TLS
334    pub enable_tls: bool,
335    /// whether to enable mTLS (requires `enable_tls` to be true)
336    pub enable_mtls: bool,
337    /// whether to generate a self-signed certificate if none is provided
338    pub generate_self_signed_certificates: bool,
339    /// Subject Alternative Names is an extension in X.509 certificates that allows a certificate to specify additional subject identifiers. It is used to support alternative names for a subject, other than its primary Common Name (CN), which is typically used to represent the primary domain name.
340    pub subject_alt_names: Vec<String>,
341    /// bind for the Massa gRPC API
342    pub bind: SocketAddr,
343    /// which compression encodings does the server accept for requests
344    pub accept_compressed: Option<String>,
345    /// which compression encodings might the server use for responses
346    pub send_compressed: Option<String>,
347    /// limits the maximum size of a decoded message. Defaults to 4MB
348    pub max_decoding_message_size: usize,
349    /// limits the maximum size of an encoded message. Defaults to 4MB
350    pub max_encoding_message_size: usize,
351    /// limits the maximum size of streaming channel
352    pub max_channel_size: usize,
353    /// set the concurrency limit applied to on requests inbound per connection. Defaults to 32
354    pub concurrency_limit_per_connection: usize,
355    /// set a timeout on for all request handlers
356    pub timeout: MassaTime,
357    /// sets the SETTINGS_INITIAL_WINDOW_SIZE spec option for HTTP2 stream-level flow control. Default is 65,535
358    pub initial_stream_window_size: Option<u32>,
359    /// sets the max connection-level flow control for HTTP2. Default is 65,535
360    pub initial_connection_window_size: Option<u32>,
361    /// sets the SETTINGS_MAX_CONCURRENT_STREAMS spec option for HTTP2 connections. Default is no limit (`None`)
362    pub max_concurrent_streams: Option<u32>,
363    /// max number of arguments per gRPC request
364    pub max_arguments: u64,
365    /// set whether TCP keepalive messages are enabled on accepted connections
366    pub tcp_keepalive: Option<MassaTime>,
367    /// set the value of `TCP_NODELAY` option for accepted connections. Enabled by default
368    pub tcp_nodelay: bool,
369    /// set whether HTTP2 Ping frames are enabled on accepted connections. Default is no HTTP2 keepalive (`None`)
370    pub http2_keepalive_interval: Option<MassaTime>,
371    /// sets a timeout for receiving an acknowledgement of the keepalive ping. Default is 20 seconds
372    pub http2_keepalive_timeout: Option<MassaTime>,
373    /// sets whether to use an adaptive flow control. Defaults to false
374    pub http2_adaptive_window: Option<bool>,
375    /// sets the maximum frame size to use for HTTP2(must be within 16,384 and 16,777,215). If not set, will default from underlying transport
376    pub max_frame_size: Option<u32>,
377    /// when looking for next draw we want to look at max `draw_lookahead_period_count`
378    pub draw_lookahead_period_count: u64,
379    /// max number of block ids that can be included in a single request
380    pub max_block_ids_per_request: u32,
381    /// max number of addresses that can be included in a single request
382    pub max_addresses_per_request: u32,
383    /// max number of slot ranges that can be included in a single request
384    pub max_slot_ranges_per_request: u32,
385    /// max number of endorsement ids that can be included in a single request
386    pub max_endorsement_ids_per_request: u32,
387    /// max number of operation ids that can be included in a single request
388    pub max_operation_ids_per_request: u32,
389    /// max op datastore entries per request
390    pub max_datastore_entries_per_request: u64,
391    /// max number of filters that can be included in a single request
392    pub max_filters_per_request: u32,
393    /// max number of query items that can be included in a single request
394    pub max_query_items_per_request: u32,
395    /// certificate authority root path
396    pub certificate_authority_root_path: PathBuf,
397    /// server certificate path
398    pub server_certificate_path: PathBuf,
399    /// server private key path
400    pub server_private_key_path: PathBuf,
401    /// client certificate authority root path
402    pub client_certificate_authority_root_path: PathBuf,
403    /// client certificate path
404    pub client_certificate_path: PathBuf,
405    /// client private key path
406    pub client_private_key_path: PathBuf,
407    /// Max keys returned by a datastore key query
408    pub max_datastore_keys_query: Option<u32>,
409    /// interval for check the connection stream
410    pub unidirectional_stream_interval_check: u64,
411}
