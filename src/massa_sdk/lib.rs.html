70/// Error when creating a new client
71#[derive(Error, Debug)]
72pub enum ClientError {
73    /// Url error
74    #[error("Invalid grpc url: {0}")]
75    Url(#[from] http::uri::InvalidUri),
76    /// Invalid URL supplied for the JSON-RPC endpoint.
77    #[error("Invalid node url: {0}")]
78    InvalidUrl(String),
79    /// Connection error
80    #[error("Cannot connect to grpc server: {0}")]
81    Connect(#[from] tonic::transport::Error),
82    /// TLS configuration error
83    #[error("Invalid grpc TLS configuration: {0}")]
84    InvalidTlsConfig(String),
85}
86
87/// Connect a gRPC channel to `addr`: in plaintext when `tls_config` is `None`, over TLS
88/// (or mTLS when a client identity is configured) otherwise.
89///
90/// Errors on an invalid TLS configuration (unreadable certificate material, or a client
91/// certificate without its key and vice versa) as well as on connection failure.
92pub async fn connect_grpc_channel(
93    addr: SocketAddr,
94    tls_config: Option<&GrpcTlsConfig>,
95) -> Result<tonic::transport::Channel, ClientError> {
96    let endpoint = match tls_config {
97        Some(tls) => {
98            let ca_pem = std::fs::read(&tls.certificate_authority_root_path).map_err(|err| {
99                ClientError::InvalidTlsConfig(format!(
100                    "failed to read certificate authority root {}: {}",
101                    tls.certificate_authority_root_path.display(),
102                    err
103                ))
104            })?;
105            let mut client_tls_config = tonic::transport::ClientTlsConfig::new()
106                .ca_certificate(tonic::transport::Certificate::from_pem(ca_pem))
107                // the certificate is verified under this name while dialing `addr`
108                .domain_name(tls.server_name.clone());
109            match (&tls.client_certificate_path, &tls.client_private_key_path) {
110                (Some(cert_path), Some(key_path)) => {
111                    let cert = std::fs::read(cert_path).map_err(|err| {
112                        ClientError::InvalidTlsConfig(format!(
113                            "failed to read client certificate {}: {}",
114                            cert_path.display(),
115                            err
116                        ))
117                    })?;
118                    let key = std::fs::read(key_path).map_err(|err| {
119                        ClientError::InvalidTlsConfig(format!(
120                            "failed to read client private key {}: {}",
121                            key_path.display(),
122                            err
123                        ))
124                    })?;
125                    client_tls_config =
126                        client_tls_config.identity(tonic::transport::Identity::from_pem(cert, key));
127                }
128                (None, None) => {}
129                _ => {
130                    return Err(ClientError::InvalidTlsConfig(
131                        "client_certificate_path and client_private_key_path must be set together"
132                            .to_string(),
133                    ));
134                }
135            }
136            tonic::transport::Channel::from_shared(format!("https://{}", addr))?
137                .tls_config(client_tls_config)?
138        }
139        None => tonic::transport::Channel::from_shared(format!("grpc://{}", addr))?,
140    };
141    endpoint.connect().await.map_err(ClientError::Connect)
142}

158impl Client {
159    /// creates a new client
160    ///
161    /// The gRPC transports are plaintext when the corresponding TLS configuration is
162    /// `None`, TLS (or mTLS) otherwise. An invalid TLS configuration is a hard error; a
163    /// connection failure leaves the corresponding gRPC client to `None`, so the JSON-RPC
164    /// clients stay usable when the node is unreachable.
165    ///
166    /// Note: the JSON-RPC transports of this constructor are plaintext `http`. For a
167    /// TLS-enabled JSON-RPC endpoint, build the client with [`RpcClient::from_url`] and
168    /// an `https` url instead.
169    #[allow(clippy::too_many_arguments)]
170    pub async fn new(
171        ip: IpAddr,
172        public_port: u16,
173        private_port: u16,
174        grpc_public_port: u16,
175        grpc_private_port: u16,
176        chain_id: u64,
177        http_config: &HttpConfig,
178        grpc_public_tls: Option<&GrpcTlsConfig>,
179        grpc_private_tls: Option<&GrpcTlsConfig>,
180    ) -> Result<Client, ClientError> {
181        let public_socket_addr = SocketAddr::new(ip, public_port);
182        let private_socket_addr = SocketAddr::new(ip, private_port);
183        let grpc_public_socket_addr = SocketAddr::new(ip, grpc_public_port);
184        let grpc_private_socket_addr = SocketAddr::new(ip, grpc_private_port);
185        let public_url = format!("http://{}", public_socket_addr);
186        let private_url = format!("http://{}", private_socket_addr);
187
188        // try to start grpc client (public api) and connect to the server
189        let grpc_pub_client =
190            match connect_grpc_channel(grpc_public_socket_addr, grpc_public_tls).await {
191                Ok(channel) => Some(PublicServiceClient::new(channel)),
192                Err(err @ ClientError::InvalidTlsConfig(_)) => return Err(err),
193                Err(err) => {
194                    tracing::error!(
195                        "unable to connect to the public grpc server at {}: {}",
196                        grpc_public_socket_addr,
197                        err
198                    );
199                    None
200                }
201            };
202
203        // try to start grpc client (private api) and connect to the server
204        let grpc_priv_client =
205            match connect_grpc_channel(grpc_private_socket_addr, grpc_private_tls).await {
206                Ok(channel) => Some(PrivateServiceClient::new(channel)),
207                Err(err @ ClientError::InvalidTlsConfig(_)) => return Err(err),
208                Err(err) => {
209                    tracing::error!(
210                        "unable to connect to the private grpc server at {}: {}",
211                        grpc_private_socket_addr,
212                        err
213                    );
214                    None
215                }
216            };
217
218        Ok(Client {
219            public: RpcClient::from_url(&public_url, http_config).await,
220            private: RpcClient::from_url(&private_url, http_config).await,
221            grpc_public: grpc_pub_client,
222            grpc_private: grpc_priv_client,
223            chain_id,
224        })
225    }
226
227    /// creates a new client from a public JSON-RPC URL.
228    ///
229    /// `public_url` is used as-is for the public JSON-RPC endpoint. The host
230    /// extracted from it is reused for the private JSON-RPC endpoint on the
231    /// provided `private_port`. The gRPC clients are left as `None`: this
232    /// constructor is intended for `massa-client`, which only uses the JSON-RPC
233    /// endpoints.
234    pub async fn from_url(
235        public_url: &str,
236        private_port: u16,
237        chain_id: u64,
238        http_config: &HttpConfig,
239    ) -> Result<Client, ClientError> {
240        const URL_EXAMPLES: &str =
241            "expected a public JSON-RPC URL such as https://mainnet.massa.net/api/v2 \
242             or http://127.0.0.1:33035";
243        let public_uri: http::Uri = public_url.parse().map_err(|err| {
244            ClientError::InvalidUrl(format!(
245                "invalid URL '{}': {} ({})",
246                public_url, err, URL_EXAMPLES
247            ))
248        })?;
249        let host = public_uri.host().ok_or_else(|| {
250            ClientError::InvalidUrl(format!(
251                "URL '{}' has no host ({})",
252                public_url, URL_EXAMPLES
253            ))
254        })?;
255        let scheme = public_uri.scheme_str().unwrap_or("http");
256
257        let private_url = format!("{}://{}:{}", scheme, host, private_port);
258
259        Ok(Client {
260            public: RpcClient::from_url(public_url, http_config).await,
261            private: RpcClient::from_url(&private_url, http_config).await,
262            grpc_public: None,
263            grpc_private: None,
264            chain_id,
265        })
266    }
267}

636impl RpcClientV2 {
637    /// Default constructor
638    pub async fn from_url(
639        socket_addr: SocketAddr,
640        http_config: &HttpConfig,
641        ws_config: &WsConfig,
642    ) -> RpcClientV2 {
643        let http_url = format!("http://{}", socket_addr);
644        let ws_url = format!("ws://{}", socket_addr);
645
646        if http_config.enabled && !ws_config.enabled {
647            let http_client = http_client_from_url(&http_url, http_config);
648            return RpcClientV2 {
649                http_client: Some(http_client),
650                ws_client: None,
651            };
652        } else if !http_config.enabled && ws_config.enabled {
653            let ws_client = ws_client_from_url(&ws_url, ws_config).await;
654            return RpcClientV2 {
655                http_client: None,
656                ws_client: Some(ws_client),
657            };
658        } else if !http_config.enabled && !ws_config.enabled {
659            panic!("wrong client configuration, you can't disable both http and ws");
660        }
661
662        let http_client = http_client_from_url(&http_url, http_config);
663        let ws_client = ws_client_from_url(&ws_url, ws_config).await;
664
665        RpcClientV2 {
666            http_client: Some(http_client),
667            ws_client: Some(ws_client),
668        }
669    }

783fn http_client_from_url(url: &str, http_config: &HttpConfig) -> HttpClient<HttpBackend> {
784    let builder = HttpClientBuilder::default()
785        .max_request_size(http_config.client_config.max_request_body_size)
786        .request_timeout(http_config.client_config.request_timeout.to_duration())
787        .max_concurrent_requests(http_config.client_config.max_concurrent_requests)
788        .id_format(get_id_kind(http_config.client_config.id_kind.as_str()))
789        .set_headers(get_headers(&http_config.client_config.headers));
790
791    // Note: `client_config.certificate_store` is ignored here, see its documentation.
792    //       TLS is selected by the `https` scheme of `url`, using the platform trust store.
793
794    builder
795        .build(url)
796        .unwrap_or_else(|_| panic!("unable to create Http client for {}", url))
797}
798
799async fn ws_client_from_url(url: &str, ws_config: &WsConfig) -> WsClient
800where
801    WsClient: SubscriptionClientT,
802{
803    let builder = WsClientBuilder::default()
804        .max_request_size(ws_config.client_config.max_request_body_size)
805        .request_timeout(ws_config.client_config.request_timeout.to_duration())
806        .max_concurrent_requests(ws_config.client_config.max_concurrent_requests)
807        .id_format(get_id_kind(ws_config.client_config.id_kind.as_str()))
808        .set_headers(get_headers(&ws_config.client_config.headers))
809        .max_buffer_capacity_per_subscription(ws_config.max_notifs_per_subscription)
810        .max_redirections(ws_config.max_redirections);
811
812    // Note: `client_config.certificate_store` is ignored here, see its documentation.
813    //       TLS is selected by the `wss` scheme of `url`, using the platform trust store.
814
815    builder
816        .build(url)
817        .await
818        .unwrap_or_else(|_| panic!("unable to create WebSocket client for {}", url))
819}

821fn get_id_kind(id_kind: &str) -> IdKind {
822    match id_kind {
823        "Number" => IdKind::Number,
824        "String" => IdKind::String,
825        _ => IdKind::Number,
826    }

829fn get_headers(headers: &[(String, String)]) -> HeaderMap {
830    let mut headers_map = HeaderMap::new();
831    headers.iter().for_each(|(key, value)| {
832        let header_name = match HeaderName::from_str(key.as_str()) {
833            Ok(header_name) => header_name,
834            Err(_) => panic!("invalid header name: {:?}", key),
835        };
836        let header_value = match HeaderValue::from_str(value.as_str()) {
837            Ok(header_name) => header_name,
838            Err(_) => panic!("invalid header value: {:?}", value),
839        };
840        headers_map.insert(header_name, header_value);
841    });
842
843    headers_map
844}
