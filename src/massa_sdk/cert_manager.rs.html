3//! This module provides functions for generating certificates for a Certificate Authority (CA) and
4//! signing certificates using the CA certificate in mutual TLS (mTLS) scenarios.
5//!
6//! The `gen_cert_for_ca` function generates a certificate for the CA, while the `gen_signed_cert`
7//! function generates a certificate signed by the CA. These functions utilize the `rcgen` crate for
8//! generating and managing certificates.
9
10use rcgen::{
11    BasicConstraints, Certificate, CertificateParams, CertificateSigningRequest, DistinguishedName,
12    DnType, IsCa, KeyUsagePurpose, RcgenError, PKCS_ECDSA_P256_SHA256,
13};
14use std::collections::HashSet;
15
16/// Generate a certificate for a certificate authority (CA).
17///
18/// # Returns
19///
20/// Returns a `Certificate` representing the generated CA certificate, or an `RcgenError` if there was an error during the certificate generation.
21pub fn gen_cert_for_ca() -> Result<Certificate, RcgenError> {
22    let mut dn = DistinguishedName::new();
23    dn.push(DnType::CommonName, "Auto-Generated Massalabs CA");
24
25    let mut params = CertificateParams::default();
26    params.is_ca = IsCa::Ca(BasicConstraints::Unconstrained);
27    params.alg = &PKCS_ECDSA_P256_SHA256;
28    params.distinguished_name = dn;
29    params.key_usages = vec![KeyUsagePurpose::KeyCertSign, KeyUsagePurpose::CrlSign];
30
31    let ca_cert = Certificate::from_params(params)?;
32    Ok(ca_cert)
33}
34
35/// Generate a certificate signed by a certificate authority (CA).
36///
37/// # Arguments
38///
39/// * `ca`: A reference to the CA certificate used to sign the generated certificate.
40/// * `subject_alt_names`: A vector of subject alternative names to include in the certificate.
41///
42/// # Returns
43///
44/// Returns a tuple containing the signed certificate PEM and the corresponding private key PEM,
45/// or an `RcgenError` if there was an error during the certificate generation or signing process.
46pub fn gen_signed_cert(
47    ca: &Certificate,
48    subject_alt_names: Vec<String>,
49) -> Result<(String, String), RcgenError> {
50    let mut dn = DistinguishedName::new();
51    dn.push(DnType::CommonName, "Auto-Generated Massa gRPC Server");
52
53    // Add "localhost" to the subject alternative names
54    let all_subject_alt_names_set: HashSet<String> = subject_alt_names
55        .into_iter()
56        .chain(vec!["localhost".to_string()])
57        .collect();
58    let all_subject_alt_names = all_subject_alt_names_set.into_iter().collect::<Vec<_>>();
59    let mut params = CertificateParams::new(all_subject_alt_names);
60    params.is_ca = IsCa::NoCa;
61    params.alg = &PKCS_ECDSA_P256_SHA256;
62    params.distinguished_name = dn;
63
64    let unsigned = Certificate::from_params(params)?;
65    let request_pem = unsigned.serialize_request_pem()?;
66    let csr = CertificateSigningRequest::from_pem(&request_pem)?;
67    let signed_pem = csr.serialize_pem_with_signer(ca)?;
68
69    Ok((signed_pem, unsigned.serialize_private_key_pem()))
70}
