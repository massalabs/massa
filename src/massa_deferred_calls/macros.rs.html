140#[cfg(test)]
141mod tests {
142    use massa_db_exports::DEFERRED_CALLS_PREFIX;
143    use massa_models::{
144        deferred_calls::{DeferredCallId, DeferredCallIdSerializer},
145        slot::Slot,
146    };
147    use massa_serialization::Serializer;
148
149    #[test]
150    fn test_deferred_call_prefix_key() {
151        let slot_ser = Slot {
152            period: 1,
153            thread: 5,
154        }
155        .to_bytes_key();
156
157        let id = DeferredCallId::new(
158            0,
159            Slot {
160                thread: 5,
161                period: 1,
162            },
163            1,
164            &[],
165        )
166        .unwrap();
167
168        let id_ser = DeferredCallIdSerializer::new();
169        let mut buf_id = Vec::new();
170        id_ser.serialize(&id, &mut buf_id).unwrap();
171
172        let prefix = ["deferred_calls/".as_bytes(), &slot_ser, &[0u8], &buf_id].concat();
173
174        assert_eq!(deferred_call_prefix_key!(buf_id, slot_ser), prefix);
175
176        // let to_check = [prefix[..], &[1u8]].concat();
177        // assert_eq!(sender_address_key!(buf_id, slot_ser), to_check);
178    }
