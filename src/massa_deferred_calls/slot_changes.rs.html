28#[derive(Default, Debug, Clone, Serialize, Deserialize)]
29pub struct DeferredRegistrySlotChanges {
30    pub calls: BTreeMap<DeferredCallId, DeferredRegistryCallChange>,
31    pub effective_slot_gas: DeferredRegistryGasChange<u64>,
32    pub base_fee: DeferredRegistryBaseFeeChange,
33}
34
35impl DeferredRegistrySlotChanges {
36    pub fn calls_len(&self) -> usize {
37        self.calls.len()
38    }
39
40    /// Validates that this structure stays within the same bounds enforced by
41    /// [`DeferredRegistrySlotChangesDeserializer`] under `config`.
42    pub fn check_bounds(&self, config: &DeferredCallsConfig) -> Result<(), SerializeError> {
43        let calls_len: u64 = self.calls.len().try_into().map_err(|_| {
44            SerializeError::GeneralError("Fail to transform usize to u64".to_string())
45        })?;
46        if calls_len > config.max_pool_changes {
47            return Err(SerializeError::NumberTooBig(format!(
48                "DeferredRegistrySlotChanges calls length {} exceeds max_pool_changes {}",
49                calls_len, config.max_pool_changes
50            )));
51        }
52        if let DeferredRegistryGasChange::Set(gas) = self.effective_slot_gas {
53            if gas > config.max_gas {
54                return Err(SerializeError::NumberTooBig(format!(
55                    "DeferredRegistrySlotChanges effective_slot_gas {} exceeds max_gas {}",
56                    gas, config.max_gas
57                )));
58            }
59        }
60        Ok(())
61    }
62
63    /// add Delete changes will delete the call from the db registry when the slot is finalized
64    pub fn delete_call(&mut self, id: &DeferredCallId) {
65        match self.calls.entry(id.clone()) {
66            std::collections::btree_map::Entry::Occupied(mut v) => {
67                *v.get_mut() = DeferredRegistryCallChange::Delete;
68            }
69            std::collections::btree_map::Entry::Vacant(v) => {
70                v.insert(DeferredRegistryCallChange::Delete);
71            }
72        }
73    }
74
75    pub fn set_call(&mut self, id: DeferredCallId, call: DeferredCall) {
76        self.calls.insert(id, DeferredRegistryCallChange::Set(call));
77    }
78
79    /// Returns the raw change entry for `id` so that callers can distinguish
80    /// `Set` (present), `Delete` (tombstoned), and `None` (no change recorded
81    /// in this layer). Required to stop speculative lookup cascades when a
82    /// deferred call has been deleted in a newer layer.
83    pub fn get_call_change(&self, id: &DeferredCallId) -> Option<&DeferredRegistryCallChange> {
84        self.calls.get(id)
85    }
86
87    pub fn set_effective_slot_gas(&mut self, gas: u64) {
88        self.effective_slot_gas = DeferredRegistryGasChange::Set(gas);
89    }
90
91    pub fn get_effective_slot_gas(&self) -> Option<u64> {
92        match self.effective_slot_gas {
93            DeferredRegistryGasChange::Set(v) => Some(v),
94            DeferredRegistryGasChange::Keep => None,
95        }
96    }
97
98    pub fn get_base_fee(&self) -> Option<Amount> {
99        match self.base_fee {
100            DeferredRegistryGasChange::Set(v) => Some(v),
101            DeferredRegistryGasChange::Keep => None,
102        }
103    }
104
105    pub fn set_base_fee(&mut self, base_fee: Amount) {
106        self.base_fee = DeferredRegistryGasChange::Set(base_fee);
107    }
108}

110pub struct DeferredRegistrySlotChangesSerializer {
111    deferred_registry_slot_changes_length: U64VarIntSerializer,
112    call_id_serializer: DeferredCallIdSerializer,
113    calls_set_or_delete_serializer: SetOrDeleteSerializer<DeferredCall, DeferredCallSerializer>,
114    gas_serializer: SetOrKeepSerializer<u64, U64VarIntSerializer>,
115    base_fee_serializer: SetOrKeepSerializer<Amount, AmountSerializer>,
116    /// Same bounds as [`DeferredRegistrySlotChangesDeserializer`].
117    config: DeferredCallsConfig,
118}
119
120impl DeferredRegistrySlotChangesSerializer {
121    pub fn new(config: DeferredCallsConfig) -> Self {
122        Self {
123            deferred_registry_slot_changes_length: U64VarIntSerializer::new(),
124            call_id_serializer: DeferredCallIdSerializer::new(),
125            calls_set_or_delete_serializer: SetOrDeleteSerializer::new(
126                DeferredCallSerializer::new(),
127            ),
128            gas_serializer: SetOrKeepSerializer::new(U64VarIntSerializer::new()),
129            base_fee_serializer: SetOrKeepSerializer::new(AmountSerializer::new()),
130            config,
131        }
132    }
133}
134
135impl Default for DeferredRegistrySlotChangesSerializer {
136    fn default() -> Self {
137        Self::new(DeferredCallsConfig::default())
138    }
139}

141impl Serializer<DeferredRegistrySlotChanges> for DeferredRegistrySlotChangesSerializer {
142    fn serialize(
143        &self,
144        value: &DeferredRegistrySlotChanges,
145        buffer: &mut Vec<u8>,
146    ) -> Result<(), massa_serialization::SerializeError> {
147        // Enforce the same bounds as DeferredRegistrySlotChangesDeserializer.
148        value.check_bounds(&self.config)?;
149        self.deferred_registry_slot_changes_length.serialize(
150            &(value.calls.len().try_into().map_err(|_| {
151                SerializeError::GeneralError("Fail to transform usize to u64".to_string())
152            })?),
153            buffer,
154        )?;
155        for (id, call) in &value.calls {
156            self.call_id_serializer.serialize(id, buffer)?;
157            self.calls_set_or_delete_serializer
158                .serialize(call, buffer)?;
159        }
160        self.gas_serializer
161            .serialize(&value.effective_slot_gas, buffer)?;
162        self.base_fee_serializer
163            .serialize(&value.base_fee, buffer)?;
164
165        Ok(())
166    }
167}

169pub struct DeferredRegistrySlotChangesDeserializer {
170    deferred_registry_slot_changes_length: U64VarIntDeserializer,
171    call_id_deserializer: DeferredCallIdDeserializer,
172    calls_set_or_delete_deserializer:
173        SetOrDeleteDeserializer<DeferredCall, DeferredCallDeserializer>,
174    gas_deserializer: SetOrKeepDeserializer<u64, U64VarIntDeserializer>,
175    base_fee_deserializer: SetOrKeepDeserializer<Amount, AmountDeserializer>,
176}
177
178impl DeferredRegistrySlotChangesDeserializer {
179    pub fn new(config: DeferredCallsConfig) -> Self {
180        Self {
181            deferred_registry_slot_changes_length: U64VarIntDeserializer::new(
182                Included(u64::MIN),
183                Included(config.max_pool_changes),
184            ),
185            call_id_deserializer: DeferredCallIdDeserializer::new(),
186            calls_set_or_delete_deserializer: SetOrDeleteDeserializer::new(
187                DeferredCallDeserializer::new(config),
188            ),
189            gas_deserializer: SetOrKeepDeserializer::new(U64VarIntDeserializer::new(
190                Included(0),
191                Included(config.max_gas),
192            )),
193            base_fee_deserializer: SetOrKeepDeserializer::new(AmountDeserializer::new(
194                Included(Amount::MIN),
195                Included(Amount::MAX),
196            )),
197        }
198    }
199}

201impl Deserializer<DeferredRegistrySlotChanges> for DeferredRegistrySlotChangesDeserializer {
202    fn deserialize<'a, E: ParseError<&'a [u8]> + ContextError<&'a [u8]>>(
203        &self,
204        buffer: &'a [u8],
205    ) -> IResult<&'a [u8], DeferredRegistrySlotChanges, E> {
206        context(
207            "Failed DeferredRegistrySlotChanges deserialization",
208            tuple((
209                length_count(
210                    context("Failed length deserialization", |input| {
211                        self.deferred_registry_slot_changes_length
212                            .deserialize(input)
213                    }),
214                    |input: &'a [u8]| {
215                        tuple((
216                            context("Failed id deserialization", |input| {
217                                self.call_id_deserializer.deserialize(input)
218                            }),
219                            context(
220                                "Failed set_update_or_delete_message deserialization",
221                                |input| self.calls_set_or_delete_deserializer.deserialize(input),
222                            ),
223                        ))(input)
224                    },
225                ),
226                context("Failed gas deserialization", |input| {
227                    self.gas_deserializer.deserialize(input)
228                }),
229                context("Failed base fee deserialize", |input| {
230                    self.base_fee_deserializer.deserialize(input)
231                }),
232            )),
233        )
234        // Note: unsorted pairs, and duplicate keys (last occurrence wins), on the wire still deserialize
235        // to a normalized BTreeMap. This serializer/deserializer pair is only used in tests: production
236        // deferred-call processing reads per-key values from the DB and never parses these map blobs.
237        // Massa is malleability-resistant by construction, this is not exploitable.
238        .map(|(vec, gas, base_fee)| {
239            let calls = vec.into_iter().collect::<BTreeMap<_, _>>();
240
241            DeferredRegistrySlotChanges {
242                calls,
243                effective_slot_gas: gas,
244                base_fee,
245            }
246        })
247        .parse(buffer)
248    }
249}

282    fn test_slot_changes_ser_deser() {
283        let config = DeferredCallsConfig::default();
284        let mut registry_slot_changes = DeferredRegistrySlotChanges::default();
285        registry_slot_changes.set_base_fee(Amount::from_str("100").unwrap());
286        registry_slot_changes.set_effective_slot_gas(100_000);
287        let target_slot = Slot {
288            thread: 5,
289            period: 1,
290        };
291
292        let call = sample_call(target_slot);
293        let id = DeferredCallId::new(
294            0,
295            Slot {
296                thread: 5,
297                period: 1,
298            },
299            1,
300            &[],
301        )
302        .unwrap();
303
304        registry_slot_changes.set_call(id, call);
305
306        let mut buffer = Vec::new();
307        let serializer = DeferredRegistrySlotChangesSerializer::new(config);
308        serializer
309            .serialize(&registry_slot_changes, &mut buffer)
310            .unwrap();
311
312        let deserializer = DeferredRegistrySlotChangesDeserializer::new(config);
313        let (rest, changes_deser) = deserializer
314            .deserialize::<DeserializeError>(&buffer)
315            .unwrap();
316        assert!(rest.is_empty());
317        assert_eq!(changes_deser.calls, registry_slot_changes.calls);
318        assert_eq!(
319            changes_deser.effective_slot_gas,
320            registry_slot_changes.effective_slot_gas
321        );
322    }
