19/// Structure gathering all elements needed by the factory thread
20pub(crate) struct EndorsementFactoryWorker {
21    cfg: FactoryConfig,
22    wallet: Arc<RwLock<Wallet>>,
23    channels: FactoryChannels,
24    factory_receiver: MassaReceiver<()>,
25    half_t0: MassaTime,
26    endorsement_serializer: EndorsementSerializer,
27}
28
29impl EndorsementFactoryWorker {
30    /// Creates the `FactoryThread` structure to gather all data and references
31    /// needed by the factory worker thread.
32    pub(crate) fn spawn(
33        cfg: FactoryConfig,
34        wallet: Arc<RwLock<Wallet>>,
35        channels: FactoryChannels,
36        factory_receiver: MassaReceiver<()>,
37    ) -> thread::JoinHandle<()> {
38        thread::Builder::new()
39            .name("endorsement-factory".into())
40            .spawn(|| {
41                let mut this = Self {
42                    half_t0: cfg
43                        .t0
44                        .checked_div_u64(2)
45                        .expect("could not compute half_t0"),
46                    cfg,
47                    wallet,
48                    channels,
49                    factory_receiver,
50                    endorsement_serializer: EndorsementSerializer::new(),
51                };
52                this.run();
53            })
54            .expect("failed to spawn thread : endorsement-factory")
55    }
56
57    /// Gets the next slot and the instant when the corresponding endorsements should be made.
58    /// Slots can be skipped if we waited too much in-between.
59    /// Extra safety against double-production caused by clock adjustments (this is the role of the `previous_slot` parameter).
60    fn get_next_slot(&self, previous_slot: Option<Slot>) -> (Slot, Instant) {
61        // get delayed time
62        let now = MassaTime::now();
63
64        // if it's the first computed slot, add a time shift to prevent double-production on node restart with clock skew
65        let base_time = if previous_slot.is_none() {
66            now.saturating_add(self.cfg.initial_delay)
67        } else {
68            now
69        };
70
71        // get closest slot according to the current absolute time
72        let mut next_slot = get_closest_slot_to_timestamp(
73            self.cfg.thread_count,
74            self.cfg.t0,
75            self.cfg.genesis_timestamp,
76            base_time,
77        );
78
79        // protection against double-production on unexpected system clock adjustment
80        if let Some(prev_slot) = previous_slot {
81            if next_slot <= prev_slot {
82                next_slot = prev_slot
83                    .get_next_slot(self.cfg.thread_count)
84                    .expect("could not compute next slot");
85            }
86        }
87
88        // ignore genesis
89        if next_slot.period <= self.cfg.last_start_period {
90            next_slot = Slot::new(self.cfg.last_start_period + 1, 0);
91        }
92
93        // get the timestamp of the target slot
94        let next_instant = get_block_slot_timestamp(
95            self.cfg.thread_count,
96            self.cfg.t0,
97            self.cfg.genesis_timestamp,
98            next_slot,
99        )
100        .expect("could not get block slot timestamp")
101        .saturating_sub(self.half_t0)
102        .estimate_instant()
103        .expect("could not estimate block slot instant");
104
105        (next_slot, next_instant)
106    }
107
108    /// Wait and interrupt or wait until an instant or a stop signal
109    ///
110    /// # Return value
111    /// Returns `true` if the instant was reached, otherwise `false` if there was an interruption.
112    fn interruptible_wait_until(&self, deadline: Instant) -> bool {
113        match self.factory_receiver.recv_deadline(deadline) {
114            // message received => quit main loop
115            Ok(()) => false,
116            // timeout => continue main loop
117            Err(crossbeam_channel::RecvTimeoutError::Timeout) => true,
118            // channel disconnected (sender dropped) => quit main loop
119            Err(crossbeam_channel::RecvTimeoutError::Disconnected) => false,
120        }
121    }
122
123    /// Process a slot: produce an endorsement at that slot if one of the managed keys is drawn.
124    fn process_slot(&mut self, slot: Slot) {
125        // get endorsement producer addresses for that slot
126        let producer_addrs = match self.channels.selector.get_selection(slot) {
127            Ok(sel) => sel.endorsements,
128            Err(err) => {
129                warn!(
130                    "endorsement factory could not get selector draws for slot {}: {}",
131                    slot, err
132                );
133                return;
134            }
135        };
136
137        // get creators if they are managed by our wallet
138        let mut producers_indices: Vec<(KeyPair, usize)> = Vec::new();
139        {
140            let wallet = self.wallet.read();
141            for (index, producer_addr) in producer_addrs.into_iter().enumerate() {
142                // check if the block producer address is handled by the wallet
143                let producer_keypair =
144                    if let Some(kp) = wallet.find_associated_keypair(&producer_addr) {
145                        // the selected block producer is managed locally => continue to attempt endorsement production
146                        kp.clone()
147                    } else {
148                        // the selected block producer is not managed locally => continue
149                        continue;
150                    };
151                producers_indices.push((producer_keypair, index));
152            }
153        }
154
155        // quit if there is nothing to produce
156        if producers_indices.is_empty() {
157            return;
158        }
159
160        // check if we need to have connections to produce a block and in this case, check if we have enough.
161        #[cfg(not(feature = "sandbox"))]
162        if self.cfg.stop_production_when_zero_connections {
163            if let Ok(stats) = self.channels.protocol.get_stats() {
164                if stats.1.is_empty() {
165                    warn!("endorsement factory could not produce endorsement for slot {} because there are no connections", slot);
166                    return;
167                }
168            }
169        }
170
171        // get consensus block ID for that slot
172        let endorsed_block: BlockId = self
173            .channels
174            .consensus
175            .get_latest_blockclique_block_at_slot(slot);
176
177        // produce endorsements
178        let mut endorsements: Vec<SecureShareEndorsement> =
179            Vec::with_capacity(producers_indices.len());
180        for (keypair, index) in producers_indices {
181            let endorsement = Endorsement::new_verifiable(
182                Endorsement {
183                    slot,
184                    index: index as u32,
185                    endorsed_block,
186                },
187                self.endorsement_serializer.clone(),
188                &keypair,
189                self.cfg.chain_id,
190            )
191            .expect("could not create endorsement");
192
193            // log endorsement creation
194            debug!(
195                "endorsement {} created at slot {} by address {}",
196                endorsement.id, endorsement.content.slot, endorsement.content_creator_address
197            );
198
199            endorsements.push(endorsement);
200        }
201
202        // store endorsements
203        let mut endo_storage = self.channels.storage.clone_without_refs();
204        endo_storage.store_endorsements(endorsements);
205
206        // send endorsement to pool for listing and propagation
207        self.channels.pool.add_endorsements(endo_storage.clone());
208
209        if let Err(err) = self.channels.protocol.propagate_endorsements(endo_storage) {
210            warn!("could not propagate endorsements to protocol: {}", err);
211        }
212    }
213
214    /// main run loop of the endorsement creator thread
215    fn run(&mut self) {
216        let mut prev_slot = None;
217        loop {
218            // get next slot
219            let (slot, endorsement_instant) = self.get_next_slot(prev_slot);
220
221            // wait until slot
222            if !self.interruptible_wait_until(endorsement_instant) {
223                break;
224            }
225
226            // process slot
227            self.process_slot(slot);
228
229            // update previous slot
230            prev_slot = Some(slot);
231        }
232    }
233}
