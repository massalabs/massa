23/// Structure gathering all elements needed by the factory thread
24pub(crate) struct BlockFactoryWorker {
25    cfg: FactoryConfig,
26    wallet: Arc<RwLock<Wallet>>,
27    channels: FactoryChannels,
28    factory_receiver: MassaReceiver<()>,
29    mip_store: MipStore,
30    op_id_serializer: OperationIdSerializer,
31}
32
33impl BlockFactoryWorker {
34    /// Creates the `FactoryThread` structure to gather all data and references
35    /// needed by the factory worker thread.
36    pub(crate) fn spawn(
37        cfg: FactoryConfig,
38        wallet: Arc<RwLock<Wallet>>,
39        channels: FactoryChannels,
40        factory_receiver: MassaReceiver<()>,
41        mip_store: MipStore,
42    ) -> thread::JoinHandle<()> {
43        thread::Builder::new()
44            .name("block-factory".into())
45            .spawn(|| {
46                let mut this = Self {
47                    cfg,
48                    wallet,
49                    channels,
50                    factory_receiver,
51                    mip_store,
52                    op_id_serializer: OperationIdSerializer::new(),
53                };
54                this.run();
55            })
56            .expect("failed to spawn thread : block-factory")
57    }

59    /// Gets the next slot and the instant when it will happen.
60    /// Slots can be skipped if we waited too much in-between.
61    /// Extra safety against double-production caused by clock adjustments (this is the role of the `previous_slot` parameter).
62    fn get_next_slot(&self, previous_slot: Option<Slot>) -> (Slot, Instant) {
63        // get current absolute time
64        let now = MassaTime::now();
65
66        // if it's the first computed slot, add a time shift to prevent double-production on node restart with clock skew
67        let base_time = if previous_slot.is_none() {
68            now.saturating_add(self.cfg.initial_delay)
69        } else {
70            now
71        };
72
73        // get closest slot according to the current absolute time
74        let mut next_slot = get_closest_slot_to_timestamp(
75            self.cfg.thread_count,
76            self.cfg.t0,
77            self.cfg.genesis_timestamp,
78            base_time,
79        );
80
81        // ignore genesis
82        if next_slot.period <= self.cfg.last_start_period {
83            next_slot = Slot::new(self.cfg.last_start_period + 1, 0);
84        }
85
86        // protection against double-production on unexpected system clock adjustment
87        if let Some(prev_slot) = previous_slot {
88            if next_slot <= prev_slot {
89                next_slot = prev_slot
90                    .get_next_slot(self.cfg.thread_count)
91                    .expect("could not compute next slot");
92            }
93        }
94
95        // get the timestamp of the target slot
96        let next_instant = get_block_slot_timestamp(
97            self.cfg.thread_count,
98            self.cfg.t0,
99            self.cfg.genesis_timestamp,
100            next_slot,
101        )
102        .expect("could not get block slot timestamp")
103        .estimate_instant()
104        .expect("could not estimate block slot instant");
105
106        (next_slot, next_instant)
107    }

109    /// Wait and interrupt or wait until an instant or a stop signal
110    ///
111    /// # Return value
112    /// Returns `true` if the instant was reached, otherwise `false` if there was an interruption.
113    fn interruptible_wait_until(&self, deadline: Instant) -> bool {
114        match self.factory_receiver.recv_deadline(deadline) {
115            // message received => quit main loop
116            Ok(()) => false,
117            // timeout => continue main loop
118            Err(crossbeam_channel::RecvTimeoutError::Timeout) => true,
119            // channel disconnected (sender dropped) => quit main loop
120            Err(crossbeam_channel::RecvTimeoutError::Disconnected) => false,
121        }
122    }

124    /// Process a slot: produce a block at that slot if one of the managed keys is drawn.
125    fn process_slot(&mut self, slot: Slot) {
126        let mut timings = Vec::new();
127        let channel_timeout = self.cfg.block_opt_channel_timeout;
128        let mut had_pool_timeout = false;
129
130        // get block producer address for that slot
131        timings.push(("get_producer START", MassaTime::now()));
132        let block_producer_addr = match self.channels.selector.get_producer(slot) {
133            Ok(addr) => addr,
134            Err(err) => {
135                warn!(
136                    "block factory could not get selector draws for slot {}: {}",
137                    slot, err
138                );
139                return;
140            }
141        };
142        timings.push(("get_producer END", MassaTime::now()));
143
144        // check if the block producer address is handled by the wallet
145        timings.push(("wallet retrieval START", MassaTime::now()));
146        let block_producer_keypair;
147        {
148            let block_producer_keypair_ref = self.wallet.read();
149            if let Some(kp) =
150                block_producer_keypair_ref.find_associated_keypair(&block_producer_addr)
151            {
152                // the selected block producer is managed locally => continue to attempt block production
153                block_producer_keypair = kp.clone();
154            } else {
155                // the selected block producer is not managed locally => quit
156                return;
157            }
158        }
159        timings.push(("wallet retrieval END", MassaTime::now()));
160
161        timings.push(("double-staking check START", MassaTime::now()));
162        let mut block_storage = self.channels.storage.clone_without_refs();
163        {
164            let block_lock = block_storage.read_blocks();
165            if let Some(block_ids) = block_lock.get_blocks_by_slot(&slot) {
166                for block_id in block_ids {
167                    if let Some(block) = block_lock.get(block_id) {
168                        if block.content_creator_address == block_producer_addr {
169                            panic!("You already created a block for slot {} with address {}, node is stopping to prevent you from losing all your stake due to double staking protection", slot, block_producer_addr);
170                        }
171                    }
172                }
173            }
174        }
175        timings.push(("double-staking check END", MassaTime::now()));
176
177        // check if we need to have connections to produce a block and in this case, check if we have enough.
178        #[cfg(not(feature = "sandbox"))]
179        if self.cfg.stop_production_when_zero_connections {
180            timings.push(("connectivity check START", MassaTime::now()));
181            if let Ok(stats) = self.channels.protocol.get_stats() {
182                if stats.1.is_empty() {
183                    warn!("block factory could not produce block for slot {} because there are no connections", slot);
184                    return;
185                }
186            }
187            timings.push(("connectivity check END", MassaTime::now()));
188        }
189
190        // get best parents and their periods
191        timings.push(("get_best_parents START", MassaTime::now()));
192        let parents: Vec<(BlockId, u64)> = self.channels.consensus.get_best_parents(); // Vec<(parent_id, parent_period)>
193        timings.push(("get_best_parents END", MassaTime::now()));
194        // generate the local storage object
195
196        // get the parent in the same thread, with its period
197        // will not panic because the thread is validated before the call
198        let (same_thread_parent_id, _) = parents[slot.thread as usize];
199
200        // gather endorsements
201        timings.push(("get_block_endorsements START", MassaTime::now()));
202        let (endorsements_ids, endo_storage) = match self.channels.pool.get_block_endorsements(
203            &same_thread_parent_id,
204            &slot,
205            Some(channel_timeout),
206        ) {
207            Ok(result) => result,
208            Err(e) => {
209                // Pool reads are best effort: a block MUST be produced on time. A late or missed block is worse
210                // than an empty one (lost block reward, production stat miss, the network loses the slot),
211                // while an empty block only loses the fees of that block. So pool lock timeouts degrade to an
212                // empty block instead of aborting the slot. This also prevents pool slowness (e.g. op flooding)
213                // from turning into missed blocks. Timeouts are exposed via metrics so the node runner can act.
214                had_pool_timeout = true;
215                massa_metrics::inc_factory_pool_timeout_endorsements();
216                warn!(
217                    "Channel timeout: get_block_endorsements took >{}ms for slot {}: {:?}. Proceeding with empty endorsements.",
218                    channel_timeout.as_millis(), slot, e
219                );
220                (
221                    vec![None; self.cfg.endorsement_count as usize],
222                    self.channels.storage.clone_without_refs(),
223                )
224            }
225        };
226        timings.push(("get_block_endorsements END", MassaTime::now()));
227
228        //TODO: Do we want to populate only with endorsement id in the future ?
229        timings.push(("add endorsements START", MassaTime::now()));
230        let endorsements: Vec<SecureShareEndorsement> = {
231            let endo_read = endo_storage.read_endorsements();
232            endorsements_ids
233                .into_iter()
234                .flatten()
235                .map(|endo_id| {
236                    endo_read
237                        .get(&endo_id)
238                        .expect("could not retrieve endorsement")
239                        .clone()
240                })
241                .collect()
242        };
243        block_storage.extend(endo_storage);
244        timings.push(("add endorsements END", MassaTime::now()));
245
246        // gather operations and compute global operations hash
247        timings.push(("get_block_operations START", MassaTime::now()));
248        let (op_ids, op_storage) = match self
249            .channels
250            .pool
251            .get_block_operations(&slot, Some(channel_timeout))
252        {
253            Ok(result) => result,
254            Err(e) => {
255                had_pool_timeout = true;
256                massa_metrics::inc_factory_pool_timeout_operations();
257                warn!(
258                    "Channel timeout: get_block_operations took >{}ms for slot {}: {:?}. Proceeding without operations.",
259                    channel_timeout.as_millis(), slot, e
260                );
261                (vec![], massa_storage::Storage::create_root())
262            }
263        };
264        if op_ids.len() > self.cfg.max_operations_per_block as usize {
265            warn!("Too many operations returned");
266            return;
267        }
268        timings.push(("get_block_operations END", MassaTime::now()));
269
270        timings.push(("add operations START", MassaTime::now()));
271        block_storage.extend(op_storage);
272        timings.push(("add operations END", MassaTime::now()));
273
274        // create header
275        let slot_ts = get_block_slot_timestamp(
276            self.cfg.thread_count,
277            self.cfg.t0,
278            self.cfg.genesis_timestamp,
279            slot,
280        )
281        .expect("could not get block slot timestamp");
282        timings.push(("mip data START", MassaTime::now()));
283        // Versions of *this slot*, not of the store's current state: the store only
284        // turns Active once finality reaches the activation slot, and stamping the
285        // current state would put the old version on every block produced during that
286        // lag -- blocks that nodes already Active then reject. Validation checks the
287        // header against get_network_version_active_at(slot_ts) too.
288        let current_version = self.mip_store.get_network_version_active_at(slot_ts);
289        let announced_version = self.mip_store.get_network_version_to_announce_at(slot_ts);
290        timings.push(("mip data END", MassaTime::now()));
291
292        timings.push(("get_block_denunciations START", MassaTime::now()));
293        let denunciations = match self
294            .channels
295            .pool
296            .get_block_denunciations(&slot, Some(channel_timeout))
297        {
298            Ok(result) => result,
299            Err(e) => {
300                had_pool_timeout = true;
301                massa_metrics::inc_factory_pool_timeout_denunciations();
302                warn!(
303                    "Channel timeout: get_block_denunciations took >{}ms for slot {}: {:?}. Proceeding without denunciations.",
304                    channel_timeout.as_millis(), slot, e
305                );
306                vec![]
307            }
308        };
309        timings.push(("get_block_denunciations END", MassaTime::now()));
310
311        timings.push(("block creation START", MassaTime::now()));
312        let sig_chain_id = sig_chain_id_for_slot(&self.mip_store, self.cfg.chain_id, slot_ts);
313        let header: SecuredHeader = BlockHeader::new_verifiable::<BlockHeaderSerializer, BlockId>(
314            BlockHeader {
315                current_version,
316                announced_version,
317                slot,
318                parents: parents.into_iter().map(|(id, _period)| id).collect(),
319                operation_merkle_root: compute_operations_hash(&op_ids, &self.op_id_serializer),
320                endorsements,
321                denunciations,
322            },
323            BlockHeaderSerializer::new(), // TODO reuse self.block_header_serializer
324            &block_producer_keypair,
325            self.cfg.chain_id,
326            sig_chain_id,
327        )
328        .expect("error while producing block header");
329        // create block
330        let block_ = Block {
331            header,
332            operations: op_ids.into_iter().collect(),
333        };
334        let block = Block::new_verifiable(
335            block_,
336            BlockSerializer::new(), // TODO reuse self.block_serializer
337            &block_producer_keypair,
338            self.cfg.chain_id,
339            None,
340        )
341        .expect("error while producing block");
342        let block_id = block.id;
343        timings.push(("block creation END", MassaTime::now()));
344
345        // store block in storage
346        timings.push(("block store START", MassaTime::now()));
347        block_storage.store_block(block);
348        timings.push(("block store END", MassaTime::now()));
349
350        // log block creation
351        info!(
352            "block {} created at slot {} by address {}",
353            block_id, slot, block_producer_addr
354        );
355
356        // send full block to consensus
357        timings.push(("register_block START", MassaTime::now()));
358        self.channels
359            .consensus
360            .register_block(block_id, slot, block_storage, true);
361        massa_metrics::inc_factory_blocks_produced(had_pool_timeout);
362        timings.push(("register_block END", MassaTime::now()));
363
364        // Check block latency
365        let block_timestamp = get_block_slot_timestamp(
366            self.cfg.thread_count,
367            self.cfg.t0,
368            self.cfg.genesis_timestamp,
369            slot,
370        )
371        .expect("could not get block slot timestamp");
372        let cur_timestamp = MassaTime::now();
373        let delay = cur_timestamp.saturating_sub(block_timestamp);
374        if delay > self.cfg.block_delay_warn {
375            let mut timing_details = String::new();
376            for (step_name, step_t) in &timings {
377                timing_details.push_str(&format!(" / {} at ts={}", step_name, step_t));
378            }
379            warn!(
380                "Block {} production at slot {} was delayed by {}ms (threshold: {}ms) [{}]",
381                block_id,
382                slot,
383                delay.as_millis(),
384                self.cfg.block_delay_warn.as_millis(),
385                timing_details
386            );
387        }
388    }
389
390    /// main run loop of the block creator thread
391    fn run(&mut self) {
392        let mut prev_slot = None;
393        loop {
394            // get next slot
395            let (slot, block_instant) = self.get_next_slot(prev_slot);
396
397            // wait until slot
398            if !self.interruptible_wait_until(block_instant) {
399                break;
400            }
401
402            // process slot
403            self.process_slot(slot);
404
405            // update previous slot
406            prev_slot = Some(slot);
407        }
408    }
