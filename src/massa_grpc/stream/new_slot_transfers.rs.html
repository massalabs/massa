22/// Creates a new stream of new slots transfers
23pub(crate) async fn new_slot_transfers(
24    grpc: &MassaPublicGrpc,
25    request: Request<Streaming<massa_proto_rs::massa::api::v1::NewSlotTransfersRequest>>,
26) -> Result<NewSlotTransfersStreamType, GrpcError> {
27    use crate::error::match_for_io_error;
28    use futures_util::StreamExt;
29    use massa_proto_rs::massa::api::v1::{self as grpc_api, FinalityLevel, TransferInfo};
30    use tokio::select;
31    use tracing::{error, warn};
32
33    let (tx, rx) = tokio::sync::mpsc::channel(grpc.grpc_config.max_channel_size);
34    // Extract the incoming stream of abi call stacks messages
35    let mut in_stream = request.into_inner();
36
37    // Subscribe to the new slot execution events channel
38    let mut subscriber = grpc
39        .execution_channels
40        .slot_execution_traces_sender
41        .subscribe();
42
43    tokio::spawn({
44        async move {
45            let mut finality = FinalityLevel::Unspecified;
46            loop {
47                select! {
48                    // Receive a new slot execution traces from the subscriber
49                    event = subscriber.recv() => {
50                        match event {
51                            Ok((massa_slot_execution_trace, slot_transfers, is_final)) => {
52                                if (finality == FinalityLevel::Final && !is_final) ||
53                                    (finality == FinalityLevel::Candidate && is_final) ||
54                                    (finality == FinalityLevel::Unspecified && !is_final) {
55                                    continue;
56                                }
57                                let mut ret_transfers = Vec::new();
58                                // flatten & filter transfer trace in asc_call_stacks
59
60                                let abi_transfer_1 = "assembly_script_transfer_coins".to_string();
61                                let abi_transfer_2 = "assembly_script_transfer_coins_for".to_string();
62                                let abi_transfer_3 = "abi_transfer_coins".to_string();
63                                let transfer_abi_names = vec![abi_transfer_1, abi_transfer_2, abi_transfer_3];
64                                for (i, asc_call_stack) in massa_slot_execution_trace.asc_call_stacks.iter().enumerate() {
65                                    for abi_trace in asc_call_stack {
66                                        let only_transfer = abi_trace.flatten_filter(&transfer_abi_names);
67                                        for transfer in only_transfer {
68                                            let (t_from, t_to, t_amount) = transfer.parse_transfer();
69                                            ret_transfers.push(TransferInfo {
70                                                from: t_from.clone(),
71                                                to: t_to.clone(),
72                                                amount: t_amount,
73                                                operation_id_or_asc_index: Some(
74                                                    grpc_api::transfer_info::OperationIdOrAscIndex::AscIndex(i as u64),
75                                                ),
76                                            });
77                                        }
78                                    }
79                                }
80
81                                for deferred_call_call_stack in massa_slot_execution_trace.deferred_call_stacks {
82                                    let deferred_call_id = deferred_call_call_stack.0;
83                                    let deferred_call_call_stack = deferred_call_call_stack.1;
84                                    for abi_trace in deferred_call_call_stack {
85                                        let only_transfer = abi_trace.flatten_filter(&transfer_abi_names);
86                                        for transfer in only_transfer {
87                                            let (t_from, t_to, t_amount) = transfer.parse_transfer();
88                                            ret_transfers.push(TransferInfo {
89                                                from: t_from.clone(),
90                                                to: t_to.clone(),
91                                                amount: t_amount,
92                                                operation_id_or_asc_index: Some(
93                                                    grpc_api::transfer_info::OperationIdOrAscIndex::DeferredCallId(deferred_call_id.to_string()),
94                                                ),
95                                            });
96                                        }
97                                    }
98                                }
99
100                                for op_call_stack in massa_slot_execution_trace.operation_call_stacks {
101                                    let op_id = op_call_stack.0;
102                                    let op_call_stack = op_call_stack.1;
103                                    for abi_trace in op_call_stack {
104                                        let only_transfer = abi_trace.flatten_filter(&transfer_abi_names);
105                                        for transfer in only_transfer {
106                                            let (t_from, t_to, t_amount) = transfer.parse_transfer();
107                                            ret_transfers.push(TransferInfo {
108                                                from: t_from.clone(),
109                                                to: t_to.clone(),
110                                                amount: t_amount,
111                                                operation_id_or_asc_index: Some(
112                                                    grpc_api::transfer_info::OperationIdOrAscIndex::OperationId(op_id.to_string()),
113                                                ),
114                                            });
115                                        }
116                                    }
117                                }
118                                // Transfers are carried in the event itself, bound to the same
119                                // execution instance as the abi call stacks above. This avoids a
120                                // separate slot-keyed lookup, which could return transfers from a
121                                // re-execution of the same slot and yield a hybrid response.
122                                for transfer in slot_transfers {
123                                    ret_transfers.push(TransferInfo {
124                                        from: transfer.from.to_string(),
125                                        to: transfer.to.to_string(),
126                                        amount: transfer.amount.to_raw(),
127                                        operation_id_or_asc_index: Some(
128                                            grpc_api::transfer_info::OperationIdOrAscIndex::OperationId(
129                                                transfer.op_id.to_string(),
130                                            ),
131                                        ),
132                                    });
133                                }
134                                let ret = grpc_api::NewSlotTransfersResponse {
135                                    slot: Some(massa_slot_execution_trace.slot.into()),
136                                    transfers: ret_transfers,
137                                };
138
139                                if let Err(e) = tx.send(Ok(ret)).await {
140                                    error!("failed to send new slot execution trace: {}", e);
141                                    break;
142                                }
143                            }
144                            Err(e) => {
145                                error!("error on receive new slot execution trace : {}", e)
146                            }
147                        }
148                    }
149                    // Receive a new message from the in_stream
150                    res = in_stream.next() => {
151                        match res {
152                            Some(res) => {
153                                match res {
154                                    Ok(message) => {
155                                        finality = message.finality_level();
156                                    },
157                                    Err(e) => {
158                                        // Any io error -> break
159                                        if let Some(io_err) = match_for_io_error(&e) {
160                                            warn!("client disconnected, broken pipe: {}", io_err);
161                                            break;
162                                        }
163                                        error!("{}", e);
164                                        if let Err(e2) = tx.send(Err(e)).await {
165                                            error!("failed to send back error response: {}", e2);
166                                            break;
167                                        }
168                                    }
169                                }
170                            }
171                            None => {
172                                // the client has disconnected
173                                break;
174                            }
175                        }
176                    }
177                }
178            }
179        }
180    });
181    let out_stream = tokio_stream::wrappers::ReceiverStream::new(rx);
182    Ok(Box::pin(out_stream) as NewSlotTransfersStreamType)
183}
