10#[cfg_attr(feature = "test-exports", mockall_wrap::wrap, mockall::automock)]
11pub trait LedgerController: Send + Sync {
12    /// Loads ledger from file
13    fn load_initial_ledger(&mut self) -> Result<(), LedgerError>;
14
15    /// Gets the balance of a ledger entry
16    ///
17    /// # Returns
18    /// The balance, or None if the ledger entry was not found
19    fn get_balance(&self, addr: &Address) -> Option<Amount>;
20
21    /// Gets a copy of the bytecode of a ledger entry
22    ///
23    /// # Returns
24    /// A copy of the found bytecode, or None if the ledger entry was not found
25    fn get_bytecode(&self, addr: &Address) -> Option<Bytecode>;
26
27    /// Checks if a ledger entry exists
28    ///
29    /// # Returns
30    /// true if it exists, false otherwise.
31    fn entry_exists(&self, addr: &Address) -> bool;
32
33    /// Gets a copy of the value of a datastore entry for a given address.
34    ///
35    /// # Arguments
36    /// * `addr`: target address
37    /// * `key`: datastore key
38    ///
39    /// # Returns
40    /// A copy of the datastore value, or `None` if the ledger entry or datastore entry was not found
41    fn get_data_entry(&self, addr: &Address, key: &[u8]) -> Option<Vec<u8>>;
42
43    /// Get every key of the datastore for a given address.
44    ///
45    /// # Returns
46    /// A `BTreeSet` of the datastore keys
47    fn get_datastore_keys(
48        &self,
49        addr: &Address,
50        prefix: &[u8],
51        start_key: Bound<Vec<u8>>,
52        end_key: Bound<Vec<u8>>,
53        count: Option<u32>,
54    ) -> Option<BTreeSet<Vec<u8>>>;
55
56    /// Reset the ledger
57    ///
58    /// USED FOR BOOTSTRAP ONLY
59    fn reset(&mut self);
60
61    fn apply_changes_to_batch(&mut self, changes: LedgerChanges, ledger_batch: &mut DBBatch);
62
63    /// Deserializes the key and value, useful after bootstrap
64    fn is_key_value_valid(&self, serialized_key: &[u8], serialized_value: &[u8]) -> bool;
65
66    /// Get every address and their corresponding balance.
67    ///
68    /// IMPORTANT: This should only be used for debug and test purposes.
69    ///
70    /// # Returns
71    /// A `BTreeMap` with the address as key and the balance as value
72    #[cfg(feature = "test-exports")]
73    fn get_every_address(&self) -> std::collections::BTreeMap<Address, Amount>;
74
75    /// Get the entire datastore for a given address.
76    ///
77    /// IMPORTANT: This should only be used for debug purposes.
78    ///
79    /// # Returns
80    /// A `BTreeMap` with the entry hash as key and the data bytes as value
81    #[cfg(feature = "test-exports")]
82    fn get_entire_datastore(&self, addr: &Address) -> std::collections::BTreeMap<Vec<u8>, Vec<u8>>;
83}
