14pub const VERSION_IDENT: u8 = 0u8;

22enum KeyTypeId {

24    Balance = 1,
25    Bytecode = 2,
26    Datastore = 3,

32    BALANCE,
33    BYTECODE,
34    DATASTORE(Vec<u8>),

37#[derive(Default, Clone)]
38pub struct KeyTypeSerializer {
39    vec_u8_serializer: VecU8Serializer,
40    // Whether is deserialized with VecU8Deserializer or not.
41    // If true, we use the VecU8Serializer to serialize the key which will add the length at the beginning.
42    // If false, we just serialize the key as is.
43    // This allows us to store the datastore key length at the beginning of the key or not.
44    // The datastore key length is useful when transferring multiple keys, like in packets,
45    // but isn't when storing a datastore key in the ledger.
46    with_datastore_key_length: bool,
47}
48
49impl KeyTypeSerializer {
50    /// Creates a new KeyTypeSerializer.
51    /// `with_datastore_key_length` if true, the datastore key is serialized with its length.
52    pub fn new(with_datastore_key_length: bool) -> Self {
53        Self {
54            vec_u8_serializer: VecU8Serializer::new(),
55            with_datastore_key_length,
56        }
57    }
58}

60impl Serializer<KeyType> for KeyTypeSerializer {
61    fn serialize(&self, value: &KeyType, buffer: &mut Vec<u8>) -> Result<(), SerializeError> {
62        match value {
63            KeyType::VERSION => buffer.extend(&[u8::from(KeyTypeId::Version)]),
64            KeyType::BALANCE => buffer.extend(&[u8::from(KeyTypeId::Balance)]),
65            KeyType::BYTECODE => buffer.extend(&[u8::from(KeyTypeId::Bytecode)]),
66            KeyType::DATASTORE(data) => {
67                buffer.extend(&[u8::from(KeyTypeId::Datastore)]);
68                if self.with_datastore_key_length {
69                    self.vec_u8_serializer.serialize(data, buffer)?;
70                } else {
71                    buffer.extend(data);
72                }
73            }
74        }
75        Ok(())
76    }
77}

80pub struct KeyTypeDeserializer {
81    vec_u8_deserializer: VecU8Deserializer,
82    max_datastore_key_length: u8,
83    // Same as in KeyTypeSerializer but for deserialization.
84    with_datastore_key_length: bool,
85}

87impl KeyTypeDeserializer {
88    /// Creates a new KeyTypeDeserializer.
89    /// `max_datastore_key_length` is the maximum length of a datastore key.
90    /// `with_datastore_key_length` if true, the datastore key is deserialized with its length.
91    pub fn new(max_datastore_key_length: u8, with_datastore_key_length: bool) -> Self {
92        Self {
93            vec_u8_deserializer: VecU8Deserializer::new(
94                Included(u64::MIN),
95                Included(max_datastore_key_length as u64),
96            ),
97            max_datastore_key_length,
98            with_datastore_key_length,
99        }
100    }
101}

103impl Deserializer<KeyType> for KeyTypeDeserializer {
104    fn deserialize<'a, E: ParseError<&'a [u8]> + ContextError<&'a [u8]>>(
105        &self,
106        input: &'a [u8],
107    ) -> nom::IResult<&'a [u8], KeyType, E> {
108        let (rest, key_type) = nom::number::complete::le_u8(input)?;
109        match KeyTypeId::try_from(key_type) {
110            Ok(KeyTypeId::Balance) => Ok((rest, KeyType::BALANCE)),
111            Ok(KeyTypeId::Bytecode) => Ok((rest, KeyType::BYTECODE)),
112            Ok(KeyTypeId::Datastore) => {
113                if self.with_datastore_key_length {
114                    let (rest, data) = self.vec_u8_deserializer.deserialize(rest)?;
115                    Ok((rest, KeyType::DATASTORE(data)))
116                } else {
117                    if rest.len() > self.max_datastore_key_length as usize {
118                        return Err(nom::Err::Error(E::from_error_kind(
119                            rest,
120                            nom::error::ErrorKind::LengthValue,
121                        )));
122                    }
123                    Ok((&[], KeyType::DATASTORE(rest.to_vec())))
124                }
125            }
126            Ok(KeyTypeId::Version) => Ok((rest, KeyType::VERSION)),
127            Err(_) => Err(nom::Err::Error(E::from_error_kind(
128                rest,
129                nom::error::ErrorKind::Tag,
130            ))),
131        }
132    }
133}

151/// Gives the general prefix of the datastore of an address while respecting a provided key prefix
152pub fn datastore_prefix_from_address(address: &Address, prefix: &[u8]) -> Vec<u8> {
153    let mut res_prefix = LEDGER_PREFIX.as_bytes().to_vec();
154    U64VarIntSerializer::new()
155        .serialize(&KEY_VERSION, &mut res_prefix)
156        .unwrap();
157    AddressSerializer::new()
158        .serialize(address, &mut res_prefix)
159        .unwrap();
160    res_prefix.push(DATASTORE_IDENT);
161    res_prefix.extend(prefix);
162    res_prefix
163}

165/// Serialized size of `value` as a u64 varint (7 bits per byte)
166const fn u64_varint_length(value: u64) -> usize {
167    let bits = (u64::BITS - value.leading_zeros()) as usize;
168    if bits == 0 {
169        1
170    } else {
171        bits.div_ceil(7)
172    }
173}
174
175/// Maximum length of a serialized ledger key as stored in the state DB:
176/// `LEDGER_PREFIX`, key version, address (type, version, hash), `DATASTORE_IDENT` and
177/// a datastore key of at most `max_datastore_key_length` bytes.
178///
179/// This is larger than `max_datastore_key_length` itself, so anything bounding raw
180/// state DB keys (e.g. bootstrap streaming) must use this instead.
181pub const fn max_ledger_db_key_length(max_datastore_key_length: u8) -> u64 {
182    // The key version is always `KEY_VERSION` (enforced by `KeyDeserializer`). The address
183    // type and version are varints that may grow with new address kinds/versions, so count
184    // them at their maximum size to keep this bound valid without having to update it.
185    (LEDGER_PREFIX.len()
186        + u64_varint_length(KEY_VERSION)
187        + 2 * u64_varint_length(u64::MAX)
188        + HASH_SIZE_BYTES
189        + 1) as u64
190        + max_datastore_key_length as u64
191}

206            address_serializer: AddressSerializer::new(),

213impl Serializer<Key> for KeySerializer {
214    /// ```
215    /// use massa_models::address::Address;
216    /// use massa_ledger_exports::{KeySerializer, KeyType, Key};
217    /// use massa_serialization::Serializer;
218    /// use massa_hash::Hash;
219    /// use std::str::FromStr;
220    ///
221    /// let mut serialized = Vec::new();
222    /// let address = Address::from_str("AU12dG5xP1RDEB5ocdHkymNVvvSJmUL9BgHwCksDowqmGWxfpm93x").unwrap();
223    /// let store_key = Hash::compute_from(b"test");
224    /// let mut key = Key::new(&address, KeyType::DATASTORE(store_key.into_bytes().to_vec()));
225    /// KeySerializer::new(true).serialize(&key, &mut serialized).unwrap();
226    /// ```
227    fn serialize(&self, value: &Key, buffer: &mut Vec<u8>) -> Result<(), SerializeError> {
228        buffer.extend(LEDGER_PREFIX.as_bytes());
229
230        self.version_byte_serializer
231            .serialize(&KEY_VERSION, buffer)?;
232        self.address_serializer.serialize(&value.address, buffer)?;
233        self.key_type_serializer
234            .serialize(&value.key_type, buffer)?;
235
236        Ok(())
237    }
238}

264impl Deserializer<Key> for KeyDeserializer {
265    /// ## Example
266    /// ```
267    /// use massa_models::address::Address;
268    /// use massa_ledger_exports::{KeyDeserializer, KeySerializer, DATASTORE_IDENT, BALANCE_IDENT, KeyType, Key};
269    /// use massa_serialization::{Deserializer, Serializer, DeserializeError};
270    /// use massa_hash::Hash;
271    /// use std::str::FromStr;
272    ///
273    /// let address = Address::from_str("AU12dG5xP1RDEB5ocdHkymNVvvSJmUL9BgHwCksDowqmGWxfpm93x").unwrap();
274    /// let store_key = Hash::compute_from(b"test");
275    ///
276    /// let mut key = Key::new(&address, KeyType::DATASTORE(store_key.into_bytes().to_vec()));
277    /// let mut serialized = Vec::new();
278    /// KeySerializer::new(true).serialize(&key, &mut serialized).unwrap();
279    /// let (rest, key_deser) = KeyDeserializer::new(255, true).deserialize::<DeserializeError>(&serialized).unwrap();
280    /// assert!(rest.is_empty());
281    /// assert_eq!(key_deser, key);
282    ///
283    /// let mut key = Key::new(&address, KeyType::BALANCE);
284    /// let mut serialized = Vec::new();
285    /// KeySerializer::new(true).serialize(&key, &mut serialized).unwrap();
286    /// let (rest, key_deser) = KeyDeserializer::new(255, true).deserialize::<DeserializeError>(&serialized).unwrap();
287    /// assert!(rest.is_empty());
288    /// assert_eq!(key_deser, key);
289    /// ```
290    fn deserialize<'a, E: ParseError<&'a [u8]> + ContextError<&'a [u8]>>(
291        &self,
292        buffer: &'a [u8],
293    ) -> nom::IResult<&'a [u8], Key, E> {
294        // Skip the ledger prefix without slicing past the end of the buffer:
295        // an undersized input must yield a clean parse error rather than an
296        // out-of-bounds panic.
297        let prefix_len = LEDGER_PREFIX.len();
298        let after_prefix = buffer.get(prefix_len..).ok_or_else(|| {
299            nom::Err::Error(E::from_error_kind(buffer, nom::error::ErrorKind::Eof))
300        })?;
301
302        let (rest, version) = self.version_byte_deserializer.deserialize(after_prefix)?;
303        // Enforce the canonical key version. `KeySerializer` always writes
304        // `KEY_VERSION`, so any other value is a non-canonical / malformed key.
305        // Rejecting it prevents version malleability where distinct encodings
306        // deserialize into the same logical `Key`.
307        if version != KEY_VERSION {
308            return Err(nom::Err::Error(E::from_error_kind(
309                after_prefix,
310                nom::error::ErrorKind::Verify,
311            )));
312        }
313        let (rest, address) = self.address_deserializer.deserialize(rest)?;
314        let (rest, key_type) = self.key_type_deserializer.deserialize(rest)?;
315
316        Ok((rest, Key { address, key_type }))
317    }
318}

387        let mut serialized = Vec::new();

390            .unwrap();

392        // The version is a u64 varint written immediately after the prefix;

400        assert!(res.is_err(), "a non-canonical key version must be rejected");

414                    let key = KeyType::DATASTORE(vec![0x80; length]);
415                    let mut bytes = Vec::new();

422                        assert_eq!(result.unwrap(), (&[][..], key));

424                        assert!(
425                            result.is_err(),
