210    /// Try load initial deferred credits from file
211    ///
212    /// Credits that would fail at execution time (same rules as `transfer_coins` when
213    /// crediting with `from_addr = None`) are skipped:
214    /// * non-existing SC address
215    /// * non-existing user address with `amount` below the ledger entry base cost
216    pub fn load_initial_deferred_credits(
217        &mut self,
218        batch: &mut DBBatch,
219        ledger: &dyn LedgerController,
220    ) -> Result<(), PosError> {
221        let Some(initial_deferred_credits_path) = &self.config.initial_deferred_credits_path else {
222            return Ok(());
223        };
224
225        use serde::Deserialize;
226        #[derive(Deserialize)]
227        struct AddressInitialDeferredCredits {
228            slot: Slot,
229            amount: Amount,
230        }
231
232        let initial_deferred_credits =
233            serde_json::from_str::<PreHashMap<Address, Vec<AddressInitialDeferredCredits>>>(
234                &std::fs::read_to_string(initial_deferred_credits_path).map_err(|err| {
235                    PosError::DeferredCreditsFileLoadingError(format!(
236                        "error while deserializing initial deferred credits file {}: {}",
237                        initial_deferred_credits_path.display(),
238                        err
239                    ))
240                })?,
241            )
242            .map_err(|err| {
243                PosError::DeferredCreditsFileLoadingError(format!(
244                    "error loading initial deferred credits file {}: {}",
245                    initial_deferred_credits_path.display(),
246                    err
247                ))
248            })?;
249
250        for (address, deferred_credits) in initial_deferred_credits {
251            let address_is_sc = matches!(address, Address::SC(..));
252            let address_exists = ledger.entry_exists(&address);
253            for AddressInitialDeferredCredits { slot, amount } in deferred_credits {
254                if address_is_sc {
255                    warn!(
256                        "deferred credits for address {} with amount {} will not execute because it is a SC address",
257                        address, amount
258                    );
259                } else if !address_exists && amount < LEDGER_ENTRY_BASE_COST {
260                    warn!(
261                        "deferred credits for address {} with amount {} will likely not execute because the address does not exist in the ledger and the amount is below the ledger entry base cost",
262                        address, amount
263                    );
264                }
265
266                // Note that we still put the deferred credits in the database even if they are likely to fail at execution time.
267                // It's for instance possible the address is created before the execution slot of the deferred credits.
268                self.put_deferred_credits_entry(&slot, &address, &amount, batch);
269            }
270        }
271
272        Ok(())
273    }

288    /// Ensures every complete cycle has the persisted fields required for selector feeding.
289    ///
290    /// A cycle marked complete must have an RNG seed and a non-empty
291    /// `final_state_hash_snapshot`. Used after load/bootstrap and during DB validation so
292    /// malformed history fails fast instead of panicking later in `feed_selector`.
293    pub fn validate_selector_history(&self) -> PosResult<()> {
294        for (cycle, complete) in self.get_cycle_history_cycles() {
295            if !complete {
296                continue;
297            }
298            if self.get_cycle_history_rng_seed(cycle).is_none() {
299                return Err(PosError::ContainerInconsistency(format!(
300                    "complete cycle {} is missing RNG seed required for selector feeding",
301                    cycle
302                )));
303            }
304            if self
305                .get_cycle_history_final_state_hash_snapshot(cycle)
306                .is_none()
307            {
308                return Err(PosError::ContainerInconsistency(format!(
309                    "complete cycle {} is missing final state hash snapshot required for selector feeding",
310                    cycle
311                )));
312            }
313        }
314        Ok(())
315    }

347    /// Create a cycle based off of another cycle_info.
348    ///
349    /// Used for downtime interpolation, when restarting from a snapshot.
350    /// Roll counts are always carried over from `last_cycle_info`.
351    ///
352    /// `rng_seed` and `production_stats` are kept only when continuing the same
353    /// incomplete cycle (`last_cycle_info.cycle` matches the target cycle and
354    /// `first_slot` is not the first slot of that cycle). Otherwise the seed
355    /// starts fresh and `production_stats` are empty, so interpolated future
356    /// cycles do not inherit stale production results from the snapshot.
357    ///
358    /// In all cases, `rng_seed` is then extended with `false` bits for every
359    /// interpolated slot in `[first_slot, last_slot]`.
360    pub fn create_new_cycle_from_last(
361        &mut self,
362        last_cycle_info: &CycleInfo,
363        first_slot: Slot,
364        last_slot: Slot,
365        batch: &mut DBBatch,
366    ) -> Result<(), PosError> {
367        let cycle = last_slot.get_cycle(self.config.periods_per_cycle);
368
369        // Keep seed/stats only when filling the gap in the same incomplete cycle.
370        // Require both same cycle id and a non-first first_slot (defense in depth:
371        // callers may reuse a snapshot CycleInfo across several target cycles).
372        let continuing_same_cycle = last_cycle_info.cycle == cycle
373            && !first_slot.is_first_of_cycle(self.config.periods_per_cycle);
374
375        let (mut rng_seed, production_stats) = if continuing_same_cycle {
376            (
377                last_cycle_info.rng_seed.clone(),
378                last_cycle_info.production_stats.clone(),
379            )
380        } else {
381            (
382                BitVec::with_capacity(self.slots_per_cycle()),
383                PreHashMap::default(),
384            )
385        };
386
387        let num_slots = match last_slot.slots_since(&first_slot, self.config.thread_count) {
388            Ok(slots_since) => slots_since.saturating_add(1),
389            // if first_slot is greater than last_slot, we have nothing to do
390            Err(_) => return Ok(()),
391        };
392
393        let slots_per_cycle = self.slots_per_cycle();
394        // Existing seed (e.g. genesis thread_count padding) may already cover some of
395        // the interpolated range. Cap so we never exceed a full cycle.
396        if rng_seed.len() > slots_per_cycle {
397            return Err(PosError::ContainerInconsistency(format!(
398                "RNG seed for cycle {} exceeds slots_per_cycle: {} > {}",
399                cycle,
400                rng_seed.len(),
401                slots_per_cycle
402            )));
403        }
404        // Pad downtime slots; not the same as leaving the seed empty.
405        let to_add = (num_slots as usize).min(slots_per_cycle.saturating_sub(rng_seed.len()));
406        rng_seed.extend(vec![false; to_add]);
407
408        let complete =
409            last_slot.is_last_of_cycle(self.config.periods_per_cycle, self.config.thread_count);
410        if rng_seed.len() > slots_per_cycle || (complete && rng_seed.len() != slots_per_cycle) {
411            return Err(PosError::ContainerInconsistency(format!(
412                "invalid RNG seed length {} for cycle {} (slots_per_cycle {}, complete={})",
413                rng_seed.len(),
414                cycle,
415                slots_per_cycle,
416                complete
417            )));
418        }
419
420        self.put_new_cycle_info(
421            &CycleInfo::new(
422                cycle,
423                complete,
424                last_cycle_info.roll_counts.clone(),
425                rng_seed,
426                production_stats,
427            ),
428            batch,
429        );
430
431        Ok(())
432    }

448    /// Sends the current draw inputs (initial or bootstrapped) to the selector.
449    /// Waits for the initial draws to be performed.
450    pub fn compute_initial_draws(&mut self) -> PosResult<()> {
451        // Reject incomplete selector inputs before recomputing draws from history.
452        self.validate_selector_history()?;
453
454        // if cycle_history starts at a cycle that is strictly higher than 0, do not feed cycles 0, 1 to selector
455        let history_starts_late = self
456            .cycle_history_cache
457            .front()
458            .map(|c_info| c_info.0 > 0)
459            .unwrap_or(false);
460
461        let mut max_cycle = None;
462
463        // feed cycles 0, 1 to selector if necessary
464        if !history_starts_late {
465            for draw_cycle in 0u64..=1 {
466                self.feed_selector(draw_cycle)?;
467                max_cycle = Some(draw_cycle);
468            }
469        }
470
471        // feed cycles available from history
472        for (idx, hist_item) in self.cycle_history_cache.iter().enumerate() {
473            if !hist_item.1 {
474                break;
475            }
476            if history_starts_late && idx == 0 {
477                // If the history starts late, the first RNG seed cannot be used to draw
478                // because the roll distribution which should be provided by the previous element is absent.
479                continue;
480            }
481            let draw_cycle = hist_item.0.checked_add(2).ok_or_else(|| {
482                PosError::OverflowError("cycle overflow in give_selector_controller".into())
483            })?;
484            self.feed_selector(draw_cycle)?;
485            max_cycle = Some(draw_cycle);
486        }
487
488        // wait for all fed cycles to be drawn
489        if let Some(wait_cycle) = max_cycle {
490            self.selector.as_mut().wait_for_draws(wait_cycle)?;
491        }
492        Ok(())
493    }

495    /// Technical specification of `apply_changes_to_batch`:
496    ///
497    /// set `self.last_final_slot` = C
498    /// if cycle C is absent from `self.cycle_history_cache`:
499    ///     `push` a new empty `CycleInfo` on disk and reflect in `self.cycle_history_cache` and set its cycle = C
500    ///     `pop_front` from `cycle_history_cache` until front() represents cycle C-4 or later (not C-3 because we might need older endorsement draws on the limit between 2 cycles)
501    ///     delete the removed cycles from disk
502    /// for the cycle C entry in the db:
503    ///     extend `seed_bits` with `changes.seed_bits`
504    ///     extend `roll_counts` with `changes.roll_changes`
505    ///         delete all entries from `roll_counts` for which the roll count is zero
506    ///     add each element of `changes.production_stats` to the cycle's `production_stats`
507    /// for each `changes.deferred_credits` targeting cycle Ct:
508    ///     overwrite `self.deferred_credits` entries of cycle Ct in `cycle_history` with the ones from change
509    ///         remove entries for which Amount = 0
510    /// if slot S was the last of cycle C:
511    ///     set complete=true for cycle C in the history
512    ///     compute the seed hash and notifies the `PoSDrawer` for cycle `C+3`
513    ///
514    pub fn apply_changes_to_batch(
515        &mut self,
516        changes: PoSChanges,
517        slot: Slot,
518        feed_selector: bool,
519        batch: &mut DBBatch,
520    ) -> PosResult<()> {
521        let slots_per_cycle = self.slots_per_cycle();
522
523        // compute the current cycle from the given slot
524        let cycle = slot.get_cycle(self.config.periods_per_cycle);
525
526        // if cycle C is absent from self.cycle_history:
527        // push a new empty CycleInfo at the back of self.cycle_history and set its cycle = C
528        // pop_front from cycle_history until front() represents cycle C-4 or later
529        // (not C-3 because we might need older endorsement draws on the limit between 2 cycles)
530        if let Some(info) = self.cycle_history_cache.back() {
531            if cycle == info.0 && !info.1 {
532                // extend the last incomplete cycle
533            } else if info.0.checked_add(1) == Some(cycle) && info.1 {
534                // the previous cycle is complete, push a new incomplete/empty one to extend
535                // By now `_finalize` should already have written its final_state_hash_snapshot.
536                // Catch a missing snapshot here (first slot of the new cycle) instead of waiting
537                // until feed_selector needs it as a lookback (~1 cycle later).
538                if self
539                    .get_cycle_history_final_state_hash_snapshot(info.0)
540                    .is_none()
541                {
542                    return Err(PosError::ContainerInconsistency(format!(
543                        "complete cycle {} is missing final state hash snapshot before starting cycle {}",
544                        info.0, cycle
545                    )));
546                }
547
548                let roll_counts = self.get_all_roll_counts(info.0);
549                self.put_new_cycle_info(
550                    &CycleInfo::new(
551                        cycle,
552                        false,
553                        roll_counts,
554                        BitVec::with_capacity(slots_per_cycle),
555                        PreHashMap::default(),
556                    ),
557                    batch,
558                );
559                while self.cycle_history_cache.len() > self.config.cycle_history_length {
560                    if let Some((old_cycle, _)) = self.cycle_history_cache.pop_front() {
561                        self.delete_cycle_info(old_cycle, batch);
562                    }
563                }
564            } else {
565                return Err(PosError::OverflowError(
566                    "invalid cycle sequence in PoS final state".into(),
567                ));
568            }
569        } else {
570            return Err(PosError::ContainerInconsistency(
571                "PoS history should never be empty here".into(),
572            ));
573        }
574
575        let complete: bool =
576            slot.is_last_of_cycle(self.config.periods_per_cycle, self.config.thread_count);
577
578        // OPTIM: we could avoid reading the previous seed bits with a cache or with an update function
579        // Validate selector inputs before marking the cycle complete so a malformed seed cannot
580        // leave `complete=true` in the in-memory cache when we return an error.
581        let mut rng_seed = self.get_cycle_history_rng_seed(cycle).ok_or_else(|| {
582            PosError::ContainerInconsistency(format!("missing RNG seed for cycle {}", cycle))
583        })?;
584        // Reject already-corrupt or overflowing seed state before mutating caches/batch.
585        if rng_seed.len() > slots_per_cycle {
586            return Err(PosError::ContainerInconsistency(format!(
587                "RNG seed for cycle {} exceeds slots_per_cycle: {} > {}",
588                cycle,
589                rng_seed.len(),
590                slots_per_cycle
591            )));
592        }
593        rng_seed.extend(changes.seed_bits);
594        if rng_seed.len() > slots_per_cycle || (complete && rng_seed.len() != slots_per_cycle) {
595            return Err(PosError::ContainerInconsistency(format!(
596                "cycle {} RNG seed length {} invalid (slots_per_cycle {}, complete={})",
597                cycle,
598                rng_seed.len(),
599                slots_per_cycle,
600                complete
601            )));
602        }
603
604        self.put_cycle_history_complete(cycle, complete, batch);
605        self.put_cycle_history_rng_seed(cycle, rng_seed, batch);
606
607        // extend roll counts
608        for (addr, roll_count) in changes.roll_changes {
609            self.put_cycle_history_address_entry(cycle, &addr, Some(&roll_count), None, batch);
610        }
611
612        // extend production stats
613        for (addr, stats) in changes.production_stats {
614            if let Some(prev_production_stats) = self.get_production_stats_for_address(cycle, &addr)
615            {
616                let mut new_production_stats = prev_production_stats;
617                new_production_stats.extend(&stats);
618                self.put_cycle_history_address_entry(
619                    cycle,
620                    &addr,
621                    None,
622                    Some(&new_production_stats),
623                    batch,
624                );
625            } else {
626                self.put_cycle_history_address_entry(cycle, &addr, None, Some(&stats), batch);
627            }
628        }
629
630        // extend deferred_credits with changes.deferred_credits and remove zeros
631        for (slot, credits) in changes.deferred_credits.credits.iter() {
632            for (address, amount) in credits.iter() {
633                self.put_deferred_credits_entry(slot, address, amount, batch);
634            }
635        }
636
637        // feed the cycle if it is complete
638        // notify the PoSDrawer about the newly ready draw data
639        // to draw cycle + 2, we use the rng data from cycle - 1 and the seed from cycle
640        debug!(
641            "After slot {} PoS cycle list is {:?}",
642            slot, self.cycle_history_cache
643        );
644        if complete && feed_selector {
645            self.feed_selector(cycle.checked_add(2).ok_or_else(|| {
646                PosError::OverflowError("cycle overflow when feeding selector".into())
647            })?)
648        } else {
649            Ok(())
650        }
651    }

653    /// Feeds the selector targeting a given draw cycle
654    pub fn feed_selector(&self, draw_cycle: u64) -> PosResult<()> {
655        // get roll lookback
656
657        let (lookback_rolls, lookback_state_hash) = match draw_cycle.checked_sub(3) {
658            // looking back in history
659            Some(c) => {
660                let index = self
661                    .get_cycle_index(c)
662                    .ok_or(PosError::CycleUnavailable(c))?;
663                let cycle_info = &self.cycle_history_cache[index];
664                if !cycle_info.1 {
665                    return Err(PosError::CycleUnfinished(c));
666                }
667                // take the final_state_hash_snapshot at cycle - 3
668                // it will later be combined with rng_seed from cycle - 2 to determine the selection seed
669                // do this here to avoid a potential attacker manipulating the selections
670                let state_hash = self
671                    .get_cycle_history_final_state_hash_snapshot(cycle_info.0)
672                    .ok_or_else(|| {
673                        PosError::ContainerInconsistency(format!(
674                            "complete cycle {} is missing final state hash snapshot required for selector feeding",
675                            c
676                        ))
677                    })?;
678                (self.get_all_roll_counts(cycle_info.0), Some(state_hash))
679            }
680            // looking back to negative cycles
681            None => (self.initial_rolls.clone(), None),
682        };
683
684        // get seed lookback
685        let lookback_seed = match draw_cycle.checked_sub(2) {
686            // looking back in history
687            Some(c) => {
688                let index = self
689                    .get_cycle_index(c)
690                    .ok_or(PosError::CycleUnavailable(c))?;
691                let cycle_info = &self.cycle_history_cache[index];
692                if !cycle_info.1 {
693                    return Err(PosError::CycleUnfinished(c));
694                }
695                let u64_ser = U64VarIntSerializer::new();
696                let mut seed = Vec::new();
697                u64_ser.serialize(&c, &mut seed).unwrap();
698                let rng_seed = self
699                    .get_cycle_history_rng_seed(cycle_info.0)
700                    .ok_or_else(|| {
701                        PosError::ContainerInconsistency(format!(
702                            "complete cycle {} is missing RNG seed required for selector feeding",
703                            c
704                        ))
705                    })?;
706                seed.extend(rng_seed.into_vec());
707                if let Some(lookback_state_hash) = lookback_state_hash {
708                    seed.extend(lookback_state_hash.to_bytes());
709                }
710                Hash::compute_from(&seed)
711            }
712            // looking back to negative cycles
713            None => self.initial_seeds[draw_cycle as usize],
714        };
715
716        // feed selector
717        self.selector
718            .as_ref()
719            .feed_cycle(draw_cycle, lookback_rolls, lookback_seed)
720    }

725        cycle: u64,

812    /// Retrieves every deferred credit in a slot range
813    /// Warning: this can be quite slow if there is tons of Deferred credits (e.g. just after genesis) to fetch
814    ///          so there is an option to filter by an address to improve the speed (cf json rpc api get_addresses)
815    pub fn get_deferred_credits_range<R>(
816        &self,
817        range: R,
818        addr_filter: Option<&Address>,
819    ) -> DeferredCredits
820    where
821        R: RangeBounds<Slot>,
822    {
823        let db = self.db.read();
824
825        let mut deferred_credits = DeferredCredits::new();
826
827        let mut start_key_buffer = Vec::new();
828        start_key_buffer.extend_from_slice(DEFERRED_CREDITS_PREFIX.as_bytes());
829
830        match range.start_bound() {
831            Included(slot) => {
832                start_key_buffer.extend_from_slice(&slot.to_bytes_key());
833            }
834            Excluded(slot) => {
835                start_key_buffer.extend_from_slice(
836                    &slot
837                        .get_next_slot(self.config.thread_count)
838                        .expect(DEFERRED_CREDITS_SER_ERROR)
839                        .to_bytes_key(),
840                );
841            }
842            Unbounded => {}
843        };
844
845        let iterator_cf = db
846            .iterator_cf(
847                STATE_CF,
848                MassaIteratorMode::From(&start_key_buffer, MassaDirection::Forward),
849            )
850            .take_while(|(k, _)| k.starts_with(DEFERRED_CREDITS_PREFIX.as_bytes()));
851
852        for (serialized_key, serialized_value) in iterator_cf {
853            // deserialize the slot
854            let rest_key = &serialized_key[DEFERRED_CREDITS_PREFIX.len()..];
855            let (rest_key, slot) = buf_to_array_ctr(rest_key, Slot::from_bytes_key)
856                .expect(DEFERRED_CREDITS_DESER_ERROR);
857
858            if !range.contains(&slot) {
859                break;
860            }
861
862            let (_, address) = self
863                .deferred_credits_deserializer
864                .credit_deserializer
865                .address_deserializer
866                .deserialize::<DeserializeError>(rest_key)
867                .expect(DEFERRED_CREDITS_DESER_ERROR);
868
869            if let Some(addr_filter_value) = &addr_filter {
870                if &&address != addr_filter_value {
871                    continue;
872                }
873            }
874
875            let (_, amount) = self
876                .deferred_credits_deserializer
877                .credit_deserializer
878                .amount_deserializer
879                .deserialize::<DeserializeError>(&serialized_value)
880                .expect(DEFERRED_CREDITS_DESER_ERROR);
881
882            deferred_credits.insert(slot, address, amount);
883        }
884
885        deferred_credits
886    }

888    /// Gets the index of a cycle in history
889    pub fn get_cycle_index(&self, cycle: u64) -> Option<usize> {
890        let first_cycle = match self.cycle_history_cache.front() {
891            Some(c) => c.0,
892            None => return None, // history empty
893        };
894        if cycle < first_cycle {
895            return None; // in the past
896        }
897        let index: usize = match (cycle - first_cycle).try_into() {
898            Ok(v) => v,
899            Err(_) => return None, // usize overflow
900        };
901        if index >= self.cycle_history_cache.len() {
902            return None; // in the future
903        }
904        Some(index)
905    }

1410    /// Helper function to construct the key prefix associated with a given cycle
1411    fn cycle_history_cycle_prefix(&self, cycle: u64) -> Vec<u8> {
1412        let mut serialized_key = Vec::new();
1413        serialized_key.extend_from_slice(CYCLE_HISTORY_PREFIX.as_bytes());
1414        serialized_key.extend_from_slice(&cycle.to_be_bytes());

1418    /// Deserializes the key and value, useful after bootstrap
1419    pub fn is_cycle_history_key_value_valid(
1420        &self,
1421        serialized_key: &[u8],
1422        serialized_value: &[u8],
1423    ) -> bool {
1424        if !serialized_key.starts_with(CYCLE_HISTORY_PREFIX.as_bytes()) {
1425            return false;
1426        }
1427
1428        // deserialize the cycle
1429        let rest_key = &serialized_key[CYCLE_HISTORY_PREFIX.len()..];
1430        let Some((rest_key, cycle)) = buf_to_array_ctr(rest_key, |v| u64::from_be_bytes(*v)) else {
1431            return false;
1432        };
1433
1434        if rest_key.is_empty() {
1435            return false;
1436        }
1437
1438        let slots_per_cycle = self.slots_per_cycle();
1439
1440        match rest_key[0] {
1441            COMPLETE_IDENT => {
1442                if rest_key.len() != 1 {
1443                    return false;
1444                }
1445                if serialized_value.len() != 1 {
1446                    return false;
1447                }
1448                if serialized_value[0] > 1 {
1449                    return false;
1450                }
1451                // A completed cycle must have an RNG seed of exactly slots_per_cycle bits.
1452                if serialized_value[0] == 1 {
1453                    let prefix = self.cycle_history_cycle_prefix(cycle);
1454                    let Ok(Some(serialized_rng_seed)) =
1455                        self.db.read().get_cf(STATE_CF, rng_seed_key!(prefix))
1456                    else {
1457                        return false;
1458                    };
1459                    let Ok((rest, rng_seed)) = self
1460                        .cycle_info_deserializer
1461                        .cycle_info_deserializer
1462                        .bitvec_deser
1463                        .deserialize::<DeserializeError>(&serialized_rng_seed)
1464                    else {
1465                        return false;
1466                    };
1467                    if !rest.is_empty() || rng_seed.len() != slots_per_cycle {
1468                        return false;
1469                    }
1470                }
1471            }
1472            RNG_SEED_IDENT => {
1473                if rest_key.len() != 1 {
1474                    return false;
1475                }
1476                let Ok((rest_key, rng_seed)) = self
1477                    .cycle_info_deserializer
1478                    .cycle_info_deserializer
1479                    .bitvec_deser
1480                    .deserialize::<DeserializeError>(serialized_value)
1481                else {
1482                    return false;
1483                };
1484                if !rest_key.is_empty() {
1485                    return false;
1486                }
1487                // Seed length must never exceed a full cycle; exact length for complete
1488                // cycles is enforced when validating the COMPLETE flag above.
1489                if rng_seed.len() > slots_per_cycle {
1490                    return false;
1491                }
1492            }
1493            FINAL_STATE_HASH_SNAPSHOT_IDENT => {
1494                if rest_key.len() != 1 {
1495                    return false;
1496                }
1497                let Ok((rest_key, _final_state_hash)) = self
1498                    .cycle_info_deserializer
1499                    .cycle_info_deserializer
1500                    .opt_hash_deser
1501                    .deserialize::<DeserializeError>(serialized_value)
1502                else {
1503                    return false;
1504                };
1505                if !rest_key.is_empty() {
1506                    return false;
1507                }
1508            }
1509            ROLL_COUNT_IDENT => {
1510                let Ok((rest_key, _addr)): std::result::Result<
1511                    (&[u8], Address),
1512                    nom::Err<massa_serialization::DeserializeError<'_>>,
1513                > = self
1514                    .cycle_info_deserializer
1515                    .cycle_info_deserializer
1516                    .rolls_deser
1517                    .address_deserializer
1518                    .deserialize::<DeserializeError>(&rest_key[1..])
1519                else {
1520                    return false;
1521                };
1522                if !rest_key.is_empty() {
1523                    return false;
1524                }
1525                let Ok((rest_key, _addr)) = self
1526                    .cycle_info_deserializer
1527                    .cycle_info_deserializer
1528                    .rolls_deser
1529                    .u64_deserializer
1530                    .deserialize::<DeserializeError>(serialized_value)
1531                else {
1532                    return false;
1533                };
1534                if !rest_key.is_empty() {
1535                    return false;
1536                }
1537            }
1538            PROD_STATS_IDENT => {
1539                let Ok((rest_key, _addr)): std::result::Result<
1540                    (&[u8], Address),
1541                    nom::Err<massa_serialization::DeserializeError<'_>>,
1542                > = self
1543                    .cycle_info_deserializer
1544                    .cycle_info_deserializer
1545                    .rolls_deser
1546                    .address_deserializer
1547                    .deserialize::<DeserializeError>(&rest_key[1..])
1548                else {
1549                    return false;
1550                };
1551                if rest_key.len() != 1 {
1552                    return false;
1553                }
1554
1555                match rest_key[0] {
1556                    PROD_STATS_FAIL_IDENT => {
1557                        let Ok((rest_key, _fail)) = self
1558                            .cycle_info_deserializer
1559                            .cycle_info_deserializer
1560                            .production_stats_deser
1561                            .u64_deserializer
1562                            .deserialize::<DeserializeError>(serialized_value)
1563                        else {
1564                            return false;
1565                        };
1566                        if !rest_key.is_empty() {
1567                            return false;
1568                        }
1569                    }
1570                    PROD_STATS_SUCCESS_IDENT => {
1571                        let Ok((rest_key, _success)) = self
1572                            .cycle_info_deserializer
1573                            .cycle_info_deserializer
1574                            .production_stats_deser
1575                            .u64_deserializer
1576                            .deserialize::<DeserializeError>(serialized_value)
1577                        else {
1578                            return false;
1579                        };
1580                        if !rest_key.is_empty() {
1581                            return false;
1582                        }
1583                    }
1584                    _ => {
1585                        return false;
1586                    }
1587                }
1588            }
1589            _ => {
1590                return false;
1591            }
1592        }
1593
1594        true
1595    }
