23/// Structure gathering all elements needed by the selector thread
24#[allow(dead_code)]
25pub(crate) struct SelectorThread {
26    // A copy of the input data allowing access to incoming requests
27    pub(crate) input_mpsc: Receiver<Command>,
28    /// Cache of computed endorsements
29    pub(crate) cache: DrawCachePtr,
30    /// Configuration
31    pub(crate) cfg: SelectorConfig,
32}
33
34impl SelectorThread {
35    /// Creates the `SelectorThread` structure to gather all data and references
36    /// needed by the selector worker thread.
37    pub(crate) fn spawn(
38        input_mpsc: Receiver<Command>,
39        cache: DrawCachePtr,
40        cfg: SelectorConfig,
41    ) -> JoinHandle<PosResult<()>> {
42        let thread_builder = thread::Builder::new().name("selector".into());
43        thread_builder
44            .spawn(|| {
45                let this = Self {
46                    input_mpsc,
47                    cache,
48                    cfg,
49                };
50                this.run()
51            })
52            .expect("failed to spawn thread : selector")
53    }
54
55    /// process the result of a draw
56    fn process_draws_result(
57        &self,
58        cycle: u64,
59        draws_result: PosResult<CycleDraws>,
60    ) -> PosResult<()> {
61        // write-lock the cache
62        let (cache_cv, cache_lock) = &*self.cache;
63        let mut cache_guard = cache_lock.write();
64
65        let out_result = {
66            // check cache validity and continuity
67            let cache = cache_guard.as_mut().map_err(|err| err.clone())?;
68            if let Some(last_cycle) = cache.0.back() {
69                if last_cycle.cycle.checked_add(1) != Some(cycle) {
70                    return Err(PosError::ContainerInconsistency(
71                        "discontinuity in cycle draws history".into(),
72                    ));
73                }
74            }
75
76            // add draw results to cache, or extract error
77            match draws_result {
78                Ok(cycle_draws) => {
79                    // add to draws
80                    cache.0.push_back(cycle_draws);
81
82                    // truncate cache to keep only the desired number of elements
83                    while cache.0.len() > self.cfg.max_draw_cache {
84                        cache.0.pop_front();
85                    }
86
87                    // no error
88                    Ok(())
89                }
90                // draw error
91                Err(err) => Err(err),
92            }
93        };
94
95        // if there was an error, save a clone of the error to the cache
96        if let Err(err) = &out_result {
97            *cache_guard = Err(err.clone());
98        }
99
100        // notify all waiters
101        cache_cv.notify_all();
102
103        out_result
104    }
105
106    /// Thread loop.
107    ///
108    /// While a `Stop` command isn't sent, pop `input_data` and compute
109    /// draws for future cycle.
110    fn run(self) -> PosResult<()> {
111        loop {
112            let Ok(Command::DrawInput {
113                cycle,
114                lookback_rolls,
115                lookback_seed,
116            }) = self.input_mpsc.recv()
117            else {
118                break;
119            };
120
121            // perform draws
122            let draws_result = perform_draws(&self.cfg, cycle, lookback_rolls, lookback_seed);
123
124            // add result to cache and notify waiters
125            self.process_draws_result(cycle, draws_result)?;
126        }
127        Ok(())
128    }
129}

131/// Launches a selector worker thread and returns a pair to interact with it.
132///
133/// # parameters
134/// * none
135///
136/// # Returns
137/// A pair `(selector_manager, selector_controller)` where:
138/// * `selector_manager`: allows to stop the worker
139/// * `selector_controller`: allows sending requests and notifications to the worker
140pub fn start_selector_worker(
141    selector_config: SelectorConfig,
142) -> PosResult<(Box<dyn SelectorManager>, Box<dyn SelectorController>)> {
143    let (input_sender, input_receiver) = sync_channel(selector_config.channel_size);
144    let cache = Arc::new((
145        RwLockCondvar::default(),
146        RwLock::new(Ok(DrawCache(VecDeque::with_capacity(
147            selector_config.max_draw_cache.saturating_add(1),
148        )))),
149    ));
150    let controller = SelectorControllerImpl {
151        input_mpsc: input_sender.clone(),
152        cache: cache.clone(),
153        periods_per_cycle: selector_config.periods_per_cycle,
154        thread_count: selector_config.thread_count,
155    };
156
157    // launch the selector thread
158    let thread_handle = SelectorThread::spawn(input_receiver, cache, selector_config);
159
160    let manager = SelectorManagerImpl {
161        thread_handle: Some(thread_handle),
162        input_mpsc: input_sender,
163    };
164    Ok((Box::new(manager), Box::new(controller)))
165}
