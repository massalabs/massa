7/// check the version of the packages in the workspace
8fn check_workspace_packages_version(
9    new_version: String,
10    workspace_path: &Path,
11) -> Result<i32, Box<dyn std::error::Error>> {
12    let mut nb_files_updated = 0;
13    // search for Cargo.toml files in the workspace
14    for entry in WalkDir::new(workspace_path)
15        .into_iter()
16        .filter_map(|e| e.ok())
17    {
18        if entry.file_name() == "Cargo.toml" {
19            // if the Cargo.toml file is found, check the version and update it if necessary
20            if check_package_version(new_version.clone(), entry.path())? {
21                // if version has been updated
22                nb_files_updated += 1;
23            }
24        }
25    }
26
27    Ok(nb_files_updated)
28}

30/// check the version of the package in the Cargo.toml file
31///
32/// Update it if version is different from the new version and if the package is a massa package
33///
34/// return Ok(true) if the version has been updated
35fn check_package_version(
36    new_version: String,
37    cargo_toml_path: &Path,
38) -> Result<bool, Box<dyn std::error::Error>> {
39    let cargo_toml_content = fs::read_to_string(cargo_toml_path)?;
40    let mut doc = cargo_toml_content.parse::<Document>()?;
41
42    let mut update = false;
43
44    if let Some(package) = doc["package"].as_table_mut() {
45        if package["name"].to_string().contains("massa") {
46            if let Some(version) = package.get_mut("version") {
47                let to_string = version.to_string().replace('\"', "");
48                let actual_version = to_string.trim();
49                if new_version.ne(actual_version) {
50                    *version = Item::Value(Value::String(Formatted::new(new_version.clone())));
51                    println!(
52                        "Updating version of package {} from {} to {}",
53                        package["name"], actual_version, new_version
54                    );
55                    update = true;
56                }
57            }
58        }
59    }
60
61    if update {
62        let updated_cargo_toml_content = doc.to_string();
63        fs::write(cargo_toml_path, updated_cargo_toml_content)?;
64    }
65
66    Ok(update)
67}

69pub(crate) fn update_package_versions() {
70    println!("Updating package versions");
71    let mut to_string = VERSION.to_string();
72
73    if to_string.contains("SECU") || to_string.contains("SAND") {
74        // TestNet and Sandbox versions < 1.0.0
75        to_string.replace_range(..4, "0");
76    } else {
77        // Main net version >= 1.0.0
78        to_string.replace_range(..5, "");
79        to_string.push_str(".0");
80    };
81
82    let workspace_path = Path::new("./");
83
84    match check_workspace_packages_version(to_string, workspace_path) {
85        Err(e) => panic!("Error updating workspace packages version: {}", e),
86        Ok(nb_files_updated) => println!("{} files updated", nb_files_updated),
87    }
