1use massa_sc_runtime::GasCosts;
2use std::collections::HashMap;
3use std::path::PathBuf;
4
5const ABI_GAS_COSTS_FILE: &str = "massa-node/base_config/gas_costs/abi_gas_costs.json";
6
7pub(crate) fn check_gas_cost_definitions() -> Result<(), String> {
8    // Check gas cost definitions between:
9    // massa-node/base_config/gas_costs/abi_gas_costs.json
10    // massa-sc-runtime GasCosts
11
12    // `GasCosts::new` fails if any cost required by the runtime is missing
13    // from the file, so this catches definitions the node needs but the
14    // shipped file lacks (the node would not start).
15    GasCosts::new(PathBuf::from(ABI_GAS_COSTS_FILE))
16        .map_err(|err| format!("failed to load {}: {}", ABI_GAS_COSTS_FILE, err))?;
17
18    // Detect stale definitions: a key that can be removed without making
19    // `GasCosts::new` fail is no longer used by the runtime. The runtime
20    // does not expose its expected key set, so probe by removing keys one
21    // at a time instead of maintaining a duplicate list here. Stale keys
22    // are harmless at runtime, so only warn about them.
23    let file_content = std::fs::read_to_string(ABI_GAS_COSTS_FILE)
24        .map_err(|err| format!("failed to read {}: {}", ABI_GAS_COSTS_FILE, err))?;
25    let costs: HashMap<String, u64> = serde_json::from_str(&file_content)
26        .map_err(|err| format!("failed to parse {}: {}", ABI_GAS_COSTS_FILE, err))?;
27
28    let probe_file = std::env::temp_dir().join("massa_xtask_abi_gas_costs_probe.json");
29    for key in costs.keys() {
30        let mut probe_costs = costs.clone();
31        probe_costs.remove(key);
32        let probe_content = serde_json::to_string(&probe_costs).map_err(|err| err.to_string())?;
33        std::fs::write(&probe_file, probe_content).map_err(|err| err.to_string())?;
34        if GasCosts::new(probe_file.clone()).is_ok() {
35            println!("Found in json but not used by the runtime: {key}");
36        }
37    }
38    let _ = std::fs::remove_file(&probe_file);
39
40    println!("Gas cost definitions OK");
41    Ok(())
42}
