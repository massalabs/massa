118        /// Initial state

120        /// Past start, can only go to LockedIn after the threshold is above a given value

213impl Defined {
214    /// Update state from state Defined
215    pub fn on_advance(self, input: Advance) -> ComponentState {
216        match input.now {
217            n if n >= input.timeout => ComponentState::failed(),

224impl Started {
225    /// Update state from state Started
226    pub fn on_advance(self, input: Advance) -> ComponentState {
227        if input.now >= input.timeout {
228            return ComponentState::failed();
229        }
230
231        if input.threshold >= VERSIONING_THRESHOLD_TRANSITION_ACCEPTED {
232            debug!("(VERSIONING LOG) transition accepted, locking in");
233            ComponentState::locked_in(input.now)
234        } else {
235            ComponentState::started(input.threshold)

242    pub fn on_advance(self, input: Advance) -> ComponentState {

246        } else {

248        }
249    }

344    /// Create a new state from an existing state - resulting state will be at state "Defined"
345    pub fn reset_from(&self) -> Option<Self> {
346        match self.history.first_key_value() {
347            Some((advance, state_id)) if *state_id == ComponentStateTypeId::Defined => {
348                Some(MipState::new(advance.now))

354    /// Advance the state
355    /// Can be called as multiple times as it will only store what changes the state in history
356    pub fn on_advance(&mut self, input: &Advance) {
357        // Check that input.now is after last item in history
358        // We don't want to go backward
359        let is_forward = self
360            .history
361            .last_key_value()
362            .map(|(adv, _)| adv.now < input.now)
363            .unwrap_or(false);
364
365        if is_forward {
366            // machines crate (for state machine) does not support passing ref :-/
367            let state = self.state.on_advance(input.clone());
368            // Update history as well
369            if state != self.state {
370                // Avoid storing too much things in history
371                // Here we avoid storing for every threshold update
372                if !(matches!(state, ComponentState::Started(Started { .. }))
373                    && matches!(self.state, ComponentState::Started(Started { .. })))
374                {
375                    self.history
376                        .insert(input.into(), ComponentStateTypeId::from(&state));
377                }
378                self.state = state;
379            }
380        }
381    }

383    /// Given a corresponding MipInfo, check if state is consistent
384    /// it is consistent
385    ///   if state can be at this position (e.g. can it be at state "Started" according to given time range)
386    ///   if history is consistent with current state
387    /// Return false for state == ComponentState::Error
388    pub fn is_consistent_with(&self, mip_info: &MipInfo) -> Result<(), IsConsistentError> {
389        // Always return false for state Error or if history is empty
390        if matches!(&self.state, &ComponentState::Error) {
391            return Err(IsConsistentError::AtError);
392        }
393
394        if mip_info.start >= mip_info.timeout {
395            return Err(IsConsistentError::Invalid);
396        }
397
398        if self.history.is_empty() {
399            return Err(IsConsistentError::EmptyHistory);
400        }
401
402        // safe to unwrap (already tested if empty or not)
403        let (initial_ts, initial_state_id) = self.history.first_key_value().unwrap();
404        if *initial_state_id != ComponentStateTypeId::Defined {
405            // self.history does not start with Defined -> (always) false
406            return Err(IsConsistentError::InvalidHistory(initial_state_id.clone()));
407        }
408
409        if mip_info.start < initial_ts.now || mip_info.timeout < initial_ts.now {
410            // MIP info start (or timeout) is before Defined timestamp??
411            return Err(IsConsistentError::InvalidHistory(initial_state_id.clone()));
412        }
413
414        // Build a new MipStateHistory from initial state, replaying the whole history
415        // but with given versioning info then compare
416        let mut vsh = MipState::new(initial_ts.now);
417        let mut advance_msg = Advance {
418            start_timestamp: mip_info.start,
419            timeout: mip_info.timeout,
420            threshold: Ratio::zero(),
421            now: initial_ts.now,
422            activation_delay: mip_info.activation_delay,
423        };
424
425        for (adv, _state) in self.history.iter().skip(1) {
426            advance_msg.now = adv.now;
427            advance_msg.threshold = adv.threshold;
428            vsh.on_advance(&advance_msg);
429        }
430
431        // Advance state if both are at 'Started' (to have the same threshold)
432        // Note: because in history we do not add entries for every threshold update
433        if let (
434            ComponentState::Started(Started {
435                vote_ratio: threshold,
436            }),
437            ComponentState::Started(Started {
438                vote_ratio: threshold_2,
439            }),
440        ) = (vsh.state, self.state)
441        {
442            if threshold_2 != threshold {
443                advance_msg.threshold = threshold_2;
444                // Need to advance now timestamp otherwise it will be ignored
445                advance_msg.now = advance_msg.now.saturating_add(MassaTime::from_millis(1));
446                vsh.on_advance(&advance_msg);
447            }
448        }
449
450        if vsh == *self {
451            Ok(())
452        } else {
453            Err(IsConsistentError::NonConsistent(self.state, vsh.state))
454        }
455    }

457    /// Query state at given timestamp
458    /// TODO: add doc for start & timeout parameter? why do we need them?
459    pub fn state_at(
460        &self,
461        ts: MassaTime,
462        start: MassaTime,
463        timeout: MassaTime,
464        activation_delay: MassaTime,
465    ) -> Result<ComponentStateTypeId, StateAtError> {
466        if self.history.is_empty() {
467            return Err(StateAtError::EmptyHistory);
468        }
469
470        // Optim: this avoids iterating over history (cheap to retrieve first item)
471        let first = self.history.first_key_value().unwrap(); // safe to unwrap
472        if ts < first.0.now {
473            // Before initial state
474            return Err(StateAtError::BeforeInitialState(first.1.clone(), ts));
475        }
476
477        // At this point, we are >= the first state in history
478        let mut lower_bound = None;
479        let mut higher_bound = None;
480        let mut is_after_last = false;
481
482        // Optim: this avoids iterating over history (cheap to retrieve first item)
483        let last = self.history.last_key_value().unwrap(); // safe to unwrap
484        if ts > last.0.now {
485            lower_bound = Some(last);
486            is_after_last = true;
487        }
488
489        if !is_after_last {
490            // We are in between two states in history, find bounds
491            for (adv, state_id) in self.history.iter() {
492                if adv.now <= ts {
493                    lower_bound = Some((adv, state_id));
494                }
495                if adv.now >= ts && higher_bound.is_none() {
496                    higher_bound = Some((adv, state_id));
497                    break;
498                }
499            }
500        }
501
502        match (lower_bound, higher_bound) {
503            (Some((_adv_1, st_id_1)), Some((_adv_2, _st_id_2))) => {
504                // Between 2 states (e.g. between Defined && Started) -> return Defined
505                Ok(st_id_1.clone())
506            }
507            (Some((adv, st_id)), None) => {
508                // After the last state in history -> need to advance the state and return
509                let threshold_for_transition = VERSIONING_THRESHOLD_TRANSITION_ACCEPTED;
510                // Note: Please update this if MipState transitions change as it might not hold true
511                if *st_id == ComponentStateTypeId::Started
512                    && adv.threshold < threshold_for_transition
513                    && ts < timeout
514                // adv.timeout - TODO: test this
515                {
516                    Err(StateAtError::Unpredictable)
517                } else {
518                    let msg = Advance {
519                        start_timestamp: start,
520                        timeout,
521                        threshold: adv.threshold,
522                        now: ts,
523                        // activation_delay: adv.activation_delay,
524                        activation_delay,
525                    };
526                    // Return the resulting state after advance
527                    let state = self.state.on_advance(msg);
528                    Ok(ComponentStateTypeId::from(&state))
529                }
530            }
531            _ => {
532                // 1. Before the first state in history: already covered
533                // 2. None, None: already covered - empty history
534                Err(StateAtError::EmptyHistory)
535            }
536        }
537    }

872    /// Update our store with another (usually after a bootstrap where we received another store)
873    /// Return list of updated / added if successful, UpdateWithError otherwise
874    #[allow(clippy::result_large_err)]
875    pub fn update_with(
876        &mut self,
877        store_raw: &MipStoreRaw,
878    ) -> Result<(Vec<MipInfo>, BTreeMap<MipInfo, MipState>), UpdateWithError> {
879        // iter over items in given store:
880        // -> 2 cases:
881        // * MipInfo is already in self store -> add to 'to_update' list
882        // * MipInfo is not in self.store -> We received a new MipInfo so we are running an out dated version
883        //                                   of the software
884        //                                   We then return the list of new MipInfo so we can warn and ask
885        //                                   to update the software
886
887        let mut component_versions: HashMap<MipComponent, u32> = self
888            .store
889            .iter()
890            .flat_map(|c| {
891                c.0.components
892                    .iter()
893                    .map(|(mip_component, component_version)| {
894                        (mip_component.clone(), *component_version)
895                    })
896            })
897            .collect();
898        let mut names: HashSet<String> = self.store.keys().map(|mi| mi.name.clone()).collect();
899        let mut to_update: BTreeMap<MipInfo, MipState> = Default::default();
900        let mut to_add: BTreeMap<MipInfo, MipState> = Default::default();
901        let mut has_error: Option<UpdateWithError> = None;
902
903        for (m_info, m_state) in store_raw.store.iter() {
904            if let Err(e) = m_state.is_consistent_with(m_info) {
905                // As soon as we found one non consistent state we abort the merge
906                has_error = Some(UpdateWithError::NonConsistent(
907                    m_info.clone(),
908                    m_state.clone(),
909                    e,
910                ));
911                break;
912            }
913
914            if let Some(m_state_orig) = self.store.get(m_info) {
915                // Given MIP info is already in self
916                // Need to check if we add it to 'to_update' list
917                let m_state_id: u32 = ComponentStateTypeId::from(&m_state.state).into();
918                let m_state_orig_id: u32 = ComponentStateTypeId::from(&m_state_orig.state).into();
919
920                // Note: we do not check for state: active OR failed OR error as they cannot change
921                if matches!(
922                    m_state_orig.state,
923                    ComponentState::Defined(_)
924                        | ComponentState::Started(_)
925                        | ComponentState::LockedIn(_)
926                ) {
927                    // Only accept 'higher' state
928                    // (e.g. 'started' if 'defined', 'locked in' if 'started'...)
929                    if m_state_id >= m_state_orig_id {
930                        to_update.insert(m_info.clone(), m_state.clone());
931                    } else {
932                        // Trying to downgrade state' (e.g. trying to go from 'active' -> 'defined')
933                        has_error = Some(UpdateWithError::Downgrade(
934                            m_info.clone(),
935                            m_state_orig.state,
936                            m_state.state,
937                        ));
938                        break;
939                    }
940                }
941            } else {
942                // Given MIP info is not in self
943                // Need to check if we add it to 'to_add' list
944
945                let last_m_info_ = to_add
946                    .last_key_value()
947                    .map(|(mi, _)| mi)
948                    .or(self.store.last_key_value().map(|(mi, _)| mi));
949
950                if let Some(last_m_info) = last_m_info_ {
951                    // check for versions of all components in v_info
952                    let mut component_version_compatible = true;
953                    for (component, component_version) in m_info.components.iter() {
954                        if component_version <= component_versions.get(component).unwrap_or(&0) {
955                            component_version_compatible = false;
956                            break;
957                        }
958                    }
959
960                    #[cfg(not(any(test, feature = "test-exports")))]
961                    if m_info.activation_delay < VERSIONING_ACTIVATION_DELAY_MIN {
962                        has_error = Some(UpdateWithError::InvalidActivationDelay(
963                            m_info.clone(),
964                            m_info.activation_delay,
965                            VERSIONING_ACTIVATION_DELAY_MIN,
966                        ));
967                        break;
968                    }
969
970                    if m_info.start > last_m_info.timeout
971                        && m_info.timeout > m_info.start
972                        && m_info.version > last_m_info.version
973                        && !names.contains(&m_info.name)
974                        && component_version_compatible
975                    {
976                        // Time range is ok / version is ok / name is unique, let's add it
977                        to_add.insert(m_info.clone(), m_state.clone());
978                        names.insert(m_info.name.clone());
979                        for (component, component_version) in m_info.components.iter() {
980                            component_versions.insert(component.clone(), *component_version);
981                        }
982                    } else {
983                        // Something is wrong (time range not ok? / version not incr? / names?
984                        // or component version not incr?)
985                        has_error = Some(UpdateWithError::Overlapping(
986                            m_info.clone(),
987                            last_m_info.clone(),
988                        ));
989                        break;
990                    }
991                } else {
992                    // to_add is empty && self.0 is empty
993                    to_add.insert(m_info.clone(), m_state.clone());
994                    names.insert(m_info.name.clone());
995                }
996            }
997        }
998
999        match has_error {
1000            None => {
1001                let updated: Vec<MipInfo> = to_update.keys().cloned().collect();
1002
1003                // Note: we only update the store with to_update collection
1004                //       having something in the to_add collection means that we need to update
1005                //       the Massa node software
1006                self.store.append(&mut to_update);
1007                Ok((updated, to_add))
1008            }
1009            Some(e) => Err(e),
1010        }
1011    }

1013    pub fn update_network_version_stats(
1014        &mut self,
1015        slot_timestamp: MassaTime,
1016        network_versions: Option<(u32, Option<u32>)>,
1017    ) {
1018        if let Some((_current_network_version, announced_network_version_)) = network_versions {
1019            let announced_network_version = announced_network_version_.unwrap_or(0);
1020
1021            let removed_version_ = match self.stats.latest_announcements.len() {
1022                n if n >= self.stats.config.block_count_considered => {
1023                    self.stats.latest_announcements.pop_front()
1024                }
1025                _ => None,
1026            };
1027            self.stats
1028                .latest_announcements
1029                .push_back(announced_network_version);
1030
1031            // We update the count of the received version (example: update counter for version 1)
1032            let mut network_version_count = *self
1033                .stats
1034                .network_version_counters
1035                .entry(announced_network_version)
1036                .and_modify(|v| *v = v.saturating_add(1))
1037                .or_insert(1);
1038
1039            // If we removed a version announcement, we decrement the corresponding counter
1040            // (example: remove a version 1, so decrement the corresponding counter)
1041            // As soon as a counter value is 0, we remove it
1042            if let Some(removed_version) = removed_version_ {
1043                if let Entry::Occupied(mut e) =
1044                    self.stats.network_version_counters.entry(removed_version)
1045                {
1046                    let entry_value = e.get_mut();
1047                    *entry_value = entry_value.saturating_sub(1);
1048                    network_version_count = *entry_value;
1049                    if *entry_value == 0 {
1050                        self.stats.network_version_counters.remove(&removed_version);
1051                    }
1052                }
1053            }
1054
1055            if announced_network_version != 0 {
1056                let vote_ratio = Ratio::new(
1057                    network_version_count,
1058                    self.stats.config.block_count_considered as u64,
1059                );
1060
1061                if vote_ratio > self.stats.config.warn_announced_version_ratio {
1062                    let last_key_value = self.store.last_key_value();
1063                    if let Some((mi, _ms)) = last_key_value {
1064                        if announced_network_version > mi.version {
1065                            // Vote ratio is > 30%
1066                            // announced version is not known (not in MIP store)
1067                            // announced version is > to the last known network version in MIP store
1068                            // -> Warn the user to update
1069                            warn!("{} our of {} last blocks advertised that they are willing to transition to version {}. You should update your node if you wish to move to that version.",
1070                                network_version_count,
1071                                self.stats.config.block_count_considered,
1072                                announced_network_version
1073                            );
1074                        }
1075                    }
1076                }
1077            }
1078        }
1079
1080        debug!(
1081            "[VERSIONING STATS] stats have {} counters and {} announcements",
1082            self.stats.network_version_counters.len(),
1083            self.stats.latest_announcements.len()
1084        );
1085
1086        // Even if stats did not move, update the states (e.g. LockedIn -> Active)
1087        self.advance_states_on_updated_stats(slot_timestamp);
1088    }

1103

1105                network_version_count,

1196        shutdown_start: Slot,
1197        shutdown_end: Slot,
1198        thread_count: u8,
1199        t0: MassaTime,
1200        genesis_timestamp: MassaTime,
1201    ) -> Result<(), IsConsistentWithShutdownPeriodError> {
1202        // let mut is_consistent = true;
1203        let mut has_error: Result<(), IsConsistentWithShutdownPeriodError> = Ok(());
1204

1206            get_block_slot_timestamp(thread_count, t0, genesis_timestamp, shutdown_start)?;
1207        let shutdown_end_ts =
1208            get_block_slot_timestamp(thread_count, t0, genesis_timestamp, shutdown_end)?;
1209        let shutdown_range = shutdown_start_ts..=shutdown_end_ts;
1210
1211        for (mip_info, mip_state) in &self.store {
1212            match mip_state.state {
1213                ComponentState::Defined(..)
1214                | ComponentState::Started(..)
1215                | ComponentState::LockedIn(..) => {

1217                    if shutdown_range.contains(&mip_info.start)
1218                        || shutdown_range.contains(&mip_info.timeout)
1219                    {
1220                        // is_consistent = false;
1221                        has_error = Err(IsConsistentWithShutdownPeriodError::NonConsistent(
1222                            mip_info.clone(),
1223                            mip_state.state,
1224                            shutdown_start_ts,
1225                            shutdown_end_ts,
1226                        ));
1227                        break;
1228                    }
1229                }
1230                _ => {
1231                    // active / failed, error, nothing to do
1232                    // locked in, nothing to do (might go from 'locked in' to 'active' during shutdown)
1233                }
1234            }
1235        }
1236
1237        has_error
1238    }

1240    fn update_for_network_shutdown(
1241        &mut self,
1242        shutdown_start: Slot,
1243        shutdown_end: Slot,
1244        thread_count: u8,
1245        t0: MassaTime,
1246        genesis_timestamp: MassaTime,
1247    ) -> Result<(), ModelsError> {
1248        let shutdown_start_ts =
1249            get_block_slot_timestamp(thread_count, t0, genesis_timestamp, shutdown_start)?;
1250        let shutdown_end_ts =
1251            get_block_slot_timestamp(thread_count, t0, genesis_timestamp, shutdown_end)?;
1252        let shutdown_range = shutdown_start_ts..=shutdown_end_ts;
1253
1254        let mut new_store: BTreeMap<MipInfo, MipState> = Default::default();
1255        let mut new_stats = self.stats.clone();
1256        new_stats.reset();
1257
1258        let next_valid_start_ = shutdown_end.get_next_slot(thread_count)?;
1259        let next_valid_start =
1260            get_block_slot_timestamp(thread_count, t0, genesis_timestamp, next_valid_start_)?;
1261
1262        let mut offset: Option<MassaTime> = None;
1263
1264        for (mip_info, mip_state) in &self.store {
1265            match mip_state.state {
1266                ComponentState::Defined(..) => {
1267                    // Defined: offset start & timeout
1268
1269                    let mut new_mip_info = mip_info.clone();
1270
1271                    if shutdown_range.contains(&new_mip_info.start) {
1272                        let offset_ts = match offset {
1273                            Some(offset_ts) => offset_ts,
1274                            None => {
1275                                let offset_ts = next_valid_start.saturating_sub(mip_info.start);
1276                                offset = Some(offset_ts);
1277                                offset_ts
1278                            }
1279                        };
1280
1281                        new_mip_info.start = new_mip_info.start.saturating_add(offset_ts);
1282                        new_mip_info.timeout = new_mip_info
1283                            .start
1284                            .saturating_add(mip_info.timeout.saturating_sub(mip_info.start));
1285                    }
1286                    new_store.insert(new_mip_info, mip_state.clone());
1287                }
1288                ComponentState::Started(..) | ComponentState::LockedIn(..) => {
1289                    // Started or LockedIn -> Reset to Defined, offset start & timeout
1290
1291                    let mut new_mip_info = mip_info.clone();
1292
1293                    let offset_ts = match offset {
1294                        Some(offset_ts) => offset_ts,
1295                        None => {

1388    /// Get MIP store changes between 2 timestamps - used by the db to update the disk
1389    pub fn update_batches(
1390        &self,
1391        batch: &mut DBBatch,
1392        versioning_batch: &mut DBBatch,
1393        between: Option<(&MassaTime, &MassaTime)>,
1394    ) -> Result<(), SerializeError> {
1395        let mip_info_ser = MipInfoSerializer::new();
1396        let mip_state_ser = MipStateSerializer::new();
1397
1398        let bounds = match between {
1399            Some(between) => (*between.0)..(*between.1),
1400            None => MassaTime::from_millis(0)..MassaTime::max(),
1401        };
1402        let mut key = Vec::new();
1403        let mut value = Vec::new();
1404
1405        for (mip_info, mip_state) in self.store.iter() {
1406            if let Some((advance, state_id)) = mip_state.history.last_key_value() {
1407                if bounds.contains(&advance.now) {
1408                    key.extend(MIP_STORE_PREFIX.as_bytes().to_vec());
1409                    mip_info_ser.serialize(mip_info, &mut key)?;
1410                    mip_state_ser.serialize(mip_state, &mut value)?;
1411                    match state_id {
1412                        ComponentStateTypeId::Active => {
1413                            batch.insert(key.clone(), Some(value.clone()));
1414                            // + "Remove key" in VERSIONING_CF
1415                            versioning_batch.insert(key.clone(), None);
1416                        }
1417                        _ => {
1418                            versioning_batch.insert(key.clone(), Some(value.clone()));
1419                        }
1420                    }
1421                    key.clear();
1422                    value.clear();
1423                }
1424            }
1425        }
1426
1427        value.clear();
1428        let mip_stats_ser = MipStoreStatsSerializer::new();
1429        mip_stats_ser.serialize(&self.stats, &mut value)?;
1430        versioning_batch.insert(
1431            MIP_STORE_STATS_PREFIX.as_bytes().to_vec(),
1432            Some(value.clone()),
1433        );
1434
1435        Ok(())
1436    }

1439    fn extend_from_db(
1440        &mut self,
1441        db: ShareableMassaDBController,
1442    ) -> Result<(Vec<MipInfo>, BTreeMap<MipInfo, MipState>), ExtendFromDbError> {
1443        let mip_info_deser = MipInfoDeserializer::new();
1444        let mip_state_deser = MipStateDeserializer::new();
1445        let mip_store_stats_deser = MipStoreStatsDeserializer::new(
1446            MIP_STORE_STATS_BLOCK_CONSIDERED,
1447            self.stats.config.warn_announced_version_ratio,
1448        );
1449
1450        let db = db.read();
1451
1452        // Get data from state cf handle
1453        let mut update_data: BTreeMap<MipInfo, MipState> = Default::default();
1454        for (ser_mip_info, ser_mip_state) in
1455            db.prefix_iterator_cf(STATE_CF, MIP_STORE_PREFIX.as_bytes())
1456        {
1457            if !ser_mip_info.starts_with(MIP_STORE_PREFIX.as_bytes()) {
1458                break;
1459            }
1460
1461            // deser
1462            let (_, mip_info) = mip_info_deser
1463                .deserialize::<DeserializeError>(&ser_mip_info[MIP_STORE_PREFIX.len()..])
1464                .map_err(|e| ExtendFromDbError::Deserialize(e.to_string()))?;
1465
1466            let (_, mip_state) = mip_state_deser
1467                .deserialize::<DeserializeError>(&ser_mip_state)
1468                .map_err(|e| ExtendFromDbError::Deserialize(e.to_string()))?;
1469
1470            update_data.insert(mip_info, mip_state);
1471        }
1472
1473        let (mut updated, mut added) = match update_data.is_empty() {
1474            true => (vec![], BTreeMap::new()),
1475            false => {
1476                let store_raw_ = MipStoreRaw {
1477                    store: update_data,
1478                    stats: MipStoreStats {
1479                        config: MipStatsConfig {
1480                            block_count_considered: MIP_STORE_STATS_BLOCK_CONSIDERED,
1481                            warn_announced_version_ratio: self
1482                                .stats
1483                                .config
1484                                .warn_announced_version_ratio,
1485                        },
1486                        latest_announcements: Default::default(),
1487                        network_version_counters: Default::default(),
1488                    },
1489                };
1490                // Only call update_with if update_data is not empty
1491                self.update_with(&store_raw_)?
1492            }
1493        };
1494
1495        let mut update_data: BTreeMap<MipInfo, MipState> = Default::default();
1496
1497        // Get data from state cf handle
1498        for (ser_mip_info, ser_mip_state) in
1499            db.prefix_iterator_cf(VERSIONING_CF, MIP_STORE_PREFIX.as_bytes())
1500        {
1501            // deser
1502
1503            match &ser_mip_info {
1504                key if key.starts_with(MIP_STORE_PREFIX.as_bytes()) => {
1505                    let (_, mip_info) = mip_info_deser
1506                        .deserialize::<DeserializeError>(&ser_mip_info[MIP_STORE_PREFIX.len()..])
1507                        .map_err(|e| ExtendFromDbError::Deserialize(e.to_string()))?;
1508
1509                    let (_, mip_state) = mip_state_deser
1510                        .deserialize::<DeserializeError>(&ser_mip_state)
1511                        .map_err(|e| ExtendFromDbError::Deserialize(e.to_string()))?;
1512
1513                    update_data.insert(mip_info, mip_state);
1514                }
1515                key if key.starts_with(MIP_STORE_STATS_PREFIX.as_bytes()) => {
1516                    let (_, mip_store_stats) = mip_store_stats_deser
1517                        .deserialize::<DeserializeError>(&ser_mip_state)
1518                        .map_err(|e| ExtendFromDbError::Deserialize(e.to_string()))?;
1519
1520                    self.stats = mip_store_stats;
1521                }
1522                _ => {
1523                    break;
1524                }
1525            }
1526        }
1527
1528        if !update_data.is_empty() {
1529            let store_raw_ = MipStoreRaw {
1530                store: update_data,
1531                stats: MipStoreStats {
1532                    config: MipStatsConfig {
1533                        block_count_considered: MIP_STORE_STATS_BLOCK_CONSIDERED,
1534                        warn_announced_version_ratio: self
1535                            .stats
1536                            .config
1537                            .warn_announced_version_ratio,
1538                    },
1539                    latest_announcements: Default::default(),
1540                    network_version_counters: Default::default(),
1541                },
1542            };
1543            // Only call update_with if update_data is not empty
1544            let (updated_2, added_2) = self.update_with(&store_raw_)?;
1545            updated.extend(updated_2);
1546            added.extend(added_2);
1547        }
1548
1549        Ok((updated, added))
1550    }

1579        let mut store = Self {

1586            store: BTreeMap::from(value),
1587            stats: MipStoreStats::new(cfg),
1588        };
1589
1590        // Use update_with ensuring that we have no overlapping time range, unique names & ...
1591        match store.update_with(&other_store) {
1592            Ok((_updated, mut added)) => {
1593                store.store.append(&mut added);
1594                Ok(store)
1595            }
1596            Err(e) => Err(e),
1597        }
