1//! Node version compatibility checked during bootstrap and protocol handshakes.
2//!
3//! Peers must run the same instance and major version. Once `MIP-0002` is
4//! active, peers older than the first release shipping it
5//! ([`MIP_0002_NODE_VERSION`]) are rejected as well: they would not follow
6//! the post-activation rules.
7
8use massa_models::version::Version;
9use massa_time::MassaTime;
10
11use crate::mips::{MIP_0002_EXECUTION_VERSION, MIP_0002_NODE_VERSION};
12use crate::versioning::{MipComponent, MipStore};
13
14/// true if a peer running `peer_version` is compatible with us at `now`
15///
16/// A node never requires its peers to be newer than itself, so the `MIP-0002`
17/// requirement only applies when `our_version` ships it.
18pub fn is_peer_version_compatible(
19    mip_store: &MipStore,
20    our_version: &Version,
21    peer_version: &Version,
22    now: MassaTime,
23) -> bool {
24    if !our_version.is_compatible(peer_version) {
25        return false;
26    }
27
28    let (major, minor) = MIP_0002_NODE_VERSION;
29    let mip_0002_active = mip_store.get_latest_component_version_at(&MipComponent::Execution, now)
30        >= MIP_0002_EXECUTION_VERSION;
31
32    !mip_0002_active
33        || !our_version.is_at_least(major, minor)
34        || peer_version.is_at_least(major, minor)
35}

37#[cfg(test)]
38mod tests {
39    use super::*;
40    use std::collections::BTreeMap;
41    use std::str::FromStr;
42
43    use massa_models::config::MIP_STORE_STATS_BLOCK_CONSIDERED;
44    use num::rational::Ratio;
45
46    use crate::test_helpers::versioning_helpers::advance_state_until;
47    use crate::versioning::{ComponentState, MipInfo, MipStatsConfig};
48
49    /// Store with `MIP-0002` active, plus a timestamp before and after activation.
50    fn store_with_mip_0002_active() -> (MipStore, MassaTime, MassaTime) {
51        let start = MassaTime::from_millis(10_000);
52        let mi = MipInfo {
53            name: "MIP-0002-BugFix".to_string(),
54            version: 2,
55            components: BTreeMap::from([(MipComponent::Execution, MIP_0002_EXECUTION_VERSION)]),
56            start,
57            timeout: MassaTime::from_millis(100_000),
58            activation_delay: MassaTime::from_millis(5_000),
59        };
60        let ms = advance_state_until(ComponentState::active(MassaTime::from_millis(0)), &mi);
61        let stats = MipStatsConfig {
62            block_count_considered: MIP_STORE_STATS_BLOCK_CONSIDERED,
63            warn_announced_version_ratio: Ratio::new_raw(30, 100),
64        };
65        let store = MipStore::try_from(([(mi, ms)], stats)).expect("mip store");
66
67        let before = start.saturating_sub(MassaTime::from_millis(1));
68        let active_at = (0..400)
69            .map(|i| start.saturating_add(MassaTime::from_millis(i * 100)))
70            .find(|ts| {
71                store.get_latest_component_version_at(&MipComponent::Execution, *ts)
72                    >= MIP_0002_EXECUTION_VERSION
73            })
74            .expect("MIP-0002 should become active after start");
75        (store, before, active_at)
76    }
77
78    fn v(s: &str) -> Version {
79        Version::from_str(s).unwrap()
80    }
81
82    fn mip_0002_release() -> Version {
83        let (major, minor) = MIP_0002_NODE_VERSION;
84        v(&format!("MAIN.{}.{}", major, minor))
85    }
86
87    fn pre_mip_0002_release() -> Version {
88        let (major, minor) = MIP_0002_NODE_VERSION;
89        assert!(minor > 0, "test assumes MIP-0002 ships in a minor release");
90        v(&format!("MAIN.{}.{}", major, minor - 1))
91    }
92
93    #[test]
94    fn same_major_required() {
95        let (store, before, active_at) = store_with_mip_0002_active();
96        let ours = mip_0002_release();
97        let (major, _) = MIP_0002_NODE_VERSION;
98        for now in [before, active_at] {
99            assert!(!is_peer_version_compatible(
100                &store,
101                &ours,
102                &v(&format!("MAIN.{}.0", major + 1)),
103                now
104            ));
105            assert!(!is_peer_version_compatible(
106                &store,
107                &ours,
108                &v(&format!("MAIN.{}.9", major - 1)),
109                now
110            ));
111            let (_, minor) = MIP_0002_NODE_VERSION;
112            assert!(!is_peer_version_compatible(
113                &store,
114                &ours,
115                &v(&format!("TEST.{}.{}", major, minor)),
116                now
117            ));
118        }
119    }
120
121    #[test]
122    fn older_peers_accepted_until_mip_0002_is_active() {
123        let (store, before, active_at) = store_with_mip_0002_active();
124        let ours = mip_0002_release();
125        let older = pre_mip_0002_release();
126
127        assert!(is_peer_version_compatible(&store, &ours, &older, before));
128        assert!(!is_peer_version_compatible(
129            &store, &ours, &older, active_at
130        ));
131        assert!(is_peer_version_compatible(&store, &ours, &ours, active_at));
132    }
133
134    #[test]
135    fn node_without_mip_0002_release_keeps_same_major_rule() {
136        let (store, _, active_at) = store_with_mip_0002_active();
137        let older = pre_mip_0002_release();
138
139        assert!(is_peer_version_compatible(
140            &store, &older, &older, active_at
141        ));
142        assert!(is_peer_version_compatible(
143            &store,
144            &older,
145            &mip_0002_release(),
146            active_at
147        ));
148    }
