54pub(crate) fn start_connectivity_thread(
55    peer_id: PeerId,
56    selector_controller: Box<dyn SelectorController>,
57    mut network_controller: Box<dyn NetworkController>,
58    consensus_controller: Box<dyn ConsensusController>,
59    pool_controller: Box<dyn PoolController>,
60    channel_blocks: (
61        MassaSender<PeerMessageTuple>,
62        MassaReceiver<PeerMessageTuple>,
63    ),
64    channel_endorsements: (
65        MassaSender<PeerMessageTuple>,
66        MassaReceiver<PeerMessageTuple>,
67    ),
68    channel_operations: (
69        MassaSender<PeerMessageTuple>,
70        MassaReceiver<PeerMessageTuple>,
71    ),
72    channel_peers: (
73        MassaSender<PeerMessageTuple>,
74        MassaReceiver<PeerMessageTuple>,
75    ),
76    initial_peers: InitialPeers,
77    peer_db: SharedPeerDB,
78    storage: Storage,
79    protocol_channels: ProtocolChannels,
80    messages_handler: MessagesHandler,
81    peer_categories: HashMap<String, (Vec<IpAddr>, PeerCategoryInfo)>,
82    default_category: PeerCategoryInfo,
83    config: ProtocolConfig,
84    mip_store: MipStore,
85    massa_metrics: MassaMetrics,
86) -> Result<(MassaSender<ConnectivityCommand>, JoinHandle<()>), ProtocolError> {
87    let handle = std::thread::Builder::new()
88    .name(THREAD_NAME.to_string())
89    .spawn({
90        let sender_endorsements_propagation_ext = protocol_channels.endorsement_handler_propagation.0.clone();
91        let sender_blocks_retrieval_ext = protocol_channels.block_handler_retrieval.0.clone();
92        let sender_blocks_propagation_ext = protocol_channels.block_handler_propagation.0.clone();
93        let sender_operations_propagation_ext = protocol_channels.operation_handler_propagation.0.clone();
94        move || {
95            for (addr, transport) in &config.listeners {
96                network_controller
97                    .start_listener(*transport, *addr)
98                    .unwrap_or_else(|_| panic!(
99                        "Failed to start listener {:?} of transport {:?} in protocol",
100                        addr, transport
101                    ));
102            }
103
104            // Little hack to be sure that listeners are started before trying to connect to peers
105            std::thread::sleep(Duration::from_millis(100));
106
107            // Create cache outside of the op handler because it could be used by other handlers
108            let total_in_slots = config.peers_categories.values().map(|v| v.max_in_connections).sum::<usize>() + config.default_category_info.max_in_connections + 1;
109            let total_out_slots = config.peers_categories.values().map(| v| v.target_out_connections).sum::<usize>() + config.default_category_info.target_out_connections + 1;
110            let operation_cache = Arc::new(RwLock::new(OperationCache::new(
111                config.max_known_ops_size.try_into().unwrap(),
112                config.max_node_known_ops_size.try_into().unwrap()
113            )));
114            let endorsement_cache = Arc::new(RwLock::new(EndorsementCache::new(
115                config.max_known_endorsements_size.try_into().unwrap(),
116                (total_in_slots + total_out_slots).try_into().unwrap()
117            )));
118
119            let block_cache = Arc::new(RwLock::new(BlockCache::new(
120                config.max_known_blocks_size.try_into().unwrap(),
121                config.max_node_known_blocks_size.try_into().unwrap(),
122            )));
123
124            // Start handlers
125            let mut peer_management_handler = PeerManagementHandler::new(
126                initial_peers,
127                peer_id,
128                peer_db.clone(),
129                channel_peers,
130                protocol_channels.peer_management_handler,
131                messages_handler,
132                network_controller.get_active_connections(),
133                peer_categories.iter().map(|(key, value)|(key.clone(), (value.0.clone(), value.1.target_out_connections))).collect(),
134                config.default_category_info.target_out_connections,
135                &config,
136                massa_metrics.clone(),
137            );
138
139            let mut operation_handler = OperationHandler::new(
140                pool_controller.clone(),
141                storage.clone_without_refs(),
142                config.clone(),
143                operation_cache.clone(),
144                network_controller.get_active_connections(),
145                channel_operations.1,
146                protocol_channels.operation_handler_retrieval.0.clone(),
147                protocol_channels.operation_handler_retrieval.1.clone(),
148                sender_operations_propagation_ext.clone(),
149                protocol_channels.operation_handler_propagation.1.clone(),
150                peer_management_handler.sender.command_sender.clone(),
151                massa_metrics.clone(),
152            );
153            let mut endorsement_handler = EndorsementHandler::new(
154                pool_controller.clone(),
155                selector_controller.clone(),
156                endorsement_cache.clone(),
157                storage.clone_without_refs(),
158                config.clone(),
159                network_controller.get_active_connections(),
160                channel_endorsements.1,
161                protocol_channels.endorsement_handler_retrieval.0,
162                protocol_channels.endorsement_handler_retrieval.1,
163                sender_endorsements_propagation_ext.clone(),
164                protocol_channels.endorsement_handler_propagation.1.clone(),
165                peer_management_handler.sender.command_sender.clone(),
166                massa_metrics.clone(),
167            );
168            let mut block_handler = BlockHandler::new(
169                network_controller.get_active_connections(),
170                selector_controller,
171                consensus_controller,
172                pool_controller,
173                channel_blocks.1,
174                sender_blocks_retrieval_ext,
175                protocol_channels.block_handler_retrieval.1.clone(),
176                protocol_channels.block_handler_propagation.1.clone(),
177                sender_blocks_propagation_ext,
178                sender_operations_propagation_ext,
179                sender_endorsements_propagation_ext,
180                peer_management_handler.sender.command_sender.clone(),
181                config.clone(),
182                endorsement_cache,
183                operation_cache,
184                block_cache,
185                storage.clone_without_refs(),
186                mip_store,
187                massa_metrics.clone(),
188            );
189
190            let tick_metrics = tick(massa_metrics.tick_delay);
191            let tick_try_connect = tick(config.try_connection_timer.to_duration());
192            let tick_unban_everyone = tick(config.unban_everyone_timer.to_duration());
193
194            //Try to connect to peers
195            loop {
196                select! {
197                    recv(protocol_channels.connectivity_thread.1) -> msg => {
198                        // update channel metrics
199                        protocol_channels.connectivity_thread.1.update_metrics();
200                        match msg {
201                            Ok(ConnectivityCommand::Stop) => {
202                                debug!("Stopping protocol");
203                                drop(network_controller);
204                                debug!("Stopped network controller");
205                                operation_handler.stop();
206                                debug!("Stopped operation handler");
207                                endorsement_handler.stop();
208                                debug!("Stopped endorsement handler");
209                                block_handler.stop();
210                                debug!("Stopped block handler");
211                                peer_management_handler.stop();
212                                debug!("Stopped peer handler");
213                                break;
214                            },
215                            Ok(ConnectivityCommand::GetStats { responder }) => {
216                                let active_node_count = network_controller.get_active_connections().get_peer_ids_connected().len() as u64;
217                                let in_connection_count = network_controller.get_active_connections().get_nb_in_connections() as u64;
218                                let out_connection_count = network_controller.get_active_connections().get_nb_out_connections() as u64;
219                                let (banned_peer_count, known_peer_count) = {
220                                    let peer_db_read = peer_db.read();
221                                    (peer_db_read.get_banned_peer_count(), peer_db_read.get_known_peer_count())
222                                };
223                                let stats = NetworkStats {
224                                    active_node_count,
225                                    in_connection_count,
226                                    out_connection_count,
227                                    banned_peer_count,
228                                    known_peer_count,
229                                };
230                                let peers: HashMap<PeerId, (SocketAddr, PeerConnectionType)> = network_controller.get_active_connections().get_peers_connected().into_iter().map(|(peer_id, peer)| {
231                                    (peer_id, (peer.0, peer.1))
232                                }).collect();
233                                responder.try_send((stats, peers)).unwrap_or_else(|_| warn!("Failed to send stats to responder"));
234                            }
235                            Err(_) => {
236                                warn!("Channel to connectivity thread is closed. Stopping the protocol");
237                                break;
238                            }
239                        }
240                    },
241                    recv(tick_metrics) -> _ => {
242                        massa_metrics.set_peernet_total_bytes_received(network_controller.get_total_bytes_received());
243                        massa_metrics.set_peernet_total_bytes_sent(network_controller.get_total_bytes_sent());
244                        let active_conn = network_controller.get_active_connections();
245                        massa_metrics.set_active_connections(active_conn.get_nb_in_connections(), active_conn.get_nb_out_connections());
246                        let peers_map = active_conn.get_peers_connections_bandwidth();
247                        massa_metrics.update_peers_tx_rx(peers_map);
248                        let peer_db_read = peer_db.read();
249                        massa_metrics.set_known_peers(peer_db_read.get_known_peer_count() as usize);
250                        massa_metrics.set_banned_peers(peer_db_read.get_banned_peer_count() as usize);
251                    },
252                    recv(tick_try_connect) -> _ => {
253                        let active_conn = network_controller.get_active_connections();
254                        let peers_connected = active_conn.get_peers_connected();
255                        let peers_connection_queue = active_conn.get_peer_ids_out_connection_queue();
256
257                        let mut connection_slots = HashMap::new();
258                        connection_slots.insert("default", config.default_category_info.target_out_connections);
259                        for (category, infos) in peer_categories.iter() {
260                            connection_slots.insert(category, infos.1.target_out_connections);
261                        }
262
263                        // Get all the addresses we can connect to, without any filter or prioritization done yet
264                        let mut addresses_can_connect  = Vec::new();
265                        {
266                            let peer_db_read = peer_db.read();
267                            for (peer_id, peer_info) in peer_db_read.get_peers() {
268
269                                // If peer already connected, decrement the slots for the given category, or default category if none
270                                if let Some(peer) = peers_connected.get(peer_id) {
271                                    if peer.1 == PeerConnectionType::OUT {
272                                        if let Some(ref peer_category) = &peer.2 {
273                                            if let Some(slots) = connection_slots.get_mut(peer_category.as_str()) {
274                                                *slots = slots.saturating_sub(1);
275                                            } else {
276                                                tracing::warn!("Category of connected peer {peer_category} not known in configuration");
277                                            }
278                                        } else {
279                                            let slots = connection_slots.get_mut("default").unwrap();
280                                            *slots = slots.saturating_sub(1);
281                                        }
282                                    }
283                                    continue;
284                                }
285
286                                if peer_info.state == PeerState::Trusted {
287                                    if let Some(ref last_announce) = peer_info.last_announce {
288                                        if last_announce.listeners.is_empty() {
289                                            continue;
290                                        }
291
292                                        if let Some((addr, _)) = last_announce.listeners.iter().next() {
293                                            let canonical_ip = to_canonical(addr.ip());
294                                            // Default category policy applies unless the peer IP
295                                            // matches an explicit peers_categories allowlist.
296                                            let mut allowed_local_ips = default_category.allow_local_peers;
297                                            // Check if the peer is in a category and we didn't reached out target yet
298                                            let mut category_found = None;
299                                            for (name, (ips, cat)) in &peer_categories {
300                                                if ips.contains(&canonical_ip) {
301                                                    category_found = Some(name);
302                                                    allowed_local_ips = cat.allow_local_peers;
303                                                }
304                                            }
305
306                                            if peers_connection_queue.contains(addr) {
307                                                if let Some(peer_category) = category_found {
308                                                    if let Some(slots) = connection_slots.get_mut(peer_category.as_str()) {
309                                                        *slots = slots.saturating_sub(1);
310                                                    }
311                                                } else if let Some(v) = connection_slots.get_mut("default") {
312                                                    *v = v.saturating_sub(1);
313                                                }
314                                                continue;
315                                            }
316
317                                            let connection_metadata = peer_db_read.get_connection_metadata_or_default(addr);
318
319                                            // check if the peer last connect attempt has not been too recent
320                                            if let ConnectionMetadata { last_try_connect: Some(lt), .. } = connection_metadata {
321                                                let last_try_connect = lt.estimate_instant().expect("Time went backward");
322                                                if last_try_connect.elapsed() < config.try_connection_timer_same_peer.to_duration() {
323                                                    continue;
324                                                }
325                                            }
326
327                                            if config.listeners.iter().any(|(local_addr, _transport)| addr == local_addr) {
328                                                continue;
329                                            }
330
331                                            // Same non-global policy as try_connect: refuse
332                                            // loopback / private / reserved destinations unless
333                                            // this peer's category (or the default) allows local peers.
334                                            if !is_routable_peer_addr(addr, allowed_local_ips) {
335                                                continue;
336                                            }
337
338                                            addresses_can_connect.push((*addr, connection_metadata, category_found));
339                                        } else {
340                                            tracing::warn!("No listeners for the peer {peer_id}");
341                                        }
342                                    }
343                                }
344                            }
345                        }
346
347                        // Sort addresses using the metadata
348                        addresses_can_connect.sort_by(|a, b| a.1.cmp(&b.1));
349
350                        // Connect to the given addresses, trying to fill all the slots available
351                        let mut addresses_connected = vec![];
352                        for (addr, _, category) in addresses_can_connect.iter() {
353                            if addresses_connected.contains(addr) {
354                                continue;
355                            }
356
357                            // Connect to the peer
358                            match category {
359                                // In case has a special category
360                                Some(cat) => {
361                                    for (name, slots) in connection_slots.iter_mut() {
362                                        if name == *cat && *slots > 0 {
363                                            // In case the connection succeeds, we take a place in a slot
364                                            if try_connect_peer(*addr, &mut network_controller, &peer_db, &config).is_ok() {
365                                                *slots = slots.saturating_sub(1);
366                                                addresses_connected.push(*addr);
367                                            }
368                                        }
369                                    }
370                                }
371
372                                // Default category
373                                None if connection_slots["default"] > 0 => {
374                                    // In case the connection succeeds, we take a place in a slot
375                                    // (mirrors the special-category branch above: the slot is
376                                    // consumed and the address recorded on success, not failure).
377                                    if try_connect_peer(*addr, &mut network_controller, &peer_db, &config).is_ok() {
378                                        if let Some(v) = connection_slots.get_mut("default") {
379                                            *v = v.saturating_sub(1);
380                                        }
381                                        addresses_connected.push(*addr);
382                                    }
383                                }
384                                None => continue,
385                            }
386
387                            // IF all slots are filled, stop
388                            if connection_slots.values().sum::<usize>() == 0 {
389                                break;
390                            }
391                        }
392                    }
393                    recv(tick_unban_everyone) -> _ => {
394                        debug!("Periodic unban of every peer");
395                        let mut peer_db_write = peer_db.write();
396                        for (peer_id, peer_status) in peer_db_write.get_peers().clone() {
397                            if peer_status.state == PeerState::Banned {
398                                peer_db_write.unban_peer(&peer_id);
399                            }
400                        }
401                    }
402                }
403            }
404        }
405    }).expect("OS failed to start connectivity thread");
406
407    // Start controller
408    Ok((protocol_channels.connectivity_thread.0, handle))
409}

411// Attempt to connect to peer
412fn try_connect_peer(
413    addr: SocketAddr,
414    network_controller: &mut Box<dyn NetworkController>,
415    peer_db: &SharedPeerDB,
416    config: &ProtocolConfig,
417) -> Result<(), ProtocolError> {
418    debug!("Trying to connect to addr {}", addr);
419
420    let conn_res = network_controller.try_connect(addr, config.timeout_connection.to_duration());
421    {
422        let mut peer_db_write = peer_db.write();
423        peer_db_write.set_try_connect_or_insert(&addr);
424        if let Err(ref err) = conn_res {
425            debug!("Failed to connect to peer {:?}: {:?}", addr, err);
426            peer_db_write.set_try_connect_failure_or_insert(&addr);
427        }
428    }
429    conn_res
