91/// This function exists because consensus need the protocol controller and we need consensus controller.
92/// Someone has to be created first.
93pub fn create_protocol_controller(
94    config: ProtocolConfig,
95) -> (Box<dyn ProtocolController>, ProtocolChannels) {
96    let (sender_operations_retrieval_ext, receiver_operations_retrieval_ext) = MassaChannel::new(
97        "operations_retrievial_ext".to_string(),
98        Some(config.max_size_channel_commands_retrieval_operations),
99    );
100    let (sender_operations_propagation_ext, receiver_operations_propagation_ext) =
101        MassaChannel::new(
102            "operations_propagation_ext".to_string(),
103            Some(config.max_size_channel_commands_propagation_operations),
104        );
105    let (sender_endorsements_retrieval_ext, receiver_endorsements_retrieval_ext) =
106        MassaChannel::new(
107            "endorsements_retrieval_ext".to_string(),
108            Some(config.max_size_channel_commands_retrieval_endorsements),
109        );
110    let (sender_endorsements_propagation_ext, receiver_endorsements_propagation_ext) =
111        MassaChannel::new(
112            "endorsements_propagation_ext".to_string(),
113            Some(config.max_size_channel_commands_propagation_endorsements),
114        );
115    let (sender_blocks_propagation_ext, receiver_blocks_propagation_ext) = MassaChannel::new(
116        "blocks_propagation_ext".to_string(),
117        Some(config.max_size_channel_commands_propagation_blocks),
118    );
119    let (sender_blocks_retrieval_ext, receiver_blocks_retrieval_ext) = MassaChannel::new(
120        "blocks_retrieval_ext".to_string(),
121        Some(config.max_size_channel_commands_retrieval_blocks),
122    );
123    let (sender_connectivity_ext, receiver_connectivity_ext) = MassaChannel::new(
124        "connectivity_ext".to_string(),
125        Some(config.max_size_channel_commands_connectivity),
126    );
127    let (sender_peer_management_ext, receiver_peer_management_ext): (
128        MassaSender<PeerManagementCmd>,
129        MassaReceiver<PeerManagementCmd>,
130    ) = MassaChannel::new(
131        "peer_management_ext".to_string(),
132        Some(config.max_size_channel_commands_peers),
133    );
134    (
135        Box::new(ProtocolControllerImpl::new(
136            sender_blocks_retrieval_ext.clone(),
137            sender_blocks_propagation_ext.clone(),
138            sender_operations_propagation_ext.clone(),
139            sender_endorsements_propagation_ext.clone(),
140            sender_connectivity_ext.clone(),
141            sender_peer_management_ext.clone(),
142        )),
143        ProtocolChannels {
144            operation_handler_retrieval: (
145                sender_operations_retrieval_ext,
146                receiver_operations_retrieval_ext,
147            ),
148            operation_handler_propagation: (
149                sender_operations_propagation_ext,
150                receiver_operations_propagation_ext,
151            ),
152            endorsement_handler_retrieval: (
153                sender_endorsements_retrieval_ext,
154                receiver_endorsements_retrieval_ext,
155            ),
156            endorsement_handler_propagation: (
157                sender_endorsements_propagation_ext,
158                receiver_endorsements_propagation_ext,
159            ),
160            block_handler_retrieval: (sender_blocks_retrieval_ext, receiver_blocks_retrieval_ext),
161            block_handler_propagation: (
162                sender_blocks_propagation_ext,
163                receiver_blocks_propagation_ext,
164            ),
165            connectivity_thread: (sender_connectivity_ext, receiver_connectivity_ext),
166            peer_management_handler: (sender_peer_management_ext, receiver_peer_management_ext),
167        },
168    )
169}

171/// start a new `ProtocolController` from a `ProtocolConfig`
172///
173/// # Arguments
174/// * `config`: protocol settings
175/// * `consensus_controller`: interact with consensus module
176/// * `bootstrap_peers`: list of peers to connect to retrieved from the bootstrap
177/// * `storage`: Shared storage to fetch data that are fetch across all modules
178#[allow(clippy::too_many_arguments)]
179pub fn start_protocol_controller(
180    config: ProtocolConfig,
181    selector_controller: Box<dyn SelectorController>,
182    consensus_controller: Box<dyn ConsensusController>,
183    bootstrap_peers: Option<BootstrapPeers>,
184    pool_controller: Box<dyn PoolController>,
185    storage: Storage,
186    protocol_channels: ProtocolChannels,
187    mip_store: MipStore,
188    massa_metrics: MassaMetrics,
189) -> Result<(Box<dyn ProtocolManager>, KeyPair, NodeId), ProtocolError> {
190    debug!("starting protocol controller");
191    let peer_db = Arc::new(RwLock::new(PeerDB::default()));
192
193    let (sender_operations, receiver_operations) = MassaChannel::new(
194        "sender_operations".to_string(),
195        Some(config.max_size_channel_network_to_operation_handler),
196    );
197    let (sender_endorsements, receiver_endorsements) = MassaChannel::new(
198        "sender_endorsements".to_string(),
199        Some(config.max_size_channel_network_to_endorsement_handler),
200    );
201    let (sender_blocks, receiver_blocks) = MassaChannel::new(
202        "sender_blocks".to_string(),
203        Some(config.max_size_channel_network_to_block_handler),
204    );
205    let (sender_peers, receiver_peers) = MassaChannel::new(
206        "sender_peers".to_string(),
207        Some(config.max_size_channel_network_to_peer_handler),
208    );
209
210    // Register channels for handlers
211    let message_handlers: MessagesHandler = MessagesHandler {
212        sender_blocks: sender_blocks.clone(),
213        sender_endorsements: sender_endorsements.clone(),
214        sender_operations: sender_operations.clone(),
215        sender_peers: sender_peers.clone(),
216        id_deserializer: U64VarIntDeserializer::new(Included(0), Included(u64::MAX)),
217    };
218
219    // try to read node keypair from file, otherwise generate it & write to file. Then derive nodeId
220    let keypair = if std::path::Path::is_file(&config.keypair_file) {
221        // file exists: try to load it
222        let keypair_bs58_check_encoded = read_to_string(&config.keypair_file).map_err(|err| {
223            std::io::Error::new(err.kind(), format!("could not load node key file: {}", err))
224        })?;
225        serde_json::from_slice::<KeyPair>(keypair_bs58_check_encoded.as_bytes())?
226    } else {
227        // node file does not exist: generate the key and save it
228        // MERGE TODO
229        let keypair_factory = KeyPairFactory {
230            mip_store: mip_store.clone(),
231        };
232        let now = MassaTime::now();
233        let keypair = keypair_factory.create(&(), FactoryStrategy::At(now))?;
234        if let Err(e) = std::fs::write(&config.keypair_file, serde_json::to_string(&keypair)?) {
235            warn!("could not generate node key file: {}", e);
236        }
237        keypair
238    };
239
240    let mut peernet_config = PeerNetConfiguration::default(
241        MassaHandshake::new(peer_db.clone(), config.clone()),
242        message_handlers.clone(),
243        Context {
244            our_keypair: keypair.clone(),
245        },
246    );
247    peernet_config.write_timeout = config.message_timeout.to_duration();
248    peernet_config.read_timeout = config.message_timeout.to_duration();
249
250    peernet_config.rate_limit = config.rate_limit;
251    peernet_config.rate_bucket_size = config.rate_limit.saturating_mul(2);
252
253    let initial_peers_infos = serde_json::from_str::<HashMap<PeerId, PeerData>>(
254        &std::fs::read_to_string(&config.initial_peers)?,
255    )?;
256
257    let initial_peers = if let Some(bootstrap_peers) = bootstrap_peers {
258        //TODO: Remove when we will be able to test the bootstrap peer even if someone else found them full
259        bootstrap_peers
260            .0
261            .into_iter()
262            .chain(
263                initial_peers_infos
264                    .iter()
265                    .map(|(peer_id, data)| (*peer_id, data.listeners.clone())),
266            )
267            .collect()
268    } else {
269        initial_peers_infos
270            .iter()
271            .map(|(peer_id, data)| (*peer_id, data.listeners.clone()))
272            .collect()
273    };
274
275    let peernet_categories = config
276        .peers_categories
277        .iter()
278        .map(|(category_name, infos)| {
279            (
280                category_name.clone(),
281                (
282                    initial_peers_infos
283                        .iter()
284                        .filter_map(|info| {
285                            if info.1.category == *category_name {
286                                //TODO: Adapt for multiple listeners
287                                Some(
288                                    info.1
289                                        .listeners
290                                        .iter()
291                                        .next()
292                                        .map(|addr| to_canonical(addr.0.ip()))
293                                        .unwrap(),
294                                )
295                            } else {
296                                None
297                            }
298                        })
299                        .collect(),
300                    PeerNetCategoryInfo {
301                        max_in_connections: infos.max_in_connections,
302                        max_in_connections_per_ip: infos.max_in_connections_per_ip,
303                        max_out_connections: infos.target_out_connections,
304                    },
305                ),
306            )
307        })
308        .collect();
309    peernet_config.peers_categories = peernet_categories;
310    peernet_config.default_category_info = PeerNetCategoryInfo {
311        max_in_connections: config.default_category_info.max_in_connections,
312        max_in_connections_per_ip: config.default_category_info.max_in_connections_per_ip,
313        max_out_connections: config.default_category_info.target_out_connections,
314    };
315    peernet_config.max_in_connections = config.max_in_connections;
316
317    let network_controller = Box::new(NetworkControllerImpl::new(
318        PeerNetManager::new(peernet_config),
319        config.allow_local_peers(),
320    ));
321
322    let connectivity_thread_handle = start_connectivity_thread(
323        PeerId::from_public_key(keypair.get_public_key()),
324        selector_controller,
325        network_controller,
326        consensus_controller,
327        pool_controller,
328        (sender_blocks, receiver_blocks),
329        (sender_endorsements, receiver_endorsements),
330        (sender_operations, receiver_operations),
331        (sender_peers, receiver_peers),
332        initial_peers,
333        peer_db,
334        storage,
335        protocol_channels,
336        message_handlers,
337        config
338            .peers_categories
339            .iter()
340            .map(|(category_name, infos)| {
341                (
342                    category_name.clone(),
343                    (
344                        initial_peers_infos
345                            .iter()
346                            .filter_map(|info| {
347                                if info.1.category == *category_name {
348                                    //TODO: Adapt for multiple listeners
349                                    Some(
350                                        info.1
351                                            .listeners
352                                            .iter()
353                                            .next()
354                                            .map(|addr| to_canonical(addr.0.ip()))
355                                            .unwrap(),
356                                    )
357                                } else {
358                                    None
359                                }
360                            })
361                            .collect(),
362                        *infos,
363                    ),
364                )
365            })
366            .collect(),
367        config.default_category_info,
368        config,
369        mip_store,
370        massa_metrics,
371    )?;
372
373    let manager = ProtocolManagerImpl::new(connectivity_thread_handle);
374
375    Ok((
376        Box::new(manager),
377        keypair.clone(),
378        NodeId::new(keypair.get_public_key()),
379    ))
