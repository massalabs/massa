12/// Maximum number of peers recorded as direct senders of a given block.
13/// Bounds the memory used by the attribution record in case many peers send us the same block.
14const MAX_SENDERS_PER_BLOCK: usize = 32;
15
16/// The kind of block data a peer sent us.
17///
18/// Used to decide what a peer can be held responsible for: a peer that only relayed a header
19/// could not have checked the block's contents, so it must not be punished for a defect that
20/// is only visible in those contents.
21#[derive(Clone, Copy, Debug, PartialEq, Eq)]
22pub enum BlockDataKind {
23    /// the peer sent us the block header
24    Header,
25    /// the peer sent us the block contents (operation ids or full operations)
26    Contents,
27}
28
29/// Cache on block knowledge by our node and its peers
30pub struct BlockCache {
31    /// cache of previously checked headers
32    pub checked_headers: LruMap<BlockId, SecuredHeader>,
33    /// cache of blocks known by peers
34    pub blocks_known_by_peer: HashMap<PeerId, LruMap<BlockId, (bool, Instant)>>,
35    /// max number of blocks known in peer knowledge cache
36    pub max_known_blocks_by_peer: u32,
37    /// For each recently seen block, the peers that directly sent us data for it, along with
38    /// the strongest kind of data they sent (`Contents` supersedes `Header`).
39    ///
40    /// This is attribution data, not propagation state: unlike `blocks_known_by_peer` it only
41    /// records peers from which we actually received the block's header or contents, it never
42    /// records peers that merely learned about the block from us, and it is not pruned when a
43    /// peer disconnects. It is therefore the only sound basis for punitive actions.
44    pub block_senders: LruMap<BlockId, HashMap<PeerId, BlockDataKind>>,
45}
46
47impl BlockCache {
48    /// Mark a given node's knowledge of a list of blocks
49    /// as either known or unknown.
50    ///
51    /// # Arguments
52    ///
53    /// * `from_peer_id` - The peer id of the peer to mark
54    /// * `block_ids` - The list of block ids to mark
55    /// * `known` - Whether the blocks are known or unknown by the peer
56    pub fn insert_peer_known_block(
57        &mut self,
58        from_peer_id: &PeerId,
59        block_ids: &[BlockId],
60        known: bool,
61    ) {
62        let now = Instant::now();
63        let known_blocks = self
64            .blocks_known_by_peer
65            .entry(*from_peer_id)
66            .or_insert_with(|| LruMap::new(ByLength::new(self.max_known_blocks_by_peer)));
67        for block_id in block_ids {
68            known_blocks.insert(*block_id, (known, now));
69        }
70    }
71
72    /// Record that a peer directly sent us data for a given block.
73    ///
74    /// # Arguments
75    ///
76    /// * `block_id` - The block the data belongs to
77    /// * `from_peer_id` - The peer that sent us the data
78    /// * `kind` - The kind of data the peer sent us
79    pub fn insert_block_sender(
80        &mut self,
81        block_id: &BlockId,
82        from_peer_id: &PeerId,
83        kind: BlockDataKind,
84    ) {
85        let senders = self
86            .block_senders
87            .get_or_insert(*block_id, HashMap::new)
88            .expect("failed to insert block senders entry");
89        if let Some(recorded_kind) = senders.get(from_peer_id).copied() {
90            // a peer that already sent us the contents stays accountable for them
91            // even if it later sends us the header again
92            if recorded_kind == BlockDataKind::Header && kind == BlockDataKind::Contents {
93                senders.insert(*from_peer_id, kind);
94            }
95            return;
96        }
97        // bound the number of recorded senders per block
98        if senders.len() >= MAX_SENDERS_PER_BLOCK {
99            // the record is full: drop a header-only sender to make room for a contents
100            // sender, so that flooding us with headers cannot shield the peers that
101            // actually sent us the block contents from being held accountable
102            if kind == BlockDataKind::Header {
103                return;
104            }
105            let header_only_sender = senders
106                .iter()
107                .find(|(_, recorded_kind)| **recorded_kind == BlockDataKind::Header)
108                .map(|(peer_id, _)| *peer_id);
109            match header_only_sender {
110                Some(peer_id) => {
111                    senders.remove(&peer_id);
112                }
113                // all the recorded senders sent us contents: keep them
114                None => return,
115            }
116        }
117        senders.insert(*from_peer_id, kind);
118    }
119
120    /// Get the peers that directly sent us any data for a given block.
121    pub fn get_block_senders(&self, block_id: &BlockId) -> Vec<PeerId> {
122        self.block_senders
123            .peek(block_id)
124            .map(|senders| senders.keys().copied().collect())
125            .unwrap_or_default()
126    }
127
128    /// Get the peers that directly sent us the contents of a given block.
129    ///
130    /// Peers that only relayed the block header are excluded: they had no way of checking
131    /// the contents, so they must not be punished for a defect found in them.
132    pub fn get_block_contents_senders(&self, block_id: &BlockId) -> Vec<PeerId> {
133        self.block_senders
134            .peek(block_id)
135            .map(|senders| {
136                senders
137                    .iter()
138                    .filter(|(_, kind)| **kind == BlockDataKind::Contents)
139                    .map(|(peer_id, _)| *peer_id)
140                    .collect()
141            })
142            .unwrap_or_default()
143    }
144}
145
146impl BlockCache {
147    pub fn new(max_known_blocks: u32, max_known_blocks_by_peer: u32) -> Self {
148        Self {
149            checked_headers: LruMap::new(ByLength::new(max_known_blocks)),
150            blocks_known_by_peer: HashMap::new(),
151            max_known_blocks_by_peer,
152            block_senders: LruMap::new(ByLength::new(max_known_blocks)),
153        }
154    }
155
156    pub fn update_cache(&mut self, peers_connected: &HashSet<PeerId>) {
157        // Remove disconnected peers from cache
158        self.blocks_known_by_peer
159            .retain(|peer_id, _| peers_connected.contains(peer_id));
160
161        // Add new connected peers to cache
162        for peer_id in peers_connected {
163            match self.blocks_known_by_peer.entry(*peer_id) {
164                std::collections::hash_map::Entry::Occupied(_) => {}
165                std::collections::hash_map::Entry::Vacant(entry) => {
166                    entry.insert(LruMap::new(ByLength::new(self.max_known_blocks_by_peer)));
167                }
168            }
169        }
170    }
171}
