18/// Endorsements need to propagate fast, so no buffering
19struct PropagationThread {
20    receiver: MassaReceiver<EndorsementHandlerPropagationCommand>,
21    config: ProtocolConfig,
22    cache: SharedEndorsementCache,
23    active_connections: Box<dyn ActiveConnectionsTrait>,
24    endorsement_serializer: MessagesSerializer,
25}
26
27impl PropagationThread {
28    fn run(&mut self) {
29        let mut next_message = None;
30        loop {
31            // get the next message to process
32            let msg = match next_message.take() {
33                Some(msg) => msg,
34                None => match self.receiver.recv() {
35                    Ok(msg) => msg,
36                    Err(_) => {
37                        info!("Stop endorsement propagation thread");
38                        return;
39                    }
40                },
41            };
42
43            match msg {
44                // endorsements to propagate
45                EndorsementHandlerPropagationCommand::PropagateEndorsements(mut endorsements) => {
46                    // also drain any remaining propagation messages that might have accumulated,
47                    // within the per-round budget
48                    next_message = drain_propagation_commands(
49                        &self.receiver,
50                        &mut endorsements,
51                        self.config.max_endorsements_per_propagation_round,
52                    );
53                    // propagate the endorsements
54                    self.propagate_endorsements(endorsements);
55                }
56                // stop the handler
57                EndorsementHandlerPropagationCommand::Stop => {
58                    info!("Stop endorsement propagation thread");
59                    return;
60                }
61            }
62        }
63    }
64
65    /// Perform propagation of endorsements to the connected peers
66    fn propagate_endorsements(&mut self, endorsements: Storage) {
67        // get all the endorsements to send
68        let endorsements: Vec<_> = {
69            let storage_lock = endorsements.read_endorsements();
70            endorsements
71                .get_endorsement_refs()
72                .iter()
73                .filter_map(|id| storage_lock.get(id).cloned())
74                .collect()
75        };
76
77        // Drop endorsements whose inclusion slot is already too old to be worth relaying.
78        // The retrieval thread already filters those out, but other callers reach us directly
79        // (the endorsement factory, the gRPC `send_endorsements` endpoint), so applying the
80        // policy here makes it hold for every source: obsolete endorsements neither consume
81        // propagation bandwidth nor pollute `checked_endorsements`.
82        let endorsements = filter_fresh_endorsements(endorsements, &self.config);
83        if endorsements.is_empty() {
84            return;
85        }
86
87        // get connected peers
88        let peers_connected = self.active_connections.get_peer_ids_connected();
89
90        // get a write lock on the cache
91        let mut cache_write = self.cache.write();
92
93        // mark that we have checked those endorsements
94        for endorsement in &endorsements {
95            cache_write.checked_endorsements.insert(endorsement.id, ());
96        }
97
98        // Add peers that potentially don't exist in cache and remove the ones that disconnected
99        cache_write.update_cache(&peers_connected);
100
101        // Propagate to peers
102        'peer_loop: for peer_id in peers_connected {
103            // write access to the cache of which endorsements are known by the peer
104            let peer_knowledge = cache_write
105                .endorsements_known_by_peer
106                .get_mut(&peer_id)
107                .expect("update_cache should have added connected peer to cache");
108
109            // get endorsements that are not known by the peer
110            let to_send: Vec<_> = endorsements
111                .iter()
112                .filter(|endorsement| peer_knowledge.peek(&endorsement.id).is_none())
113                .collect();
114
115            if to_send.is_empty() {
116                // nothing to send to that peer, try the next one
117                continue 'peer_loop;
118            }
119
120            // send by chunks
121            for chunk in to_send.chunks(self.config.max_endorsements_per_message as usize) {
122                if let Err(err) = self.active_connections.send_to_peer(
123                    &peer_id,
124                    &self.endorsement_serializer,
125                    EndorsementMessage::Endorsements(chunk.iter().map(|&e| e.clone()).collect())
126                        .into(),
127                    false,
128                ) {
129                    warn!(
130                        "could not send endorsements batch to node {}: {}",
131                        peer_id, err
132                    );
133                    // try with next peer, this one is probably congested
134                    continue 'peer_loop;
135                }
136                // sent successfully: mark peer as knowing the endorsements that were sent to it
137                for endorsement in chunk {
138                    peer_knowledge.insert(endorsement.id, ());
139                }
140            }
141        }
142    }
143}

145/// Keep only the endorsements that are still fresh enough to be propagated, logging the
146/// discarded ones (see `is_endorsement_fresh`).
147fn filter_fresh_endorsements(
148    endorsements: Vec<SecureShareEndorsement>,
149    config: &ProtocolConfig,
150) -> Vec<SecureShareEndorsement> {
151    let now = MassaTime::now();
152    endorsements
153        .into_iter()
154        .filter(|endorsement| {
155            let fresh = is_endorsement_fresh(endorsement, config, now);
156            if !fresh {
157                debug!(
158                    "not propagating endorsement {}: its inclusion slot {} is too old",
159                    endorsement.id, endorsement.content.slot
160                );
161            }
162            fresh
163        })
164        .collect()
165}

167/// Merge the queued `PropagateEndorsements` commands into `endorsements` until the channel is
168/// empty or `max_endorsements_per_propagation_round` endorsements have been accumulated.
169///
170/// Returns a message that was pulled from the channel but not consumed by this round (a
171/// non-propagation command), which the caller must process next.
172///
173/// Bounding the drain keeps a single propagation round from growing unboundedly: without it a
174/// sustained endorsement source could make the batch (and the per-peer work derived from it)
175/// arbitrarily large. Endorsements left in the channel are not dropped, they are propagated by
176/// the next loop iteration.
177fn drain_propagation_commands(
178    receiver: &MassaReceiver<EndorsementHandlerPropagationCommand>,
179    endorsements: &mut Storage,
180    max_endorsements_per_propagation_round: usize,
181) -> Option<EndorsementHandlerPropagationCommand> {
182    while endorsements.get_endorsement_refs().len() < max_endorsements_per_propagation_round {
183        match receiver.try_recv() {
184            // we got more endorsements to propagate: extend the buffer
185            Ok(EndorsementHandlerPropagationCommand::PropagateEndorsements(new_endorsements)) => {
186                endorsements.extend(new_endorsements);
187            }
188            // we grabbed a message that is not a propagation message, mark it for processing
189            Ok(other_msg) => return Some(other_msg),
190            // nothing left to merge
191            Err(_) => break,
192        }
193    }
194    None
195}

197pub fn start_propagation_thread(
198    receiver: MassaReceiver<EndorsementHandlerPropagationCommand>,
199    cache: SharedEndorsementCache,
200    config: ProtocolConfig,
201    active_connections: Box<dyn ActiveConnectionsTrait>,
202) -> JoinHandle<()> {
203    std::thread::Builder::new()
204        .name(THREAD_NAME.to_string())
205        .spawn(move || {
206            let endorsement_serializer = MessagesSerializer::new()
207                .with_endorsement_message_serializer(EndorsementMessageSerializer::new());
208            let mut propagation_thread = PropagationThread {
209                receiver,
210                config,
211                active_connections,
212                cache,
213                endorsement_serializer,
214            };
215            propagation_thread.run();
216        })
217        .expect("OS failed to start endorsement propagation thread")
218}

233    /// Build an endorsement, `index` making it unique so that the storage does not deduplicate it
234    fn endorsement(index: u32) -> SecureShareEndorsement {
235        endorsement_at_slot(index, Slot::new(1, 0))

238    /// Build an endorsement for the given inclusion slot
239    fn endorsement_at_slot(index: u32, slot: Slot) -> SecureShareEndorsement {
240        let keypair = KeyPair::generate(0).unwrap();
241        let content = Endorsement {
242            slot,
243            index,
244            endorsed_block: BlockId::generate_from_hash(Hash::compute_from(b"block")),
245        };
246        Endorsement::new_verifiable(content, EndorsementSerializer::new(), &keypair, 0, None)
247            .unwrap()
248    }

250    /// Build a storage sharing `root`'s objects and holding `count` distinct endorsements
251    fn storage_with(root: &Storage, count: u32) -> Storage {
252        let mut storage = root.clone_without_refs();
253        storage.store_endorsements((0..count).map(endorsement).collect());
254        storage
255    }

279        let root = Storage::create_root();
