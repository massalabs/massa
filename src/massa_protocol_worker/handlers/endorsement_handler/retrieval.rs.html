44    receiver_ext: MassaReceiver<EndorsementHandlerRetrievalCommand>,

50    storage: Storage,

57impl RetrievalThread {
58    fn run(&mut self) {
59        // regular interval ticks for metrics
60        let tick_metrics = tick(self.metrics.tick_delay);
61
62        loop {
63            select! {
64                recv(self.receiver) -> msg => {
65                    self.receiver.update_metrics();
66                    match msg {
67                        Ok((peer_id, message)) => self.process_message(peer_id, message),
68                        Err(_) => {
69                            info!("Stop endorsement retrieval thread");
70                            return;
71                        }
72                    }
73                },
74                recv(self.receiver_ext) -> msg => {
75                    self.receiver_ext.update_metrics();
76                    match msg {
77                        Ok(msg) => {
78                            match msg {
79                                EndorsementHandlerRetrievalCommand::Stop => {
80                                    info!("Stop endorsement retrieval thread");
81                                    return;
82                                }
83                            }
84                        }
85                        Err(_) => {
86                            info!("Stop endorsement retrieval thread");
87                            return;
88                        }
89                    }
90                },
91                recv(tick_metrics) -> _ => {
92                    // update metrics
93                    let cache_lock = self.cache.read();
94                    let count = cache_lock
95                        .endorsements_known_by_peer
96                        .values()
97                        .map(|v| v.len())
98                        .sum();
99                    self.metrics
100                        .set_endorsements_cache_metrics(cache_lock.checked_endorsements.len(), count);
101                }
102            }
103        }
104    }
105
106    /// Process incoming message
107    fn process_message(&mut self, peer_id: PeerId, message: Vec<u8>) {
108        let (rest, message) = match self
109            .endorsement_message_deserializer
110            .deserialize::<DeserializeError>(&message)
111        {
112            Ok((rest, message)) => (rest, message),
113            Err(err) => {
114                debug!(
115                    "Error while deserializing message from peer {} err: {:?}",
116                    peer_id, err
117                );
118                return;
119            }
120        };
121        if !rest.is_empty() {
122            // A compliant peer never sends trailing bytes, so ban the sender.
123            warn!(
124                "peer {} sent an endorsement message with {} unexpected trailing byte(s); banning it",
125                peer_id,
126                rest.len()
127            );
128            if let Err(err) = self.ban_peer(&peer_id) {
129                warn!("Error while banning peer {} err: {:?}", peer_id, err);
130            }
131            return;
132        }
133        match message {
134            EndorsementMessage::Endorsements(endorsements) => {
135                debug!("Received endorsement message: Endorsement from {}", peer_id);
136                // Discard endorsements that are already too old to be useful before doing
137                // any expensive work on them (signature verification, PoS draw lookup):
138                // otherwise a peer could replay arbitrary amounts of old endorsements
139                // to make us burn CPU, cycling through IDs faster than the
140                // `checked_endorsements` cache can remember them.
141                let now = MassaTime::now();
142                let endorsements: Vec<_> = endorsements
143                    .into_iter()
144                    .filter(|endorsement| is_endorsement_fresh(endorsement, &self.config, now))
145                    .collect();
146                if endorsements.is_empty() {
147                    return;
148                }
149                if let Err(err) = note_endorsements_from_peer(
150                    endorsements,
151                    &peer_id,
152                    &self.cache,
153                    self.selector_controller.as_ref(),
154                    &self.storage,
155                    &self.config,
156                    &self.internal_sender,
157                    self.pool_controller.as_mut(),
158                    &self.mip_store,
159                ) {
160                    warn!(
161                        "peer {} sent us critically incorrect endorsements, \
162                        which may be an attack attempt by the remote node or a \
163                        loss of sync between us and the remote node. Err = {}",
164                        peer_id, err
165                    );
166                    if let Err(err) = self.ban_peer(&peer_id) {
167                        warn!("Error while banning peer {} err: {:?}", peer_id, err);
168                    }
169                }
170            }
171        }
172    }
173
174    /// send a ban peer command to the peer handler
175    fn ban_peer(&mut self, peer_id: &PeerId) -> Result<(), ProtocolError> {
176        massa_trace!("ban node from retrieval thread", { "peer_id": peer_id.to_string() });
177        self.peer_cmd_sender
178            .try_send(PeerManagementCmd::Ban(vec![*peer_id]))
179            .map_err(|err| ProtocolError::SendError(err.to_string()))
180    }
181}

183/// Note endorsements coming from a given node,
184/// and propagate them when they were received outside of a header.
185///
186/// Caches knowledge of valid ones.
187///
188/// Returns an error if the peer must be banned.
189///
190/// Checks performed:
191/// - Valid signature.
192/// - Endorsement slot not too far in the future (peer offense if it is).
193/// - The creator is the endorser drawn for the endorsement's `(slot, index)` pair, when the draws
194///   of that slot are locally available.
195/// - At most `MAX_ENDORSEMENTS_PER_SLOT_INDEX` distinct endorsements are noted for a given draw,
196///   so that an equivocating endorser cannot have us relay unlimited variants of its endorsement.
197#[allow(clippy::too_many_arguments)]
198pub(crate) fn note_endorsements_from_peer(
199    endorsements: Vec<SecureShareEndorsement>,
200    from_peer_id: &PeerId,
201    cache: &SharedEndorsementCache,
202    selector_controller: &dyn SelectorController,
203    storage: &Storage,
204    config: &ProtocolConfig,
205    endorsement_propagation_sender: &MassaSender<EndorsementHandlerPropagationCommand>,
206    pool_controller: &mut dyn PoolController,
207    mip_store: &MipStore,
208) -> Result<(), ProtocolError> {
209    let mut new_endorsements = PreHashMap::with_capacity(endorsements.len());
210    let mut all_endorsement_ids = PreHashSet::with_capacity(endorsements.len());
211
212    // cache check
213    {
214        let cache_read = cache.read();
215        for endorsement in endorsements.into_iter() {
216            let endorsement_id = endorsement.id;
217            all_endorsement_ids.insert(endorsement_id);
218
219            // only consider the endorsement as new if we have not already checked it
220            if cache_read
221                .checked_endorsements
222                .peek(&endorsement_id)
223                .is_none()
224            {
225                new_endorsements.insert(endorsement_id, endorsement);
226            }
227        }
228    }
229
230    // Batch signature verification. Layout is per-endorsement slot: the
231    // network-agreed Execution component version at that slot timestamp.
232    let sig_batch = new_endorsements
233        .values()
234        .map(|endorsement| {
235            let slot_ts = get_block_slot_timestamp(
236                config.thread_count,
237                config.t0,
238                config.genesis_timestamp,
239                endorsement.content.slot,
240            )?;
241            let sig_chain_id = sig_chain_id_for_slot(mip_store, config.chain_id, slot_ts);
242            Ok((
243                endorsement.compute_signed_hash(sig_chain_id),
244                endorsement.signature,
245                endorsement.content_creator_pub_key,
246            ))
247        })
248        .collect::<Result<Vec<_>, ProtocolError>>()?;
249    verify_sigs_batch(&sig_batch)?;
250
251    let now = MassaTime::now();
252
253    // Check the endorsement slots and the PoS draws.
254    //
255    // An endorsement is created at its own slot, so its slot can only be as far ahead of our clock
256    // as the propagation window, plus one period of tolerance for clock drift. Beyond that the
257    // endorsement is invalid whatever our local state is, and propagating it would amplify it, so we
258    // fail the whole batch and the caller bans the peer.
259    //
260    // Endorsements that are too old, and endorsements whose cycle draws are not available locally,
261    // are only dropped: both cases happen to honest peers (respectively a slow propagation and a
262    // selector of ours that has not caught up yet). They are not marked as checked either, so that
263    // they get validated again if we see them later.
264    let slot_acceptance_margin = config
265        .max_endorsements_propagation_time
266        .saturating_add(config.t0);
267    let mut unchecked_endorsements = PreHashSet::with_capacity(new_endorsements.len());
268    for (endorsement_id, endorsement) in new_endorsements.iter() {
269        let slot = endorsement.content.slot;
270        let slot_timestamp = get_block_slot_timestamp(
271            config.thread_count,
272            config.t0,
273            config.genesis_timestamp,
274            slot,
275        )
276        .map_err(|err| {
277            ProtocolError::GeneralProtocolError(format!(
278                "Endorsement slot {} does not have a valid timestamp: {}",
279                slot, err
280            ))
281        })?;
282        if slot_timestamp > now.saturating_add(slot_acceptance_margin) {
283            return Err(ProtocolError::GeneralProtocolError(format!(
284                "Endorsement slot {} is too far in the future",
285                slot
286            )));
287        }
288        if slot_timestamp.saturating_add(slot_acceptance_margin) < now {
289            debug!(
290                "Ignoring endorsement {} from peer {}: its slot {} is too old",
291                endorsement_id, from_peer_id, slot
292            );
293            unchecked_endorsements.insert(*endorsement_id);
294            continue;
295        }
296
297        let selection = match selector_controller.get_selection(slot) {
298            Ok(selection) => selection.endorsements,
299            Err(err) => {
300                debug!(
301                    "Could not check the PoS draws of endorsement {} at slot {} because our local \
302                    selector is not ready: {}. Ignoring it without banning peer {}.",
303                    endorsement_id, slot, err, from_peer_id
304                );
305                unchecked_endorsements.insert(*endorsement_id);
306                continue;
307            }
308        };
309        let Some(address) = selection.get(endorsement.content.index as usize) else {
310            return Err(ProtocolError::GeneralProtocolError(format!(
311                "No selection on slot {} for index {}",
312                slot, endorsement.content.index
313            )));
314        };
315        if address != &endorsement.content_creator_address {
316            return Err(ProtocolError::GeneralProtocolError(format!(
317                "Invalid endorsement producer selection: expected address {}, got {}",
318                address, endorsement.content_creator_address
319            )));
320        }
321    }
322    new_endorsements.retain(|endorsement_id, _| !unchecked_endorsements.contains(endorsement_id));
323
324    // From there we note new endorsements and propagate them
325
326    // Filter out endorsements if they are too old (max age of the inclusion slot: `max_endorsements_propagation_time`)
327    new_endorsements.retain(|_id, endorsement| is_endorsement_fresh(endorsement, config, now));
328
329    {
330        let mut cache_write = cache.write();
331
332        // add to the cache of endorsements we have checked
333        for endorsement_id in all_endorsement_ids
334            .iter()
335            .filter(|endorsement_id| !unchecked_endorsements.contains(endorsement_id))
336        {
337            cache_write.insert_checked_endorsement(*endorsement_id);
338        }
339
340        // add to the cache of endorsements known by the source node
341        cache_write.insert_peer_known_endorsements(
342            from_peer_id,
343            &all_endorsement_ids.iter().copied().collect::<Vec<_>>(),
344        );
345
346        // Drop conflicting endorsements beyond the per-draw bound. Nothing here constrains the
347        // endorsed block, so the drawn endorser of a `(slot, index)` pair can sign arbitrarily many
348        // valid endorsements differing only by that field, each with its own id: without this bound
349        // they would all be treated as new data, noted and gossiped further. We keep a few variants
350        // so the denunciation pool still gets to see the equivocation, and silently ignore the
351        // rest: equivocating is denounceable, not a reason to ban the peer that relayed it to us.
352        new_endorsements.retain(|id, endorsement| {
353            let accepted = cache_write.register_draw_endorsement(
354                endorsement.content.slot,
355                endorsement.content.index,
356                *id,
357            );
358            if !accepted {
359                debug!(
360                    "ignoring endorsement {} from peer {}: too many conflicting endorsements for the draw at slot {} index {}",
361                    id, from_peer_id, endorsement.content.slot, endorsement.content.index
362                );
363            }
364            accepted
365        });
366    }
367
368    if new_endorsements.is_empty() {
369        // no endorsements to note or propagate
370        return Ok(());
371    }
372
373    // Store new endorsements
374    let mut endorsement_store = storage.clone_without_refs();
375    endorsement_store.store_endorsements(new_endorsements.into_values().collect());
376
377    // Propagate to other peers
378    if let Err(err) = endorsement_propagation_sender.try_send(
379        EndorsementHandlerPropagationCommand::PropagateEndorsements(endorsement_store.clone()),
380    ) {
381        warn!(
382            "Failed to send from retrieval thread of endorsement handler to propagation: {:?}",
383            err
384        );
385    }
386
387    // Add to pool
388    pool_controller.add_endorsements(endorsement_store);
389
390    Ok(())
391}

393#[allow(clippy::too_many_arguments)]
394pub fn start_retrieval_thread(
395    receiver: MassaReceiver<PeerMessageTuple>,
396    receiver_ext: MassaReceiver<EndorsementHandlerRetrievalCommand>,
397    internal_sender: MassaSender<EndorsementHandlerPropagationCommand>,
398    peer_cmd_sender: MassaSender<PeerManagementCmd>,
399    cache: SharedEndorsementCache,
400    selector_controller: Box<dyn SelectorController>,
401    pool_controller: Box<dyn PoolController>,
402    config: ProtocolConfig,
403    storage: Storage,
404    metrics: MassaMetrics,
405    mip_store: MipStore,
406) -> JoinHandle<()> {
407    let endorsement_message_deserializer =
408        EndorsementMessageDeserializer::new(EndorsementMessageDeserializerArgs {
409            thread_count: config.thread_count,
410            max_length_endorsements: config.max_endorsements_per_message,
411            endorsement_count: config.endorsement_count,
412            chain_id: config.chain_id,
413        });
414    std::thread::Builder::new()
415        .name(THREAD_NAME.to_string())
416        .spawn(move || {
417            let mut retrieval_thread = RetrievalThread {
418                receiver,
419                receiver_ext,
420                peer_cmd_sender,
421                cache,
422                internal_sender,
423                selector_controller,
424                pool_controller,
425                config,
426                storage,
427                metrics,
428                mip_store,
429                endorsement_message_deserializer,
430            };
431            retrieval_thread.run();
432        })
433        .expect("OS failed to start endorsement retrieval thread")
434}
