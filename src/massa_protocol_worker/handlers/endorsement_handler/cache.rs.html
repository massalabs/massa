6use massa_models::{
7    config::MAX_ENDORSEMENTS_PER_SLOT_INDEX, endorsement::EndorsementId, prehash::PreHashSet,
8    slot::Slot,
9};
10use massa_protocol_exports::PeerId;
11use parking_lot::RwLock;
12use schnellru::{ByLength, LruMap};
13
14/// Cache of endorsements
15pub struct EndorsementCache {
16    /// List of endorsements we checked recently
17    pub checked_endorsements: LruMap<EndorsementId, ()>,
18    /// Endorsements accepted for a given `(inclusion slot, index)` draw, used to bound how many
19    /// conflicting variants a single drawn endorser can push through us
20    pub endorsements_by_draw: LruMap<(Slot, u32), PreHashSet<EndorsementId>>,
21    /// List of endorsements known by peers
22    pub endorsements_known_by_peer: HashMap<PeerId, LruMap<EndorsementId, ()>>,
23    /// Maximum number of endorsements known by a peer
24    pub max_known_endorsements_by_peer: u32,
25}
26
27impl EndorsementCache {
28    /// Create a new EndorsementCache
29    pub fn new(max_known_endorsements: u32, max_known_endorsements_by_peer: u32) -> Self {
30        Self {
31            checked_endorsements: LruMap::new(ByLength::new(max_known_endorsements)),
32            // one entry per draw holds up to `MAX_ENDORSEMENTS_PER_SLOT_INDEX` ids, so scaling the
33            // capacity down keeps this index's memory in line with `checked_endorsements` while
34            // covering the same horizon of endorsements
35            endorsements_by_draw: LruMap::new(ByLength::new(std::cmp::max(
36                max_known_endorsements / MAX_ENDORSEMENTS_PER_SLOT_INDEX as u32,
37                1,
38            ))),
39            endorsements_known_by_peer: HashMap::new(),
40            max_known_endorsements_by_peer,
41        }
42    }
43
44    /// Mark a list of endorsement IDs prefixes as known by a peer
45    pub fn insert_peer_known_endorsements(
46        &mut self,
47        peer_id: &PeerId,
48        endorsements: &[EndorsementId],
49    ) {
50        let known_endorsements = self
51            .endorsements_known_by_peer
52            .entry(*peer_id)
53            .or_insert_with(|| LruMap::new(ByLength::new(self.max_known_endorsements_by_peer)));
54        for endorsement in endorsements {
55            known_endorsements.insert(*endorsement, ());
56        }
57    }
58
59    /// Mark an endorsement ID as checked by us
60    pub fn insert_checked_endorsement(&mut self, enrodsement_id: EndorsementId) {
61        self.checked_endorsements.insert(enrodsement_id, ());
62    }
63
64    /// Register an endorsement against its `(slot, index)` draw and tell whether we accept it.
65    ///
66    /// The endorsement id is derived from the whole signed content, so an equivocating endorser can
67    /// mint unlimited distinct-but-valid endorsements for the draw it won just by varying the
68    /// endorsed block: deduplicating on `checked_endorsements` alone does not stop them from being
69    /// noted and gossiped further. Returns `false` once `MAX_ENDORSEMENTS_PER_SLOT_INDEX` distinct
70    /// endorsements have already been registered for that draw.
71    ///
72    /// Registering the id rather than only counting keeps this idempotent, so an endorsement that
73    /// gets re-checked after being evicted from `checked_endorsements` does not consume the budget
74    /// twice.
75    pub fn register_draw_endorsement(
76        &mut self,
77        slot: Slot,
78        index: u32,
79        endorsement_id: EndorsementId,
80    ) -> bool {
81        let Some(draw_endorsements) = self
82            .endorsements_by_draw
83            .get_or_insert((slot, index), PreHashSet::default)
84        else {
85            return false;
86        };
87        if draw_endorsements.contains(&endorsement_id) {
88            return true;
89        }
90        if draw_endorsements.len() >= MAX_ENDORSEMENTS_PER_SLOT_INDEX {
91            return false;
92        }
93        draw_endorsements.insert(endorsement_id);
94        true
95    }
96
97    /// Update caches to remove all data from disconnected peers
98    pub fn update_cache(&mut self, peers_connected: &HashSet<PeerId>) {
99        // Remove disconnected peers from cache
100        self.endorsements_known_by_peer
101            .retain(|peer_id, _| peers_connected.contains(peer_id));
102
103        // Add new connected peers to cache
104        for peer_id in peers_connected {
105            match self.endorsements_known_by_peer.entry(*peer_id) {
106                std::collections::hash_map::Entry::Occupied(_) => {}
107                std::collections::hash_map::Entry::Vacant(entry) => {
108                    entry.insert(LruMap::new(ByLength::new(
109                        self.max_known_endorsements_by_peer,
110                    )));
111                }
112            }
113        }
114    }
115}
116
117pub type SharedEndorsementCache = Arc<RwLock<EndorsementCache>>;
