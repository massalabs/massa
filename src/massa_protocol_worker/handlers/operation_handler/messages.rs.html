15#[derive(Debug)]
16pub enum OperationMessage {
17    /// Batch of operation ids
18    OperationsAnnouncement(OperationPrefixIds),
19    /// Someone ask for operations.
20    AskForOperations(OperationPrefixIds),
21    /// A list of operations
22    Operations(Vec<SecureShareOperation>),
23}
24
25#[derive(IntoPrimitive, Debug, Eq, PartialEq, TryFromPrimitive)]
26#[repr(u64)]
27pub enum MessageTypeId {
28    OperationsAnnouncement = 0,
29    AskForOperations = 1,
30    Operations = 2,
31}
32
33impl From<&OperationMessage> for MessageTypeId {
34    fn from(message: &OperationMessage) -> Self {
35        match message {
36            OperationMessage::OperationsAnnouncement(_) => MessageTypeId::OperationsAnnouncement,
37            OperationMessage::AskForOperations(_) => MessageTypeId::AskForOperations,
38            OperationMessage::Operations(_) => MessageTypeId::Operations,
39        }
40    }
41}

43#[derive(Default, Clone)]
44pub struct OperationMessageSerializer {
45    id_serializer: U64VarIntSerializer,
46    operation_prefix_ids_serializer: OperationPrefixIdsSerializer,
47    operations_serializer: OperationsSerializer,
48}
49
50impl OperationMessageSerializer {
51    pub fn new() -> Self {
52        Self {
53            id_serializer: U64VarIntSerializer::new(),
54            operation_prefix_ids_serializer: OperationPrefixIdsSerializer::new(),
55            operations_serializer: OperationsSerializer::new(),
56        }
57    }
58}
59
60impl Serializer<OperationMessage> for OperationMessageSerializer {
61    fn serialize(
62        &self,
63        value: &OperationMessage,
64        buffer: &mut Vec<u8>,
65    ) -> Result<(), SerializeError> {
66        self.id_serializer
67            .serialize(&MessageTypeId::from(value).into(), buffer)?;
68        match value {
69            OperationMessage::OperationsAnnouncement(operations) => {
70                self.operation_prefix_ids_serializer
71                    .serialize(operations, buffer)?;
72            }
73            OperationMessage::AskForOperations(operations) => {
74                self.operation_prefix_ids_serializer
75                    .serialize(operations, buffer)?;
76            }
77            OperationMessage::Operations(operations) => {
78                self.operations_serializer.serialize(operations, buffer)?;
79            }
80        }
81        Ok(())
82    }
83}

91/// Limits used in the deserialization of `OperationMessage`
92pub struct OperationMessageDeserializerArgs {
93    /// Maximum number of prefix ids that can be asked to propagate or sent
94    pub max_operations_prefix_ids: u32,
95    /// Maximum of full operations sent in one message
96    pub max_operations: u32,
97    //TODO: All of this arguments should be in a `OperationDeserializer` struct that would be used here

99    pub max_bytecode_size: u64,
100    /// Maximum size of a function name
101    pub max_function_name_length: u16,
102    /// Maximum size of parameters
103    pub max_parameters_size: u32,
104    /// Maximum number of entries in the op datastore
105    pub max_op_datastore_entry_count: u64,
106    /// Maximum size of a op datastore key
107    pub max_op_datastore_key_length: u8,
108    /// Maximum size of a op datastore value
109    pub max_op_datastore_value_length: u64,
110    /// Chain id
111    pub chain_id: u64,

114impl OperationMessageDeserializer {

131        }

135impl Deserializer<OperationMessage> for OperationMessageDeserializer {
136    fn deserialize<'a, E: ParseError<&'a [u8]> + ContextError<&'a [u8]>>(
137        &self,
138        buffer: &'a [u8],
139    ) -> IResult<&'a [u8], OperationMessage, E> {
140        context("Failed OperationMessage deserialization", |buffer| {
141            let (buffer, raw_id) = self.id_deserializer.deserialize(buffer)?;
142            let id = MessageTypeId::try_from(raw_id).map_err(|_| {
143                nom::Err::Error(ParseError::from_error_kind(
144                    buffer,
145                    nom::error::ErrorKind::Eof,
146                ))
147            })?;
148            match id {
149                MessageTypeId::AskForOperations => {
150                    context("Failed AskForOperations deserialization", |input| {
151                        self.operation_prefix_ids_deserializer.deserialize(input)
152                    })
153                    .map(OperationMessage::AskForOperations)
154                    .parse(buffer)
155                }
156                MessageTypeId::OperationsAnnouncement => {
157                    context("Failed OperationsAnnouncement deserialization", |input| {
158                        self.operation_prefix_ids_deserializer.deserialize(input)
159                    })
160                    .map(OperationMessage::OperationsAnnouncement)
161                    .parse(buffer)
162                }
163                MessageTypeId::Operations => {
164                    context("Failed Operations deserialization", |input| {
165                        self.operations_deserializer.deserialize(input)
166                    })
167                    .map(OperationMessage::Operations)
168                    .parse(buffer)
169                }
170            }
171        })
172        .parse(buffer)
173    }
