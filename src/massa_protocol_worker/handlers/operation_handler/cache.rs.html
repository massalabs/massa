1use std::{
2    collections::{HashMap, HashSet},
3    sync::Arc,
4};
5
6use massa_models::operation::{OperationId, OperationPrefixId};
7use massa_protocol_exports::PeerId;
8use parking_lot::RwLock;
9use schnellru::{ByLength, LruMap};
10
11/// Cache for operations
12pub struct OperationCache {
13    /// List of operations we checked recently
14    pub checked_operations: LruMap<OperationId, ()>,
15    /// List of operation ID prefixes we checked recently
16    pub checked_operations_prefix: LruMap<OperationPrefixId, ()>,
17    /// List of operations known by peers
18    pub ops_known_by_peer: HashMap<PeerId, LruMap<OperationPrefixId, ()>>,
19    /// Maximum number of operations known by a peer
20    pub max_known_ops_by_peer: u32,
21}
22
23impl OperationCache {
24    /// Create a new OperationCache
25    pub fn new(max_known_ops: u32, max_known_ops_by_peer: u32) -> Self {
26        Self {
27            checked_operations: LruMap::new(ByLength::new(max_known_ops)),
28            checked_operations_prefix: LruMap::new(ByLength::new(max_known_ops)),
29            ops_known_by_peer: HashMap::new(),
30            max_known_ops_by_peer,
31        }
32    }
33
34    /// Mark a list of operation ID prefixes as known by a peer
35    pub fn insert_peer_known_ops(&mut self, peer_id: &PeerId, ops: &[OperationPrefixId]) {
36        let known_ops = self
37            .ops_known_by_peer
38            .entry(*peer_id)
39            .or_insert_with(|| LruMap::new(ByLength::new(self.max_known_ops_by_peer)));
40        for op in ops {
41            known_ops.insert(*op, ());
42        }
43    }
44
45    /// Mark an operation ID as checked by us
46    pub fn insert_checked_operation(&mut self, operation_id: OperationId) {
47        self.checked_operations.insert(operation_id, ());
48        self.checked_operations_prefix
49            .insert(operation_id.prefix(), ());
50    }
51
52    /// Update caches to remove all data from disconnected peers
53    pub fn update_cache(&mut self, peers_connected: &HashSet<PeerId>) {
54        // Remove disconnected peers from cache
55        self.ops_known_by_peer
56            .retain(|peer_id, _| peers_connected.contains(peer_id));
57
58        // Add new connected peers to cache
59        for peer_id in peers_connected {
60            match self.ops_known_by_peer.entry(*peer_id) {
61                std::collections::hash_map::Entry::Occupied(_) => {}
62                std::collections::hash_map::Entry::Vacant(entry) => {
63                    entry.insert(LruMap::new(ByLength::new(self.max_known_ops_by_peer)));
64                }
65            }
66        }
67    }
68}
69
70pub type SharedOperationCache = Arc<RwLock<OperationCache>>;
