30/// Returns the subset of operation ids in `storage` whose serialized size is
31/// within `max_serialized_operations_size_per_block`, and releases `storage`'s
32/// references to the rejected operations.
33///
34/// Operations exceeding the protocol size limit are excluded (and logged) so
35/// they are never marked checked, stored for propagation, or announced. This
36/// mirrors the size check already enforced on the peer-received (retrieval)
37/// path, and protects locally submitted operations (JSON-RPC / gRPC), which are
38/// not validated against this limit on submission.
39///
40/// The rejected references must not be merged into the propagation storage:
41/// only queued operations are ever pruned from it, so kept-alive refs of
42/// filtered-out operations would pin their content until shutdown.
43fn filter_propagatable_ops(
44    storage: &mut Storage,
45    max_serialized_operations_size_per_block: usize,
46) -> PreHashSet<OperationId> {
47    let kept: PreHashSet<OperationId> = {
48        let ops_read = storage.read_operations();
49        storage
50            .get_op_refs()
51            .iter()
52            .copied()
53            .filter(|op_id| match ops_read.get(op_id) {
54                Some(op) => {
55                    let size = op.serialized_size();
56                    let ok = size <= max_serialized_operations_size_per_block;
57                    if !ok {
58                        warn!(
59                            "not propagating operation {}: serialized size {} exceeds max {} bytes",
60                            op_id, size, max_serialized_operations_size_per_block
61                        );
62                    }
63                    ok
64                }
65                None => false,
66            })
67            .collect()
68    };
69
70    let rejected: PreHashSet<OperationId> = storage
71        .get_op_refs()
72        .iter()
73        .copied()
74        .filter(|op_id| !kept.contains(op_id))
75        .collect();
76    storage.drop_operation_refs(&rejected);
77
78    kept
79}

81struct PropagationThread {
82    internal_receiver: MassaReceiver<OperationHandlerPropagationCommand>,
83    active_connections: Box<dyn ActiveConnectionsTrait>,
84    // times at which previous ops were announced
85    stored_for_propagation: VecDeque<(std::time::Instant, PreHashSet<OperationId>)>,
86    // ids of the operations held in `stored_for_propagation`, maintained incrementally
87    // so pruning does not need to scan the whole queue on every call. The queue batches
88    // are disjoint, so this is both the set of operations referenced in `op_storage` and
89    // the number of operations the cache size is capped on.
90    stored_ops: PreHashSet<OperationId>,
91    op_storage: Storage,
92    next_batch: PreHashSet<OperationId>,
93    config: ProtocolConfig,
94    cache: SharedOperationCache,
95    operation_message_serializer: MessagesSerializer,
96    _massa_metrics: MassaMetrics,
97}
98
99impl PropagationThread {
100    /// Time at which the next announcement should happen if the batch is not full before that.
101    fn next_batch_deadline(&self) -> std::time::Instant {
102        std::time::Instant::now()
103            .checked_add(self.config.operation_announcement_interval.to_duration())
104            .expect("Can't init interval op propagation")
105    }
106
107    fn run(&mut self) {
108        let mut batch_deadline = self.next_batch_deadline();
109        loop {
110            match self.internal_receiver.recv_deadline(batch_deadline) {
111                Ok(internal_message) => {
112                    match internal_message {
113                        OperationHandlerPropagationCommand::PropagateOperations(mut operations) => {
114                            // Re-validate the protocol size limit before propagating.
115                            // Locally submitted operations (JSON-RPC / gRPC) are not
116                            // checked against `max_serialized_operations_size_per_block`
117                            // on submission, so an oversized-but-otherwise-valid
118                            // operation could be marked checked, stored and announced,
119                            // making honest peers ban us for advertising invalid data.
120                            // This mirrors the check already enforced on the retrieval
121                            // (peer-received) path. The rejected operations' references
122                            // are released here too, so they are never merged into
123                            // `op_storage` (which only prunes operations it queued).
124                            let new_ops = filter_propagatable_ops(
125                                &mut operations,
126                                self.config.max_serialized_operations_size_per_block,
127                            );
128
129                            // Note valid operations as checked.
130                            {
131                                let mut cache_write = self.cache.write();
132                                for op_id in new_ops.iter().copied() {
133                                    cache_write.insert_checked_operation(op_id);
134                                }
135                            }
136
137                            // an operation can be propagated again while it is still
138                            // queued: only queue the ones that are not already tracked, so
139                            // that queue membership stays aligned with the `op_storage`
140                            // refs and duplicates do not consume the cache budget twice
141                            let newly_tracked: PreHashSet<OperationId> = new_ops
142                                .iter()
143                                .filter(|op_id| self.stored_ops.insert(**op_id))
144                                .copied()
145                                .collect();
146                            if !newly_tracked.is_empty() {
147                                // add to propagation storage
148                                self.stored_for_propagation
149                                    .push_back((std::time::Instant::now(), newly_tracked));
150                            }
151                            self.op_storage.extend(operations);
152                            self.prune_propagation_storage();
153
154                            for op_id in new_ops {
155                                self.next_batch.insert(op_id);
156                                if self.next_batch.len()
157                                    >= self.config.operation_announcement_buffer_capacity
158                                {
159                                    self.announce_ops();
160                                    batch_deadline = self.next_batch_deadline();
161                                }
162                            }
163
164                            // `recv_deadline` returns immediately whenever a message is ready,
165                            // so a continuously non-empty channel would never let the timeout
166                            // branch below fire. Enforce the deadline here as well to make sure
167                            // announcements keep happening under a constant flow of commands.
168                            if std::time::Instant::now() >= batch_deadline {
169                                self.announce_ops();
170                                batch_deadline = self.next_batch_deadline();
171                            }
172                        }
173                        OperationHandlerPropagationCommand::Stop => {
174                            info!("Stop operation propagation thread");
175                            return;
176                        }
177                    }
178                }
179                Err(RecvTimeoutError::Timeout) => {
180                    self.announce_ops();
181                    batch_deadline = self.next_batch_deadline();
182                }
183                Err(RecvTimeoutError::Disconnected) => {
184                    return;
185                }
186            }
187        }
188    }
189
190    /// Prune the list of operations kept for propagation.
191    fn prune_propagation_storage(&mut self) {
192        let removed = Self::prune_stored_for_propagation(
193            &mut self.stored_for_propagation,
194            &mut self.stored_ops,
195            self.config.max_operations_propagation_time.to_duration(),
196            self.config.max_ops_kept_for_propagation,
197        );
198
199        // remove from storage
200        self.op_storage.drop_operation_refs(&removed);
201    }
202
203    /// Remove expired and excess operation batches from the propagation queue,
204    /// keeping `stored_ops` in sync with the operations queued for propagation.
205    ///
206    /// `stored_ops` is maintained incrementally (updated here on removal and by the
207    /// caller on insertion) so that pruning costs O(number of removed operations)
208    /// instead of scanning the whole queue on every call. This prevents an attacker
209    /// from making each propagation command increasingly expensive by flooding the
210    /// queue with many tiny batches.
211    ///
212    /// Because the caller never queues an operation that is already tracked, the queue
213    /// batches are disjoint: the cap is enforced on distinct operations and each removed
214    /// id is released exactly once, so an operation is referenced in `op_storage` if and
215    /// only if it is still held by a batch.
216    ///
217    /// Returns the set of operation ids that were removed from the queue.
218    fn prune_stored_for_propagation(
219        stored_for_propagation: &mut VecDeque<(std::time::Instant, PreHashSet<OperationId>)>,
220        stored_ops: &mut PreHashSet<OperationId>,
221        max_operations_propagation_time: std::time::Duration,
222        max_ops_kept_for_propagation: usize,
223    ) -> PreHashSet<OperationId> {
224        let mut removed = PreHashSet::default();
225
226        // remove expired
227        while let Some((t, _)) = stored_for_propagation.front() {
228            if t.elapsed() > max_operations_propagation_time {
229                let (_, op_ids) = stored_for_propagation
230                    .pop_front()
231                    .expect("there should be at least one element, checked above");
232                for op_id in op_ids {
233                    stored_ops.remove(&op_id);
234                    removed.insert(op_id);
235                }
236            } else {
237                break;
238            }
239        }
240
241        // Cap cache size
242        // Note that we directly remove batches of operations, not individual operations
243        // to favor simplicity and performance over precision.
244        while stored_ops.len() > max_ops_kept_for_propagation {
245            if let Some((_t, op_ids)) = stored_for_propagation.pop_front() {
246                for op_id in op_ids {
247                    stored_ops.remove(&op_id);
248                    removed.insert(op_id);
249                }
250            } else {
251                break;
252            }
253        }
254
255        removed
256    }

258    fn announce_ops(&mut self) {
259        // Quit if empty  to avoid iterating on nodes
260        if self.next_batch.is_empty() {
261            return;
262        }
263        let operation_ids = mem::take(&mut self.next_batch);
264        massa_trace!("protocol.protocol_worker.announce_ops.begin", {
265            "operation_ids": operation_ids
266        });
267        // Operations whose announcement could not be delivered to at least one peer.
268        // They are put back into the next batch so that a later round retries them.
269        let mut to_retry: PreHashSet<OperationId> = PreHashSet::default();
270        {
271            let mut cache_write = self.cache.write();
272            let peers_connected = self.active_connections.get_peer_ids_connected();
273            cache_write.update_cache(&peers_connected);
274
275            // Propagate to peers
276            let all_keys: Vec<PeerId> = cache_write.ops_known_by_peer.keys().cloned().collect();
277            let chunk_size = self.config.max_operations_per_message as usize;
278            'peer_loop: for peer_id in all_keys {
279                let ops = cache_write.ops_known_by_peer.get_mut(&peer_id).unwrap();
280                let new_ops: Vec<OperationId> = operation_ids
281                    .iter()
282                    .filter(|id| ops.peek(&id.prefix()).is_none())
283                    .copied()
284                    .collect();
285                if !new_ops.is_empty() {
286                    debug!(
287                        "Send operations announcement of len {} to {}",
288                        new_ops.len(),
289                        peer_id
290                    );
291                    for (chunk_index, sub_list) in new_ops.chunks(chunk_size).enumerate() {
292                        if let Err(err) = self.active_connections.send_to_peer(
293                            &peer_id,
294                            &self.operation_message_serializer,
295                            OperationMessage::OperationsAnnouncement(
296                                sub_list.iter().map(|id| id.into_prefix()).collect(),
297                            )
298                            .into(),
299                            false,
300                        ) {
301                            warn!(
302                                "Failed to send OperationsAnnouncement message to peer: {}",
303                                err
304                            );
305                            // Nothing was delivered from this chunk on: the peer's cache is left
306                            // untouched for those operations, and they are retried later.
307                            to_retry.extend(
308                                new_ops[chunk_index.saturating_mul(chunk_size)..]
309                                    .iter()
310                                    .copied(),
311                            );
312                            // this peer is disconnected or congested, try with the next one
313                            // (a disconnected peer's cache is removed by the next
314                            // cache_write.update_cache call)
315                            continue 'peer_loop;
316                        }
317                        // sent successfully: only now mark the peer as knowing those operations
318                        for id in sub_list {
319                            ops.insert(id.prefix(), ());
320                        }
321                    }
322                }
323            }
324        }
325
326        // Requeue the operations that still need to be announced. Only the ones we still
327        // hold for propagation are worth retrying, and the batch is kept bounded so that a
328        // permanently congested peer cannot make it grow without limit.
329        if !to_retry.is_empty() {
330            let stored_ops = self.op_storage.get_op_refs();
331            let retry: Vec<OperationId> = to_retry
332                .into_iter()
333                .filter(|op_id| stored_ops.contains(op_id))
334                .take(
335                    self.config
336                        .operation_announcement_buffer_capacity
337                        .saturating_sub(self.next_batch.len()),
338                )
339                .collect();
340            self.next_batch.extend(retry);
341        }
342    }
343}

345pub fn start_propagation_thread(
346    internal_receiver: MassaReceiver<OperationHandlerPropagationCommand>,
347    active_connections: Box<dyn ActiveConnectionsTrait>,
348    config: ProtocolConfig,
349    cache: SharedOperationCache,
350    op_storage: Storage,
351    massa_metrics: MassaMetrics,
352) -> JoinHandle<()> {
353    std::thread::Builder::new()
354        .name(THREAD_NAME.to_string())
355        .spawn(move || {
356            let mut propagation_thread = PropagationThread {
357                internal_receiver,
358                active_connections,
359                stored_for_propagation: VecDeque::with_capacity(
360                    config.max_ops_kept_for_propagation,
361                ),
362                stored_ops: PreHashSet::with_capacity(config.max_ops_kept_for_propagation),
363                op_storage,
364                next_batch: PreHashSet::with_capacity(
365                    config
366                        .operation_announcement_buffer_capacity
367                        .saturating_add(1),
368                ),
369                config,
370                cache,
371                _massa_metrics: massa_metrics,
372                operation_message_serializer: MessagesSerializer::new()
373                    .with_operation_message_serializer(OperationMessageSerializer::new()),
374            };
375            propagation_thread.run();
376        })
377        .expect("OS failed to start operation propagation thread")
378}

406    /// Push a batch of operations at the given time, mirroring the accounting done by
407    /// the propagation thread on `PropagateOperations`: only the operations that are not
408    /// already tracked are queued, and an empty batch is not pushed at all.
409    fn push_batch_at(
410        queue: &mut VecDeque<(Instant, PreHashSet<OperationId>)>,
411        stored_ops: &mut PreHashSet<OperationId>,
412        time: Instant,
413        ops: impl IntoIterator<Item = OperationId>,
414    ) {
415        let newly_tracked: PreHashSet<OperationId> = ops
416            .into_iter()
417            .filter(|op_id| stored_ops.insert(*op_id))
418            .collect();
419        if !newly_tracked.is_empty() {
420            queue.push_back((time, newly_tracked));
421        }
422    }
423
424    fn push_batch(
425        queue: &mut VecDeque<(Instant, PreHashSet<OperationId>)>,
426        stored_ops: &mut PreHashSet<OperationId>,
427        ops: impl IntoIterator<Item = OperationId>,
428    ) {
429        push_batch_at(queue, stored_ops, Instant::now(), ops);
430    }

432    /// The ground-truth set of queued operations, computed by scanning the whole queue
433    /// (the O(n) operation the incrementally maintained set is meant to replace).
434    fn real_stored_ops(
435        queue: &VecDeque<(Instant, PreHashSet<OperationId>)>,
436    ) -> PreHashSet<OperationId> {
437        queue
438            .iter()
439            .flat_map(|(_, ops)| ops.iter().copied())
440            .collect()
441    }

443    /// A flood of tiny (single-op) batches must stay bounded by the op cap, and the
444    /// tracked set must stay exact so pruning never falls back to scanning.
445    #[test]
446    fn running_count_stays_exact_and_caps_total_ops() {
447        let mut queue = VecDeque::new();
448        let mut stored_ops = PreHashSet::default();
449        let max_ops = 100;
450        let never_expires = Duration::from_secs(3600);
451
452        for i in 0..1000u64 {
453            push_batch(&mut queue, &mut stored_ops, [op_id(i)]);
454            let removed = PropagationThread::prune_stored_for_propagation(
455                &mut queue,
456                &mut stored_ops,
457                never_expires,
458                max_ops,
459            );
460
461            // the tracked set must always match the ground truth
462            assert_eq!(stored_ops, real_stored_ops(&queue));
463            // once the cap is reached, each new 1-op batch evicts exactly one old batch
464            if i >= max_ops as u64 {
465                assert_eq!(removed.len(), 1);
466            }
467        }
468
469        // stored ops (== queue length for distinct 1-op batches) stays bounded by the cap
470        assert_eq!(stored_ops.len(), max_ops);
471        assert_eq!(queue.len(), max_ops);
472    }

644    /// A failed announcement must not mark the peer as knowing the operations, and the
645    /// unsent operations must be requeued so a later round retries them. Otherwise the
646    /// filtering done on the next round would suppress them forever for that peer.
647    #[test]
648    fn failed_announcement_is_not_cached_and_is_requeued() {
649        let peer_id = PeerId::from_public_key(KeyPair::generate(0).unwrap().get_public_key());
650        let ops: Vec<_> = (0..3).map(test_operation).collect();
651        let op_ids: PreHashSet<OperationId> = ops.iter().map(|op| op.id).collect();
652
653        let mut active_connections = MockActiveConnectionsTrait::new();
654        active_connections
655            .expect_get_peer_ids_connected()
656            .returning(move || HashSet::from_iter([peer_id]));
657        active_connections
658            .expect_send_to_peer()
659            .returning(|_, _, _, _| Err(ProtocolError::SendError("congested".to_string())));
660
661        let mut propagation_thread = propagation_thread_with_batch(active_connections, &ops);
662        propagation_thread.announce_ops();
663
664        // the peer knows none of the operations, so they will be announced again
665        let cache_read = propagation_thread.cache.read();
666        let known_by_peer = cache_read.ops_known_by_peer.get(&peer_id).unwrap();
667        for op_id in &op_ids {
668            assert!(known_by_peer.peek(&op_id.prefix()).is_none());
669        }
670        drop(cache_read);
671
672        // the operations are queued again for the next announcement round
673        assert_eq!(propagation_thread.next_batch, op_ids);
674    }

705    #[test]
706    fn oversized_operations_are_filtered_out_of_propagation() {
707        let keypair = KeyPair::generate(0).unwrap();
708        let op = create_operation_with_expire_period(&keypair, 1);
709        let op_id = op.id;
710        let op_size = op.serialized_size();
711
712        let mut storage = Storage::create_root();
713        storage.store_operations(vec![op.clone()]);
714
715        // With a generous limit the operation is kept for propagation,
716        // and its reference is retained.
717        let kept = filter_propagatable_ops(&mut storage, op_size);
718        assert!(kept.contains(&op_id), "op within the limit must be kept");
719        assert!(storage.get_op_refs().contains(&op_id));
720
721        // With a limit below its size the operation is excluded, so it is never
722        // marked checked, stored, or announced.
723        let dropped = filter_propagatable_ops(&mut storage, op_size - 1);
724        assert!(
725            dropped.is_empty(),
726            "oversized op must be filtered out of propagation"
727        );
728
729        // The rejected operation's reference must be released, so that merging
730        // the input storage into the propagation storage does not pin it.
731        assert!(!storage.get_op_refs().contains(&op_id));
732
733        // The submission owned the only reference, so the rejection also
734        // orphaned the content: it must no longer be present in shared storage.
735        assert!(
736            storage.read_operations().get(&op_id).is_none(),
737            "rejected op must not stay pinned in shared storage"
738        );
739    }
