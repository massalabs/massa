42    pub fn run(
43        config: &ProtocolConfig,
44        active_connections: Box<dyn ActiveConnectionsTrait>,
45        peer_db: SharedPeerDB,
46        messages_handler: MessagesHandler,
47        target_out_connections: HashMap<String, (Vec<IpAddr>, usize)>,
48        default_target_out_connections: usize,
49        massa_metrics: MassaMetrics,
50    ) -> (
51        (
52            MassaSender<(PeerId, HashMap<SocketAddr, TransportType>)>,
53            MassaReceiver<(PeerId, HashMap<SocketAddr, TransportType>)>,
54        ),
55        Vec<Tester>,
56    ) {
57        let mut testers = Vec::new();
58
59        // create shared channel between thread for launching test
60        let (test_sender, test_receiver) = MassaChannel::new(
61            "test_sender".to_string(),
62            Some(config.max_size_channel_commands_peer_testers),
63        );
64
65        for _ in 0..config.thread_tester_count {
66            testers.push(Tester::new(
67                peer_db.clone(),
68                active_connections.clone(),
69                config.clone(),
70                test_receiver.clone(),
71                messages_handler.clone(),
72                target_out_connections.clone(),
73                default_target_out_connections,
74                massa_metrics.clone(),
75            ));
76        }
77
78        ((test_sender, test_receiver), testers)
79    }

81    #[allow(clippy::too_many_arguments)]
82    pub fn tcp_handshake(
83        messages_handler: MessagesHandler,
84        peer_db: SharedPeerDB,
85        announcement_deserializer: AnnouncementDeserializer,
86        version_deserializer: VersionDeserializer,
87        peer_id_deserializer: PeerIdDeserializer,
88        addr: SocketAddr,
89        config: &ProtocolConfig,
90        massa_metrics: MassaMetrics,
91    ) -> PeerNetResult<PeerId> {
92        let our_version = config.version;
93
94        let exec_handshake = || {
95            let mut socket =
96                std::net::TcpStream::connect_timeout(&addr, config.tester_timeout.into())
97                    .map_err(|e| PeerNetError::PeerConnectionError.new("connect", e, None))?;
98            socket
99                .set_read_timeout(Some(config.tester_timeout.into()))
100                .map_err(|err| PeerNetError::PeerConnectionError.new("read timeout", err, None))?;
101            socket
102                .set_write_timeout(Some(config.tester_timeout.into()))
103                .map_err(|err| PeerNetError::PeerConnectionError.new("write timeout", err, None))?;
104            // data.receive() from Endpoint
105            let mut len_bytes = vec![0u8; 4];
106            socket
107                .read_exact(&mut len_bytes)
108                .map_err(|err| PeerNetError::PeerConnectionError.new("recv len", err, None))?;
109
110            let res_size = u32::from_be_bytes(len_bytes.try_into().map_err(|err| {
111                PeerNetError::PeerConnectionError.error("recv len", Some(format!("{:?}", err)))
112            })?);
113            if res_size > 1048576000 {
114                return Err(PeerNetError::InvalidMessage
115                    .error("len too long", Some(format!("{:?}", res_size))));
116            }
117            let mut data = vec![0u8; res_size as usize];
118            socket
119                .read_exact(&mut data)
120                .map_err(|err| PeerNetError::PeerConnectionError.new("recv data", err, None))?;
121
122            // handshake
123            if data.is_empty() {
124                return Err(PeerNetError::HandshakeError.error(
125                    "Tester Handshake",
126                    Some(String::from("Peer didn't accepted us")),
127                ));
128            }
129            let (data, peer_id) = peer_id_deserializer
130                .deserialize::<DeserializeError>(&data)
131                .map_err(|_| {
132                    PeerNetError::HandshakeError.error(
133                        "Massa Handshake",
134                        Some("Failed to deserialize PeerId".to_string()),
135                    )
136                })?;
137            let res = {
138                {
139                    // check if peer is banned
140                    let peer_db_read = peer_db.read();
141                    if let Some(info) = peer_db_read.get_peers().get(&peer_id) {
142                        if info.state == super::PeerState::Banned {
143                            return Err(PeerNetError::HandshakeError
144                                .error("Tester Handshake", Some(String::from("Peer is banned"))));
145                        }
146                    }
147                }
148
149                let (data, version) = version_deserializer
150                    .deserialize::<DeserializeError>(data)
151                    .map_err(|err| {
152                        PeerNetError::HandshakeError.error(
153                            "Tester Handshake",
154                            Some(format!("Failed to deserialize version: {}", err)),
155                        )
156                    })?;
157                if !our_version.is_compatible(&version) {
158                    return Err(PeerNetError::HandshakeError.error(
159                        "Massa Handshake",
160                        Some(format!("Received version incompatible: {}", version)),
161                    ));
162                }
163                let id = data.first().ok_or(
164                    PeerNetError::HandshakeError
165                        .error("Massa Handshake", Some("Failed to get id".to_string())),
166                )?;
167                match id {
168                    0 => {
169                        let (_, announcement) = announcement_deserializer
170                            .deserialize::<DeserializeError>(data.get(1..).ok_or(
171                                PeerNetError::HandshakeError.error(
172                                    "Massa Handshake",
173                                    Some("Failed to get buffer".to_string()),
174                                ),
175                            )?)
176                            .map_err(|err| {
177                                PeerNetError::HandshakeError.error(
178                                    "Tester Handshake",
179                                    Some(format!("Failed to deserialize announcement: {}", err)),
180                                )
181                            })?;
182
183                        if peer_id
184                            .verify_signature(&announcement.hash, &announcement.signature)
185                            .is_err()
186                        {
187                            return Err(PeerNetError::HandshakeError.error(
188                                "Tester Handshake",
189                                Some(String::from("Invalid signature")),
190                            ));
191                        }
192                        // the announcement timestamp is the freshness signal used to decide
193                        // which peers we keep sharing, so refuse one that is dated too far
194                        // ahead of us
195                        if announcement.is_future_dated(MassaTime::now().as_millis()) {
196                            return Err(PeerNetError::HandshakeError.error(
197                                "Tester Handshake",
198                                Some(String::from("Announcement timestamp too far in the future")),
199                            ));
200                        }
201                        //TODO: Check ip we are connected match one of the announced ips
202                        {
203                            // Store only endpoints we are allowed to dial later.
204                            let mut announcement = announcement;
205                            announcement.listeners = filter_routable_listeners(
206                                announcement.listeners,
207                                config.allow_local_peers(),
208                            );
209                            let mut peer_db_write = peer_db.write();
210                            peer_db_write
211                                .get_peers_mut()
212                                .entry(peer_id)
213                                .and_modify(|info| {
214                                    if let Some(last_announce) = &info.last_announce {
215                                        if last_announce.timestamp < announcement.timestamp {
216                                            info.last_announce = Some(announcement.clone());
217                                        }
218                                    } else {
219                                        info.last_announce = Some(announcement.clone());
220                                    }
221                                    info.state = super::PeerState::Trusted;
222                                })
223                                .or_insert(PeerInfo {
224                                    last_announce: Some(announcement),
225                                    state: super::PeerState::Trusted,
226                                });
227                        }
228                        Ok(peer_id)
229                    }
230                    1 => {
231                        messages_handler.handle(
232                            data.get(1..).ok_or(PeerNetError::HandshakeError.error(
233                                "Massa Handshake",
234                                Some("Failed to get buffer".to_string()),
235                            ))?,
236                            &peer_id,
237                        )?;
238                        Err(PeerNetError::HandshakeError.error(
239                                "Massa Handshake",
240                                Some("Tester Handshake failed received a message that our connection has been refused".to_string()),
241                            ))
242                        //TODO: Add the peerdb but for now impossible as we don't have announcement and we need one to place in peerdb
243                    }
244                    _ => Err(PeerNetError::HandshakeError
245                        .error("Massa handshake", Some("Invalid id".to_string()))),
246                }
247            };
248
249            {
250                let mut peer_db_write = peer_db.write();
251
252                // if handshake failed, we set the peer state to HandshakeFailed
253                if res.is_err() {
254                    peer_db_write
255                        .get_peers_mut()
256                        .entry(peer_id)
257                        .and_modify(|info| {
258                            info.state = super::PeerState::HandshakeFailed;
259                        })
260                        .or_insert(PeerInfo {
261                            last_announce: None,
262                            state: super::PeerState::HandshakeFailed,
263                        });
264                    peer_db_write.set_try_connect_test_failure_or_insert(&addr);
265                } else {
266                    peer_db_write.set_try_connect_test_success_or_insert(&addr);
267                }
268            }
269
270            if let Err(e) = socket.shutdown(std::net::Shutdown::Both) {
271                // the tested peer usually closes the connection on its side as soon as it has
272                // sent us its announcement, so the socket is often already fully torn down here:
273                // nothing left to shut down, and it gets closed anyway when dropped
274                if e.kind() == std::io::ErrorKind::NotConnected {
275                    debug!("Socket for {} was already closed by the peer", addr);
276                } else {
277                    tracing::error!("Failed to shutdown socket for {} : {}", addr, e);
278                }
279            }
280            res
281        };
282
283        let result = exec_handshake();
284
285        if result.is_ok() {
286            massa_metrics.inc_protocol_tester_success();
287        } else {
288            massa_metrics.inc_protocol_tester_failed();
289        }
290
291        result
292    }

294    /// Create a new tester (spawn a thread)
295    #[allow(clippy::too_many_arguments)]
296    pub fn new(
297        peer_db: SharedPeerDB,
298        active_connections: Box<dyn ActiveConnectionsTrait>,
299        protocol_config: ProtocolConfig,
300        receiver: MassaReceiver<(PeerId, HashMap<SocketAddr, TransportType>)>,
301        messages_handler: MessagesHandler,
302        target_out_connections: HashMap<String, (Vec<IpAddr>, usize)>,
303        default_target_out_connections: usize,
304        massa_metrics: MassaMetrics,
305    ) -> Self {
306        let handle = std::thread::Builder::new()
307        .name(THREAD_NAME.to_string())
308        .spawn(move || {
309            let db = peer_db;
310            let active_connections = active_connections.clone();
311
312            let announcement_deser = AnnouncementDeserializer::new(
313                AnnouncementDeserializerArgs {
314                    max_listeners: protocol_config.max_size_listeners_per_peer,
315                },
316            );
317
318
319            //let mut network_manager = PeerNetManager::new(config);
320            let protocol_config = protocol_config.clone();
321            let allow_local_peers = protocol_config.allow_local_peers();
322            'main_loop: loop {
323                crossbeam::select! {
324                    recv(receiver) -> res => {
325                        receiver.update_metrics();
326                        match res {
327                            Ok(listener) => {
328                                if listener.1.is_empty() {
329                                    continue;
330                                }
331                                //Test
332                                let peers_connected = active_connections.get_peers_connected();
333                                let slots_out_connections: HashMap<String, (Vec<IpAddr>, usize)> = target_out_connections
334                                    .iter()
335                                    .map(|(key, value)| {
336                                        let mut value = value.clone();
337                                        value.1 = value.1.saturating_sub(peers_connected.iter().filter(|(_, (_, ty, category))| {
338                                            if ty == &PeerConnectionType::IN {
339                                                return false;
340                                            }
341                                            if let Some(category) = category {
342                                                category == key
343                                            } else {
344                                                false
345                                            }
346                                        }).count());
347                                        (key.clone(), value)
348                                    })
349                                    .collect();
350                                let slot_default_category = default_target_out_connections.saturating_sub(peers_connected.iter().filter(|(_, (_, ty, category))| {
351                                    if ty == &PeerConnectionType::IN {
352                                        return false;
353                                    }
354                                    if category.is_some() {
355                                        return false;
356                                    }
357                                    true
358                                }).count());
359                                {
360                                    let now = MassaTime::now();
361                                    let db = db.clone();
362                                    // receive new listener to test
363                                    for (addr, _) in listener.1.iter() {
364                                        // Announced endpoints are peer controlled: never probe one
365                                        // that isn't a routable address (loopback, private, ...)
366                                        if !is_routable_peer_addr(addr, allow_local_peers) {
367                                            continue;
368                                        }
369                                        if !db.write().insert_peer_in_test(addr) {
370                                            // if the peer is already in test, we skip it
371                                            continue;
372                                        }
373
374                                        //Find category of that address
375                                        let ip_canonical = to_canonical(addr.ip());
376                                        let cooldown = 'cooldown: {
377                                            for category in &slots_out_connections {
378                                                if category.1.0.contains(&ip_canonical) {
379                                                    if category.1.1 == 0 {
380                                                        break 'cooldown Duration::from_secs(60 * 60 * 2);
381                                                    } else {
382                                                        break 'cooldown Duration::from_secs(30);
383                                                    }
384                                                }
385                                            }
386                                            if slot_default_category == 0 {
387                                                Duration::from_secs(60 * 60 * 2)
388                                            } else {
389                                                Duration::from_secs(30)
390                                            }
391                                        };
392                                        //TODO: Change it to manage multiple listeners SAFETY: Check above
393                                        {
394                                            let mut db_write = db.write();
395                                            if let Some(last_tested_time) = db_write.get_tested_addresses().get(addr) {
396                                                let last_tested_time = last_tested_time.estimate_instant().expect("Time went backward");
397                                                if last_tested_time.elapsed() < cooldown {
398                                                    db_write.remove_peer_in_test(addr);
399                                                    continue;
400                                                }
401                                            }
402                                            db_write.insert_tested_address(addr, now);
403                                        }
404                                        // TODO:  Don't launch test if peer is already connected to us as a normal connection.
405                                        // Maybe we need to have a way to still update his last announce timestamp because he is a great peer
406                                        if !active_connections.get_peers_connected().iter().any(|(_, (addr, _, _))| to_canonical(addr.ip()) == ip_canonical) {
407                                            //Don't test our local addresses
408                                            if protocol_config.listeners.iter().any(|(local_addr, _transport)| addr == local_addr) {
409                                                db.write().remove_peer_in_test(addr);
410                                                continue 'main_loop;
411                                            }
412
413                                            //Don't test our proper ip
414                                            if let Some(ip) = protocol_config.routable_ip {
415                                                if to_canonical(ip) == ip_canonical {
416                                                    db.write().remove_peer_in_test(addr);
417                                                    continue 'main_loop;
418                                                }
419                                            }
420                                            debug!("testing peer {} listener addr: {}", &listener.0, &addr);
421
422                                            let res = Tester::tcp_handshake(
423                                                messages_handler.clone(),
424                                                db.clone(),
425                                                announcement_deser.clone(),
426                                                VersionDeserializer::new(),
427                                                PeerIdDeserializer::new(),
428                                                *addr,
429                                                &protocol_config,
430                                                massa_metrics.clone(),
431                                            );
432
433                                            db.write().remove_peer_in_test(addr);
434
435                                            // let _res =  network_manager.try_connect(
436                                            //     *addr,
437                                            //     protocol_config.timeout_connection.to_duration(),
438                                            //     &OutConnectionConfig::Tcp(Box::new(TcpOutConnectionConfig::new(protocol_config.read_write_limit_bytes_per_second / 10, Duration::from_millis(100)))),
439                                            // );
440
441                                            tracing::debug!("{:?}", res);
442                                        }
443                                    };
444                                }
445                            },
446                            Err(_e) => break,
447                        }
448                    }
449                    default(Duration::from_secs(2)) => {
450                        // If no message in 2 seconds they will test a peer that hasn't been tested for long time
451
452                        let listener = {
453                            let mut db_write = db.write();
454                            if let Some(listener) = db_write.get_oldest_peer(
455                                protocol_config.test_oldest_peer_cooldown.into(),
456                                db_write.get_peers_in_test(),
457                            ) {
458                                db_write.insert_peer_in_test(&listener);
459                                db_write.insert_tested_address(&listener, MassaTime::now());
460                                listener
461                            } else {
462                                continue;
463                            }
464                        };
465
466                        // we try to connect to all peer listener (For now we have only one listener)
467                        if !is_routable_peer_addr(&listener, allow_local_peers) {
468                            db.write().remove_peer_in_test(&listener);
469                            continue;
470                        }
471                        let ip_canonical = to_canonical(listener.ip());
472                        if active_connections.get_peers_connected().iter().any(|(_, (addr, _, _))| to_canonical(addr.ip()) == ip_canonical) {
473                            db.write().remove_peer_in_test(&listener);
474                            continue;
475                        }
476                        //Don't test our local addresses
477                        for (local_addr, _transport) in protocol_config.listeners.iter() {
478                            if listener == *local_addr {
479                                db.write().remove_peer_in_test(&listener);
480                                continue;
481                            }
482                        }
483                        //Don't test our proper ip
484                        if let Some(ip) = protocol_config.routable_ip {
485                            if to_canonical(ip) == ip_canonical {
486                                db.write().remove_peer_in_test(&listener);
487                                continue;
488                            }
489                        }
490                        debug!("testing listener addr: {}", &listener);
491
492                        let _ = Tester::tcp_handshake(
493                            messages_handler.clone(),
494                            db.clone(),
495                            announcement_deser.clone(),
496                            VersionDeserializer::new(),
497                            PeerIdDeserializer::new(),
498                            listener,
499                            &protocol_config,
500                            massa_metrics.clone(),
501                        );
502                        // let res =  network_manager.try_connect(
503                        //     listener,
504                        //     protocol_config.timeout_connection.to_duration(),
505                        //     &OutConnectionConfig::Tcp(Box::new(TcpOutConnectionConfig::new(protocol_config.read_write_limit_bytes_per_second / 10, Duration::from_millis(100)))),
506                        // );
507                        db.write().remove_peer_in_test(&listener);
508                        // debug!("{:?}", res);
509                    }
510                }
511            }
512        }).expect("OS failed to start peer tester thread");
513
514        Self {
515            handler: Some(handle),
516        }
517    }
