3use massa_deferred_calls::config::DeferredCallsConfig;
4use massa_models::amount::Amount;
5use massa_signature::KeyPair;
6use massa_time::MassaTime;
7use std::net::SocketAddr;
8use std::path::PathBuf;
9
10use serde::Deserialize;
11
12/// API settings.
13/// the API settings
14#[derive(Debug, Deserialize, Clone)]
15pub struct APIConfig {
16    /// when looking for next draw we want to look at max `draw_lookahead_period_count`
17    pub draw_lookahead_period_count: u64,
18    /// bind for the private API
19    pub bind_private: SocketAddr,
20    /// bind for the public API
21    pub bind_public: SocketAddr,
22    /// bind for the Massa API
23    pub bind_api: SocketAddr,
24    /// max argument count
25    pub max_arguments: u64,
26    /// openrpc specification path
27    pub openrpc_spec_path: PathBuf,
28    /// bootstrap whitelist path
29    pub bootstrap_whitelist_path: PathBuf,
30    /// bootstrap blacklist path
31    pub bootstrap_blacklist_path: PathBuf,
32    /// maximum size in bytes of a request.
33    pub max_request_body_size: u32,
34    /// maximum size in bytes of a response.
35    pub max_response_body_size: u32,
36    /// maximum number of incoming connections allowed.
37    pub max_connections: u32,
38    /// maximum number of subscriptions per connection.
39    pub max_subscriptions_per_connection: u32,
40    /// max length for logging for requests and responses. Logs bigger than this limit will be truncated.
41    pub max_log_length: u32,
42    /// host filtering.
43    pub allow_hosts: Vec<String>,
44    /// batch request limit. 0 means disabled.
45    pub batch_request_limit: u32,
46    /// the interval at which `Ping` frames are submitted.
47    pub ping_interval: MassaTime,
48    /// whether to enable HTTP.
49    pub enable_http: bool,
50    /// whether to enable WS.
51    pub enable_ws: bool,
52    /// max size of a smart contract bytecode in an `ExecuteSC` operation
53    pub max_bytecode_size: u64,
54    /// max datastore key length
55    pub max_datastore_key_length: u8,
56    /// max op datastore entry
57    pub max_op_datastore_entry_count: u64,
58    /// max datastore key length
59    pub max_op_datastore_key_length: u8,
60    /// max datastore value length
61    pub max_op_datastore_value_length: u64,
62    /// max function name length
63    pub max_function_name_length: u16,
64    /// max parameter size
65    pub max_parameter_size: u32,
66    /// max gas in a block
67    pub max_gas_per_block: u64,
68    /// maximum serialized size of a single operation: an operation bigger than
69    /// the whole operations payload of a block can never be propagated by the
70    /// protocol nor included in a block (`MAX_BLOCK_SIZE`)
71    pub max_serialized_operation_size: usize,
72    /// base gas used by any operation
73    pub base_operation_gas_cost: u64,
74    /// Amount required for a SinglePass compilation (ExecuteSC operation)
75    pub sp_compilation_cost: u64,
76    /// thread count
77    pub thread_count: u8,
78    /// `genesis_timestamp`
79    pub genesis_timestamp: MassaTime,
80    /// t0
81    pub t0: MassaTime,
82    /// periods per cycle
83    pub periods_per_cycle: u64,
84    /// keypair file
85    pub keypair: KeyPair,
86    /// last_start_period value, used to know if we are during a restart or not
87    pub last_start_period: u64,
88    /// chain id
89    pub chain_id: u64,
90    /// Delta to compute upper bounds when fetching deferred credits
91    pub deferred_credits_delta: MassaTime,
92    /// minimal fees to include an operation in a block
93    pub minimal_fees: Amount,
94    /// deferred calls config
95    pub deferred_calls_config: DeferredCallsConfig,
96    /// max datastore keys queries
97    pub max_datastore_keys_queries: Option<u32>,
98    /// Limit the number of addresses in a single datastore keys request
99    pub max_addresses_datastore_keys_query: Option<u32>,
100    /// Max events returned per query_state batch
101    pub max_event_per_query: u32,
102    /// wall-clock deadline for a query_state batch, in ms (`None` = no deadline)
103    pub query_state_deadline_ms: Option<u64>,
104    /// timeout for API calls to pools (in milliseconds)
105    pub pool_api_timeout: MassaTime,
106}
