13/// Errors of the api component.
14#[non_exhaustive]
15#[derive(Display, thiserror::Error, Debug)]
16pub enum ApiError {
17    /// Send channel error: {0}
18    SendChannelError(String),
19    /// Receive channel error: {0}
20    ReceiveChannelError(String),
21    /// `massa_hash` error: {0}
22    MassaHashError(#[from] MassaHashError),
23    /// consensus error: {0}
24    ConsensusError(String),
25    /// execution error: {0}
26    ExecutionError(String),
27    /// Protocol error: {0}
28    ProtocolError(String),
29    /// Models error: {0}
30    ModelsError(#[from] ModelsError),
31    /// Time error: {0}
32    TimeError(#[from] TimeError),
33    /// Wallet error: {0}
34    WalletError(#[from] WalletError),
35    /// Not found
36    NotFound,
37    /// Inconsistency error: {0}
38    InconsistencyError(String),
39    /// Missing command sender: {0}
40    MissingCommandSender(String),
41    /// Missing configuration: {0}
42    MissingConfig(String),
43    /// The wrong API (either Public or Private) was called
44    WrongAPI,
45    /// Bad request: {0}
46    BadRequest(String),
47    /// Internal server error: {0}
48    InternalServerError(String),
49    /// Versioning Factory error: {0}
50    FactoryError(#[from] FactoryError),
51    /// Pool error: {0}
52    PoolError(#[from] PoolError),
53}
54
55impl From<ApiError> for ErrorObjectOwned {
56    fn from(err: ApiError) -> Self {
57        // JSON-RPC Server errors codes must be between -32099 to -32000
58        let code = match err {
59            ApiError::BadRequest(_) => -32000,
60            ApiError::InternalServerError(_) => -32001,
61            ApiError::NotFound => -32004,
62            ApiError::SendChannelError(_) => -32006,
63            ApiError::ReceiveChannelError(_) => -32007,
64            ApiError::MassaHashError(_) => -32008,
65            ApiError::ConsensusError(_) => -32009,
66            ApiError::ExecutionError(_) => -32010,
67            ApiError::ProtocolError(_) => -32012,
68            ApiError::ModelsError(_) => -32013,
69            ApiError::TimeError(_) => -32014,
70            ApiError::WalletError(_) => -32015,
71            ApiError::InconsistencyError(_) => -32016,
72            ApiError::MissingCommandSender(_) => -32017,
73            ApiError::MissingConfig(_) => -32018,
74            ApiError::WrongAPI => -32019,
75            ApiError::FactoryError(_) => -32020,
76            ApiError::PoolError(_) => -32021,
77        };
78
79        ErrorObject::owned(code, err.to_string(), None::<()>)
80    }
81}
