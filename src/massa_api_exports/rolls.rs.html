3use serde::{Deserialize, Serialize};
4
5/// Roll counts
6#[derive(Debug, Deserialize, Serialize, Clone, Copy)]
7pub struct RollsInfo {
8    /// count taken into account for the current cycle
9    pub active_rolls: u64,
10    /// at final blocks
11    pub final_rolls: u64,
12    /// at latest blocks
13    pub candidate_rolls: u64,
14}
15
16impl std::fmt::Display for RollsInfo {
17    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
18        writeln!(f, "\tActive rolls: {}", self.active_rolls)?;
19        writeln!(f, "\tFinal rolls: {}", self.final_rolls)?;
20        writeln!(f, "\tCandidate rolls: {}", self.candidate_rolls)?;
21        Ok(())
22    }
23}
