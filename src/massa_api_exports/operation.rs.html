13/// operation input
14#[derive(Serialize, Deserialize, Debug)]
15pub struct OperationInput {
16    /// The public key of the creator of the TX
17    pub creator_public_key: PublicKey,
18    /// The signature of the operation
19    pub signature: Signature,
20    /// The serialized version of the content `base58` encoded
21    pub serialized_content: Vec<u8>,
22}
23
24/// Operation and contextual info about it
25#[derive(Debug, Clone, Deserialize, Serialize)]
26pub struct OperationInfo {
27    /// id
28    pub id: OperationId,
29    /// true if operation is still in pool
30    pub in_pool: bool,
31    /// the operation appears in `in_blocks`
32    /// if it appears in multiple blocks, these blocks are in different cliques
33    pub in_blocks: Vec<BlockId>,
34    /// true if the operation is final (for example in a final block)
35    pub is_operation_final: Option<bool>,
36    /// Thread in which the operation can be included
37    pub thread: u8,
38    /// the operation itself
39    pub operation: SecureShareOperation,
40    /// true if the operation execution succeeded, false if failed, None means unknown
41    pub op_exec_status: Option<bool>,
42}
43
44impl std::fmt::Display for OperationInfo {
45    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
46        writeln!(
47            f,
48            "Operation {}{}{}{}",
49            self.id,
50            display_if_true(self.in_pool, "in pool"),
51            display_option_bool(
52                self.is_operation_final,
53                "operation is final",
54                "operation is not final",
55                "finality unknown"
56            ),
57            display_option_bool(self.op_exec_status, "success", "failed", "status unknown")
58        )?;
59        writeln!(f, "In blocks:")?;
60        for block_id in &self.in_blocks {
61            writeln!(f, "\t- {}", block_id)?;
62        }
63        writeln!(f, "{}", self.operation)?;
64        Ok(())
65    }
66}

68#[cfg(test)]
69mod tests {
70    use jsonrpsee::core::__reexports::serde_json::{self, Value};
71    use massa_models::{amount::Amount, operation::OperationType};
72    use serial_test::serial;
73    use std::collections::BTreeMap;
74    use std::str::FromStr;
75
76    #[test]
77    #[serial]
78    fn test_execute_sc_with_datastore() {
79        let expected_op = OperationType::ExecuteSC {
80            max_gas: 123,
81            max_coins: Amount::from_str("5000000").unwrap(),
82            data: vec![23u8, 123u8, 44u8],
83            datastore: BTreeMap::from([
84                (vec![1, 2, 3], vec![4, 5, 6, 7, 8, 9]),
85                (vec![22, 33, 44, 55, 66, 77], vec![11]),
86                (vec![2, 3, 4, 5, 6, 7], vec![1]),
87            ]),
88        };
89
90        let op_json_str = serde_json::to_string(&expected_op).unwrap();
91        let op_json_value: Value = serde_json::from_str(&op_json_str).unwrap();
92        let datastore = op_json_value["ExecuteSC"]
93            .as_object()
94            .unwrap()
95            .get("datastore")
96            .unwrap()
97            .as_array()
98            .unwrap();
99        assert_eq!(datastore.len(), 3);
100        let first_entry = datastore[0].as_array().unwrap();
101        assert_eq!(first_entry.len(), 2);
102        let first_key = first_entry[0].as_array().unwrap();
103        let first_value = first_entry[1].as_array().unwrap();
104        assert_eq!(first_key.len(), 3);
105        assert_eq!(first_value.len(), 6);
106
107        let actual_op: OperationType = serde_json::from_str(&op_json_str).unwrap();
108        assert_eq!(actual_op, expected_op);
109    }
