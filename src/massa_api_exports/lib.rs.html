39/// Dumb utils function to display nicely boolean value
40fn display_if_true(value: bool, text: &str) -> String {
41    if value {
42        format!("[{}]", text)
43    } else {
44        String::from("")
45    }
46}
47
48/// Help to format Optional bool
49fn display_option_bool(
50    value: Option<bool>,
51    text_true: &str,
52    text_false: &str,
53    text_none: &str,
54) -> String {
55    match value {
56        Some(true) => {
57            format!("[{}]", text_true)
58        }
59        Some(false) => {
60            format!("[{}]", text_false)
61        }
62        None => {
63            format!("[{}]", text_none)
64        }
65    }
66}
