3use massa_models::amount::Amount;
4use massa_models::node::NodeId;
5use massa_models::stats::{ConsensusStats, ExecutionStats, NetworkStats};
6use massa_models::{config::CompactConfig, slot::Slot, version::Version};
7use massa_time::MassaTime;
8use serde::{Deserialize, Serialize};
9use std::collections::BTreeMap;
10use std::net::IpAddr;
11
12/// node status
13#[derive(Debug, Clone, Deserialize, Serialize)]
14pub struct NodeStatus {
15    /// our node id
16    pub node_id: NodeId,
17    /// optional node ip
18    pub node_ip: Option<IpAddr>,
19    /// node version
20    pub version: Version,
21    /// now
22    pub current_time: MassaTime,
23    /// current cycle
24    pub current_cycle: u64,
25    /// current cycle starting timestamp
26    pub current_cycle_time: MassaTime,
27    /// next cycle starting timestamp
28    pub next_cycle_time: MassaTime,
29    /// connected nodes (node id, ip address, true if the connection is outgoing, false if incoming)
30    pub connected_nodes: BTreeMap<NodeId, (IpAddr, bool)>,
31    /// latest slot, none if now is before genesis timestamp
32    pub last_slot: Option<Slot>,
33    /// next slot
34    pub next_slot: Slot,
35    /// consensus stats
36    pub consensus_stats: ConsensusStats,
37    /// pool stats (operation count and endorsement count)
38    pub pool_stats: (usize, usize),
39    /// network stats
40    pub network_stats: NetworkStats,
41    /// execution stats
42    pub execution_stats: ExecutionStats,
43    /// compact configuration
44    pub config: CompactConfig,
45    /// chain id
46    pub chain_id: u64,
47    /// minimal fees to include an operation in a block
48    pub minimal_fees: Amount,
49    /// current mip version
50    pub current_mip_version: u32,
51    /// max number of datastore keys returned by a single datastore key query
52    pub max_datastore_keys_query: Option<u32>,
53}
54
55impl std::fmt::Display for NodeStatus {
56    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
57        writeln!(f, "Node's ID: {}", self.node_id)?;
58        if self.node_ip.is_some() {
59            writeln!(f, "Node's IP: {}", self.node_ip.unwrap())?;
60        } else {
61            writeln!(f, "No routable IP set")?;
62        }
63        writeln!(f)?;
64
65        writeln!(f, "Version: {}", self.version)?;
66        writeln!(f, "Config:\n{}", self.config)?;
67        if let Some(max_datastore_keys_query) = self.max_datastore_keys_query {
68            writeln!(
69                f,
70                "Max datastore keys per query: {}",
71                max_datastore_keys_query
72            )?;
73        }
74        writeln!(f)?;
75
76        writeln!(f, "Current time: {}", self.current_time.format_instant())?;
77        writeln!(f, "Current cycle: {}", self.current_cycle)?;
78        if self.last_slot.is_some() {
79            writeln!(f, "Last slot: {}", self.last_slot.unwrap())?;
80        }
81        writeln!(f, "Next slot: {}", self.next_slot)?;
82        writeln!(f)?;
83
84        writeln!(f, "{}", self.consensus_stats)?;
85
86        writeln!(f, "Pool stats:")?;
87        writeln!(f, "\tOperations count: {}", self.pool_stats.0)?;
88        writeln!(f, "\tEndorsements count: {}", self.pool_stats.1)?;
89        writeln!(f)?;
90
91        writeln!(f, "{}", self.network_stats)?;
92
93        writeln!(f, "{}", self.execution_stats)?;
94
95        writeln!(f, "Connected nodes:")?;
96        for (node_id, (ip_addr, is_outgoing)) in &self.connected_nodes {
97            writeln!(
98                f,
99                "Node's ID: {} / IP address: {} / {} connection",
100                node_id,
101                ip_addr,
102                if *is_outgoing { "Out" } else { "In" }
103            )?
104        }
105        Ok(())
106    }
107}
