1//! One-time, versioning-gated patch that replaces the bytecode of the deployed
2//! WMAS contract to fix the storage-cost drain (a `transfer`/`increaseAllowance`
3//! to a fresh address makes WMAS pay, from its own locked MAS, for the new
4//! datastore balance entry).
5//!
6//! WMAS is a widely-used singleton that cannot be redeployed without forcing
7//! every holder and integration to migrate. Instead, at the activation of
8//! [`MipComponent::Execution`] version `MIP_0002_EXECUTION_VERSION`, every node
9//! deterministically overwrites the WMAS bytecode with an audited fixed build.
10//!
11//! Safety of the approach:
12//! - Bytecode is stored separately from the datastore and the coin balance, so
13//!   all WMAS balances and all locked MAS are preserved by the swap.
14//! - The module cache is keyed by bytecode hash (`Hash::compute_from`), so no
15//!   explicit invalidation is required: the new bytecode hashes differently and
16//!   is recompiled on first use.
17//! - The change is applied inside `execute_slot`, which is shared by both
18//!   candidate and final execution, so speculative and final states agree. The
19//!   resulting bytecode change is part of the slot's ledger changes and thus of
20//!   the final state, so nodes bootstrapping after activation receive the
21//!   already-patched bytecode.
22//!
23//! WMAS is the same contract deployed at a different address on each network,
24//! so a single embedded bytecode serves all of them; only the target address is
25//! selected per network via `chain_id` (see [`wmas_address`]).
26//!
27//! Activation: this patch rides on the release's `MIP-0002-BugFix`, which already
28//! bumps `MipComponent::Execution` to `MIP_0002_EXECUTION_VERSION` in
29//! `massa-versioning/src/mips.rs`. It therefore applies at that MIP's activation
30//! slot, together with the rest of the next release's breaking changes — there
31//! is no separate WMAS MIP.
32//!
33//! IMPORTANT: because it is bundled with `MIP-0002-BugFix` (which has real
34//! activation dates), this patch is NOT inert until a future date. Before that
35//! release ships, `wmas_bytecode.rs` MUST be regenerated from the reproducible,
36//! audited build of the deployed WMAS contract (it currently embeds a dev build)
37//! and the per-network `WMAS_ADDRESS_*` constants verified. The fail-safe in
38//! `execute_slot` only skips on an unknown chain_id or missing contract, so
39//! a wrong-but-valid blob would still be applied.
40
41use massa_models::{address::Address, bytecode::Bytecode};
42use std::str::FromStr;
43
44/// Chain ids of the networks where WMAS is deployed (see
45/// `massa_models::config::CHAINID`).
46const MAINNET_CHAIN_ID: u64 = 77658377;
47const BUILDNET_CHAIN_ID: u64 = 77658366;
48
49/// Deployed WMAS address on mainnet.
50pub const WMAS_ADDRESS_MAINNET: &str = "AS12U4TZfNK7qoLyEERBBRDMu8nm5MKoRzPXDXans4v9wdATZedz9";
51/// Deployed WMAS address on buildnet.
52pub const WMAS_ADDRESS_BUILDNET: &str = "AS12FW5Rs5YN2zdpEnqwj4iHUUPt9R4Eqjq2qtpJFNKW3mn33RuLU";
53
54/// Audited fixed WMAS bytecode, embedded as an in-source byte array (see
55/// [`crate::wmas_bytecode`]).
56///
57/// TODO(release): regenerate `wmas_bytecode.rs` from the reproducible, audited
58/// build of the actually deployed WMAS contract before enabling the activation
59/// MIP.
60pub const PATCHED_WMAS_BYTECODE: &[u8] = crate::wmas_bytecode::PATCHED_WMAS_BYTECODE;
61
62// Some checks at compile time that should not be ignored!
63#[allow(clippy::const_is_empty)]
64const _: () = {
65    assert!(!PATCHED_WMAS_BYTECODE.is_empty());
66};
67
68/// Returns the WMAS address to patch for the network identified by `chain_id`,
69/// or `None` if the network has no known WMAS deployment (e.g. sandbox/labnet)
70/// or the address constant is malformed. Returning `None` rather than panicking
71/// keeps a misconfiguration from aborting consensus execution.
72pub fn wmas_address(chain_id: u64) -> Option<Address> {
73    let addr = match chain_id {
74        MAINNET_CHAIN_ID => WMAS_ADDRESS_MAINNET,
75        BUILDNET_CHAIN_ID => WMAS_ADDRESS_BUILDNET,
76        _ => return None,
77    };
78    Address::from_str(addr).ok()
79}
80
81/// Returns the patched WMAS bytecode.
82pub fn patched_wmas_bytecode() -> Bytecode {
83    Bytecode(PATCHED_WMAS_BYTECODE.to_vec())
