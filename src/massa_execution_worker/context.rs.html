365    /// Resets context to an existing snapshot.
366    /// Optionally emits an error as an event after restoring the snapshot.
367    /// Note that the snapshot does not include slot-level information such as the slot number or block ID.
368    ///
369    /// # Arguments
370    /// * `snapshot`: a saved snapshot to be restored
371    /// * `error`: an execution error to emit as an event conserved after snapshot reset.
372    pub fn reset_to_snapshot(&mut self, snapshot: ExecutionContextSnapshot, error: ExecutionError) {
373        // Emit the error event.
374        // Note that the context event counter is properly handled by event_emit (see doc).
375        let mut event = self.event_create(
376            serde_json::json!({ "massa_execution_error": format!("{}", error) }).to_string(),
377            true,
378        );
379        let max_event_size = self.config.max_event_size;
380        if event.data.len() > max_event_size {
381            event.data = truncate_string(event.data, max_event_size);
382        }
383        self.event_emit(event);
384
385        // Reset context to snapshot.
386        self.speculative_ledger
387            .reset_to_snapshot(snapshot.ledger_changes);
388        self.speculative_async_pool
389            .reset_to_snapshot((snapshot.async_pool_changes, snapshot.message_infos));
390        self.speculative_deferred_calls
391            .reset_to_snapshot(snapshot.deferred_calls_changes);
392        self.speculative_roll_state
393            .reset_to_snapshot(snapshot.pos_changes);
394        self.speculative_executed_ops
395            .reset_to_snapshot(snapshot.executed_ops);
396        self.speculative_executed_denunciations
397            .reset_to_snapshot(snapshot.executed_denunciations);
398        self.created_addr_index = snapshot.created_addr_index;
399        self.created_event_index = snapshot.created_event_index;
400        self.created_message_index = snapshot.created_message_index;
401        self.stack = snapshot.stack;
402        self.unsafe_rng = snapshot.unsafe_rng;
403        self.gas_remaining_before_subexecution = snapshot.gas_remaining_before_subexecution;
404        self.recursion_counter = snapshot.recursion_counter;
405        self.user_event_count_in_current_exec = snapshot.user_event_count_in_current_exec;
406        self.execution_component_version = snapshot.execution_component_version;
407        self.execution_component_version_upgraded = snapshot.execution_component_version_upgraded;
408
409        {
410            let mut transfers_history = self.transfers_history.write();
411            transfers_history.truncate(snapshot.transfer_history_len);
412        }
413
414        // For events, set snapshot delta to error events.
415        for event in self.events.0.range_mut(snapshot.event_count..) {
416            event.context.is_error = true;
417        }
418    }

420    /// Create a new `ExecutionContext` for read-only execution
421    /// This should be used before performing a read-only execution.
422    ///
423    /// # arguments
424    /// * `slot`: slot at which the execution will happen
425    /// * `req`: parameters of the read only execution
426    /// * `final_state`: thread-safe access to the final state. Note that this will be used only for reading, never for writing
427    ///
428    /// # returns
429    /// A `ExecutionContext` instance ready for a read-only execution
430    #[allow(clippy::too_many_arguments)]
431    pub(crate) fn readonly(
432        config: ExecutionConfig,
433        slot: Slot,
434        call_stack: Vec<ExecutionStackElement>,
435        final_state: Arc<RwLock<dyn FinalStateController>>,
436        active_history: Arc<RwLock<ActiveHistory>>,
437        module_cache: Arc<RwLock<ModuleCache>>,
438        mip_store: MipStore,
439    ) -> Self {
440        // Get the execution hash trail
441        let prev_execution_trail_hash = active_history.read().get_execution_trail_hash();
442        let prev_execution_trail_hash = match prev_execution_trail_hash {
443            HistorySearchResult::Present(h) => h,
444            _ => final_state.read().get_execution_trail_hash(),
445        };
446        let execution_trail_hash =
447            generate_execution_trail_hash(&prev_execution_trail_hash, &slot, None, true);
448
449        let (execution_component_version, execution_component_version_upgraded) =
450            execution_component_version_info(
451                &mip_store,
452                &slot,
453                config.thread_count,
454                config.t0,
455                config.genesis_timestamp,
456            );
457
458        // return readonly context
459        ExecutionContext {
460            slot,
461            stack: call_stack,
462            read_only: true,
463            execution_component_version,
464            execution_component_version_upgraded,
465            ..ExecutionContext::new(
466                config,
467                final_state,
468                active_history,
469                module_cache,
470                mip_store,
471                execution_trail_hash,
472            )
473        }

591    /// Creates a new smart contract address with initial bytecode, and returns this address
592    pub fn create_new_sc_address(&mut self, bytecode: Bytecode) -> Result<Address, ExecutionError> {
593        // deterministically generate a new unique smart contract address
594        let slot_timestamp = get_block_slot_timestamp(
595            self.config.thread_count,
596            self.config.t0,
597            self.config.genesis_timestamp,
598            self.slot,
599        )
600        .expect("could not compute current slot timestamp");
601
602        // Loop over nonces until we find an address that doesn't exist in the speculative ledger.
603        // Note that this loop is here for robustness, and should not be looping because
604        // even through the SC addresses are predictable, nobody can create them beforehand because:
605        // - their category is "SC" and not "USER" so they can't be derived from a public key
606        // - sending tokens to the target SC address to create it by funding is not allowed because transactions towards SC addresses are not allowed
607        let mut nonce = 0u64;
608        let address = loop {
609            // get a deterministic seed hash
610            let hash = massa_hash::Hash::compute_from_tuple(&[
611                "SC_ADDRESS".as_bytes(),
612                self.execution_trail_hash.to_bytes(),
613                &self.created_addr_index.to_be_bytes(),
614                &nonce.to_be_bytes(),
615            ]);
616
617            // deduce the address
618            let addr = self.address_factory.create(
619                &AddressArgs::SC { hash },
620                FactoryStrategy::At(slot_timestamp),
621            )?;
622
623            // check if this address already exists in the speculative ledger
624            if !self.speculative_ledger.entry_exists(&addr) {
625                // if not, we can use it
626                break addr;
627            }
628
629            // otherwise, increment the nonce to get a new hash and try again
630            nonce = nonce.checked_add(1).ok_or_else(|| {
631                ExecutionError::RuntimeError("nonce overflow when creating SC address".into())
632            })?;
633        };
634
635        // add this address with its bytecode to the speculative ledger
636        self.speculative_ledger.create_new_sc_address(
637            self.get_current_address()?,
638            address,
639            bytecode,
640        )?;
641
642        // add the address to owned addresses
643        // so that the current call has write access to it
644        // from now and for its whole duration,
645        // in order to allow initializing newly created ledger entries.
646        match self.stack.last_mut() {
647            Some(v) => {
648                v.owned_addresses.push(address);
649            }
650            None => {
651                return Err(ExecutionError::RuntimeError(
652                    "owned addresses not found in context stack".into(),
653                ));
654            }
655        };
656
657        // increment the address creation counter at this slot
658        self.created_addr_index += 1;
659
660        Ok(address)
661    }

668    /// gets the datastore keys of an address if it exists in the speculative ledger, or returns None
669    pub fn get_keys(
670        &self,
671        addr: &Address,
672        prefix: &[u8],
673        start_key: std::ops::Bound<Vec<u8>>,
674        end_key: std::ops::Bound<Vec<u8>>,
675        count: Option<u32>,
676    ) -> Result<Option<BTreeSet<Vec<u8>>>, ExecutionError> {
677        // From Execution v2 (MIP-0002), the SC datastore-key path is bounded by
678        // a protocol constant (massa #5284). Below v2 the query stays uncapped.
679        let v2 = self.is_execution_component_version_at_least(MIP_0002_EXECUTION_VERSION);
680        // `count = None` (the only query the deprecated ABI can express) is
681        // probed at `cap + 1` so that matching more than `cap` keys fails
682        // loudly instead of being silently truncated. An explicit over-cap
683        // count is rejected by the cleanup below.
684        let (effective_count, max_query) = match (v2, count) {
685            (false, _) => (None, None),
686            (true, Some(_)) => (count, Some(MAX_SC_DATASTORE_KEY_COUNT)),
687            (true, None) => (Some(MAX_SC_DATASTORE_KEY_COUNT.saturating_add(1)), None),
688        };
689
690        // cleanup bounds
691        let (prefix, start_key, end_key) = cleanup_datastore_key_range_query(
692            prefix,
693            start_key,
694            end_key,
695            count,
696            self.config.max_datastore_key_length,
697            max_query,
698        )?;
699
700        let keys =
701            self.speculative_ledger
702                .get_keys(addr, &prefix, start_key, end_key, effective_count);
703
704        // over-cap probe: more matches than the cap is a hard error, never a
705        // silent truncation
706        if v2 && count.is_none() {
707            if let Some(keys) = keys.as_ref() {
708                if keys.len() > MAX_SC_DATASTORE_KEY_COUNT as usize {
709                    return Err(ExecutionError::RuntimeError(format!(
710                        "datastore key query matched more than the maximum of {} keys",
711                        MAX_SC_DATASTORE_KEY_COUNT
712                    )));
713                }
714            }
715        }
716
717        Ok(keys)
718    }

870    /// Cancels an asynchronous message, reimbursing `msg.coins` to the sender
871    ///
872    /// # Arguments
873    /// * `msg`: the asynchronous message to cancel
874    ///
875    /// Note that if the sender address is missing in the ledger (either from a network restart with an edited ledger or from a potential future delete address feature),
876    /// the transfer will fail and the message coins will be burned. We still want to cancel the message, to avoid clogging the pool with a message that will be cancelled again.
877    pub fn cancel_async_message(
878        &mut self,
879        msg: &AsyncMessage,
880    ) -> Option<(Address, Result<Amount, String>)> {
881        #[allow(unused_assignments, unused_mut)]
882        let mut result = None;
883        let transfer_result = self.transfer_coins(
884            None,
885            Some(msg.sender),
886            msg.coins,
887            false,
888            TransferContext::AyncMsgCancel(OriginTransferContext {
889                async_message_id: Some(msg.compute_id()),
890                ..Default::default()
891            }),
892        );
893        if let Err(e) = transfer_result.as_ref() {
894            warn!(
895                "async message cancel: reimbursement of {} failed (burning coins): {}",
896                msg.sender, e
897            );
898        }
899
900        #[cfg(feature = "execution-info")]
901        if let Err(e) = transfer_result {
902            result = Some((msg.sender, Err(e.to_string())))
903        } else {
904            result = Some((msg.sender, Ok(msg.coins)));
905        }
906
907        result
908    }

943    /// Try to slash `roll_count` rolls from the denounced address. If not enough rolls,
944    /// slash the available amount and return the result
945    ///
946    /// # Arguments
947    /// * `denounced_addr`: address to sell the rolls from
948    /// * `roll_count`: number of rolls to slash
949    pub fn try_slash_rolls(
950        &mut self,
951        denounced_addr: &Address,
952        roll_count: u64,
953        denunciation_idx: &DenunciationIndex,
954    ) -> Result<Amount, ExecutionError> {
955        // try to slash as many roll as available
956        let slashed_rolls = self
957            .speculative_roll_state
958            .try_slash_rolls(denounced_addr, roll_count);
959
960        // convert slashed rolls to coins (as deferred credits => coins)
961        let slashed_coins_from_rolls = self
962            .config
963            .roll_price
964            .checked_mul_u64(slashed_rolls.unwrap_or_default())
965            .ok_or_else(|| {
966                ExecutionError::RuntimeError(format!(
967                    "Cannot multiply roll price by {}",
968                    roll_count
969                ))
970            })?;
971
972        // what remains to slash (then will try to slash as many deferred credits as avail/what remains to be slashed)
973        let amount_remaining_to_slash = self
974            .config
975            .roll_price
976            .checked_mul_u64(roll_count)
977            .ok_or_else(|| {
978                ExecutionError::RuntimeError(format!(
979                    "Cannot multiply roll price by {}",
980                    roll_count
981                ))
982            })?
983            .saturating_sub(slashed_coins_from_rolls);
984
985        let mut total_slashed_coins = slashed_coins_from_rolls;
986
987        if amount_remaining_to_slash > Amount::zero() {
988            // There is still an amount to slash for this denunciation so we need to slash
989            // in deferred credits
990            let slashed_coins_in_deferred_credits =
991                self.speculative_roll_state.try_slash_deferred_credits(
992                    &self.slot,
993                    denounced_addr,
994                    &amount_remaining_to_slash,
995                    denunciation_idx,
996                );
997
998            total_slashed_coins =
999                total_slashed_coins.saturating_add(slashed_coins_in_deferred_credits);
1000
1001            let amount_remaining_to_slash_2 =
1002                amount_remaining_to_slash.saturating_sub(slashed_coins_in_deferred_credits);
1003            if amount_remaining_to_slash_2 > Amount::zero() {
1004                // Use saturating_mul_u64 to avoid an error (for just a warn!(..))
1005                warn!("Slashed {} coins (by selling rolls) and {} coins from deferred credits of address: {} but cumulative amount is lower than expected: {} coins",
1006                    slashed_coins_from_rolls, slashed_coins_in_deferred_credits, denounced_addr,
1007                    self.config.roll_price.saturating_mul_u64(roll_count)
1008                );
1009            }
1010        }
1011
1012        Ok(total_slashed_coins)
1013    }

1035    ///  
1036    /// Note that if the deferred credits's receiver address is missing in the ledger (either from a network restart with an edited ledger or from a potential future delete address feature),
1037    /// the transfer will fail and the coins of the deferred credits will be burned. We still want to consume the deferred credit, to avoid clogging the system with deferred credits that fail to execute.

1056                    TransferContext::DeferredCredits(OriginTransferContext {
1057                        ..Default::default()

1061                if let Err(e) = transfer_result.as_ref() {

1080    /// Finishes a slot and generates the execution output.
1081    /// Settles emitted asynchronous messages, reimburse the senders of deleted messages.
1082    /// Moves the output of the execution out of the context,
1083    /// resetting some context fields in the process.
1084    ///
1085    /// This is used to get the output of an execution before discarding the context.
1086    /// Note that we are not taking self by value to consume it because the context is shared.
1087    pub fn settle_slot(&mut self, block_info: Option<ExecutedBlockInfo>) -> ExecutionOutput {
1088        let slot = self.slot;
1089
1090        // execute the deferred credits coming from roll sells
1091        let deferred_credits_transfers = self.execute_deferred_credits(&slot);
1092
1093        // settle emitted async messages and reimburse the senders of deleted messages
1094        let deleted_messages = self.speculative_async_pool.settle_slot(
1095            &slot,
1096            &self.speculative_ledger.added_changes,
1097            self.config.async_msg_cst_gas_cost,
1098            self.is_execution_component_version_at_least(MIP_0002_EXECUTION_VERSION),
1099        );
1100
1101        let mut cancel_async_message_transfers = vec![];
1102        for (_msg_id, msg) in deleted_messages {
1103            if let Some(t) = self.cancel_async_message(&msg) {
1104                cancel_async_message_transfers.push(t)
1105            }
1106        }
1107
1108        // update module cache
1109        // Skip this for read-only executions: their speculative bytecode is never
1110        // finalized, so importing it would only pollute the shared RAM + on-disk module
1111        // cache (potentially evicting legitimately warmed modules) with no cache-warming
1112        // benefit. Real executions (read_only == false) still warm the cache as intended.
1113        if !self.read_only {
1114            let bc_updates = self.speculative_ledger.added_changes.get_bytecode_updates();
1115            let mut cache_write_lock = self.module_cache.write();
1116            for bytecode in bc_updates {
1117                cache_write_lock.save_module(&bytecode.0, self.config.condom_limits.clone());
1118            }
1119        }
1120
1121        // if the current slot is last in cycle check the production stats and act accordingly
1122        let auto_sell_rolls = if self
1123            .slot
1124            .is_last_of_cycle(self.config.periods_per_cycle, self.config.thread_count)
1125        {
1126            self.speculative_roll_state.settle_production_stats(
1127                &slot,
1128                self.config.periods_per_cycle,
1129                self.config.thread_count,
1130                self.config.roll_price,
1131                self.config.max_miss_ratio,
1132                self.is_execution_component_version_at_least(MIP_0002_EXECUTION_VERSION),
1133            )
1134        } else {
1135            vec![]
1136        };
1137
1138        // generate the execution output
1139        let state_changes = StateChanges {
1140            ledger_changes: self.speculative_ledger.take(),
1141            async_pool_changes: self.speculative_async_pool.take(),
1142            deferred_call_changes: self.speculative_deferred_calls.take(),
1143            pos_changes: self.speculative_roll_state.take(),
1144            executed_ops_changes: self.speculative_executed_ops.take(),
1145            executed_denunciations_changes: self.speculative_executed_denunciations.take(),
1146            execution_trail_hash_change: SetOrKeep::Set(self.execution_trail_hash),
1147        };
1148
1149        let transfers_history = std::mem::take(&mut *self.transfers_history.write());
1150
1151        std::mem::take(&mut self.opt_block_id);
1152        ExecutionOutput {
1153            slot,
1154            block_info,
1155            state_changes,
1156            events: std::mem::take(&mut self.events),
1157            #[cfg(feature = "execution-trace")]
1158            slot_trace: None,
1159            #[cfg(feature = "dump-block")]
1160            storage: None,
1161            deferred_credits_execution: deferred_credits_transfers,
1162            cancel_async_message_execution: cancel_async_message_transfers,
1163            auto_sell_execution: auto_sell_rolls,
1164            transfers_history,
1165            execution_info: None,
1166        }
1167    }

1385    /// when a deferred call execution fails we need to refund the coins to the caller.
1386    ///
1387    /// Note that if the sender address is missing in the ledger (either from a network restart with an edited ledger or from a potential future delete address feature),
1388    /// the transfer will fail and the coins of the deferred call will be burned. We still want to consume the deferred call, to avoid clogging the registry with failed deferred calls.
1389    pub fn deferred_call_fail_exec(
1390        &mut self,
1391        id: &DeferredCallId,
1392        call: &DeferredCall,
1393    ) -> Option<(Address, Result<Amount, String>)> {
1394        #[allow(unused_assignments, unused_mut)]
1395        let mut result = None;
1396
1397        let transfer_result = self.transfer_coins(
1398            None,
1399            Some(call.sender_address),
1400            call.coins,
1401            false,
1402            TransferContext::DeferredCallFail(OriginTransferContext {
1403                deferred_call_id: Some(id.clone()),
1404                operation_id: self.origin_operation_id,
1405                async_message_id: self.async_msg_id,
1406                ..Default::default()
1407            }),
1408        );
1409        if let Err(e) = transfer_result.as_ref() {
1410            warn!(
1411                "deferred call {} fail: reimbursement of {} to {} failed (burning coins): {}",
1412                id, call.coins, call.sender_address, e
1413            );
1414        }
1415
1416        #[cfg(feature = "execution-info")]
1417        if let Err(e) = transfer_result {
1418            result = Some((call.sender_address, Err(e.to_string())))
1419        } else {
1420            result = Some((call.sender_address, Ok(call.coins)));
1421        }
1422
1423        result
1424    }

1426    /// when a deferred call is cancelled we need to refund the coins to the caller
1427    ///
1428    /// Note that if the sender address is missing in the ledger (either from a network restart with an edited ledger or from a potential future delete address feature),
1429    /// the transfer will fail and the coins of the deferred call will be burned. We still want to cancel the deferred call, to avoid clogging the registry with cancelled deferred calls.
1430    pub fn deferred_call_cancel(
1431        &mut self,
1432        call_id: &DeferredCallId,
1433        caller_address: Address,
1434    ) -> Result<(), ExecutionError> {
1435        match self.speculative_deferred_calls.get_call(call_id) {
1436            Some(call) => {
1437                // check that the caller is the one who registered the deferred call
1438                if call.sender_address != caller_address {
1439                    return Err(ExecutionError::DeferredCallsError(format!(
1440                        "only the caller {} can cancel the deferred call (cancel caller: {})",
1441                        call.sender_address, caller_address
1442                    )));
1443                }
1444
1445                let (address, amount) = self.speculative_deferred_calls.cancel_call(call_id)?;
1446
1447                // refund the coins to the caller
1448                let transfer_result = self.transfer_coins(
1449                    None,
1450                    Some(address),
1451                    amount,
1452                    false,
1453                    TransferContext::DeferredCallCancel(OriginTransferContext {
1454                        deferred_call_id: Some(call_id.clone()),
1455                        operation_id: self.origin_operation_id,
1456                        async_message_id: self.async_msg_id,
1457                        ..Default::default()
1458                    }),
1459                );
1460                if let Err(e) = transfer_result.as_ref() {
1461                    warn!(
1462                        "deferred call cancel: reimbursement of {} failed (burning coins): {}",
1463                        address, e
1464                    );
1465                }
1466
1467                Ok(())
1468            }
1469            _ => Err(ExecutionError::DeferredCallsError(format!(
1470                "deferred call {} does not exist",
1471                call_id
1472            )))?,
1473        }
1474    }

1507/// Execution component version active at `slot`, and whether that version is
1508/// strictly greater than at the previous slot (any bump). Genesis counts as an
1509/// upgrade when the version is greater than 0.
1510///
1511/// Returns `(version, upgraded)`.
1512pub(crate) fn execution_component_version_info(
1513    mip_store: &MipStore,
1514    slot: &Slot,
1515    thread_count: u8,
1516    t0: MassaTime,
1517    genesis_timestamp: MassaTime,
1518) -> (u32, bool) {
1519    let version =
1520        execution_component_version_at(mip_store, slot, thread_count, t0, genesis_timestamp);
1521    let upgraded = match slot.get_prev_slot(thread_count) {
1522        Ok(prev) => {
1523            version
1524                > execution_component_version_at(
1525                    mip_store,
1526                    &prev,
1527                    thread_count,
1528                    t0,
1529                    genesis_timestamp,
1530                )
1531        }
1532        Err(_) => version > 0,
1533    };
1534    (version, upgraded)
1535}
1536
1537fn execution_component_version_at(
1538    mip_store: &MipStore,
1539    slot: &Slot,
1540    thread_count: u8,
1541    t0: MassaTime,
1542    genesis_timestamp: MassaTime,
1543) -> u32 {
1544    let ts = get_block_slot_timestamp(thread_count, t0, genesis_timestamp, *slot)
1545        .unwrap_or(genesis_timestamp);
1546    mip_store.get_latest_component_version_at(&MipComponent::Execution, ts)
1547}

1549/// Generate the execution trail hash
1550fn generate_execution_trail_hash(
1551    previous_execution_trail_hash: &massa_hash::Hash,
1552    slot: &Slot,
1553    opt_block_id: Option<&BlockId>,
1554    read_only: bool,
1555) -> massa_hash::Hash {
1556    match opt_block_id {
1557        None => massa_hash::Hash::compute_from_tuple(&[
1558            previous_execution_trail_hash.to_bytes(),
1559            &slot.to_bytes_key(),
1560            &[if read_only { 1u8 } else { 0u8 }, 0u8],
1561        ]),
1562        Some(block_id) => {
1563            let mut bytes = Vec::new();
1564            let block_id_serializer = BlockIdSerializer::new();
1565            block_id_serializer.serialize(block_id, &mut bytes).unwrap();
1566            massa_hash::Hash::compute_from_tuple(&[
1567                previous_execution_trail_hash.to_bytes(),
1568                &slot.to_bytes_key(),
1569                &[if read_only { 1u8 } else { 0u8 }, 1u8],
1570                &bytes,
1571            ])
1572        }
1573    }
1574}

1576/// Initializes and seeds the PRNG with the given execution trail hash.
1577fn init_prng(execution_trail_hash: &massa_hash::Hash) -> Xoshiro256PlusPlus {
1578    // Deterministically seed the unsafe RNG to allow the bytecode to use it.
1579    // Note that consecutive read-only calls for the same slot will get the same random seed.
1580    let seed = massa_hash::Hash::compute_from_tuple(&[
1581        "PRNG_SEED".as_bytes(),
1582        execution_trail_hash.to_bytes(),
1583    ])
1584    .into_bytes();
1585
1586    // We use Xoshiro256PlusPlus because it is very fast,
1587    // has a period long enough to ensure no repetitions will ever happen,
1588    // of decent quality (given the unsafe constraints)
1589    // but not cryptographically secure (and that's ok because the internal state is exposed anyway)
1590    Xoshiro256PlusPlus::from_seed(seed)
1591}

1593/// Truncates a string to a maximum byte length, ensuring it remains valid UTF-8 in the process.
1594pub(crate) fn truncate_string(string: String, max_byte_len: usize) -> String {
1595    let mut truncated_bytes = string.as_bytes().to_vec();
1596    truncated_bytes.truncate(max_byte_len);
1597    let mut res_str = String::from_utf8_lossy(&truncated_bytes);
1598    while res_str.len() > max_byte_len {
1599        truncated_bytes.pop();
1600        res_str = String::from_utf8_lossy(&truncated_bytes);
1601    }
1602    res_str.to_string()
1603}
