111/// Structure holding consistent speculative and final execution states,
112/// and allowing access to them.
113pub(crate) struct ExecutionState {
114    // execution config
115    config: ExecutionConfig,
116    // History of the outputs of recently executed slots. Slots should be consecutive, newest at the back.
117    // Whenever an active slot is executed, it is appended at the back of active_history.
118    // Whenever an executed active slot becomes final,
119    // its output is popped from the front of active_history and applied to the final state.
120    // It has atomic R/W access.
121    pub active_history: Arc<RwLock<ActiveHistory>>,
122    // a cursor pointing to the highest executed slot
123    pub active_cursor: Slot,
124    // a cursor pointing to the highest executed final slot
125    pub final_cursor: Slot,
126    // store containing execution events that became final
127    final_events_cache: Box<dyn EventCacheController>,
128    // final state with atomic R/W access
129    final_state: Arc<RwLock<dyn FinalStateController>>,
130    // execution context (see documentation in context.rs)
131    execution_context: Arc<Mutex<ExecutionContext>>,
132    // execution interface allowing the VM runtime to access the Massa context
133    execution_interface: Box<dyn Interface>,
134    // execution statistics
135    stats_counter: ExecutionStatsCounter,
136    // cache of pre compiled sc modules
137    pub module_cache: Arc<RwLock<ModuleCache>>,
138    // MipStore (Versioning)
139    mip_store: MipStore,
140    // wallet used to verify double staking on local addresses
141    wallet: Arc<RwLock<Wallet>>,
142    // selector controller to get draws
143    selector: Box<dyn SelectorController>,
144    // channels used by the execution worker

146    /// prometheus metrics

157    /// Create a new execution state. This should be called only once at the start of the execution worker.
158    ///
159    /// # Arguments
160    /// * `config`: execution configuration
161    /// * `final_state`: atomic access to the final state
162    ///
163    /// # returns
164    /// A new `ExecutionState`
165    #[allow(clippy::too_many_arguments)]
166    pub fn new(
167        config: ExecutionConfig,
168        final_state: Arc<RwLock<dyn FinalStateController>>,
169        mip_store: MipStore,
170        selector: Box<dyn SelectorController>,
171        channels: ExecutionChannels,
172        wallet: Arc<RwLock<Wallet>>,
173        massa_metrics: MassaMetrics,
174        event_cache: Box<dyn EventCacheController>,
175        #[cfg(feature = "dump-block")] block_storage_backend: Arc<RwLock<dyn StorageBackend>>,
176    ) -> ExecutionState {
177        // Get the slot at the output of which the final state is attached.
178        // This should be among the latest final slots.
179        let last_final_slot;
180        let execution_trail_hash;
181        {
182            let final_state_read = final_state.read();
183            last_final_slot = final_state_read.get_slot();
184            execution_trail_hash = final_state_read.get_execution_trail_hash();
185        }
186
187        // Create default active history
188        let active_history: Arc<RwLock<ActiveHistory>> = Default::default();
189
190        // Initialize the SC module cache
191        let module_cache = Arc::new(RwLock::new(ModuleCache::new(ModuleCacheConfig {
192            hd_cache_path: config.hd_cache_path.clone(),
193            gas_costs: config.gas_costs.clone(),
194            lru_cache_size: config.lru_cache_size,
195            hd_cache_size: config.hd_cache_size,
196            snip_amount: config.snip_amount,
197            max_module_length: config.max_bytecode_size,
198            condom_limits: config.condom_limits.clone(),
199        })));
200
201        // Create an empty placeholder execution context, with shared atomic access
202        let execution_context = ExecutionContext::new(
203            config.clone(),
204            final_state.clone(),
205            active_history.clone(),
206            module_cache.clone(),
207            mip_store.clone(),
208            execution_trail_hash,
209        );
210        let execution_context = Arc::new(Mutex::new(execution_context));
211
212        // Instantiate the interface providing ABI access to the VM, share the execution context with it
213        let execution_interface = Box::new(InterfaceImpl::new(
214            config.clone(),
215            execution_context.clone(),
216        ));
217
218        // build the execution state
219        ExecutionState {
220            final_state,
221            execution_context,
222            execution_interface,
223            // empty execution output history: it is not recovered through bootstrap
224            active_history,
225            // empty final event store: it is not recovered through bootstrap
226            // final_events: Default::default(),
227            final_events_cache: event_cache,
228            // no active slots executed yet: set active_cursor to the last final block
229            active_cursor: last_final_slot,
230            final_cursor: last_final_slot,
231            stats_counter: ExecutionStatsCounter::new(config.stats_time_window_duration),
232            module_cache,
233            mip_store,
234            selector,
235            channels,
236            wallet,
237            massa_metrics,
238            #[cfg(feature = "execution-trace")]
239            trace_history: Arc::new(RwLock::new(TraceHistory::new(
240                config.max_execution_traces_slot_limit as u32,
241                std::cmp::min(
242                    MAX_OPERATIONS_PER_BLOCK,
243                    (MAX_GAS_PER_BLOCK / BASE_OPERATION_GAS_COST) as u32,
244                ),
245            ))),
246            // #[cfg(feature = "execution-info")]
247            // execution_info: Arc::new(RwLock::new(ExecutionInfo::new(
248            //     config.max_execution_traces_slot_limit as u32,
249            // ))),
250            config,
251            #[cfg(feature = "dump-block")]
252            block_storage_backend,
253        }
254    }

267    /// Applies the output of an execution to the final execution state.
268    /// The newly applied final output should be from the slot just after the last executed final slot
269    ///
270    /// # Arguments
271    /// * `exec_out`: execution output to apply
272    pub fn apply_final_execution_output(&mut self, mut exec_out: ExecutionOutput) {
273        if self.final_cursor >= exec_out.slot {
274            panic!("attempting to apply a final execution output at or before the current final_cursor");
275        }
276
277        // count stats
278        if exec_out.block_info.is_some() {
279            self.stats_counter.register_final_blocks(1);
280            self.stats_counter.register_final_executed_operations(
281                exec_out.state_changes.executed_ops_changes.len(),
282            );
283            self.stats_counter.register_final_executed_denunciations(
284                exec_out.state_changes.executed_denunciations_changes.len(),
285            );
286        }
287
288        let exec_out_2 = exec_out.clone();
289        #[cfg(feature = "slot-replayer")]
290        {
291            println!(">>> Execution changes");
292            println!("{:#?}", serde_json::to_string_pretty(&exec_out));
293            println!("<<<");
294        }
295        // apply state changes to the final ledger
296        let network_versions = exec_out
297            .block_info
298            .as_ref()
299            .map(|i| (i.current_version, i.announced_version));
300        self.final_state
301            .write()
302            .finalize(exec_out.slot, exec_out.state_changes, network_versions);
303
304        // update the final ledger's slot
305        self.final_cursor = exec_out.slot;
306
307        // update active cursor:
308        // if it was at the previous latest final block, set it to point to the new one
309        if self.active_cursor < self.final_cursor {
310            self.active_cursor = self.final_cursor;
311        }
312
313        // append generated events to the final event store
314        exec_out.events.finalize();
315
316        self.final_events_cache.save_events(exec_out.events.0);
317
318        // update the prometheus metrics
319        self.massa_metrics
320            .set_active_cursor(self.active_cursor.period, self.active_cursor.thread);
321        self.massa_metrics
322            .set_final_cursor(self.final_cursor.period, self.final_cursor.thread);
323        self.massa_metrics.inc_operations_final_counter(
324            exec_out_2.state_changes.executed_ops_changes.len() as u64,
325        );
326        self.massa_metrics
327            .set_active_history(self.active_history.read().0.len());
328
329        self.massa_metrics
330            .inc_sc_messages_final_by(exec_out_2.state_changes.async_pool_changes.0.len());
331
332        self.massa_metrics.set_async_message_pool_size(
333            self.final_state.read().get_async_pool().message_cache.len(),
334        );
335
336        self.massa_metrics.inc_executed_final_slot();
337        if exec_out.block_info.is_some() {
338            self.massa_metrics.inc_executed_final_slot_with_block();
339        }
340
341        // Broadcast a final slot execution output to active channel subscribers.
342        if self.config.broadcast_enabled {
343            let slot_exec_out = SlotExecutionOutput::FinalizedSlot(exec_out_2);
344            if let Err(err) = self
345                .channels
346                .slot_execution_output_sender
347                .send(slot_exec_out)
348            {
349                trace!(
350                    "error, failed to broadcast final execution output for slot {} due to: {}",
351                    exec_out.slot,
352                    err
353                );
354            }
355        }
356
357        #[cfg(feature = "execution-trace")]
358        {
359            if self.config.broadcast_traces_enabled {
360                if let Some((slot_trace, transfers)) = exec_out.slot_trace.clone() {
361                    if let Err(err) = self
362                        .channels
363                        .slot_execution_traces_sender
364                        .send((slot_trace, transfers, true))
365                    {
366                        trace!(
367                            "error, failed to broadcast abi trace for slot {} due to: {}",
368                            exec_out.slot.clone(),
369                            err
370                        );
371                    }
372                }
373            }
374        }
375
376        #[cfg(feature = "execution-info")]
377        {
378            if self.config.broadcast_enabled {
379                // The exec_info was attached to `exec_out` at the end of
380                // `execute_slot`, so it carries the slot, block_id,
381                // timestamp and execution_trail_hash of the *finalized*
382                // slot (rather than whatever slot the execution context
383                // happens to be on right now). Transfers are still moved
384                // in from `exec_out.transfers_history` at this point:
385                // that vec is only populated now because it was left on
386                // `exec_out` by `settle_slot`.
387                if let Some(mut exec_info_for_slot) = exec_out.execution_info.take() {
388                    exec_info_for_slot.transfers = exec_out.transfers_history;
389                    exec_info_for_slot.build_transfer_list();
390                    if let Err(err) = self
391                        .channels
392                        .slot_execution_info_sender
393                        .send(exec_info_for_slot)
394                    {
395                        trace!(
396                            "error, failed to broadcast execution info for slot {} due to: {}",
397                            exec_out.slot.clone(),
398                            err
399                        );
400                    }
401                }
402            }
403        }
404
405        #[cfg(feature = "dump-block")]
406        {
407            let mut block_ser = vec![];
408            if let Some(block_info) = exec_out.block_info {
409                let block_id = block_info.block_id;
410                let storage = exec_out.storage.unwrap();
411                let guard = storage.read_blocks();
412                let secured_block = guard
413                    .get(&block_id)
414                    .unwrap_or_else(|| panic!("Unable to get block for block id: {}", block_id));
415
416                let operations: Vec<(OperationId, Option<SecureShare<Operation, OperationId>>)> =
417                    secured_block
418                        .content
419                        .operations
420                        .iter()
421                        .map(|operation_id| {
422                            match storage.read_operations().get(operation_id).cloned() {
423                                Some(verifiable_operation) => {
424                                    (*operation_id, Some(verifiable_operation))
425                                }
426                                None => (*operation_id, None),
427                            }
428                        })
429                        .collect();
430
431                let filled_block = FilledBlock {
432                    header: secured_block.content.header.clone(),
433                    operations,
434                };
435
436                let grpc_filled_block = grpc_model::FilledBlock::from(filled_block);
437                grpc_filled_block.encode(&mut block_ser).unwrap();
438            }
439
440            self.block_storage_backend
441                .write()
442                .write(&exec_out.slot, &block_ser);
443        }
444    }

446    /// Applies an execution output to the active (non-final) state
447    /// The newly active final output should be from the slot just after the last executed active slot
448    ///
449    /// # Arguments
450    /// * `exec_out`: execution output to apply
451    pub fn apply_active_execution_output(&mut self, exec_out: ExecutionOutput) {
452        if self.active_cursor >= exec_out.slot {
453            panic!("attempting to apply an active execution output at or before the current active_cursor");
454        }
455        if exec_out.slot <= self.final_cursor {
456            panic!("attempting to apply an active execution output at or before the current final_cursor");
457        }
458
459        // update active cursor to reflect the new latest active slot
460        self.active_cursor = exec_out.slot;
461
462        // add the execution output at the end of the output history
463        self.active_history.write().0.push_back(exec_out);
464
465        // update the prometheus metrics
466        self.massa_metrics
467            .set_active_history(self.active_history.read().0.len())
468    }

470    /// Helper function.
471    /// Within a locked execution context (lock is taken at the beginning of the function then released at the end):
472    /// - if not yet executed then transfer fee and add the operation to the context then return a context snapshot
473    ///
474    /// # Arguments
475    /// * `operation`: operation to be schedule
476    /// * `sender_addr`: sender address for the operation (for fee transfer)
477    fn prepare_operation_for_execution(
478        &self,
479        operation: &SecureShareOperation,
480        sender_addr: Address,
481    ) -> Result<ExecutionContextSnapshot, ExecutionError> {
482        let operation_id = operation.id;
483
484        // lock execution context
485        let mut context = context_guard!(self);
486
487        // ignore the operation if it was already executed
488        if context.is_op_executed(&operation_id) {
489            return Err(ExecutionError::IncludeOperationError(
490                "operation was executed previously".to_string(),
491            ));
492        }
493
494        // Compute the minimal amount of coins the sender is allowed to have after the execution of this op based on `op.max_spending`.
495        // Note that the max spending might exceed the sender's balance.
496        let creator_initial_balance = context
497            .get_balance(&sender_addr)
498            .unwrap_or_else(Amount::zero);
499        context.creator_min_balance = Some(
500            creator_initial_balance
501                .saturating_sub(operation.get_max_spending(self.config.roll_price)),
502        );
503
504        // set the context origin operation ID
505        // Note: set operation ID early as if context.transfer_coins fails, event_create will use
506        // operation ID in the event message
507        context.origin_operation_id = Some(operation_id);
508
509        // debit the fee from the operation sender
510        if let Err(err) = context.transfer_coins(
511            Some(sender_addr),
512            None,
513            operation.content.fee,
514            false,
515            TransferContext::OperationFee(OriginTransferContext {
516                operation_id: Some(operation_id),
517                ..Default::default()
518            }),
519        ) {
520            let mut error = format!("could not spend fees: {}", err);
521            let max_event_size = self.config.max_event_size;
522            if error.len() > max_event_size {
523                error = truncate_string(error, max_event_size);
524            }
525            let event = context.event_create(error.clone(), true);
526            context.event_emit(event);
527            return Err(ExecutionError::IncludeOperationError(error));
528        }
529
530        // from here, fees have been transferred.
531        // Op will be executed just after in the context of a snapshot.
532
533        // save a snapshot of the context to revert any further changes on error
534        let context_snapshot = context.get_snapshot();
535
536        // set the creator address
537        context.creator_address = Some(operation.content_creator_address);
538        context.gas_remaining_before_subexecution = None;
539        context.recursion_counter = 0;
540        context.user_event_count_in_current_exec = 0;
541
542        Ok(context_snapshot)
543    }

545    /// Execute an operation in the context of a block.
546    /// Assumes the execution context was initialized at the beginning of the slot.
547    ///
548    /// # Arguments
549    /// * `operation`: operation to execute
550    /// * `block_slot`: slot of the block in which the op is included
551    /// * `remaining_block_gas`: mutable reference towards the remaining gas in the block
552    /// * `block_credits`: mutable reference towards the total block reward/fee credits
553    pub fn execute_operation(
554        &self,
555        operation: &SecureShareOperation,
556        block_slot: Slot,
557        remaining_block_gas: &mut u64,
558        block_credits: &mut Amount,
559    ) -> Result<ExecutionResult, ExecutionError> {
560        // check validity period
561        if !(operation
562            .get_validity_range(self.config.operation_validity_period)
563            .contains(&block_slot.period))
564        {
565            return Err(ExecutionError::InvalidSlotRange);
566        }
567
568        // check remaining block gas
569        let op_gas = operation.get_gas_usage(
570            self.config.base_operation_gas_cost,
571            self.config.gas_costs.sp_compilation_cost,
572        );
573        let new_remaining_block_gas = remaining_block_gas.checked_sub(op_gas).ok_or_else(|| {
574            ExecutionError::NotEnoughGas(
575                "not enough remaining block gas to execute operation".to_string(),
576            )
577        })?;
578
579        // get the operation's sender address
580        let sender_addr = operation.content_creator_address;
581
582        // get the thread to which the operation belongs
583        let op_thread = sender_addr.get_thread(self.config.thread_count);
584
585        // check block/op thread compatibility
586        if op_thread != block_slot.thread {
587            return Err(ExecutionError::IncludeOperationError(
588                "operation vs block thread mismatch".to_string(),
589            ));
590        }
591
592        // get operation ID
593        let operation_id = operation.id;
594
595        // Add fee from operation.
596        let new_block_credits = block_credits.saturating_add(operation.content.fee);
597
598        let context_snapshot = self.prepare_operation_for_execution(operation, sender_addr)?;
599
600        // update block gas
601        *remaining_block_gas = new_remaining_block_gas;
602
603        // update block credits
604        *block_credits = new_block_credits;
605
606        #[cfg(feature = "execution-trace")]
607        let res = vec![];
608        #[allow(clippy::let_unit_value)]
609        #[cfg(not(feature = "execution-trace"))]
610        let res = ();
611        // Call the execution process specific to the operation type.
612        let mut execution_result = match &operation.content.op {
613            OperationType::ExecuteSC { .. } => {
614                self.execute_executesc_op(&operation.content.op, sender_addr)
615            }
616            OperationType::CallSC { .. } => {
617                self.execute_callsc_op(&operation.content.op, sender_addr, operation_id)
618            }
619            OperationType::RollBuy { .. } => self
620                .execute_roll_buy_op(&operation.content.op, sender_addr, operation_id)
621                .map(|_| res),
622            OperationType::RollSell { .. } => self
623                .execute_roll_sell_op(&operation.content.op, sender_addr, operation_id)
624                .map(|_| res),
625            OperationType::Transaction { .. } => self
626                .execute_transaction_op(&operation.content.op, sender_addr, operation_id)
627                .map(|_| res),
628        };
629
630        {
631            // lock execution context
632            let mut context = context_guard!(self);
633
634            if execution_result.is_ok() {
635                // check that the `max_coins` spending limit was respected by the sender
636                if let Some(creator_min_balance) = &context.creator_min_balance {
637                    let creator_balance = context
638                        .get_balance(&sender_addr)
639                        .unwrap_or_else(Amount::zero);
640                    if &creator_balance < creator_min_balance {
641                        execution_result = Err(ExecutionError::RuntimeError(format!(
642                            "at the end of the execution of the operation, the sender {} was expected to have at least {} coins according to the operation's max spending, but has only {}.",
643                            sender_addr, creator_min_balance, creator_balance
644                        )));
645                    }
646                }
647            }
648
649            // check execution results
650            match execution_result {
651                Ok(_value) => {
652                    context.insert_executed_op(
653                        operation_id,
654                        true,
655                        Slot::new(operation.content.expire_period, op_thread),
656                    );
657                    #[cfg(feature = "execution-trace")]
658                    {
659                        Ok((_value, true))
660                    }
661                    #[cfg(not(feature = "execution-trace"))]
662                    {
663                        Ok(())
664                    }
665                }
666                Err(err) => {
667                    // an error occurred: emit error event and reset context to snapshot
668                    let err = ExecutionError::RuntimeError(format!(
669                        "runtime error when executing operation {}: {}",
670                        operation_id, &err
671                    ));
672                    debug!("{}", &err);
673                    context.reset_to_snapshot(context_snapshot, err);
674
675                    // Insert op AFTER the context has been restored (otherwise it would be overwritten)
676                    context.insert_executed_op(
677                        operation_id,
678                        false,
679                        Slot::new(operation.content.expire_period, op_thread),
680                    );
681                    #[cfg(feature = "execution-trace")]
682                    {
683                        Ok((vec![], false))
684                    }
685                    #[cfg(not(feature = "execution-trace"))]
686                    {
687                        Ok(())
688                    }
689                }
690            }
691        }
692    }

694    /// Execute a denunciation in the context of a block.
695    ///
696    /// # Arguments
697    /// * `denunciation`: denunciation to process
698    /// * `block_credits`: mutable reference towards the total block reward/fee credits
699    fn execute_denunciation(
700        &self,
701        denunciation: &Denunciation,
702        block_slot: &Slot,
703        block_credits: &mut Amount,
704    ) -> Result<DenunciationResult, ExecutionError> {
705        let addr_denounced = Address::from_public_key(denunciation.get_public_key());
706
707        // acquire write access to the context
708        let mut context = context_guard!(self);
709
710        let de_slot = denunciation.get_slot();
711
712        if de_slot.period <= self.config.last_start_period {
713            // denunciation created before last restart (can be 0 or >= 0 after a network restart) - ignored
714            // Note: as we use '<=', also ignore denunciation created for genesis block
715            return Err(ExecutionError::IncludeDenunciationError(format!(
716                "Denunciation target ({}) is before the last start period: {}",
717                de_slot, self.config.last_start_period
718            )));
719        }
720
721        // ignore denunciation if not valid
722        //
723        // The signature reconstruction is chain-scoped once MIP-0002's `Execution`
724        // component v2 is active for the denounced slot: the signed hash then folds
725        // in the local chain id. This prevents an attacker from combining two
726        // legitimate same-(slot, index) endorsements / block headers produced by
727        // the same validator on two different chains into a denunciation on either
728        // chain (F90 / PDF #11). Pre-activation `sig_chain_id` is `None` and the
729        // legacy (chain-agnostic) layout is used, so existing denunciations keep
730        // verifying. The layout is decided by the network-agreed active version for
731        // the denounced slot, not any self-claimed header field.
732        let de_slot_ts = get_block_slot_timestamp(
733            self.config.thread_count,
734            self.config.t0,
735            self.config.genesis_timestamp,
736            *de_slot,
737        )
738        .map_err(|e| {
739            ExecutionError::IncludeDenunciationError(format!(
740                "cannot compute denounced slot timestamp: {e}"
741            ))
742        })?;
743        let sig_chain_id = massa_versioning::consensus_signature::sig_chain_id_for_slot(
744            &self.mip_store,
745            self.config.chain_id,
746            de_slot_ts,
747        );
748        if !denunciation.is_valid_with_chain(sig_chain_id) {
749            return Err(ExecutionError::IncludeDenunciationError(
750                "denunciation is not valid".to_string(),
751            ));
752        }
753
754        // ignore denunciation if too old or expired
755
756        if Denunciation::is_expired(
757            &de_slot.period,
758            &block_slot.period,
759            &self.config.denunciation_expire_periods,
760        ) {
761            // too old - cannot be denounced anymore
762            return Err(ExecutionError::IncludeDenunciationError(format!(
763                "Denunciation target ({}) is too old with respect to the block ({})",
764                de_slot, block_slot
765            )));
766        }
767
768        if de_slot > block_slot {
769            // too much in the future - ignored
770            // Note: de_slot == block_slot is OK,
771            //       for example if the block producer wants to denounce someone who multi-endorsed
772            //       for the block's slot
773            return Err(ExecutionError::IncludeDenunciationError(format!(
774                "Denunciation target ({}) is at a later slot than the block slot ({})",
775                de_slot, block_slot
776            )));
777        }
778
779        // ignore the denunciation if it was already executed
780        let de_idx = DenunciationIndex::from(denunciation);
781        if context.is_denunciation_executed(&de_idx) {
782            return Err(ExecutionError::IncludeDenunciationError(
783                "Denunciation was already executed".to_string(),
784            ));
785        }
786
787        // Check selector
788        // Note 1: Has to be done after slot limit and executed check
789        // Note 2: that this is done for a node to create a Block with 'fake' denunciation thus
790        //       include them in executed denunciation and prevent (by occupying the corresponding entry)
791        //       any further 'real' denunciation.
792
793        match &denunciation {
794            Denunciation::Endorsement(_de) => {
795                // Get selected address from selector and check
796                let selection = self
797                    .selector
798                    .get_selection(*de_slot)
799                    .expect("Could not get producer from selector");
800                let selected_addr = selection
801                    .endorsements
802                    .get(*denunciation.get_index().unwrap_or(&0) as usize)
803                    .expect("could not get selection for endorsement at index");
804
805                if *selected_addr != addr_denounced {
806                    return Err(ExecutionError::IncludeDenunciationError(
807                        "Attempt to execute a denunciation but address was not selected"
808                            .to_string(),
809                    ));
810                }
811            }
812            Denunciation::BlockHeader(_de) => {
813                let selected_addr = self
814                    .selector
815                    .get_producer(*de_slot)
816                    .expect("Cannot get producer from selector");
817
818                if selected_addr != addr_denounced {
819                    return Err(ExecutionError::IncludeDenunciationError(
820                        "Attempt to execute a denunciation but address was not selected"
821                            .to_string(),
822                    ));
823                }
824            }
825        }
826
827        context.insert_executed_denunciation(&de_idx);
828
829        // Punishment is roll slashing only; same-slot block/endorsement rewards are not forfeited
830        // (see comment after the denunciation loop in execute_slot).
831        let slashed = context.try_slash_rolls(
832            &addr_denounced,
833            self.config.roll_count_to_slash_on_denunciation,
834            &de_idx,
835        );
836
837        match slashed.as_ref() {
838            Ok(slashed_amount) => {
839                // Add slashed amount / 2 to block reward
840                let amount = slashed_amount.checked_div_u64(2).ok_or_else(|| {
841                    ExecutionError::RuntimeError(format!(
842                        "Unable to divide slashed amount: {} by 2",
843                        slashed_amount
844                    ))
845                })?;
846                *block_credits = block_credits.saturating_add(amount);
847            }
848            Err(e) => {
849                warn!("Unable to slash rolls or deferred credits: {}", e);
850            }
851        }
852
853        if self
854            .wallet
855            .read()
856            .get_wallet_address_list()
857            .contains(&addr_denounced)
858        {
859            match &denunciation.is_for_block_header() {
860                true => panic!("You are being slashed at slot {} for double-staking using address {}. The node is stopping to prevent any further loss. Block header denunciation of block at slot {:?}. Denunciation's public key: {:?}", block_slot, addr_denounced, denunciation.get_slot(), denunciation.get_public_key()),
861                false => panic!("You are being slashed at slot {} for double-staking using address {}. The node is stopping to prevent any further loss. Endorsement denunciation of endorsement at slot {:?} and index {:?}. Denunciation's public key: {:?}", block_slot, addr_denounced, denunciation.get_slot(), denunciation.get_index(), denunciation.get_public_key())
862            }
863        }
864
865        Ok(DenunciationResult {
866            address_denounced: addr_denounced,
867            slot: *de_slot,
868            slashed: slashed.unwrap_or_default(),
869        })
870    }

1125                param,
1126                coins,
1127                ..
1128            } => (*max_gas, *target_addr, target_func, param, *coins),
1129            _ => panic!("unexpected operation type"),
1130        };
1131
1132        // prepare the current slot context for executing the operation
1133        let bytecode;
1134        let condom_limits;
1135        {
1136            // acquire write access to the context
1137            let mut context = context_guard!(self);
1138
1139            condom_limits = context.get_condom_limits();
1140
1141            // Set the call stack
1142            // This needs to be defined before anything can fail, so that the emitted event contains the right stack
1143            context.stack = vec![

1155                },

1159            // Ensure that the target address exists

1222        }

1225            Ok(())

1229    /// Tries to execute an asynchronous message
1230    /// If the execution failed reimburse the message sender.
1231    ///
1232    /// # Arguments
1233    /// * message: message information
1234    /// * bytecode: executable target bytecode, or None if unavailable
1235    pub fn execute_async_message(
1236        &self,
1237        message: AsyncMessage,
1238        bytecode: Option<Bytecode>,
1239    ) -> Result<AsyncMessageExecutionResult, ExecutionError> {
1240        let mut result = AsyncMessageExecutionResult::new();
1241        #[cfg(feature = "execution-info")]
1242        {
1243            // TODO: From impl + no ::new -> no cfg feature
1244            result.sender = Some(message.sender);
1245            result.destination = Some(message.destination);
1246        }
1247
1248        // prepare execution context
1249        let context_snapshot;
1250        let bytecode = {
1251            let mut context = context_guard!(self);
1252            context_snapshot = context.get_snapshot();
1253            context.creator_address = None;
1254            context.creator_min_balance = None;
1255            context.stack = vec![
1256                ExecutionStackElement {
1257                    address: message.sender,
1258                    coins: Default::default(),
1259                    owned_addresses: vec![message.sender],
1260                    operation_datastore: None,
1261                },
1262                ExecutionStackElement {
1263                    address: message.destination,
1264                    coins: message.coins,
1265                    owned_addresses: vec![message.destination],
1266                    operation_datastore: None,
1267                },
1268            ];
1269            context.origin_operation_id = None;
1270            context.gas_remaining_before_subexecution = None;
1271            context.recursion_counter = 0;
1272            context.user_event_count_in_current_exec = 0;
1273            context.async_msg_id = Some(message.compute_id());
1274
1275            // check the target address
1276            if let Err(err) = context.check_target_sc_address(message.destination) {
1277                context.reset_to_snapshot(context_snapshot, err.clone());
1278                context.cancel_async_message(&message);
1279                return Err(err);
1280            }
1281
1282            // if there is no bytecode: fail
1283            let bytecode = match bytecode {
1284                Some(bytecode) => bytecode,
1285                None => {
1286                    let err = ExecutionError::RuntimeError("no target bytecode found".into());
1287                    context.reset_to_snapshot(context_snapshot, err.clone());
1288                    context.cancel_async_message(&message);
1289                    return Err(err);
1290                }
1291            };
1292
1293            // credit coins to the target address
1294            if let Err(err) = context.transfer_coins(
1295                None,
1296                Some(message.destination),
1297                message.coins,
1298                false,
1299                TransferContext::AsyncMsgCoins(OriginTransferContext {
1300                    async_message_id: Some(message.compute_id()),
1301                    ..Default::default()
1302                }),
1303            ) {
1304                // coin crediting failed: reset context to snapshot and reimburse sender
1305                let err = ExecutionError::RuntimeError(format!(
1306                    "could not credit coins to target of async execution: {}",
1307                    err
1308                ));
1309
1310                context.reset_to_snapshot(context_snapshot, err.clone());
1311                context.cancel_async_message(&message);
1312                return Err(err);
1313            } else {
1314                result.coins = Some(message.coins);
1315            }
1316
1317            bytecode.0
1318        };
1319
1320        // load and execute the compiled module
1321        // IMPORTANT: do not keep a lock here as `run_function` uses the `get_module` interface
1322        let module = match self.module_cache.write().load_module(
1323            &bytecode,
1324            message.max_gas,
1325            self.config.condom_limits.clone(),
1326        ) {
1327            Ok(module) => module,
1328            Err(err) => {
1329                let err = ExecutionError::RuntimeError(format!(
1330                    "could not load module for async execution: {}",
1331                    err
1332                ));
1333                let mut context = context_guard!(self);
1334                context.reset_to_snapshot(context_snapshot, err.clone());
1335                context.cancel_async_message(&message);
1336                return Err(err);
1337            }
1338        };
1339
1340        let response = massa_sc_runtime::run_function(
1341            &*self.execution_interface,
1342            module,
1343            &message.function,
1344            &message.function_params,
1345            message.max_gas,
1346            self.config.gas_costs.clone(),
1347            self.config.condom_limits.clone(),
1348        );
1349        match response {
1350            Ok(res) => {
1351                self.module_cache
1352                    .write()
1353                    .set_init_cost(&bytecode, res.init_gas_cost);
1354                #[cfg(feature = "execution-trace")]
1355                {
1356                    result.traces = Some((res.trace.into_iter().map(|t| t.into()).collect(), true));
1357                }
1358                #[cfg(feature = "execution-info")]
1359                {
1360                    result.success = true;
1361                }
1362                Ok(result)
1363            }
1364            Err(error) => {
1365                if let VMError::ExecutionError { init_gas_cost, .. } = error {
1366                    self.module_cache
1367                        .write()
1368                        .set_init_cost(&bytecode, init_gas_cost);
1369                }
1370                // execution failed: reset context to snapshot and reimburse sender
1371                let err = ExecutionError::VMError {
1372                    context: "Asynchronous Message".to_string(),
1373                    error,
1374                };
1375                let mut context = context_guard!(self);
1376                context.reset_to_snapshot(context_snapshot, err.clone());
1377                context.cancel_async_message(&message);
1378                Err(err)
1379            }
1380        }
1381    }

1383    fn execute_deferred_call(
1384        &self,
1385        id: &DeferredCallId,
1386        call: DeferredCall,
1387    ) -> Result<DeferredCallExecutionResult, ExecutionError> {
1388        let mut result = DeferredCallExecutionResult::new(&call);
1389
1390        let snapshot = {
1391            let mut context = context_guard!(self);
1392
1393            // refund the sender for the storage costs
1394            let amount = DeferredCall::get_storage_cost(
1395                self.config.storage_costs_constants.ledger_cost_per_byte,
1396                call.parameters.len() as u64,
1397                self.config.max_function_length,
1398            );
1399            if let Err(e) = context.transfer_coins(
1400                None,
1401                Some(call.sender_address),
1402                amount,
1403                false,
1404                TransferContext::DeferredCallStorageRefund(OriginTransferContext {
1405                    deferred_call_id: Some(id.clone()),
1406                    ..Default::default()
1407                }),
1408            ) {
1409                warn!(
1410                    "could not refund storage costs to sender: {} - amount: {} - e:{}",
1411                    call.sender_address,
1412                    amount,
1413                    e.to_string()
1414                );
1415            }
1416
1417            context.get_snapshot()
1418        };
1419
1420        if call.cancelled {
1421            Ok(result)
1422        } else {
1423            let deferred_call_execution = || {
1424                let bytecode = {
1425                    // acquire write access to the context
1426                    let mut context = context_guard!(self);
1427
1428                    // Set the call stack
1429                    // This needs to be defined before anything can fail, so that the emitted event contains the right stack
1430                    context.stack = vec![
1431                        ExecutionStackElement {
1432                            address: call.sender_address,
1433                            coins: Default::default(),
1434                            owned_addresses: vec![call.sender_address],
1435                            operation_datastore: None,
1436                        },
1437                        ExecutionStackElement {
1438                            address: call.target_address,
1439                            coins: call.coins,
1440                            owned_addresses: vec![call.target_address],
1441                            operation_datastore: None,
1442                        },
1443                    ];
1444                    context.origin_operation_id = None;
1445                    context.gas_remaining_before_subexecution = None;
1446                    context.recursion_counter = 0;
1447                    context.user_event_count_in_current_exec = 0;
1448                    context.deferred_call_id = Some(id.clone());
1449
1450                    // Ensure that the target address is an SC address
1451                    // Ensure that the target address exists
1452                    context.check_target_sc_address(call.target_address)?;
1453
1454                    // credit coins to the target address
1455                    if let Err(err) = context.transfer_coins(
1456                        None,
1457                        Some(call.target_address),
1458                        call.coins,
1459                        false,
1460                        TransferContext::DeferredCallCoins(OriginTransferContext {
1461                            deferred_call_id: Some(id.clone()),
1462                            ..Default::default()
1463                        }),
1464                    ) {
1465                        // coin crediting failed: reset context to snapshot and reimburse sender
1466                        return Err(ExecutionError::DeferredCallsError(format!(
1467                            "could not credit coins to target of deferred call execution: {}",
1468                            err
1469                        )));
1470                    }
1471
1472                    // quit if there is no function to be called
1473                    if call.target_function.is_empty() {
1474                        return Err(ExecutionError::DeferredCallsError(
1475                            "no function to call in the deferred call".to_string(),
1476                        ));
1477                    }
1478
1479                    // Load bytecode. Assume empty bytecode if not found.
1480                    context
1481                        .get_bytecode(&call.target_address)
1482                        .ok_or(ExecutionError::DeferredCallsError(
1483                            "no bytecode found".to_string(),
1484                        ))?
1485                        .0
1486                };
1487
1488                let module = self.module_cache.write().load_module(
1489                    &bytecode,
1490                    call.get_effective_gas(self.config.deferred_calls_config.call_cst_gas_cost),
1491                    self.config.condom_limits.clone(),
1492                )?;
1493                let response = massa_sc_runtime::run_function(
1494                    &*self.execution_interface,
1495                    module,
1496                    &call.target_function,
1497                    &call.parameters,
1498                    call.get_effective_gas(self.config.deferred_calls_config.call_cst_gas_cost),
1499                    self.config.gas_costs.clone(),
1500                    self.config.condom_limits.clone(),
1501                );
1502
1503                match response {
1504                    Ok(res) => {
1505                        self.module_cache
1506                            .write()
1507                            .set_init_cost(&bytecode, res.init_gas_cost);
1508                        #[cfg(feature = "execution-trace")]
1509                        {
1510                            result.traces =
1511                                Some((res.trace.into_iter().map(|t| t.into()).collect(), true));
1512                        }
1513                        // #[cfg(feature = "execution-info")]
1514                        // {
1515                        // result.success = true;
1516                        // }
1517                        result.success = true;
1518                        Ok(result)
1519                    }
1520                    Err(error) => {
1521                        if let VMError::ExecutionError { init_gas_cost, .. } = error {
1522                            self.module_cache
1523                                .write()
1524                                .set_init_cost(&bytecode, init_gas_cost);
1525                        }
1526                        // execution failed: reset context to snapshot and reimburse sender
1527                        Err(ExecutionError::VMError {
1528                            context: "Deferred Call".to_string(),
1529                            error,
1530                        })
1531                    }
1532                }
1533            };
1534
1535            // execute the deferred call
1536            let execution_result = deferred_call_execution();
1537
1538            // if the execution failed, reset the context to the snapshot
1539            if let Err(err) = &execution_result {
1540                let mut context = context_guard!(self);
1541                context.reset_to_snapshot(snapshot, err.clone());
1542                context.deferred_call_fail_exec(id, &call);
1543            }
1544            execution_result
1545        }
1546    }
1547    /// Executes a full slot (with or without a block inside) without causing any changes to the state,
1548    /// just yielding the execution output.
1549    ///
1550    /// # Arguments
1551    /// * `slot`: slot to execute
1552    /// * `exec_target`: metadata of the block to execute, if not miss
1553    /// * `selector`: Reference to the selector
1554    ///
1555    /// # Returns
1556    /// An `ExecutionOutput` structure summarizing the output of the executed slot
1557    pub fn execute_slot(
1558        &mut self,
1559        slot: &Slot,
1560        exec_target: Option<&(BlockId, ExecutionBlockMetadata)>,
1561        selector: Box<dyn SelectorController>,
1562    ) -> ExecutionOutput {
1563        // Start slot execution timer
1564        let slot_start_time = Instant::now();
1565        #[cfg(feature = "execution-trace")]
1566        let mut slot_trace = SlotAbiCallStack {
1567            slot: *slot,
1568            operation_call_stacks: PreHashMap::default(),
1569            asc_call_stacks: vec![],
1570            deferred_call_stacks: HashMap::default(),
1571        };
1572        #[cfg(feature = "execution-trace")]
1573        let mut transfers = vec![];
1574
1575        // Create a new execution context for the whole active slot
1576        let mut execution_context = ExecutionContext::active_slot(
1577            self.config.clone(),
1578            *slot,
1579            exec_target.as_ref().map(|(b_id, _)| *b_id),
1580            self.final_state.clone(),
1581            self.active_history.clone(),
1582            self.module_cache.clone(),
1583            self.mip_store.clone(),
1584        );
1585
1586        #[cfg(feature = "execution-info")]
1587        let mut exec_info = ExecutionInfoForSlot::new(
1588            slot.clone(),
1589            execution_context.execution_trail_hash.clone(),
1590            exec_target.as_ref().map(|(b_id, _)| *b_id),
1591        );
1592
1593        // (success, fail, cancel)
1594        let mut deferred_calls_stats = (0, 0, 0);
1595
1596        // Deferred calls execution
1597        let calls = execution_context.deferred_calls_advance_slot(*slot);
1598
1599        let deferred_calls_slot_gas = calls.effective_slot_gas;
1600
1601        // One-time, versioning-gated WMAS bytecode patch (see `wmas_patch`).
1602        // Applied here so it is part of the slot's ledger changes and shared by
1603        // both candidate and final execution (this function is called by both).
1604        if execution_context.is_execution_component_version_activation(MIP_0002_EXECUTION_VERSION) {
1605            match wmas_patch::wmas_address(self.config.chain_id) {
1606                Some(addr) => {
1607                    // Existence guard: only overwrite an already-deployed WMAS
1608                    // contract, never create a new entry. Keeps the patch a
1609                    // no-op on networks that share a known chain_id but do not
1610                    // actually have WMAS deployed (e.g. a private fork or a
1611                    // sandbox misconfigured with a mainnet/buildnet chain_id).
1612                    if execution_context.get_bytecode(&addr).is_some() {
1613                        execution_context
1614                            .override_bytecode(&addr, wmas_patch::patched_wmas_bytecode());
1615                        info!("applied WMAS bytecode patch at slot {}", slot);
1616                    } else {
1617                        warn!(
1618                            "WMAS bytecode patch activation slot {} reached but no contract exists at {}; skipping",
1619                            slot, addr
1620                        );
1621                    }
1622                }
1623                None => {
1624                    // Fail-safe: the patch is not applicable on this network
1625                    // (unknown chain_id). Leave WMAS untouched rather than abort
1626                    // consensus.
1627                    warn!(
1628                        "WMAS bytecode patch activation slot {} reached but no address for chain_id {}; skipping",
1629                        slot, self.config.chain_id
1630                    );
1631                }
1632            }
1633        }
1634
1635        // Apply the created execution context for slot execution
1636        *context_guard!(self) = execution_context;
1637
1638        for (id, call) in calls.slot_calls {
1639            let cancelled = call.cancelled;
1640            match self.execute_deferred_call(&id, call) {
1641                Ok(_exec) => {
1642                    if cancelled {
1643                        deferred_calls_stats.2 += 1;
1644                        continue;
1645                    }
1646                    deferred_calls_stats.0 += 1;
1647                    info!("executed deferred call: {:?}", id);
1648                    cfg_if::cfg_if! {
1649                        if #[cfg(feature = "execution-trace")] {
1650                            // Safe to unwrap
1651                            slot_trace.deferred_call_stacks.insert(id, _exec.traces.unwrap().0);
1652                        } else if #[cfg(feature = "execution-info")] {
1653                            slot_trace.deferred_call_stacks.insert(id, _exec.traces.unwrap().0);
1654                            exec_info.deferred_calls_messages.push(Ok(_exec));
1655                        }
1656                    }
1657                }
1658                Err(err) => {
1659                    deferred_calls_stats.1 += 1;
1660                    let msg = format!("failed executing deferred call: {}", err);
1661                    #[cfg(feature = "execution-info")]
1662                    exec_info.deferred_calls_messages.push(Err(msg.clone()));
1663                    debug!(msg);
1664                }
1665            }
1666        }
1667
1668        // Block execution
1669
1670        let mut block_info: Option<ExecutedBlockInfo> = None;
1671        // Set block gas (deferred calls have their own budget, see async gas below)
1672        let mut remaining_block_gas = self.config.max_gas_per_block;
1673
1674        // Check if there is a block at this slot
1675        if let Some((block_id, block_metadata)) = exec_target {
1676            let block_store = block_metadata
1677                .storage
1678                .as_ref()
1679                .expect("Cannot execute a block for which the storage is missing");
1680
1681            // Retrieve the block from storage
1682            let stored_block = block_store
1683                .read_blocks()
1684                .get(block_id)
1685                .expect("Missing block in storage.")
1686                .clone();
1687
1688            block_info = Some(ExecutedBlockInfo {
1689                block_id: *block_id,
1690                current_version: stored_block.content.header.content.current_version,
1691                announced_version: stored_block.content.header.content.announced_version,
1692            });
1693
1694            // gather all operations
1695            let operations = {
1696                let ops = block_store.read_operations();
1697                stored_block
1698                    .content
1699                    .operations
1700                    .into_iter()
1701                    .map(|op_id| {
1702                        ops.get(&op_id)
1703                            .expect("block operation absent from storage")
1704                            .clone()
1705                    })
1706                    .collect::<Vec<_>>()
1707            };
1708
1709            debug!("executing {} operations at slot {}", operations.len(), slot);
1710
1711            // gather all available endorsement creators and target blocks
1712            let endorsement_creators: Vec<Address> = stored_block
1713                .content
1714                .header
1715                .content
1716                .endorsements
1717                .iter()
1718                .map(|endo| endo.content_creator_address)
1719                .collect();
1720            let endorsement_target_creator = block_metadata
1721                .same_thread_parent_creator
1722                .expect("same thread parent creator missing");
1723
1724            // Block credits count every operation fee, denunciation slash and endorsement reward.
1725            // We initialize the block credits with the block reward to stimulate block production
1726            // even in the absence of operations and denunciations.
1727            let mut block_credits = self.config.block_reward;
1728
1729            // Try executing the operations of this block in the order in which they appear in the block.
1730            // Errors are logged but do not interrupt the execution of the slot.
1731            for operation in operations.into_iter() {
1732                match self.execute_operation(
1733                    &operation,
1734                    stored_block.content.header.content.slot,
1735                    &mut remaining_block_gas,
1736                    &mut block_credits,
1737                ) {
1738                    Ok(_op_return) => {
1739                        #[cfg(feature = "execution-trace")]
1740                        {
1741                            slot_trace
1742                                .operation_call_stacks
1743                                .insert(operation.id, _op_return.0);
1744                            match &operation.content.op {
1745                                OperationType::Transaction {
1746                                    recipient_address,
1747                                    amount,
1748                                } => {
1749                                    let receiver_balance = {
1750                                        let context = context_guard!(self);
1751                                        context.get_balance(recipient_address).unwrap_or_default()
1752                                    };
1753                                    let mut effective_received_amount = *amount;
1754                                    if receiver_balance
1755                                        == amount
1756                                            .checked_sub(
1757                                                self.config
1758                                                    .storage_costs_constants
1759                                                    .ledger_entry_base_cost,
1760                                            )
1761                                            .unwrap_or_default()
1762                                    {
1763                                        effective_received_amount = amount
1764                                            .checked_sub(
1765                                                self.config
1766                                                    .storage_costs_constants
1767                                                    .ledger_entry_base_cost,
1768                                            )
1769                                            .unwrap_or_default();
1770                                    }
1771                                    transfers.push(Transfer {
1772                                        from: operation.content_creator_address,
1773                                        to: *recipient_address,
1774                                        amount: *amount,
1775                                        effective_received_amount,
1776                                        op_id: operation.id,
1777                                        succeed: _op_return.1,
1778                                        fee: operation.content.fee,
1779                                    });
1780                                }
1781                                OperationType::CallSC {
1782                                    target_addr, coins, ..
1783                                } => {
1784                                    transfers.push(Transfer {
1785                                        from: operation.content_creator_address,
1786                                        to: *target_addr,
1787                                        amount: *coins,
1788                                        effective_received_amount: *coins,
1789                                        op_id: operation.id,
1790                                        succeed: _op_return.1,
1791                                        fee: operation.content.fee,
1792                                    });
1793                                }
1794                                _ => {}
1795                            }
1796                        }
1797
1798                        #[cfg(feature = "execution-info")]
1799                        {
1800                            match &operation.content.op {
1801                                OperationType::RollBuy { roll_count } => {
1802                                    exec_info.roll_operations.push(RollOperationInfo::RollBuy(
1803                                        operation.content_creator_address,
1804                                        *roll_count,
1805                                    ))
1806                                }
1807                                OperationType::RollSell { roll_count } => {
1808                                    exec_info.roll_operations.push(RollOperationInfo::RollSell(
1809                                        operation.content_creator_address,
1810                                        *roll_count,
1811                                    ))
1812                                }
1813                                _ => {}
1814                            }
1815                        }
1816                    }
1817                    Err(err) => {
1818                        debug!(
1819                            "failed executing operation {} in block {}: {}",
1820                            operation.id, block_id, err
1821                        );
1822                    }
1823                }
1824            }
1825
1826            // Try executing the denunciations of this block
1827            for denunciation in &stored_block.content.header.content.denunciations {
1828                match self.execute_denunciation(
1829                    denunciation,
1830                    &stored_block.content.header.content.slot,
1831                    &mut block_credits,
1832                ) {
1833                    Ok(_de_res) => {
1834                        #[cfg(feature = "execution-info")]
1835                        exec_info.denunciations.push(Ok(_de_res));
1836                    }
1837                    Err(e) => {
1838                        let msg = format!(
1839                            "Failed processing denunciation: {:?}, in block: {}: {}",
1840                            denunciation, block_id, e
1841                        );
1842                        #[cfg(feature = "execution-info")]
1843                        exec_info.denunciations.push(Err(msg.clone()));
1844                        debug!(msg);
1845                    }
1846                }
1847            }
1848
1849            // Denunciation punishment is limited to slashing locked rolls (see execute_denunciation).
1850            // Addresses denounced in this block are intentionally still eligible for the slot's
1851            // EndorsementCreatorReward, EndorsementTargetReward, and BlockCreatorReward below.
1852            // Excluding them would make the effective penalty depend on whether the denounced
1853            // address was also selected by draws for this slot, which is not a coherent punishment model.
1854
1855            // Get block creator address
1856            let block_creator_addr = stored_block.content_creator_address;
1857
1858            // acquire lock on execution context
1859            let mut context = context_guard!(self);
1860
1861            // Update speculative rolls state production stats
1862            context.update_production_stats(&block_creator_addr, *slot, Some(*block_id));
1863
1864            // Divide the total block credits into parts + remainder
1865            let block_credit_part_count = 3 * (1 + self.config.endorsement_count);
1866            let block_credit_part = block_credits
1867                .checked_div_u64(block_credit_part_count)
1868                .expect("critical: block_credits checked_div factor is 0");
1869            let remainder = block_credits
1870                .checked_rem_u64(block_credit_part_count)
1871                .expect("critical: block_credits checked_rem factor is 0");
1872
1873            // Give 3 parts + remainder to the block producer to stimulate block production
1874            // even in the absence of endorsements.
1875            let mut block_producer_credit = block_credit_part
1876                .saturating_mul_u64(3)
1877                .saturating_add(remainder);
1878
1879            for endorsement_creator in endorsement_creators {
1880                // Credit the creator of the block with 1 part to stimulate endorsement inclusion of endorsements,
1881                // and dissuade from emitting the block too early (before the endorsements have propagated).
1882                block_producer_credit = block_producer_credit.saturating_add(block_credit_part);
1883
1884                // Credit creator of the endorsement with 1 part to stimulate the production of endorsements.
1885                // This also motivates endorsers to not publish their endorsements too early (will not endorse the right block),
1886                // and to not publish too late (will not be included in the block).
1887                match context.transfer_coins(
1888                    None,
1889                    Some(endorsement_creator),
1890                    block_credit_part,
1891                    false,
1892                    TransferContext::EndorsementCreatorReward,
1893                ) {
1894                    Ok(_) => {
1895                        #[cfg(feature = "execution-info")]
1896                        exec_info
1897                            .endorsement_creator_rewards
1898                            .insert(endorsement_creator, block_credit_part);
1899                    }
1900                    Err(err) => {
1901                        debug!(
1902                            "failed to credit {} coins to endorsement creator {} for an endorsed block execution: {}",
1903                            block_credit_part, endorsement_creator, err
1904                        )
1905                    }
1906                }
1907
1908                // Credit the creator of the endorsed block with 1 part.
1909                // This is done to incentivize block producers to be endorsed,
1910                // typically by not publishing their blocks too late.
1911                match context.transfer_coins(
1912                    None,
1913                    Some(endorsement_target_creator),
1914                    block_credit_part,
1915                    false,
1916                    TransferContext::EndorsementTargetReward,
1917                ) {
1918                    Ok(_) => {
1919                        #[cfg(feature = "execution-info")]
1920                        {
1921                            exec_info.endorsement_target_reward =
1922                                Some((endorsement_target_creator, block_credit_part));
1923                        }
1924                    }
1925                    Err(err) => {
1926                        debug!(
1927                            "failed to credit {} coins to endorsement target creator {} on block execution: {}",
1928                            block_credit_part, endorsement_target_creator, err
1929                        )
1930                    }
1931                }
1932            }
1933
1934            // Credit block producer
1935            if let Err(err) = context.transfer_coins(
1936                None,
1937                Some(block_creator_addr),
1938                block_producer_credit,
1939                false,
1940                TransferContext::BlockCreatorReward,
1941            ) {
1942                debug!(
1943                    "failed to credit {} coins to block creator {} on block execution: {}",
1944                    block_producer_credit, block_creator_addr, err
1945                )
1946            } else {
1947                #[cfg(feature = "execution-info")]
1948                {
1949                    exec_info.block_producer_reward =
1950                        Some((block_creator_addr, block_producer_credit));
1951                }
1952            }
1953        } else {
1954            // the slot is a miss, check who was supposed to be the creator and update production stats
1955            let producer_addr = selector
1956                .get_producer(*slot)
1957                .expect("couldn't get the expected block producer for a missed slot");
1958            context_guard!(self).update_production_stats(&producer_addr, *slot, None);
1959        }
1960
1961        // Async msg execution
1962
1963        // Get asynchronous messages to execute
1964        // The gas available for async messages is the remaining block gas + async remaining gas (max_async - gas used by deferred calls)
1965        // Reusing the gas the block left unused is intentional (activated by MIP-0001, MipComponent::Execution v1):
1966        // it keeps the whole-slot budget bounded by max_gas_per_block + max_async_gas while letting
1967        // underfilled slots do more async work. Changing this formula is a consensus change.
1968        let async_msg_gas_available = self
1969            .config
1970            .max_async_gas
1971            .saturating_sub(deferred_calls_slot_gas)
1972            .saturating_add(remaining_block_gas);
1973
1974        // Get asynchronous messages to execute
1975        let messages = context_guard!(self)
1976            .take_async_batch(async_msg_gas_available, self.config.async_msg_cst_gas_cost);
1977
1978        // clear operation id (otherwise events will be generated using this operation id)
1979        self.execution_context.lock().origin_operation_id = None;
1980
1981        // Try executing asynchronous messages.
1982        // Effects are cancelled on failure and the sender is reimbursed.
1983        for (_message_id, message) in messages {
1984            let opt_bytecode = context_guard!(self).get_bytecode(&message.destination);
1985
1986            match self.execute_async_message(message, opt_bytecode) {
1987                Ok(_message_return) => {
1988                    cfg_if::cfg_if! {
1989                        if #[cfg(feature = "execution-trace")] {
1990                            // Safe to unwrap
1991                            slot_trace.asc_call_stacks.push(_message_return.traces.unwrap().0);
1992                        } else if #[cfg(feature = "execution-info")] {
1993                            slot_trace.asc_call_stacks.push(_message_return.traces.clone().unwrap().0);
1994                            exec_info.async_messages.push(Ok(_message_return));
1995                        }
1996                    }
1997                }
1998                Err(err) => {
1999                    let msg = format!("failed executing async message: {}", err);
2000                    #[cfg(feature = "execution-info")]
2001                    exec_info.async_messages.push(Err(msg.clone()));
2002                    debug!(msg);
2003                }
2004            }
2005        }
2006
2007        #[cfg(feature = "execution-trace")]
2008        self.trace_history
2009            .write()
2010            .save_for_slot(*slot, slot_trace.clone(), transfers.clone());
2011
2012        // Finish slot
2013        #[allow(unused_mut)]
2014        let mut exec_out = context_guard!(self).settle_slot(block_info);
2015        #[cfg(feature = "execution-trace")]
2016        {
2017            exec_out.slot_trace = Some((slot_trace, transfers));
2018        };
2019        #[cfg(feature = "dump-block")]
2020        {
2021            exec_out.storage = match exec_target {
2022                Some((_block_id, block_metadata)) => block_metadata.storage.clone(),
2023                _ => None,
2024            }
2025        }
2026
2027        // Attach the per-slot execution info to the output so it travels
2028        // with `exec_out` through `active_history`. This guarantees that
2029        // when the slot is eventually finalized, the broadcast exposes
2030        // the correct slot metadata (slot, block_id, timestamp,
2031        // execution_trail_hash) — previously the info was stored on the
2032        // execution context and therefore reflected the *latest* active
2033        // slot at finalization time, not the one being finalized.
2034        #[cfg(feature = "execution-info")]
2035        {
2036            exec_info.deferred_credits_execution =
2037                std::mem::take(&mut exec_out.deferred_credits_execution);
2038            exec_info.cancel_async_message_execution =
2039                std::mem::take(&mut exec_out.cancel_async_message_execution);
2040            exec_info.auto_sell_execution = std::mem::take(&mut exec_out.auto_sell_execution);
2041            exec_out.execution_info = Some(exec_info);
2042        }
2043
2044        // Broadcast a slot execution output to active channel subscribers.
2045        if self.config.broadcast_enabled {
2046            let slot_exec_out = SlotExecutionOutput::ExecutedSlot(exec_out.clone());
2047            if let Err(err) = self
2048                .channels
2049                .slot_execution_output_sender
2050                .send(slot_exec_out)
2051            {
2052                trace!(
2053                    "error, failed to broadcast execution output for slot {} due to: {}",
2054                    exec_out.slot.clone(),
2055                    err
2056                );
2057            }
2058        }
2059
2060        exec_out.state_changes.deferred_call_changes.exec_stats = deferred_calls_stats;
2061
2062        // Warn if the slot execution took more than the maximum slot duration
2063        let max_slot_duration = self
2064            .config
2065            .t0
2066            .checked_div_u64(self.config.thread_count as u64)
2067            .expect("could not deduce slot duration")
2068            .to_duration();
2069        let total_slot_duration = slot_start_time.elapsed();
2070        if total_slot_duration > max_slot_duration {
2071            warn!(
2072                "slot {} execution took {:?} (above the {:?} maximum slot duration)",
2073                slot, total_slot_duration, max_slot_duration
2074            );
2075        }
2076
2077        // Return the execution output
2078        exec_out
2079    }

2207    /// Runs a read-only execution request.
2208    /// The executed bytecode appears to be able to read and write the consensus state,
2209    /// but all accumulated changes are simply returned as an `ExecutionOutput` object,

2213    /// * `req`: a read-only execution request

2246    /// Inner implementation of `execute_readonly_request`.
2247    /// The caller is responsible for resetting the shared execution context
2248    /// after this function returns, on both success and error paths.
2249    fn execute_readonly_request_inner(
2250        &self,
2251        req: ReadOnlyExecutionRequest,
2252    ) -> Result<ReadOnlyExecutionOutput, ExecutionError> {
2253        // check if read only request max gas is above the threshold
2254        if req.max_gas > self.config.max_read_only_gas {
2255            return Err(ExecutionError::TooMuchGas(format!(
2256                "execution gas for read-only call is {} which is above the maximum allowed {}",
2257                req.max_gas, self.config.max_read_only_gas
2258            )));
2259        }
2260
2261        // set the execution slot to be the one after the latest executed active slot
2262        let slot = self
2263            .active_cursor
2264            .get_next_slot(self.config.thread_count)

2267        // create a readonly execution context
2268        let execution_context = ExecutionContext::readonly(
2269            self.config.clone(),
2270            slot,
2271            req.call_stack,
2272            self.final_state.clone(),
2273            self.active_history.clone(),
2274            self.module_cache.clone(),
2275            self.mip_store.clone(),
2276        );

2279        let exec_response = match req.target {
2280            ReadOnlyExecutionTarget::BytecodeExecution(bytecode) => {
2281                let condom_limits = execution_context.get_condom_limits();
2282                {
2283                    let mut context = context_guard!(self);
2284                    *context = execution_context;
2285
2286                    let call_stack_addr = context.get_call_stack();
2287
2288                    // transfer fee
2289                    if let (Some(fee), Some(addr)) = (req.fee, call_stack_addr.first()) {
2290                        context.transfer_coins(
2291                            Some(*addr),
2292                            None,
2293                            fee,
2294                            false,
2295                            TransferContext::ReadOnlyBytecodeExecutionFee,
2296                        )?;
2297                    }
2298                }
2299
2300                // load the tmp module

2302                    &bytecode,
2303                    req.max_gas,
2304                    condom_limits.clone(),
2305                )?;
2306
2307                // run the VM
2308                massa_sc_runtime::run_main(
2309                    &*self.execution_interface,
2310                    module,
2311                    req.max_gas,
2312                    self.config.gas_costs.clone(),
2313                    condom_limits,
2314                )

2316                    context: "ReadOnlyExecutionTarget::BytecodeExecution".to_string(),

2318                })?
2319            }
2320
2321            ReadOnlyExecutionTarget::FunctionCall {
2322                target_addr,
2323                target_func,
2324                parameter,
2325            } => {
2326                // get the bytecode, default to an empty vector
2327                let bytecode = execution_context
2328                    .get_bytecode(&target_addr)
2329                    .unwrap_or_default()
2330                    .0;
2331
2332                let condom_limits = execution_context.get_condom_limits();
2333                {
2334                    let mut context = context_guard!(self);
2335                    *context = execution_context;
2336
2337                    // Ensure that the target address is an SC address and exists
2338                    context.check_target_sc_address(target_addr)?;
2339
2340                    let call_stack_addr = context.get_call_stack();
2341
2342                    // transfer fee
2343                    if let (Some(fee), Some(addr)) = (req.fee, call_stack_addr.first()) {
2344                        context.transfer_coins(
2345                            Some(*addr),
2346                            None,
2347                            fee,
2348                            false,
2349                            TransferContext::ReadOnlyFunctionCallFee,
2350                        )?;
2351                    }
2352
2353                    // transfer coins
2354                    if let (Some(coins), Some(from), Some(to)) =
2355                        (req.coins, call_stack_addr.first(), call_stack_addr.get(1))
2356                    {
2357                        context.transfer_coins(
2358                            Some(*from),
2359                            Some(*to),
2360                            coins,
2361                            false,
2362                            TransferContext::ReadOnlyFunctionCallCoins,
2363                        )?;
2364                    }
2365                }
2366
2367                // load and execute the compiled module
2368                // IMPORTANT: do not keep a lock here as `run_function` uses the `get_module` interface
2369                let module = self.module_cache.write().load_module(
2370                    &bytecode,
2371                    req.max_gas,
2372                    condom_limits.clone(),
2373                )?;
2374
2375                let response = massa_sc_runtime::run_function(
2376                    &*self.execution_interface,
2377                    module,
2378                    &target_func,
2379                    &parameter,
2380                    req.max_gas,
2381                    self.config.gas_costs.clone(),
2382                    condom_limits,
2383                );
2384
2385                match response {
2386                    Ok(Response { init_gas_cost, .. })
2387                    | Err(VMError::ExecutionError { init_gas_cost, .. }) => {
2388                        self.module_cache
2389                            .write()
2390                            .set_init_cost(&bytecode, init_gas_cost);
2391                    }
2392                    _ => (),
2393                }
2394
2395                response.map_err(|error| ExecutionError::VMError {
2396                    context: "ReadOnlyExecutionTarget::FunctionCall".to_string(),
2397                    error,
2398                })?
2399            }
2400        };
2401
2402        // return the execution output
2403        let execution_output = context_guard!(self).settle_slot(None);
2404        let exact_exec_cost = req.max_gas.saturating_sub(exec_response.remaining_gas);
2405
2406        // compute a gas cost, estimating the gas of the last SC call to be max_instance_cost
2407        let corrected_cost = match (context_guard!(self)).gas_remaining_before_subexecution {
2408            Some(gas_remaining) => req
2409                .max_gas
2410                .saturating_sub(gas_remaining) // yield gas used until last subexecution
2411                .saturating_add(self.config.gas_costs.max_instance_cost),
2412            None => self.config.gas_costs.max_instance_cost, // no subexecution, just max_instance_cost
2413        };
2414
2415        // keep the max of the two so the last SC call has at least max_instance_cost of gas
2416        let estimated_cost = u64::max(exact_exec_cost, corrected_cost);
2417        debug!(
2418            "execute_readonly_request:
2419            exec_response.remaining_gas: {}
2420            exact_exec_cost: {}
2421            corrected_cost: {}
2422            estimated_cost: {}",
2423            exec_response.remaining_gas, exact_exec_cost, corrected_cost, estimated_cost
2424        );
2425
2426        Ok(ReadOnlyExecutionOutput {
2427            out: execution_output,
2428            gas_cost: estimated_cost,
2429            call_result: exec_response.ret,
2430        })
2431    }

2706    /// Get future deferred credits of an address
2707    /// Returns tuple: (speculative, final)
2708    pub fn get_address_deferred_credits(
2709        &self,
2710        address: &Address,
2711    ) -> (BTreeMap<Slot, Amount>, BTreeMap<Slot, Amount>) {
2712        // get values from final state
2713        let res_final: BTreeMap<Slot, Amount> = self
2714            .final_state
2715            .read()
2716            .get_pos_state()
2717            .get_deferred_credits_range(.., Some(address))
2718            .credits
2719            .iter()
2720            .filter_map(|(slot, addr_amount)| {
2721                addr_amount.get(address).map(|amount| (*slot, *amount))
2722            })
2723            .collect();
2724
2725        // get values from active history, backwards
2726        let mut res_speculative: BTreeMap<Slot, Amount> = BTreeMap::default();
2727        for hist_item in self.active_history.read().0.iter().rev() {
2728            for (slot, addr_amount) in &hist_item.state_changes.pos_changes.deferred_credits.credits
2729            {
2730                if let Some(amount) = addr_amount.get(address) {
2731                    res_speculative.entry(*slot).or_insert(*amount);
2732                };
2733            }
2734        }
2735
2736        // fill missing speculative entries with final entries
2737        for (slot, amount) in &res_final {
2738            res_speculative.entry(*slot).or_insert(*amount);
2739        }
2740
2741        // remove zero entries from speculative
2742        res_speculative.retain(|_s, a| !a.is_zero());
2743
2744        (res_speculative, res_final)
2745    }

2748    ///
2749    ///  Return value: vector of
2750    ///  `(Option<speculative_status>, Option<final_status>)`

2758            .into_iter()
2759            .zip(final_exec)

2762                    (None, Some(f)) => (Some(f), Some(f)), // special case: a final execution should also appear as speculative

2764                }
2765            })
2766            .collect()
2767    }
