36/// structure used to communicate with execution thread
37pub(crate) struct ExecutionInputData {
38    /// set stop to true to stop the thread
39    pub stop: bool,
40    /// list of newly finalized blocks
41    pub finalized_blocks: HashMap<Slot, BlockId>,
42    /// new blockclique (if there is a new one)
43    pub new_blockclique: Option<HashMap<Slot, BlockId>>,
44    /// storage instances for previously unprocessed blocks
45    pub block_metadata: PreHashMap<BlockId, ExecutionBlockMetadata>,
46    /// queue for read-only execution requests and response MPSCs to send back their outputs
47    pub readonly_requests: RequestQueue<ReadOnlyExecutionRequest, ReadOnlyExecutionOutput>,
48    /// set by the worker after its final drain: new read-only requests are refused.
49    /// Unlike the other fields, it is not reset by `take`.
50    pub closed: bool,
51}
52
53impl Display for ExecutionInputData {
54    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
55        write!(
56            f,
57            "stop={:?}, finalized={:?}, blockclique={:?}, readonly={:?}, storage={:?}",
58            self.stop,
59            self.finalized_blocks
60                .iter()
61                .map(|(slot, id)| (*slot, *id))
62                .collect::<BTreeMap<Slot, BlockId>>(),
63            self.new_blockclique.as_ref().map(|bq| bq
64                .iter()
65                .map(|(slot, id)| (*slot, *id))
66                .collect::<BTreeMap<Slot, BlockId>>()),
67            self.readonly_requests,
68            self.block_metadata.keys().collect::<Vec<&BlockId>>(),
69        )
70    }
71}

73impl ExecutionInputData {
74    /// Creates a new empty `ExecutionInputData`
75    pub fn new(config: ExecutionConfig) -> Self {
76        ExecutionInputData {
77            stop: Default::default(),
78            finalized_blocks: Default::default(),
79            new_blockclique: Default::default(),
80            block_metadata: Default::default(),
81            readonly_requests: RequestQueue::new(config.readonly_queue_length),
82            closed: false,
83        }
84    }
85
86    /// Checks whether a new read-only request can be queued
87    pub fn check_readonly_admission(&self) -> Result<(), ExecutionError> {
88        if self.closed {
89            return Err(ExecutionError::ChannelError(
90                "readonly execution refused because the execution worker is closing".into(),
91            ));
92        }
93        if self.readonly_requests.is_full() {
94            return Err(ExecutionError::ChannelError(
95                "too many queued readonly requests".into(),
96            ));
97        }
98        Ok(())
99    }
100
101    /// Takes the current input data into a clone that is returned,
102    /// and resets self.
103    pub fn take(&mut self) -> Self {
104        let readonly_queue_length = self.readonly_requests.capacity();
105        ExecutionInputData {
106            stop: std::mem::take(&mut self.stop),
107            finalized_blocks: std::mem::take(&mut self.finalized_blocks),
108            new_blockclique: std::mem::take(&mut self.new_blockclique),
109            block_metadata: std::mem::take(&mut self.block_metadata),
110            readonly_requests: std::mem::replace(
111                &mut self.readonly_requests,
112                RequestQueue::new(readonly_queue_length),
113            ),
114            closed: self.closed,
115        }
116    }
117}

129impl ExecutionController for ExecutionControllerImpl {
130    /// Updates blockclique status by signaling newly finalized blocks and the latest blockclique.
131    ///
132    /// # Arguments
133    /// * `finalized_blocks`: newly finalized blocks indexed by slot.
134    /// * `blockclique`: new blockclique (if changed). Indexed by slot.
135    /// * `block_metadata`: storage instances and metadata for new blocks. Each storage owns refs to the block and its ops/endorsements.
136    fn update_blockclique_status(
137        &self,
138        finalized_blocks: HashMap<Slot, BlockId>,
139        new_blockclique: Option<HashMap<Slot, BlockId>>,
140        block_metadata: PreHashMap<BlockId, ExecutionBlockMetadata>,
141    ) {
142        // lock input data
143        let mut input_data = self.input_data.1.lock();
144
145        // extend block info
146        input_data.block_metadata.extend(block_metadata);
147
148        // extend finalized blocks
149        input_data.finalized_blocks.extend(finalized_blocks);
150
151        // update blockclique
152        if new_blockclique.is_some() {
153            input_data.new_blockclique = new_blockclique;
154        }
155
156        // wake up execution controller loop
157        self.input_data.0.notify_one();
158    }

160    /// Atomically query the execution state with multiple requests.
161    ///
162    /// INVARIANT: all items MUST be computed under the single `execution_state` read lock taken below,
163    /// because the returned cursors and fingerprint describe the whole batch and clients rely on this
164    /// to read a consistent state. Never release the lock between items to reduce contention:
165    /// bound the cost of a batch via budgets (per-item errors) instead.
166    ///
167    /// This invariant is enforced by borrowing: item reads go through `&ExecutionState`
168    /// borrowed from the guard, so releasing the guard mid-batch is a compile error.
169    ///
170    /// Large payload responses (`Bytecode`, `DatastoreValue`, `Events`, datastore keys)
171    /// are counted toward `req.max_response_size`. Other response variants are treated
172    /// as 0 bytes.
173    /// When appending a large payload would exceed the budget, that item is returned
174    /// as `ExecutionQueryError::TooLargeResponse` and its bytes are not retained.
175    fn query_state(&self, req: ExecutionQueryRequest) -> ExecutionQueryResponse {
176        let execution_guard = self.execution_state.read();
177        // `exec_state` borrows `execution_guard`: the guard cannot be released
178        // while batch reads go through it.
179        let exec_state: &ExecutionState = &execution_guard;
180        let mut resp: ExecutionQueryResponse = ExecutionQueryResponse {
181            responses: Vec::with_capacity(req.requests.len()),
182            candidate_cursor: exec_state.active_cursor,
183            final_cursor: exec_state.final_cursor,
184            final_state_fingerprint: exec_state.get_final_state_fingerprint(),
185        };
186        // Cumulative size of counted payloads kept in `resp` (bytecode,
187        // datastore values and keys, events).
188        let mut response_payload_size: usize = 0;
189        // Events still returnable by this batch; see `max_event_count`.
190        // `None` means unbounded and stays unbounded.
191        let mut remaining_events = req.max_event_count;
192        // Safety net against lock contention: a wall-clock deadline for the whole
193        // batch. Not the main bound (that is the cumulative budget). `None` = no
194        // deadline.
195        let deadline = req
196            .query_state_deadline_ms
197            .map(|ms| Instant::now() + Duration::from_millis(ms));
198        let mut req_items = req.requests.into_iter().enumerate();
199        while let Some((idx, req_item)) = req_items.next() {
200            // The first item is always evaluated (a 0 ms deadline still lets one
201            // item through — deterministic test); the deadline gates the rest.
202            if idx > 0 && deadline.is_some_and(|d| Instant::now() >= d) {
203                // Remaining items (current and onwards) error out; already
204                // computed items stay valid and cursors are untouched.
205                let err = ExecutionQueryError::TooLargeResponse(
206                    "query_state deadline exceeded".to_string(),
207                );
208                resp.responses.push(Err(err.clone()));
209                for _ in &mut req_items {
210                    resp.responses.push(Err(err.clone()));
211                }
212                break;
213            }
214            let resp_item = eval_query_item(
215                exec_state,
216                req_item,
217                &mut response_payload_size,
218                req.max_response_size,
219                &mut remaining_events,
220            );
221            resp.responses.push(resp_item);
222        }
223        resp
224    }

270    /// Executes a read-only request
271    /// Read-only requests do not modify consensus state
272    fn execute_readonly_request(
273        &self,
274        req: ReadOnlyExecutionRequest,
275    ) -> Result<ReadOnlyExecutionOutput, ExecutionError> {
276        let resp_rx = {
277            let mut input_data = self.input_data.1.lock();
278
279            // refuse the request if the queue is full or the worker is closing
280            input_data.check_readonly_admission()?;
281
282            // prepare the channel to send back the result of the read-only execution
283            let (resp_tx, resp_rx) = MassaChannel::new("read_only_request".to_string(), None);
284
285            // append the request to the queue of input read-only requests
286            input_data
287                .readonly_requests
288                .push(RequestWithResponseSender::new(req, resp_tx));
289
290            // wake up the execution main loop
291            self.input_data.0.notify_one();
292
293            resp_rx
294        };
295
296        // Wait for the result of the execution
297        match resp_rx.recv() {
298            Ok(result) => result,
299            Err(err) => Err(ExecutionError::ChannelError(format!(
300                "readonly execution response channel readout failed: {}",
301                err
302            ))),
303        }
304    }

447fn get_addresses_infos_under(
448    exec_state: &ExecutionState,
449    addresses: &[Address],
450    deferred_credits_max_slot: std::ops::Bound<Slot>,
451    max_keys: Option<u32>,
452) -> Vec<ExecutionAddressInfo> {
453    let mut res = Vec::with_capacity(addresses.len());
454    for addr in addresses {
455        let (final_datastore_keys, candidate_datastore_keys) = exec_state
456            .get_final_and_candidate_datastore_keys(
457                addr,
458                &[],
459                std::ops::Bound::Unbounded,
460                std::ops::Bound::Unbounded,
461                max_keys,
462            );
463        let (final_balance, candidate_balance) = exec_state.get_final_and_candidate_balance(addr);
464        let (final_roll_count, candidate_roll_count) =
465            exec_state.get_final_and_candidate_rolls(addr);
466        let future_deferred_credits =
467            exec_state.get_address_future_deferred_credits(addr, deferred_credits_max_slot);
468        res.push(ExecutionAddressInfo {
469            final_datastore_keys: final_datastore_keys.unwrap_or_default(),
470            candidate_datastore_keys: candidate_datastore_keys.unwrap_or_default(),
471            final_balance: final_balance.unwrap_or_default(),
472            candidate_balance: candidate_balance.unwrap_or_default(),
473            final_roll_count,
474            candidate_roll_count,
475            future_deferred_credits,
476            cycle_infos: exec_state.get_address_cycle_infos(addr),
477        });
478    }
479    res
480}

482fn eval_query_item(
483    exec_state: &ExecutionState,
484    req_item: ExecutionQueryRequestItem,
485    response_payload_size: &mut usize,
486    max_response_size: usize,
487    remaining_events: &mut Option<usize>,
488) -> Result<ExecutionQueryResponseItem, ExecutionQueryError> {
489    match req_item {
490        ExecutionQueryRequestItem::AddressExistsCandidate(addr) => {
491            Ok(ExecutionQueryResponseItem::Boolean(
492                exec_state
493                    .get_final_and_candidate_balance(&addr)
494                    .1
495                    .is_some(),
496            ))
497        }
498        ExecutionQueryRequestItem::AddressExistsFinal(addr) => {
499            Ok(ExecutionQueryResponseItem::Boolean(
500                exec_state
501                    .get_final_and_candidate_balance(&addr)
502                    .0
503                    .is_some(),
504            ))
505        }
506        ExecutionQueryRequestItem::AddressBalanceCandidate(addr) => {
507            let (_final_v, speculative_v) = exec_state.get_final_and_candidate_balance(&addr);
508            match speculative_v {
509                Some(balance) => Ok(ExecutionQueryResponseItem::Amount(balance)),
510                None => Err(ExecutionQueryError::NotFound(format!("Account {}", addr))),
511            }
512        }
513        ExecutionQueryRequestItem::AddressBalanceFinal(addr) => {
514            let (final_v, _speculative_v) = exec_state.get_final_and_candidate_balance(&addr);
515            match final_v {
516                Some(balance) => Ok(ExecutionQueryResponseItem::Amount(balance)),
517                None => Err(ExecutionQueryError::NotFound(format!("Account {}", addr))),
518            }
519        }
520        ExecutionQueryRequestItem::AddressBytecodeCandidate(addr) => {
521            let (_final_v, speculative_v) = exec_state.get_final_and_active_bytecode(&addr);
522            match speculative_v {
523                Some(bytecode) => account_large_payload(
524                    &mut *response_payload_size,
525                    max_response_size,
526                    bytecode.0.len(),
527                )
528                .map(|()| ExecutionQueryResponseItem::Bytecode(bytecode)),
529                None => Err(ExecutionQueryError::NotFound(format!("Account {}", addr))),
530            }
531        }
532        ExecutionQueryRequestItem::AddressBytecodeFinal(addr) => {
533            let (final_v, _speculative_v) = exec_state.get_final_and_active_bytecode(&addr);
534            match final_v {
535                Some(bytecode) => account_large_payload(
536                    &mut *response_payload_size,
537                    max_response_size,
538                    bytecode.0.len(),
539                )
540                .map(|()| ExecutionQueryResponseItem::Bytecode(bytecode)),
541                None => Err(ExecutionQueryError::NotFound(format!("Account {}", addr))),
542            }
543        }
544        ExecutionQueryRequestItem::AddressDatastoreKeysCandidate {
545            address,
546            prefix,
547            start_key,
548            end_key,
549            count,
550        } => {
551            let (_final_v, speculative_v) = exec_state.get_final_and_candidate_datastore_keys(
552                &address, &prefix, start_key, end_key, count,
553            );
554            match speculative_v {
555                Some(keys) => {
556                    let bytes = datastore_keys_byte_len(&keys);
557                    account_large_payload(&mut *response_payload_size, max_response_size, bytes)
558                        .map(|()| {
559                            ExecutionQueryResponseItem::AddressDatastoreKeys(keys, address, false)
560                        })
561                }
562                None => Err(ExecutionQueryError::NotFound(format!(
563                    "Account {}",
564                    address
565                ))),
566            }
567        }
568        ExecutionQueryRequestItem::AddressDatastoreKeysFinal {
569            address,
570            prefix,
571            start_key,
572            end_key,
573            count,
574        } => {
575            let final_v =
576                exec_state.get_final_datastore_keys(&address, &prefix, start_key, end_key, count);
577            match final_v {
578                Some(keys) => {
579                    let bytes = datastore_keys_byte_len(&keys);
580                    account_large_payload(&mut *response_payload_size, max_response_size, bytes)
581                        .map(|()| {
582                            ExecutionQueryResponseItem::AddressDatastoreKeys(keys, address, true)
583                        })
584                }
585                None => Err(ExecutionQueryError::NotFound(format!(
586                    "Account {}",
587                    address
588                ))),
589            }
590        }
591        ExecutionQueryRequestItem::AddressDatastoreValueCandidate { addr, key } => {
592            let (_final_v, speculative_v) = exec_state.get_final_and_active_data_entry(&addr, &key);
593            match speculative_v {
594                Some(value) => account_large_payload(
595                    &mut *response_payload_size,
596                    max_response_size,
597                    value.len(),
598                )
599                .map(|()| ExecutionQueryResponseItem::DatastoreValue(value)),
600                None => Err(ExecutionQueryError::NotFound(format!(
601                    "Account {} datastore entry {:?}",
602                    addr, key
603                ))),
604            }
605        }
606        ExecutionQueryRequestItem::AddressDatastoreValueFinal { addr, key } => {
607            let (final_v, _speculative_v) = exec_state.get_final_and_active_data_entry(&addr, &key);
608            match final_v {
609                Some(value) => account_large_payload(
610                    &mut *response_payload_size,
611                    max_response_size,
612                    value.len(),
613                )
614                .map(|()| ExecutionQueryResponseItem::DatastoreValue(value)),
615                None => Err(ExecutionQueryError::NotFound(format!(
616                    "Account {} datastore entry {:?}",
617                    addr, key
618                ))),
619            }
620        }
621        ExecutionQueryRequestItem::OpExecutionStatusCandidate(id) => {
622            let (speculative_v, _final_v) = exec_state
623                .get_ops_exec_status(&[id])
624                .first()
625                .map(|(s_v, f_v)| (*s_v, *f_v))
626                .expect("expected one return value");
627            match speculative_v {
628                Some(true) => Ok(ExecutionQueryResponseItem::ExecutionStatus(
629                    ExecutionQueryExecutionStatus::AlreadyExecutedWithSuccess,
630                )),
631                Some(false) => Ok(ExecutionQueryResponseItem::ExecutionStatus(
632                    ExecutionQueryExecutionStatus::AlreadyExecutedWithFailure,
633                )),
634                None => Ok(ExecutionQueryResponseItem::ExecutionStatus(
635                    ExecutionQueryExecutionStatus::ExecutableOrExpired,
636                )),
637            }
638        }
639        ExecutionQueryRequestItem::OpExecutionStatusFinal(id) => {
640            let (_speculative_v, final_v) = exec_state
641                .get_ops_exec_status(&[id])
642                .first()
643                .map(|(s_v, f_v)| (*s_v, *f_v))
644                .expect("expected one return value");
645            match final_v {
646                Some(true) => Ok(ExecutionQueryResponseItem::ExecutionStatus(
647                    ExecutionQueryExecutionStatus::AlreadyExecutedWithSuccess,
648                )),
649                Some(false) => Ok(ExecutionQueryResponseItem::ExecutionStatus(
650                    ExecutionQueryExecutionStatus::AlreadyExecutedWithFailure,
651                )),
652                None => Ok(ExecutionQueryResponseItem::ExecutionStatus(
653                    ExecutionQueryExecutionStatus::ExecutableOrExpired,
654                )),
655            }
656        }
657        ExecutionQueryRequestItem::DenunciationExecutionStatusCandidate(id) => {
658            let (speculative_v, _final_v) = exec_state.get_denunciation_execution_status(&id);
659            match speculative_v {
660                true => Ok(ExecutionQueryResponseItem::ExecutionStatus(
661                    ExecutionQueryExecutionStatus::AlreadyExecutedWithSuccess,
662                )),
663                false => Ok(ExecutionQueryResponseItem::ExecutionStatus(
664                    ExecutionQueryExecutionStatus::ExecutableOrExpired,
665                )),
666            }
667        }
668        ExecutionQueryRequestItem::DenunciationExecutionStatusFinal(id) => {
669            let (_speculative_v, final_v) = exec_state.get_denunciation_execution_status(&id);
670            match final_v {
671                true => Ok(ExecutionQueryResponseItem::ExecutionStatus(
672                    ExecutionQueryExecutionStatus::AlreadyExecutedWithSuccess,
673                )),
674                false => Ok(ExecutionQueryResponseItem::ExecutionStatus(
675                    ExecutionQueryExecutionStatus::ExecutableOrExpired,
676                )),
677            }
678        }
679        ExecutionQueryRequestItem::AddressRollsCandidate(addr) => {
680            let (_final_rolls, candidate_rolls) = exec_state.get_final_and_candidate_rolls(&addr);
681            Ok(ExecutionQueryResponseItem::RollCount(candidate_rolls))
682        }
683        ExecutionQueryRequestItem::AddressRollsFinal(addr) => {
684            let (final_rolls, _candidate_rolls) = exec_state.get_final_and_candidate_rolls(&addr);
685            Ok(ExecutionQueryResponseItem::RollCount(final_rolls))
686        }
687        ExecutionQueryRequestItem::AddressDeferredCreditsCandidate(addr) => {
688            let (candidate_v, _final_v) = exec_state.get_address_deferred_credits(&addr);
689            Ok(ExecutionQueryResponseItem::DeferredCredits(candidate_v))
690        }
691        ExecutionQueryRequestItem::AddressDeferredCreditsFinal(addr) => {
692            let (_candidate_v, final_v) = exec_state.get_address_deferred_credits(&addr);
693            Ok(ExecutionQueryResponseItem::DeferredCredits(final_v))
694        }
695        ExecutionQueryRequestItem::CycleInfos {
696            cycle,
697            restrict_to_addresses,
698        } => {
699            let cycle_infos = exec_state.get_cycle_infos(cycle, restrict_to_addresses.as_ref());
700            match cycle_infos {
701                Some(cycle_infos) => Ok(ExecutionQueryResponseItem::CycleInfos(cycle_infos)),
702                None => Err(ExecutionQueryError::NotFound(format!("Cycle {}", cycle))),
703            }
704        }
705        ExecutionQueryRequestItem::Events(filter) => {
706            if *remaining_events == Some(0) {
707                return Err(ExecutionQueryError::TooLargeResponse(
708                    "event budget for this batch is exhausted".to_string(),
709                ));
710            }
711            let limit = *remaining_events;
712            let events = exec_state.get_filtered_sc_output_event(filter, limit);
713            if let Some(remaining) = remaining_events {
714                *remaining -= events.len();
715            }
716            let bytes = events
717                .iter()
718                .map(|event| {
719                    event
720                        .data
721                        .len()
722                        .saturating_add(EVENT_RESPONSE_OVERHEAD_BYTES)
723                })
724                .fold(0usize, |acc, len| acc.saturating_add(len));
725            account_large_payload(&mut *response_payload_size, max_response_size, bytes)
726                .map(|()| ExecutionQueryResponseItem::Events(events))
727        }
728        ExecutionQueryRequestItem::DeferredCallQuote {
729            target_slot,
730            max_gas_request,
731            params_size,
732        } => {
733            let result = exec_state.deferred_call_quote(target_slot, max_gas_request, params_size);
734            Ok(ExecutionQueryResponseItem::DeferredCallQuote(
735                result.0, result.1, result.2, result.3,
736            ))
737        }
738        ExecutionQueryRequestItem::DeferredCallInfo(deferred_call_id) => exec_state
739            .deferred_call_info(&deferred_call_id)
740            .ok_or_else(|| {
741                ExecutionQueryError::NotFound(format!("Deferred call id {}", deferred_call_id))
742            })
743            .map(|call| ExecutionQueryResponseItem::DeferredCallInfo(deferred_call_id, call)),
744        ExecutionQueryRequestItem::DeferredCallsBySlot(slot) => {
745            let res = exec_state.get_deferred_calls_by_slot(slot);
746            Ok(ExecutionQueryResponseItem::DeferredCallsBySlot(slot, res))
747        }
748    }
749}

751/// Estimated serialized size (bytes) of one event's context on top of `data.len()`.
752/// Covers slot, emitter/caller/operation ids and Vec bookkeeping.
753// ponytail: overhead estimated, measure real serialized SCOutputEvent context size and adjust;
754// revisit if the legacy 50 KB events leave the cache and the estimate dominates the budget.
755const EVENT_RESPONSE_OVERHEAD_BYTES: usize = 128;
756
757/// Summed key lengths of a datastore-keys response, for the cumulative size budget.
758fn datastore_keys_byte_len(keys: &BTreeSet<Vec<u8>>) -> usize {
759    keys.iter()
760        .map(Vec::len)
761        .fold(0usize, usize::saturating_add)
762}
763
764/// Account a large payload against the cumulative response budget.
765/// Only bytecode and datastore-value lengths are counted; other items are ignored.
766fn account_large_payload(
767    accumulated: &mut usize,
768    max_response_size: usize,
769    payload_len: usize,
770) -> Result<(), ExecutionQueryError> {
771    let new_size = accumulated.saturating_add(payload_len);
772    if new_size > max_response_size {
773        return Err(ExecutionQueryError::TooLargeResponse(format!(
774            "query response size would exceed limit of {} bytes (current {}, item {})",
775            max_response_size, accumulated, payload_len
776        )));
777    }
778    *accumulated = new_size;
779    Ok(())
780}

782/// Execution manager
783/// Allows stopping the execution worker
784pub struct ExecutionManagerImpl {
785    /// input data to process in the Execution manager loop
786    /// with a wake-up condition variable that needs to be triggered when the data changes
787    pub(crate) input_data: Arc<(Condvar, Mutex<ExecutionInputData>)>,
788    /// handle used to join the worker thread
789    pub(crate) thread_handle: Option<std::thread::JoinHandle<()>>,
790}
791
792impl ExecutionManager for ExecutionManagerImpl {
793    /// stops the worker
794    fn stop(&mut self) {
795        info!("Stopping Execution manager...");
796        // notify the worker thread to stop
797        {
798            let mut input_wlock = self.input_data.1.lock();
799            input_wlock.stop = true;
800            self.input_data.0.notify_one();
801        }
802        // join the execution thread
803        if let Some(join_handle) = self.thread_handle.take() {
804            join_handle
805                .join()
806                .expect("Execution manager thread panicked");
807        }
808        info!("Execution manager stopped");
809    }
810}
