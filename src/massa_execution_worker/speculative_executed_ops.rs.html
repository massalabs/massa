3//! Speculative list of previously executed operations, to prevent reuse.
4
5use crate::active_history::{ActiveHistory, HistorySearchResult};
6use massa_executed_ops::ExecutedOpsChanges;
7use massa_final_state::FinalStateController;
8use massa_models::{operation::OperationId, slot::Slot};
9use parking_lot::RwLock;
10use std::sync::Arc;
11
12/// Speculative state of executed operations
13pub(crate) struct SpeculativeExecutedOps {
14    /// Thread-safe shared access to the final state. For reading only.
15    final_state: Arc<RwLock<dyn FinalStateController>>,
16
17    /// History of the outputs of recently executed slots.
18    /// Slots should be consecutive, newest at the back.
19    active_history: Arc<RwLock<ActiveHistory>>,
20
21    /// executed operations: maps the operation ID to its validity slot end - included
22    executed_ops: ExecutedOpsChanges,
23}
24
25impl SpeculativeExecutedOps {
26    /// Creates a new `SpeculativeExecutedOps`
27    ///
28    /// # Arguments
29    /// * `final_state`: thread-safe shared access the final state
30    /// * `active_history`: thread-safe shared access the speculative execution history
31    pub fn new(
32        final_state: Arc<RwLock<dyn FinalStateController>>,
33        active_history: Arc<RwLock<ActiveHistory>>,
34    ) -> Self {
35        SpeculativeExecutedOps {
36            final_state,
37            active_history,
38            executed_ops: Default::default(),
39        }
40    }
41
42    /// Returns the set of operation IDs caused to the `SpeculativeExecutedOps` since its creation,
43    /// and resets their local value to nothing
44    pub fn take(&mut self) -> ExecutedOpsChanges {
45        std::mem::take(&mut self.executed_ops)
46    }
47
48    /// Takes a snapshot (clone) of the changes caused to the `SpeculativeExecutedOps` since its creation
49    pub fn get_snapshot(&self) -> ExecutedOpsChanges {
50        self.executed_ops.clone()
51    }
52
53    /// Resets the `SpeculativeRollState` to a snapshot (see `get_snapshot` method)
54    pub fn reset_to_snapshot(&mut self, snapshot: ExecutedOpsChanges) {
55        self.executed_ops = snapshot;
56    }
57
58    /// Checks if an operation was executed previously
59    pub fn is_op_executed(&self, op_id: &OperationId) -> bool {
60        // check in the current changes
61        if self.executed_ops.contains_key(op_id) {
62            return true;
63        }
64
65        // check in the active history, backwards
66        match self.active_history.read().fetch_executed_op(op_id) {
67            HistorySearchResult::Present(_) => {
68                return true;
69            }
70            HistorySearchResult::NoInfo => {}
71            HistorySearchResult::Absent => unreachable!(), // fetch_executed_op does not return Absent
72        }
73
74        // check in the final state
75        self.final_state.read().executed_ops_contains(op_id)
76    }
77
78    /// Insert an executed operation.
79    /// Does not check for reuse, please use `SpeculativeExecutedOps::is_op_executed` before.
80    ///
81    /// # Arguments
82    /// * `op_id`: operation ID
83    /// * `op_exec_status` : the status of the execution of the operation.
84    /// * `op_valid_until_slot`: slot until which the operation remains valid (included)
85    pub fn insert_executed_op(
86        &mut self,
87        op_id: OperationId,
88        op_exec_status: bool,
89        op_valid_until_slot: Slot,
90    ) {
91        self.executed_ops
92            .insert(op_id, (op_exec_status, op_valid_until_slot));
93    }
94}
