183    fn bypass_event_limitation(&self, call_stack: Vec<Address>) -> bool {
184        // NOTE: the router addresses are smart contract SC_1 that will call a smart contract SC_2,
185        // and SC_2 will generate the event.
186        // This is why we do not check against the last address in the call stack, but the one before (call_stack.len() - 2)
187        // These addresses are respectively the the Mainnet and Buildnet routers for Dusa
188        let allowed_router_addresses = [
189            Address::from_str("AS12UMSUxgpRBB6ArZDJ19arHoxNkkpdfofQGekAiAJqsuE6PEFJy").unwrap(),
190            Address::from_str("AS1XqtvX3rz2RWbnqLfaYVKEjM3VS5pny9yKDdXcmJ5C1vrcLEFd").unwrap(),
191        ];
192        call_stack.len() > 1 && allowed_router_addresses.contains(&call_stack[call_stack.len() - 2])
193    }

293    /// Initialize the call when bytecode calls a function from another bytecode
294    /// This function transfers the coins passed as parameter,
295    /// prepares the current execution context by pushing a new element on the top of the call stack,
296    /// and returns the target bytecode from the ledger.
297    ///
298    /// # Arguments
299    /// * `address`: string representation of the target address on which the bytecode will be called
300    /// * `raw_coins`: raw representation (without decimal factor) of the amount of coins to transfer from the caller address to the target address at the beginning of the call
301    ///
302    /// # Returns
303    /// The target bytecode or an error
304    fn init_call(&self, address: &str, raw_coins: u64) -> Result<Vec<u8>> {
305        // get target address
306        let to_address = Address::from_str(address).map_err(|e| e.to_string())?;
307
308        // write-lock context
309        let mut context = context_guard!(self);
310
311        // check that the target address is a SC address and if it exists
312        context
313            .check_target_sc_address(to_address)
314            .map_err(|e| e.to_string())?;
315
316        // get target bytecode
317        let bytecode = match context.get_bytecode(&to_address) {
318            Some(bytecode) => bytecode,
319            None => bail!("bytecode not found for address {}", to_address),
320        };
321
322        // get caller address
323        let from_address = match context.stack.last() {
324            Some(addr) => addr.address,
325            _ => bail!("failed to read call stack current address"),
326        };
327
328        // transfer coins from caller to target address
329        let coins = Amount::from_raw(raw_coins);
330        // note: rights are not checked here we checked that to_address is an SC address above
331        // and we know that the sender is at the top of the call stack
332        if let Err(err) = context.transfer_coins(
333            Some(from_address),
334            Some(to_address),
335            coins,
336            false,
337            TransferContext::AbiCallCoins,
338        ) {
339            bail!(
340                "error transferring {} coins from {} to {}: {}",
341                coins,
342                from_address,
343                to_address,
344                err
345            );
346        }
347
348        // push a new call stack element on top of the current call stack
349        context.stack.push(ExecutionStackElement {
350            address: to_address,
351            coins,
352            owned_addresses: vec![to_address],
353            operation_datastore: None,
354        });
355
356        // return the target bytecode
357        Ok(bytecode.0)
358    }

364
365        if context.stack.pop().is_none() {
366            bail!("call stack out of bounds")
367        }

1011    /// Verify an EVM signature
1012    ///
1013    /// Information:
1014    /// * Expects a SECP256K1 signature in full ETH format.
1015    ///   Format: (r, s, v) v will be ignored
1016    ///   Length: 65 bytes
1017    /// * Expects a public key in raw secp256k1 format.
1018    ///   Length: 64 bytes
1019    fn evm_signature_verify(
1020        &self,
1021        message_: &[u8],
1022        signature_: &[u8],
1023        public_key_: &[u8],
1024    ) -> Result<bool> {
1025        // check the signature length
1026        if signature_.len() != 65 {
1027            return Err("invalid signature length in evm_signature_verify".into());
1028        }
1029
1030        // parse the public key
1031        let public_key =
1032            libsecp256k1::PublicKey::parse_slice(public_key_, None).map_err(|e| e.to_string())?;
1033
1034        // build the message
1035        let prefix = format!("\x19Ethereum Signed Message:\n{}", message_.len());
1036        let to_hash = [prefix.as_bytes(), message_].concat();
1037        let full_hash = sha3::Keccak256::digest(to_hash);
1038        let message = libsecp256k1::Message::parse_slice(&full_hash)
1039            .expect("message could not be parsed from a hash slice");
1040
1041        // parse the signature as being (r, s, v)
1042        // r is the R.x value of the signature's R point (32 bytes)
1043        // s is the signature proof for R.x (32 bytes)
1044        // v is a recovery parameter used to ease the signature verification (1 byte)
1045        // see test_evm_verify for an example of its usage
1046        let signature = libsecp256k1::Signature::parse_standard_slice(&signature_[..64])
1047            .map_err(|e| e.to_string())?;
1048
1049        let recovery_id: u8 = libsecp256k1::RecoveryId::parse_rpc(signature_[64])
1050            .map_err(|e| e.to_string())?
1051            .into();
1052        // Note: parse_rpc returns p - 27 and allow for 27, 28, 29, 30
1053        //       restrict to only 27 & 28 (=> 0 & 1)
1054        if recovery_id != 0 && recovery_id != 1 {
1055            // Note:
1056            // The v value in an EVM signature serves as a recovery ID,
1057            // aiding in the recovery of the public key from the signature.
1058            // Typically, v should be either 27 or 28
1059            // (or sometimes 0 or 1, depending on the implementation).
1060            // Ensuring that v is within the expected range is crucial
1061            // for correctly recovering the public key.
1062            // the Ethereum yellow paper specifies only 27 and 28, requiring additional checks.
1063            return Err(
1064                "invalid recovery id value (v = {recovery_id}) in evm_signature_verify".into(),
1065            );
1066        }
1067
1068        // Note:
1069        // The s value in an EVM signature should be in the lower half of the elliptic curve
1070        // in order to prevent malleability attacks.
1071        // If s is in the high-order range, it can be converted to its low-order equivalent,
1072        // which should be enforced during signature verification.
1073        if signature.s.is_high() {
1074            return Err(
1075                "High-Order s Value are prohibited in evm_get_pubkey_from_signature".into(),
1076            );
1077        }
1078
1079        // verify the signature
1080        Ok(libsecp256k1::verify(&message, &signature, &public_key))
1081    }

1105    fn evm_get_pubkey_from_signature(&self, hash_: &[u8], signature_: &[u8]) -> Result<Vec<u8>> {
1106        // check the signature length
1107        if signature_.len() != 65 {
1108            return Err("invalid signature length in evm_get_pubkey_from_signature".into());
1109        }
1110
1111        // parse the message
1112        let message = libsecp256k1::Message::parse_slice(hash_).map_err(|e| e.to_string())?;
1113
1114        // parse the signature as being (r, s, v) use only r and s
1115        let signature = libsecp256k1::Signature::parse_standard_slice(&signature_[..64])
1116            .map_err(|e| e.to_string())?;
1117
1118        // Note:
1119        // See evm_signature_verify explanation
1120        if signature.s.is_high() {
1121            return Err(
1122                "High-Order s Value are prohibited in evm_get_pubkey_from_signature".into(),
1123            );
1124        }
1125
1126        // parse v as a recovery id
1127        let recovery_id =
1128            libsecp256k1::RecoveryId::parse_rpc(signature_[64]).map_err(|e| e.to_string())?;
1129
1130        let recovery_id_: u8 = recovery_id.into();
1131        if recovery_id_ != 0 && recovery_id_ != 1 {

1133            // See evm_signature_verify explanation

1137            );
1138        }

1140        // recover the public key
1141        let recovered =
1142            libsecp256k1::recover(&message, &signature, &recovery_id).map_err(|e| e.to_string())?;
1143
1144        // return its serialized value

1423    /// Adds an asynchronous message to the context speculative asynchronous pool
1424    ///
1425    /// # Arguments
1426    /// * `target_address`: Destination address hash in format string
1427    /// * `target_function`: Name of the message handling function
1428    /// * `validity_start`: Tuple containing the period and thread of the validity start slot
1429    /// * `validity_end`: Tuple containing the period and thread of the validity end slot
1430    /// * `max_gas`: Maximum gas for the message execution.
1431    ///   Bounded below by `max_instance_cost`, and — from execution component version
1432    ///   [`MIP_0002_EXECUTION_VERSION`] on — above by the largest budget any slot
1433    ///   can ever offer. `take_batch_to_execute` only schedules a message when
1434    ///   `max_gas + async_msg_cst_gas_cost` fits the slot's async gas budget, which never
1435    ///   exceeds `max_async_gas + max_gas_per_block` (see `execute_slot`). Before that
1436    ///   version, a message above the ceiling was accepted but permanently unschedulable:
1437    ///   it occupied async pool capacity until its validity end, and on expiry
1438    ///   `cancel_async_message` refunded `raw_coins` but not `raw_fee`.
1439    /// * `fee`: Fee to pay
1440    /// * `raw_coins`: Coins given by the sender
1441    /// * `data`: Message data
1442    fn send_message(
1443        &self,
1444        target_address: &str,
1445        target_function: &str,
1446        validity_start: (u64, u8),
1447        validity_end: (u64, u8),
1448        max_gas: u64,
1449        raw_fee: u64,
1450        raw_coins: u64,
1451        data: &[u8],
1452        filter: Option<(&str, Option<&[u8]>)>,
1453    ) -> Result<()> {
1454        if validity_start.1 >= self.config.thread_count {
1455            bail!("validity start thread exceeds the configuration thread count")
1456        }
1457        if validity_end.1 >= self.config.thread_count {
1458            bail!("validity end thread exceeds the configuration thread count")
1459        }
1460
1461        let target_addr = Address::from_str(target_address).map_err(|e| e.to_string())?;
1462
1463        // check that the target address is an SC address
1464        if !matches!(target_addr, Address::SC(..)) {
1465            bail!("target address is not a smart contract address")
1466        }
1467
1468        // Length verifications
1469        if target_function.len() > self.config.max_function_length as usize {
1470            bail!("Function name is too large");
1471        }
1472        if data.len() > self.config.max_parameter_length as usize {
1473            bail!("Parameter size is too large");
1474        }
1475
1476        let mut execution_context = context_guard!(self);
1477        let emission_slot = execution_context.slot;
1478
1479        if max_gas < self.config.gas_costs.max_instance_cost {
1480            bail!("max gas is lower than the minimum instance cost")
1481        }
1482        // Reject messages that no slot could ever schedule. Gated: rejecting here changes
1483        // execution results, so it only applies once the MIP has activated.
1484        if execution_context.is_execution_component_version_at_least(MIP_0002_EXECUTION_VERSION) {
1485            let max_schedulable_gas = self
1486                .config
1487                .max_async_gas
1488                .saturating_add(self.config.max_gas_per_block)
1489                .saturating_sub(self.config.async_msg_cst_gas_cost);
1490            if max_gas > max_schedulable_gas {
1491                bail!(
1492                    "max gas is higher than the maximum schedulable async gas ({})",
1493                    max_schedulable_gas
1494                )
1495            }
1496        }
1497        if Slot::new(validity_end.0, validity_end.1) < Slot::new(validity_start.0, validity_start.1)
1498        {
1499            bail!("validity end is earlier than the validity start")
1500        }
1501        if Slot::new(validity_end.0, validity_end.1) < emission_slot {
1502            bail!("validity end is earlier than the current slot")
1503        }
1504
1505        let emission_index = execution_context.created_message_index;
1506        let sender = execution_context
1507            .get_current_address()
1508            .map_err(|e| e.to_string())?;
1509        let coins = Amount::from_raw(raw_coins);
1510        execution_context
1511            .transfer_coins(
1512                Some(sender),
1513                None,
1514                coins,
1515                true,
1516                TransferContext::AbiSendMsgCoins,
1517            )
1518            .map_err(|e| e.to_string())?;
1519        let fee = Amount::from_raw(raw_fee);
1520        execution_context
1521            .transfer_coins(
1522                Some(sender),
1523                None,
1524                fee,
1525                true,
1526                TransferContext::AbiSendMsgFee,
1527            )
1528            .map_err(|e| e.to_string())?;
1529        execution_context.push_new_message(AsyncMessage::new(
1530            emission_slot,
1531            emission_index,
1532            sender,
1533            target_addr,
1534            target_function.to_string(),
1535            max_gas,
1536            fee,
1537            coins,
1538            Slot::new(validity_start.0, validity_start.1),
1539            Slot::new(validity_end.0, validity_end.1),
1540            data.to_vec(),
1541            filter
1542                .map(|(addr, key)| {
1543                    let datastore_key = key.map(|k| k.to_vec());
1544                    if let Some(ref k) = datastore_key {
1545                        if k.len() > self.config.max_datastore_key_length as usize {
1546                            bail!("datastore key is too long")
1547                        }
1548                    }
1549                    Ok::<AsyncMessageTrigger, _>(AsyncMessageTrigger {
1550                        address: Address::from_str(addr).map_err(|e| e.to_string())?,
1551                        datastore_key,
1552                    })
1553                })
1554                .transpose()?,
1555            None,
1556        ));
1557        execution_context.created_message_index += 1;
1558        Ok(())
1559    }

1814
1815        let call_id = DeferredCallId::from_str(id).map_err(|e| e.to_string())?;
1816

1818            .deferred_call_cancel(&call_id, caller)
1819            .map_err(|e| e.to_string())?;

1821    }

1823    #[allow(unused_variables)]
1824    fn init_call_wasmv1(&self, address: &str, raw_coins: NativeAmount) -> Result<Vec<u8>> {
1825        // get target address
1826        let to_address = Address::from_str(address).map_err(|e| e.to_string())?;
1827
1828        // check that the target address is an SC address
1829        if !matches!(to_address, Address::SC(..)) {
1830            bail!("called address {} is not an SC address", to_address);
1831        }
1832
1833        // write-lock context
1834        let mut context = context_guard!(self);
1835
1836        // get target bytecode
1837        let bytecode = match context.get_bytecode(&to_address) {
1838            Some(bytecode) => bytecode,
1839            None => bail!("bytecode not found for address {}", to_address),
1840        };
1841
1842        // get caller address
1843        let from_address = match context.stack.last() {
1844            Some(addr) => addr.address,
1845            _ => bail!("failed to read call stack current address"),
1846        };
1847
1848        // transfer coins from caller to target address
1849        let coins = amount_from_native_amount(&raw_coins)?;
1850        // note: rights are not checked here we checked that to_address is an SC address above
1851        // and we know that the sender is at the top of the call stack
1852        if let Err(err) = context.transfer_coins(
1853            Some(from_address),
1854            Some(to_address),
1855            coins,
1856            false,
1857            TransferContext::AbiCallCoins,
1858        ) {
1859            bail!(
1860                "error transferring {} coins from {} to {}: {}",
1861                coins,
1862                from_address,
1863                to_address,
1864                err
1865            );
1866        }
1867
1868        // push a new call stack element on top of the current call stack
1869        context.stack.push(ExecutionStackElement {
1870            address: to_address,
1871            coins,
1872            owned_addresses: vec![to_address],
1873            operation_datastore: None,
1874        });
1875
1876        // return the target bytecode
1877        Ok(bytecode.0)
1878    }
