15/// Parse a block-dump file name of the form `block_slot_{thread}_{period}.bin`
16/// back into its [`Slot`]. Returns `None` for names that don't match.
17fn parse_block_dump_file_name(name: &str) -> Option<Slot> {
18    let rest = name.strip_prefix("block_slot_")?.strip_suffix(".bin")?;
19    let (thread_str, period_str) = rest.split_once('_')?;
20    let thread: u8 = thread_str.parse().ok()?;
21    let period: u64 = period_str.parse().ok()?;
22    if thread >= THREAD_COUNT {
23        None
24    } else {
25        Some(Slot::new(period, thread))
26    }
27}

29/// Rebuild the retention queue (oldest first) from block dumps already present
30/// on disk, so that the `max_blocks` cap keeps accounting for pre-restart entries.
31fn load_saved_slots_from_folder(folder: &Path) -> VecDeque<Slot> {
32    let mut slots: Vec<Slot> = match std::fs::read_dir(folder) {
33        Ok(entries) => entries
34            .flatten()
35            .filter_map(|entry| parse_block_dump_file_name(entry.file_name().to_str()?))
36            .collect(),
37        // Folder missing/unreadable (e.g. fresh start): nothing persisted yet.
38        Err(_) => Vec::new(),
39    };
40    slots.sort_unstable();
41    VecDeque::from(slots)
42}

44/// Rebuild the retention queue (oldest first) from block dumps already present
45/// in the RocksDB store, so that the `max_blocks` cap survives restarts.
46///
47/// This runs on node startup, before the execution worker starts, over a store
48/// that can hold millions of dumped blocks, so the scan is tuned for a single
49/// sequential pass: only keys are read (values are never copied), and since
50/// keys are sortable slot keys they already come out oldest first.
51fn load_saved_slots_from_db(db: &rocksdb::DB) -> VecDeque<Slot> {
52    let mut read_opts = ReadOptions::default();
53    // one-shot scan: don't evict useful blocks from the block cache
54    read_opts.fill_cache(false);
55    read_opts.set_verify_checksums(false);
56    read_opts.set_readahead_size(8 * 1024 * 1024);
57    read_opts.set_async_io(true);
58
59    let mut slots = VecDeque::new();
60    let mut iter = db.raw_iterator_opt(read_opts);
61    iter.seek_to_first();
62    while let Some(key) = iter.key() {
63        if let Ok(key) = <[u8; SLOT_KEY_SIZE]>::try_from(key) {
64            let slot = Slot::from_bytes_key(&key);
65            if slot.thread < THREAD_COUNT {
66                slots.push_back(slot);
67            }
68        }
69        iter.next();
70    }
71    if let Err(e) = iter.status() {
72        warn!("block dump store scan stopped early: {}", e);
73    }
74    slots
75}

77/// A trait that defines the interface for a storage backend for the dump-block
78/// feature.
79pub trait StorageBackend: Send + Sync {
80    /// Writes the given value to the storage backend.
81    /// The slot is used as the key to the value.
82    fn write(&mut self, slot: &Slot, value: &[u8]);
83
84    /// Reads the value from the storage backend.
85    /// The slot is used as the key to the value.
86    fn read(&self, slot: &Slot) -> Option<Vec<u8>>;
87}

89/// A storage backend that uses the file system as the underlying storage engine.
90pub struct FileStorageBackend {
91    folder: PathBuf,
92    slots_saved: VecDeque<Slot>,
93    max_blocks: u64,
94}
95impl FileStorageBackend {
96    /// Creates a new instance of `FileStorageBackend` with the given path.
97    pub fn new(path: PathBuf, max_blocks: u64) -> Self {
98        let slots_saved = load_saved_slots_from_folder(&path);
99        Self {
100            folder: path,
101            slots_saved,
102            max_blocks,
103        }
104    }
105}
106
107impl StorageBackend for FileStorageBackend {
108    fn write(&mut self, slot: &Slot, value: &[u8]) {
109        if self.slots_saved.len() >= self.max_blocks as usize {
110            let slot_to_remove = self.slots_saved.pop_front().unwrap();
111            let block_file_path = self.folder.join(format!(
112                "block_slot_{}_{}.bin",
113                slot_to_remove.thread, slot_to_remove.period
114            ));
115            std::fs::remove_file(block_file_path).expect("Unable to delete block from disk");
116        }
117        let block_file_path = self
118            .folder
119            .join(format!("block_slot_{}_{}.bin", slot.thread, slot.period));
120
121        let mut file = File::create(block_file_path.clone())
122            .unwrap_or_else(|_| panic!("Cannot create file: {:?}", block_file_path));
123
124        file.write_all(value).expect("Unable to write to disk");
125        self.slots_saved.push_back(*slot);
126    }
127
128    fn read(&self, slot: &Slot) -> Option<Vec<u8>> {
129        let block_file_path = self
130            .folder
131            .join(format!("block_slot_{}_{}.bin", slot.thread, slot.period));
132
133        let file = File::open(block_file_path.clone())
134            .unwrap_or_else(|_| panic!("Cannot open file: {:?}", block_file_path));
135        let mut reader = std::io::BufReader::new(file);
136        let mut buffer = Vec::new();
137        reader
138            .read_to_end(&mut buffer)
139            .expect("Unable to read from disk");
140
141        Some(buffer)
142    }
143}

145/// A storage backend that uses RocksDB as the underlying storage engine.
146pub struct RocksDBStorageBackend {
147    db: rocksdb::DB,
148    slots_saved: VecDeque<Slot>,
149    max_blocks: u64,
150}
151
152impl RocksDBStorageBackend {
153    /// Creates a new instance of `RocksDBStorageBackend` with the given path.
154    pub fn new(path: PathBuf, max_blocks: u64) -> Self {
155        let mut opts = Options::default();
156        opts.create_if_missing(true);
157        opts.set_compression_type(DBCompressionType::Lz4);
158        opts.set_max_open_files(8);
159
160        let db = rocksdb::DB::open(&opts, path.clone())
161            .unwrap_or_else(|_| panic!("Failed to create storage db at {:?}", path));
162
163        let slots_saved = load_saved_slots_from_db(&db);
164        Self {
165            db,
166            slots_saved,
167            max_blocks,
168        }
169    }
170}
171
172impl StorageBackend for RocksDBStorageBackend {
173    fn write(&mut self, slot: &Slot, value: &[u8]) {
174        if self.slots_saved.len() >= self.max_blocks as usize {
175            let slot_to_remove = self.slots_saved.pop_front().unwrap();
176            self.db
177                .delete(slot_to_remove.to_bytes_key())
178                .expect("Unable to delete block from db");
179        }
180        self.db
181            .put(slot.to_bytes_key(), value)
182            .expect("Unable to write block to db");
183        self.slots_saved.push_back(*slot);
184    }
185
186    fn read(&self, slot: &Slot) -> Option<Vec<u8>> {
187        match self.db.get(slot.to_bytes_key()) {
188            Ok(val) => val,
189            Err(e) => {
190                println!("Error: {} reading key {:?}", e, slot.to_bytes_key());
191                None
192            }
193        }
194    }
195}

218    fn file_backend_retention_survives_restart() {
219        // Regression test for F59: after a restart the retention cap must keep
220        // accounting for block dumps already present on disk.
221        let dir = std::env::temp_dir().join(format!("massa_f59_test_{}", std::process::id()));
222        std::fs::create_dir_all(&dir).unwrap();
223        // start from a clean folder
224        for entry in std::fs::read_dir(&dir).unwrap().flatten() {
225            let _ = std::fs::remove_file(entry.path());
226        }
227
228        let value = vec![1, 2, 3];
229        let s1 = Slot::new(1, 0);
230        let s2 = Slot::new(2, 0);
231        let s3 = Slot::new(3, 0);
232
233        // First run: persist two dumps (cap = 2, so both are kept).
234        {
235            let mut storage = FileStorageBackend::new(dir.clone(), 2);
236            storage.write(&s1, &value);
237            storage.write(&s2, &value);
238        }
239
240        // Simulated restart: the queue must be rebuilt from the two files on disk,
241        // so writing a third dump evicts the oldest one instead of bypassing the cap.
242        {
243            let mut storage = FileStorageBackend::new(dir.clone(), 2);
244            storage.write(&s3, &value);
245        }
246
247        assert!(
248            !dir.join("block_slot_0_1.bin").exists(),
249            "oldest dump should have been evicted after restart"
250        );
251        assert!(dir.join("block_slot_0_2.bin").exists());
252        assert!(dir.join("block_slot_0_3.bin").exists());
253
254        // cleanup
255        for entry in std::fs::read_dir(&dir).unwrap().flatten() {
256            let _ = std::fs::remove_file(entry.path());
257        }
258        let _ = std::fs::remove_dir(&dir);
259    }
