19/// Converts a gRPC `grpc_model::DenunciationIndex` into a DenunciationIndex
20pub fn to_denunciation_index(
21    value: grpc_model::DenunciationIndex,
22) -> Result<DenunciationIndex, ModelsError> {
23    match value
24        .entry
25        .ok_or_else(|| ModelsError::ErrorRaised("no denunciation_index found".to_string()))?
26    {
27        grpc_model::denunciation_index::Entry::BlockHeader(block_header) => {
28            Ok(DenunciationIndex::BlockHeader {
29                slot: block_header
30                    .slot
31                    .ok_or_else(|| ModelsError::ErrorRaised("no slot found".to_string()))?
32                    .into(),
33            })
34        }
35        grpc_model::denunciation_index::Entry::Endorsement(endorsement) => {
36            Ok(DenunciationIndex::Endorsement {
37                slot: endorsement
38                    .slot
39                    .ok_or_else(|| ModelsError::ErrorRaised("no slot found".to_string()))?
40                    .into(),
41                index: endorsement.index,
42            })
43        }
44    }
45}

172impl From<OperationType> for grpc_model::OperationType {
173    fn from(operation_type: OperationType) -> grpc_model::OperationType {
174        let mut grpc_operation_type = grpc_model::OperationType::default();
175        match operation_type {
176            OperationType::Transaction {
177                recipient_address,
178                amount,
179            } => {
180                let transaction = grpc_model::Transaction {
181                    recipient_address: recipient_address.to_string(),
182                    amount: Some(amount.into()),
183                };
184                grpc_operation_type.r#type =
185                    Some(grpc_model::operation_type::Type::Transaction(transaction));
186            }
187            OperationType::RollBuy { roll_count } => {
188                let roll_buy = grpc_model::RollBuy { roll_count };
189                grpc_operation_type.r#type =
190                    Some(grpc_model::operation_type::Type::RollBuy(roll_buy));
191            }
192            OperationType::RollSell { roll_count } => {
193                let roll_sell = grpc_model::RollSell { roll_count };
194                grpc_operation_type.r#type =
195                    Some(grpc_model::operation_type::Type::RollSell(roll_sell));
196            }
197            OperationType::ExecuteSC {
198                data,
199                max_gas,
200                max_coins,
201                datastore,
202            } => {
203                let execute_sc = grpc_model::ExecuteSc {
204                    data,
205                    max_coins: max_coins.to_raw(),
206                    max_gas,
207                    datastore: datastore
208                        .into_iter()
209                        .map(|(key, value)| grpc_model::BytesMapFieldEntry { key, value })
210                        .collect(),
211                };
212                grpc_operation_type.r#type =
213                    Some(grpc_model::operation_type::Type::ExecuteSc(execute_sc));
214            }
215            OperationType::CallSC {
216                target_addr,
217                target_func,
218                param,
219                max_gas,
220                coins,
221            } => {
222                let call_sc = grpc_model::CallSc {
223                    target_address: target_addr.to_string(),
224                    target_function: target_func,
225                    parameter: param,
226                    max_gas,
227                    coins: Some(coins.into()),
228                };
229                grpc_operation_type.r#type =
230                    Some(grpc_model::operation_type::Type::CallSc(call_sc));
231            }
232        }
233
234        grpc_operation_type
235    }
236}
