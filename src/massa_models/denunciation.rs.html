3//! Denunciation intro
4//!
5//! Currently, nothing prevents a user from producing multiple blocks (for the same Slot) or endorsing a block multiple times.
6//! If an invalid Denunciation will just be ignored, a valid Denunciation will slash some locked rolls.
7//! Note that this proposal aims at dissuading 'rational' users from causing damage but does help against 'byzantine' actors
8//! that just want to disturb or break the blockclique (at any cost).
9//!
10//! Denunciation structure
11//!
12//! A denunciation embeds some information such as
13//! Slot, the slot from the block header or the endorsements (+ index)
14//! A public key: the public key of the secure share endorsement or secured header
15//! 2 Hashes & 2 Signatures
16//!
17//! All of this constitutes a proof (and is verifiable) that a user produced multiple (at least 2) blocks or endorsements
18//!
19//! Denunciation creation
20//!
21//! Denunciations are created in a Denunciation pool, receiving new blocks & block headers & endorsements from various places.
22//! The denunciation pool is also responsible of returning a list of denunciations to insert into a new block header.
23//! After execution, the denunciation is kept for some time inside `executed_denunciations`
24//! in order to prevent multiple executions. Note that this structure is part of the final state hash and thus is bootstrapped.
25//!
26//! Denunciation execution
27//!
28//! When a Denunciation is proven valid, we then need to ensure that the user has sufficient funds. If we cannot deduct the funds on node
29//! balance or staked rolls, we will use 'locked' rolls` (or Deferred credits).
30//! Selling a roll will lock it for some time (== 4 cycles). Note that it restricts the time,
31//! a denunciation can be produced (A constant value will be defined for this).
32//!
33//! Punishment model
34//!
35//! A valid denunciation slashes locked rolls only. It does not forfeit slot rewards
36//! (`EndorsementCreatorReward`, `EndorsementTargetReward`, `BlockCreatorReward`) for the
37//! denouncing block. A validator denounced in a block may still receive those rewards in the
38//! same slot. That is intentional: excluding denounced addresses from same-slot rewards would
39//! make the effective penalty depend on whether they were also selected by draws for that slot,
40//! which is not a coherent punishment model.

106    }

209    /// Check if it is a Denunciation for this endorsement
210    ///
211    /// `sig_chain_id` selects the signed-hash layout used to reconstruct the
212    /// signature hash (see [`EndorsementDenunciation::compute_hash_for_sig_verif`]).
213    pub fn is_also_for_endorsement(
214        &self,
215        s_endorsement: &SecureShareEndorsement,
216        sig_chain_id: Option<u64>,
217    ) -> Result<bool, DenunciationError> {
218        match self {
219            Denunciation::BlockHeader(_) => Ok(false),
220            Denunciation::Endorsement(endo_de) => {
221                let content_hash = s_endorsement.id.get_hash();
222
223                let hash = EndorsementDenunciation::compute_hash_for_sig_verif(
224                    &endo_de.public_key,
225                    &endo_de.slot,
226                    &endo_de.index,
227                    content_hash,
228                    sig_chain_id,
229                );
230
231                Ok(endo_de.slot == s_endorsement.content.slot
232                    && endo_de.index == s_endorsement.content.index
233                    && endo_de.public_key == s_endorsement.content_creator_pub_key
234                    && endo_de.hash_1 != *content_hash
235                    && endo_de.hash_2 != *content_hash
236                    && endo_de
237                        .public_key
238                        .verify_signature(&hash, &s_endorsement.signature)
239                        .is_ok())
240            }
241        }
242    }

246    /// `sig_chain_id` selects the signed-hash layout used to reconstruct the

277    /// Check if Denunciation is valid using the legacy (chain-agnostic) signed-hash
278    /// layout. Kept for backward compatibility with tests / pre-activation data.
279    /// Production code should use [`Self::is_valid_with_chain`] with the
280    /// `sig_chain_id` derived from the denounced slot's network-agreed active
281    /// version.
282    pub fn is_valid(&self) -> bool {
283        self.is_valid_with_chain(None)
284    }
285
286    /// Check if Denunciation is valid, reconstructing signatures with the
287    /// chain-scoped signed-hash layout selected by `sig_chain_id` (`None` = legacy,
288    /// `Some(chain_id)` = chain-scoped).
289    ///
290    /// This is the chain-aware variant that prevents cross-chain replay of
291    /// endorsements / block headers as denunciations (F90 / PDF #11): when
292    /// `sig_chain_id` is `Some(local_chain_id)`, a denunciation combining two
293    /// messages signed on different chains no longer verifies, because the second
294    /// message's signature was produced over a hash folding in a different chain id.
295    pub fn is_valid_with_chain(&self, sig_chain_id: Option<u64>) -> bool {
296        let (signature_1, signature_2, hash_1, hash_2, public_key) = match self {
297            Denunciation::Endorsement(de) => {
298                let hash_1 = EndorsementDenunciation::compute_hash_for_sig_verif(
299                    &de.public_key,
300                    &de.slot,
301                    &de.index,
302                    &de.hash_1,
303                    sig_chain_id,
304                );
305                let hash_2 = EndorsementDenunciation::compute_hash_for_sig_verif(
306                    &de.public_key,
307                    &de.slot,
308                    &de.index,
309                    &de.hash_2,
310                    sig_chain_id,
311                );
312
313                (
314                    de.signature_1,
315                    de.signature_2,
316                    hash_1,
317                    hash_2,
318                    de.public_key,
319                )
320            }
321            Denunciation::BlockHeader(de) => {
322                let hash_1 = BlockHeaderDenunciation::compute_hash_for_sig_verif(
323                    &de.public_key,
324                    &de.slot,
325                    &de.hash_1,
326                    sig_chain_id,
327                );
328                let hash_2 = BlockHeaderDenunciation::compute_hash_for_sig_verif(
329                    &de.public_key,
330                    &de.slot,
331                    &de.hash_2,
332                    sig_chain_id,
333                );
334
335                (
336                    de.signature_1,
337                    de.signature_2,
338                    hash_1,
339                    hash_2,
340                    de.public_key,
341                )
342            }
343        };
344
345        hash_1 != hash_2
346            && public_key.verify_signature(&hash_1, &signature_1).is_ok()
347            && public_key.verify_signature(&hash_2, &signature_2).is_ok()
348    }

376    /// * A final slot period (for example in order to cleanup denunciation pool caches)

400/// Create a new Denunciation from 2 SecureShareEndorsement using the chain-scoped
401/// signed-hash layout selected by `sig_chain_id` (`None` = legacy, `Some(chain_id)`
402/// = chain-scoped). The same `sig_chain_id` must be used to sign the endorsements
403/// and to verify the resulting denunciation.
404impl
405    TryFrom<(
406        &SecureShareEndorsement,
407        &SecureShareEndorsement,
408        Option<u64>,
409    )> for Denunciation
410{
411    type Error = DenunciationError;
412
413    fn try_from(
414        (s_e1, s_e2, sig_chain_id): (
415            &SecureShareEndorsement,
416            &SecureShareEndorsement,
417            Option<u64>,
418        ),
419    ) -> Result<Self, Self::Error> {
420        // In order to create a Denunciation, there should be the same
421        // slot, index & public key
422        if s_e1.content.slot != s_e2.content.slot
423            || s_e1.content.index != s_e2.content.index
424            || s_e1.content_creator_pub_key != s_e2.content_creator_pub_key
425            || s_e1.id == s_e2.id
426        {
427            return Err(DenunciationError::InvalidInput(format!(
428                "Not the same slot, index or public key or same hash for {:?} & {:?}",
429                s_e1, s_e2
430            )));
431        }
432
433        // Check sig of s_e1 with s_e1.public_key, s_e1.slot, s_e1.index
434        let s_e1_hash_content = s_e1.id.get_hash();
435        let s_e1_hash = EndorsementDenunciation::compute_hash_for_sig_verif(
436            &s_e1.content_creator_pub_key,
437            &s_e1.content.slot,
438            &s_e1.content.index,
439            s_e1_hash_content,
440            sig_chain_id,
441        );
442        // Check sig of s_e2 but with s_e1.public_key, s_e1.slot, s_e1.index
443        let s_e2_hash_content = s_e2.id.get_hash();
444        let s_e2_hash = EndorsementDenunciation::compute_hash_for_sig_verif(
445            &s_e1.content_creator_pub_key,
446            &s_e1.content.slot,
447            &s_e1.content.index,
448            s_e2_hash_content,
449            sig_chain_id,
450        );
451
452        s_e1.content_creator_pub_key
453            .verify_signature(&s_e1_hash, &s_e1.signature)?;
454        s_e1.content_creator_pub_key
455            .verify_signature(&s_e2_hash, &s_e2.signature)?;
456
457        Ok(Denunciation::Endorsement(EndorsementDenunciation {
458            public_key: s_e1.content_creator_pub_key,
459            slot: s_e1.content.slot,
460            index: s_e1.content.index,
461            signature_1: s_e1.signature,
462            signature_2: s_e2.signature,
463            hash_1: *s_e1_hash_content,
464            hash_2: *s_e2_hash_content,
465        }))
466    }
467}

1219/// Create a new Denunciation from 2 DenunciationPrecursor using the chain-scoped
1220/// signed-hash layout selected by `sig_chain_id` (`None` = legacy, `Some(chain_id)`
1221/// = chain-scoped).
1222impl TryFrom<(&DenunciationPrecursor, &DenunciationPrecursor, Option<u64>)> for Denunciation {
1223    type Error = DenunciationError;
1224
1225    fn try_from(
1226        (de_p_1, de_p_2, sig_chain_id): (
1227            &DenunciationPrecursor,
1228            &DenunciationPrecursor,
1229            Option<u64>,
1230        ),
1231    ) -> Result<Self, Self::Error> {
1232        match (de_p_1, de_p_2) {
1233            (
1234                DenunciationPrecursor::BlockHeader(de_p_blkh_1),
1235                DenunciationPrecursor::BlockHeader(de_p_blkh_2),
1236            ) => {
1237                // Cannot use the same block header (here: block header denunciation precursor) twice
1238                if de_p_blkh_1.slot != de_p_blkh_2.slot
1239                    || de_p_blkh_1.public_key != de_p_blkh_2.public_key
1240                    || de_p_blkh_1.hash == de_p_blkh_2.hash
1241                {
1242                    return Err(DenunciationError::InvalidInput(
1243                        format!("Not the same slot or public key or same hash for de precursor: {:?} & {:?}", de_p_blkh_1, de_p_blkh_2)
1244                    ));
1245                }
1246
1247                // Check sig
1248                let de_p_blkh_1_hash = BlockHeaderDenunciation::compute_hash_for_sig_verif(
1249                    &de_p_blkh_1.public_key,
1250                    &de_p_blkh_1.slot,
1251                    &de_p_blkh_1.hash,
1252                    sig_chain_id,
1253                );
1254                let de_p_blkh_2_hash = BlockHeaderDenunciation::compute_hash_for_sig_verif(
1255                    &de_p_blkh_2.public_key,
1256                    &de_p_blkh_2.slot,
1257                    &de_p_blkh_2.hash,
1258                    sig_chain_id,
1259                );
1260
1261                de_p_blkh_1
1262                    .public_key
1263                    .verify_signature(&de_p_blkh_1_hash, &de_p_blkh_1.signature)?;
1264                de_p_blkh_1
1265                    .public_key
1266                    .verify_signature(&de_p_blkh_2_hash, &de_p_blkh_2.signature)?;
1267
1268                Ok(Denunciation::BlockHeader(BlockHeaderDenunciation {
1269                    public_key: de_p_blkh_1.public_key,
1270                    slot: de_p_blkh_1.slot,
1271                    signature_1: de_p_blkh_1.signature,
1272                    signature_2: de_p_blkh_2.signature,
1273                    hash_1: de_p_blkh_1.hash,
1274                    hash_2: de_p_blkh_2.hash,
1275                }))
1276            }
1277            (
1278                DenunciationPrecursor::Endorsement(de_p_endo_1),
1279                DenunciationPrecursor::Endorsement(de_p_endo_2),
1280            ) => {
1281                // Cannot use the same endorsement (here: endorsement denunciation) twice
1282                if de_p_endo_1.slot != de_p_endo_2.slot
1283                    || de_p_endo_1.index != de_p_endo_2.index
1284                    || de_p_endo_1.public_key != de_p_endo_2.public_key
1285                    || de_p_endo_1.hash == de_p_endo_2.hash
1286                {
1287                    return Err(DenunciationError::InvalidInput(
