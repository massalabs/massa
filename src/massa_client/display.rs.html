61pub enum Style {
62    /// Any information that identifies an element
63    Id,
64    /// If a process is ongoing, not final, will change in the future
65    Pending,
66    /// If a process is finished, fixed, and won't evolve in the future
67    Finished,
68    /// Good things in general, success of an operation
69    Good,
70    /// Bad things in general, failure of an operation
71    Bad,
72    /// For any information that is unknown
73    Unknown,
74    /// Any amount of Massa coin displayed
75    Coins,
76    /// Any information related to the protocol, staking, peers and the consensus
77    Protocol,
78    /// Any information concerning a block of the blockchain
79    Block,
80    /// For cryptographic signature
81    Signature,
82    /// For any information related to the wallet, addresses or public keys
83    Wallet,
84    /// For any secret information
85    Secret,
86    /// To separate some information on the screen by barely visible characters
87    Separator,
88    /// When displaying a timestamp or date
89    Time,
90}

92impl Style {
93    fn style<T: ToString>(&self, msg: T) -> console::StyledObject<std::string::String> {
94        style(msg.to_string()).color256(match self {
95            Style::Id => 218,        // #ffafd7
96            Style::Pending => 172,   // #d78700
97            Style::Finished => 81,   // #5fd7ff
98            Style::Good => 112,      // #87d700
99            Style::Bad => 160,       // #d70000
100            Style::Unknown => 248,   // #a8a8a8
101            Style::Coins => 141,     // #af87ff
102            Style::Protocol => 151,  // #afd7af
103            Style::Block => 158,     // #afffd7
104            Style::Signature => 220, // #ffd700
105            Style::Wallet => 213,    // #ff87ff
106            Style::Secret => 64,     // #5f8700
107            Style::Separator => 239, // #4e4e4e
108            Style::Time => 117,      // #87d7ff
109        })
110    }
111}

117impl dyn Output {
118    pub(crate) fn stdout_json(&self) -> anyhow::Result<()> {
119        let json = &mut serde_json::Serializer::new(std::io::stdout());
120        let mut format: Box<dyn Serializer> = Box::new(<dyn Serializer>::erase(json));
121        self.erased_serialize(&mut format)?;
122        Ok(())
123    }

132impl Output for ExtendedWallet {
133    fn pretty_print(&self) {
134        if self.0.is_empty() {
135            client_warning!("your wallet does not contain any key, use 'wallet_generate_secret_key' to generate a new key and add it to your wallet");
136        }
137        println!("{}", Style::Separator.style("====="));
138        for entry in self.0.values() {
139            if entry.show_keys {
140                println!("Secret key: {}", Style::Secret.style(&entry.keypair));
141                println!(
142                    "Public key: {}",
143                    Style::Wallet.style(entry.keypair.get_public_key())
144                );
145            }
146            println!(
147                "Address: {} (thread {}):",
148                Style::Wallet.style(entry.address_info.address),
149                Style::Protocol.style(entry.address_info.thread),
150            );
151            println!(
152                "\tBalance: {}={}, {}={}",
153                Style::Finished.style("final"),
154                Style::Coins.style(entry.address_info.final_balance),
155                Style::Pending.style("candidate"),
156                Style::Coins.style(entry.address_info.candidate_balance),
157            );
158            println!(
159                "\tRolls: {}={}, {}={}, {}={}",
160                Style::Good.style("active"),
161                Style::Protocol.style(entry.address_info.active_rolls),
162                Style::Finished.style("final"),
163                Style::Protocol.style(entry.address_info.final_rolls),
164                Style::Pending.style("candidate"),
165                Style::Protocol.style(entry.address_info.candidate_rolls),
166            );
167            println!("{}", Style::Separator.style("====="));
168        }
169    }
170}

218impl Output for NodeStatus {
219    fn pretty_print(&self) {
220        println!("Node's ID: {}", Style::Id.style(self.node_id));
221        if self.node_ip.is_some() {
222            println!(
223                "Node's IP: {}",
224                Style::Protocol.style(self.node_ip.unwrap())
225            );
226        } else {
227            println!("{}", Style::Unknown.style("No routable IP set"));
228        }
229        println!();
230
231        println!("Version: {}", Style::Id.style(self.version));
232        self.config.pretty_print();
233        println!();
234
235        println!("Current time: {}", self.current_time.format_instant());
236        println!(
237            "Current cycle: {}",
238            Style::Protocol.style(self.current_cycle)
239        );
240        if self.last_slot.is_some() {
241            println!(
242                "Last slot: {}",
243                Style::Protocol.style(self.last_slot.unwrap())
244            );
245        }
246        println!("Next slot: {}", Style::Protocol.style(self.next_slot));
247        println!();
248
249        self.consensus_stats.pretty_print();
250
251        println!("Pool stats:");
252        println!(
253            "\tOperations count: {}",
254            Style::Protocol.style(self.pool_stats.0)
255        );
256        println!(
257            "\tEndorsements count: {}",
258            Style::Protocol.style(self.pool_stats.1)
259        );
260        println!();
261
262        self.network_stats.pretty_print();
263        self.execution_stats.pretty_print();
264
265        if !self.connected_nodes.is_empty() {
266            println!("Connected nodes:");
267            for (node_id, (ip_addr, is_outgoing)) in &self.connected_nodes {
268                println!(
269                    "Node's ID: {} / IP address: {} / {} connection",
270                    Style::Id.style(node_id),
271                    Style::Protocol.style(ip_addr),
272                    if *is_outgoing { "Out" } else { "In" }
273                )
274            }
275        }
276
277        println!();
278        println!("Chain id: {}", self.chain_id);
279        println!("Current MIP version: {}", self.current_mip_version);
280        if let Some(max_datastore_keys_query) = self.max_datastore_keys_query {
281            println!(
282                "Max datastore keys per query: {}",
283                Style::Protocol.style(max_datastore_keys_query)
284            );
285        }
286    }
287}

420impl Output for Vec<AddressInfo> {
421    fn pretty_print(&self) {
422        for info in self {
423            println!("{}", Style::Separator.style("========"));
424            println!(
425                "Address {} (thread {}):",
426                Style::Wallet.style(info.address),
427                Style::Protocol.style(info.thread),
428            );
429            println!(
430                "\tBalance: {}={}, {}={}",
431                Style::Finished.style("final"),
432                Style::Coins.style(info.final_balance),
433                Style::Pending.style("candidate"),
434                Style::Coins.style(info.candidate_balance),
435            );
436            println!(
437                "\tRolls: {}={}, {}={}",
438                Style::Finished.style("final"),
439                Style::Protocol.style(info.final_roll_count),
440                Style::Pending.style("candidate"),
441                Style::Protocol.style(info.candidate_roll_count),
442            );
443
444            print!("\tLocked coins:");
445            if info.deferred_credits.is_empty() {
446                println!(" {}", Style::Coins.style("0"));
447            } else {
448                println!();
449                for slot_amount in &info.deferred_credits {
450                    println!(
451                        "\t\t{} locked coins will be unlocked at slot {}",
452                        Style::Coins.style(slot_amount.amount),
453                        Style::Protocol.style(slot_amount.slot),
454                    );
455                }
456            }
457            if !info.cycle_infos.is_empty() {
458                println!("\tCycle infos:");
459            }
460            for cycle_info in &info.cycle_infos {
461                println!(
462                    "\t\tCycle {} ({}): produced {} and missed {} blocks{}",
463                    Style::Protocol.style(cycle_info.cycle),
464                    if cycle_info.is_final {
465                        Style::Finished.style("final")
466                    } else {
467                        Style::Pending.style("candidate")
468                    },
469                    Style::Good.style(cycle_info.ok_count),
470                    Style::Bad.style(cycle_info.nok_count),
471                    match cycle_info.active_rolls {
472                        Some(rolls) => format!(" with {} active rolls", Style::Good.style(rolls)),
473                        None => "".into(),
474                    },
475                );
476            }
477        }
478    }
479}

505impl Output for Vec<OperationInfo> {
506    fn pretty_print(&self) {
507        for info in self {
508            println!("{}", style("==========").color256(237));
509            print!("Operation {}", Style::Id.style(info.id));
510            if info.in_pool {
511                print!(", {}", Style::Pending.style("in pool"));
512            }
513            if let Some(f) = info.is_operation_final {
514                print!(
515                    ", operation is {}",
516                    if f {
517                        Style::Finished.style("final")
518                    } else {
519                        Style::Pending.style("not final")
520                    }
521                );
522            } else {
523                print!(", finality {}", Style::Unknown.style("unknown"));
524            }
525            println!(
526                ", {}",
527                match info.op_exec_status {
528                    Some(true) => Style::Good.style("success"),
529                    Some(false) => Style::Bad.style("failed"),
530                    None => Style::Unknown.style("unknown status"),
531                }
532            );
533            if info.in_blocks.is_empty() {
534                println!("{}", Style::Block.style("Not in any blocks"));
535            } else {
536                println!("In blocks:");
537                for bid in info.in_blocks.iter() {
538                    println!("\t- {}", Style::Block.style(bid));
539                }
540            }
541            println!(
542                "Signature: {}",
543                Style::Signature.style(info.operation.signature)
544            );
545            println!(
546                "Creator pubkey: {}",
547                Style::Wallet.style(info.operation.content_creator_pub_key)
548            );
549            println!(
550                "Creator address: {}",
551                Style::Wallet.style(info.operation.content_creator_address)
552            );
553            println!("Fee: {}", Style::Coins.style(info.operation.content.fee));
554            println!(
555                "Expire period: {}",
556                Style::Pending.style(info.operation.content.expire_period)
557            );
558            println!(
559                "Operation type: {}",
560                Style::Id.style(&info.operation.content.op)
561            );
562        }
563    }
564}

610impl Output for MasOGBalanceResponse {
611    fn pretty_print(&self) {
612        println!("{}", Style::Separator.style("========"));
613        println!(
614            "MASOG Total supply: {}",
615            Style::Coins.style(self.total_supply)
616        );
617        for balance in self.balances.iter() {
618            println!("{}", Style::Separator.style("========"));
619            println!("Address {} :", Style::Wallet.style(balance.address),);
620            println!(
621                "\tBalance: {}={}, {}={}",
622                Style::Finished.style("final"),
623                Style::Coins.style(balance.final_balance),
624                Style::Pending.style("candidate"),
625                Style::Coins.style(balance.candidate_balance),
626            );
627            println!(
628                "\tVoting power: {}%",
629                Style::Protocol.style(format!("{:.4}", balance.voting_power))
630            );
631        }
632    }
