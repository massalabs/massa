21fn group_parameters(parameters: Vec<String>) -> Vec<String> {
22    let mut new_parameters = Vec::new();
23    let mut has_opening_simple_quote = false;
24    let mut temp_simple_quote = String::new();
25    let mut has_opening_double_quote = false;
26    let mut temp_double_quote = String::new();
27    for param in parameters {
28        let mut chars = param.chars();
29        match chars.next() {
30            Some('\'') if !has_opening_double_quote => {
31                has_opening_simple_quote = true;
32                temp_simple_quote = param.clone();
33                temp_simple_quote.remove(0);
34            }
35            Some('"') if !has_opening_simple_quote => {
36                has_opening_double_quote = true;
37                temp_double_quote = param.clone();
38                temp_double_quote.remove(0);
39            }
40            Some(_) if has_opening_simple_quote => {
41                temp_simple_quote.push(' ');
42                temp_simple_quote.push_str(&param);
43            }
44            Some(_) if has_opening_double_quote => {
45                temp_double_quote.push(' ');
46                temp_double_quote.push_str(&param);
47            }
48            Some(_) => new_parameters.push(param.clone()),
49            None => continue,
50        };
51        match chars.last() {
52            Some('\'') if has_opening_simple_quote => {
53                has_opening_simple_quote = false;
54                let mut to_add = temp_simple_quote.clone();
55                to_add.pop();
56                new_parameters.push(to_add);
57            }
58            Some('"') if has_opening_double_quote => {
59                has_opening_double_quote = false;
60                let mut to_add = temp_double_quote.clone();
61                to_add.pop();
62                new_parameters.push(to_add);
63            }
64            Some(_) => continue,
65            None => continue,
66        }
67    }
68    new_parameters
69}

79pub(crate) async fn run(
80    client: &mut Client,
81    wallet_path: &Path,
82    args_password: Option<String>,
83) -> Result<()> {
84    massa_fancy_ascii_art_logo!();
85    println!("Use 'exit' or 'CTRL+D or CTRL+C' to quit the prompt");
86    println!("Use the Up/Down arrows to scroll through history");
87    println!("Use the Right arrow or Tab to complete your command");
88    println!("Use the Enter key to execute your command");
89    crate::cmds::help();
90    let h = MyHelper {
91        completer: MassaCompleter::new(),
92        validator: MatchingBracketValidator::new(),
93    };
94    let config = Config::builder()
95        .auto_add_history(false)
96        .completion_prompt_limit(100)
97        .completion_type(CompletionType::List)
98        .max_history_size(SETTINGS.history)?
99        .build();
100
101    let mut rl = Editor::with_config(config)?;
102    rl.set_helper(Some(h));
103    if rl.load_history(&SETTINGS.history_file_path).is_err() {
104        println!("No previous history.");
105    }
106
107    let mut wallet_opt = None;
108
109    loop {
110        let readline = rl.readline("command > ");
111        match readline {
112            Ok(line) => {
113                if line.is_empty() {
114                    continue;
115                }
116                let input: Vec<String> =
117                    group_parameters(line.split_whitespace().map(|x| x.to_string()).collect());
118                let cmd: Result<Command, ParseError> = input[0].parse();
119                let parameters = input[1..].to_vec();
120                // Print result of evaluated command
121                match cmd {
122                    Ok(command) => {
123                        // Only add to history if the command does not contain sensitive information
124                        if !command.hide_from_history() {
125                            if let Err(e) = rl.add_history_entry(line.as_str()) {
126                                println!("Failed to append commands history {}", e);
127                            }
128                            if let Err(e) = rl.append_history(&SETTINGS.history_file_path) {
129                                println!("Failed to append commands file history: {}", e);
130                            }
131                        }
132
133                        // Check if we need to prompt the user for their wallet password
134                        if command.is_pwd_needed() && wallet_opt.is_none() {
135                            let password =
136                                match (args_password.clone(), env::var("MASSA_CLIENT_PASSWORD")) {
137                                    (Some(pwd), _) => pwd,
138                                    (_, Ok(pwd)) => pwd,
139                                    _ => ask_password(wallet_path),
140                                };
141
142                            let wallet = match Wallet::new(
143                                wallet_path.to_path_buf(),
144                                password,
145                                client.chain_id,
146                            ) {
147                                Ok(wallet) => wallet,
148                                Err(e) => {
149                                    println!("Could not open wallet: {}", e);
150                                    continue;
151                                }
152                            };
153                            wallet_opt = Some(wallet);
154                        }
155
156                        match command
157                            .run(client, &mut wallet_opt, &parameters, false)
158                            .await
159                        {
160                            Ok(output) => output.pretty_print(),
161                            Err(e) => println!("{}", style(format!("Error: {}", e)).red()),
162                        }
163                    }
164                    Err(_) => {
165                        println!("Command not found!\ntype \"help\" to get the list of commands")
166                    }
167                }
168            }
169            Err(ReadlineError::Interrupted) => {
170                break;
171            }
172            Err(ReadlineError::Eof) => {
173                break;
174            }
175            Err(err) => {
176                println!("Error: {err:?}");
177                break;
178            }
179        }
180    }
181    Ok(())
182}

184struct MassaCompleter {
185    file_completer: FilenameCompleter,
186}
187
188impl MassaCompleter {
189    fn new() -> Self {
190        Self {
191            file_completer: FilenameCompleter::new(),
192        }
193    }
194}
195
196impl Completer for MassaCompleter {
197    type Candidate = Pair;
198    fn complete(
199        &self,
200        line: &str,
201        pos: usize,
202        ctx: &rustyline::Context<'_>,
203    ) -> rustyline::Result<(usize, Vec<Self::Candidate>)> {
204        if line.contains(' ') {
205            self.file_completer.complete(line, pos, ctx)
206        } else {
207            let mut candidates = Vec::new();
208            for cmd in Command::iter() {
209                if cmd.to_string().starts_with(line) {
210                    candidates.push(Pair {
211                        display: cmd.to_string(),
212                        replacement: cmd.to_string(),
213                    });
214                }
215            }
216            Ok((0, candidates))
217        }
218    }
