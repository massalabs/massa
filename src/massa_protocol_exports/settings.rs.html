189impl ProtocolConfig {
190    /// Whether the node is configured to deal with peers that do not have a
191    /// globally routable address, i.e. local networks. Mirrors the
192    /// `allow_local_peers` category setting: as soon as one category accepts
193    /// local peers, non-global endpoints are kept in the peer management
194    /// pipeline and may be dialed. Per-category checks still apply when
195    /// selecting peers to connect out; `NetworkController::try_connect`
196    /// refuses non-global destinations when this returns false.
197    pub fn allow_local_peers(&self) -> bool {
198        self.default_category_info.allow_local_peers
199            || self
200                .peers_categories
201                .values()
202                .any(|category| category.allow_local_peers)
203    }
